**1. Quick Sort**

* **Prompt:** Implement a recursive quick sort function that sorts an array of numbers in ascending order.
* **Example:**

```javascript
function quickSort(arr) {
  if (arr.length <= 1) return arr;

  const pivot = arr[arr.length - 1];
  const left = [];
  const right = [];

  for (let i = 0; i < arr.length - 1; i++) {
    if (arr[i] < pivot) {
      left.push(arr[i]);
    } else {
      right.push(arr[i]);
    }
  }

  return [...quickSort(left), pivot, ...quickSort(right)];
}

// Usage:
// quickSort([5, 3, 7, 6, 2, 9]); // [2, 3, 5, 6, 7, 9]

```

**2. Selection Sort**

* **Prompt:** Implement an in-place selection sort algorithm that iteratively finds the minimum element and places it at the sorted boundary.
* **Example:**

```javascript
function selectionSort(arr) {
  const n = arr.length;
  for (let i = 0; i < n - 1; i++) {
    let minIdx = i;
    for (let j = i + 1; j < n; j++) {
      if (arr[j] < arr[minIdx]) {
        minIdx = j;
      }
    }
    if (minIdx !== i) {
      [arr[i], arr[minIdx]] = [arr[minIdx], arr[i]];
    }
  }
  return arr;
}

// Usage:
// selectionSort([64, 25, 12, 22, 11]); // [11, 12, 22, 25, 64]

```

**12. Debounce II**

* **Prompt:** Implement a debounce utility with `.cancel()` and `.flush()` methods to abort pending executions or invoke them immediately.
* **Example:**

```javascript
function debounce(fn, delay) {
  let timerId = null;
  let lastArgs = null;
  let lastThis = null;

  function debounced(...args) {
    lastArgs = args;
    lastThis = this;
    clearTimeout(timerId);
    timerId = setTimeout(() => {
      fn.apply(lastThis, lastArgs);
      timerId = null;
    }, delay);
  }

  debounced.cancel = () => {
    clearTimeout(timerId);
    timerId = null;
  };

  debounced.flush = () => {
    if (timerId) {
      clearTimeout(timerId);
      fn.apply(lastThis, lastArgs);
      timerId = null;
    }
  };

  return debounced;
}

```

**14. Queue**

* **Prompt:** Implement a standard Queue data structure with `enqueue`, `dequeue`, `peek`, `isEmpty`, and `size` methods.
* **Example:**

```javascript
class Queue {
  constructor() {
    this.items = {};
    this.head = 0;
    this.tail = 0;
  }

  enqueue(item) {
    this.items[this.tail++] = item;
  }

  dequeue() {
    if (this.isEmpty()) return undefined;
    const item = this.items[this.head];
    delete this.items[this.head++];
    return item;
  }

  peek() {
    return this.items[this.head];
  }

  isEmpty() {
    return this.size() === 0;
  }

  size() {
    return this.tail - this.head;
  }
}
```

**17. Event Emitter / Event Emitter II**

* **Prompt:** Implement an EventEmitter class supporting `on(eventName, listener)`, `emit(eventName, ...args)`, and an `unsubscribe()` method returned by subscription.
* **Example:**

```javascript
class EventEmitter {
  constructor() {
    this.events = new Map();
  }

  subscribe(eventName, callback) {
    if (!this.events.has(eventName)) {
      this.events.set(eventName, new Set());
    }
    const listeners = this.events.get(eventName);
    listeners.add(callback);

    return {
      unsubscribe: () => listeners.delete(callback)
    };
  }

  emit(eventName, ...args) {
    if (!this.events.has(eventName)) return [];
    const results = [];
    for (const callback of this.events.get(eventName)) {
      results.push(callback(...args));
    }
    return results;
  }
}
```

**18. Deep Equal**

* **Prompt:** Implement a function `deepEqual(a, b)` that deeply compares primitives, arrays, nested objects, and distinct reference types for equality.
* **Example:**

```javascript
function deepEqual(a, b) {
  if (Object.is(a, b)) return true;
  if (typeof a !== 'object' || a === null || typeof b !== 'object' || b === null) return false;

  const keysA = Object.keys(a);
  const keysB = Object.keys(b);
  if (keysA.length !== keysB.length) return false;

  for (const key of keysA) {
    if (!Object.prototype.hasOwnProperty.call(b, key) || !deepEqual(a[key], b[key])) {
      return false;
    }
  }

  return true;
}

// Usage:
// deepEqual({ x: [1, 2] }, { x: [1, 2] }); // true
```

* **Prompt:** Implement `textSearch(text, queries)` highlighting all matching query words within a text string by wrapping them with `<b>` tags.
* **Example:**

```javascript
function textSearch(text, queries) {
  if (!text || !queries || queries.length === 0) return text;
  const words = Array.isArray(queries) ? queries : [queries];
  const regex = new RegExp(`(${words.join('|')})`, 'gi');
  return text.replace(regex, '<b>$1</b>');
}

// Usage:
// textSearch('The quick brown fox', ['quick', 'fox']);
// Output: "The <b>quick</b> brown <b>fox</b>"

```

**29. Array.prototype.filter**

* **Prompt:** Implement a custom `myFilter(callback, thisArg)` on the Array prototype.
* **Example:**

```javascript
Array.prototype.myFilter = function (callback, thisArg) {
  const result = [];
  for (let i = 0; i < this.length; i++) {
    if (i in this && callback.call(thisArg, this[i], i, this)) {
      result.push(this[i]);
    }
  }
  return result;
};

// Usage:
// [1, 2, 3, 4].myFilter(x => x % 2 === 0); // [2, 4]

```

**32. Promise.race**

* **Prompt:** Implement `promiseRace(iterable)` that settles immediately with the value or reason of the first resolving or rejecting promise.
* **Example:**

```javascript
function promiseRace(iterable) {
  return new Promise((resolve, reject) => {
    for (const item of iterable) {
      Promise.resolve(item).then(resolve, reject);
    }
  });
}

// Usage:
// promiseRace([new Promise(res => setTimeout(() => res('fast'), 100)), new Promise(res => setTimeout(() => res('slow'), 500))]).then(console.log);
// Output: 'fast'
```

**33. Promise.any**

* **Prompt:** Implement `promiseAny(iterable)` that resolves as soon as any promise fulfills, or rejects with an `AggregateError` if all reject.
* **Example:**

```javascript
function promiseAny(iterable) {
  return new Promise((resolve, reject) => {
    const promises = Array.from(iterable);
    if (promises.length === 0) {
      return reject(new AggregateError([], 'All promises were rejected'));
    }

    const errors = [];
    let rejectedCount = 0;

    promises.forEach((p, idx) => {
      Promise.resolve(p).then(resolve).catch((err) => {
        errors[idx] = err;
        rejectedCount += 1;
        if (rejectedCount === promises.length) {
          reject(new AggregateError(errors, 'All promises were rejected'));
        }
      });
    });
  });
}

```

**34. Data Selection**

* **Prompt:** Implement a query evaluation function `selectData(rows, criteria)` to filter an array of dataset objects matching various logical constraints.
* **Example:**

```javascript
function selectData(rows, { minAge, role } = {}) {
  return rows.filter((row) => {
    if (minAge !== undefined && row.age < minAge) return false;
    if (role !== undefined && row.role !== role) return false;
    return true;
  });
}

// Usage:
// const users = [{ age: 20, role: 'admin' }, { age: 30, role: 'user' }];
// selectData(users, { minAge: 25 }); // [{ age: 30, role: 'user' }]

```
