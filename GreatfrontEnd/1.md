```typescript
function search(nums: number[], target: number): number {
  let left = 0;
  let right = nums.length - 1;

  while (left <= right) {
    const mid = (left + right) >> 1;

    if (nums[mid] === target) return mid;

    // Check if the left half is sorted
    if (nums[left] <= nums[mid]) {
      if (nums[left] <= target && target < nums[mid]) {
        right = mid - 1;
      } else {
        left = mid + 1;
      }
    } else {
      // Right half is sorted
      if (nums[mid] < target && target <= nums[right]) {
        left = mid + 1;
      } else {
        right = mid - 1;
      }
    }
  }

  return -1;
}
```

```typescript
function hasCycle(head: ListNode | null): boolean {

  let slow: ListNode | null = head;
  let fast: ListNode | null = head;
```

```typescript
export function lengthOfLIS(nums: number[]): number {
  if (nums.length === 0) return 0;

  const tails: number[] = [];

  for (const num of nums) {
    let left = 0;
    let right = tails.length;

    // Binary search for insertion point
    while (left < right) {
      const mid = (left + right) >> 1;
      if (tails[mid] < num) {
        left = mid + 1;
      } else {
        right = mid;
      }
    }

    tails[left] = num;
  }

  return tails.length;
```

```typescript
export function coinChange(coins: number[], amount: number): number {
  const dp: number[] = new Array(amount + 1).fill(Infinity);
  dp[0] = 0;

  for (let i = 1; i <= amount; i++) {
    for (const coin of coins) {
      if (i - coin >= 0) {
```

### 6. Task Coordination (Task Scheduler with Cooldown)

> **Problem Statement:** Implement a function to find minimum intervals for tasks with cooldown.

```typescript
export function leastInterval(tasks: string[], n: number): number {
  const counts = new Map<string, number>();
  let maxFreq = 0;

  for (const task of tasks) {
    const count = (counts.get(task) || 0) + 1;
    counts.set(task, count);
    maxFreq = Math.max(maxFreq, count);
  }

  let maxFreqCount = 0;
  for (const count of counts.values()) {
    if (count === maxFreq) {
      maxFreqCount++;
    }
  }

  // Calculate required slots framed by max frequency tasks
  const emptySlots = (maxFreq - 1) * (n + 1) + maxFreqCount;

  return Math.max(tasks.length, emptySlots);
}
```

```typescript
type WhereClause<T> = Partial<T>;

class ModelDelegate<T extends { id: string | number }> {
  private records: Map<string | number, T> = new Map();

  async create(args: { data: T }): Promise<T> {
    const record = { ...args.data };
    this.records.set(record.id, record);
    return record;
  }

  async findUnique(args: { where: { id: string | number } }): Promise<T | null> {
    return this.records.get(args.where.id) ?? null;
  }

  async findMany(args?: { where?: WhereClause<T> }): Promise<T[]> {
    const all = Array.from(this.records.values());
    if (!args?.where) return all;

    return all.filter((item) =>
      Object.entries(args.where!).every(([k, v]) => item[k as keyof T] === v)
    );
  }

  async update(args: { where: { id: string | number }; data: Partial<T> }): Promise<T> {
    const record = this.records.get(args.where.id);
    if (!record) throw new Error('Record not found');

    const updated = { ...record, ...args.data };
    this.records.set(args.where.id, updated);
    return updated;
  }

  async delete(args: { where: { id: string | number } }): Promise<T> {
    const record = this.records.get(args.where.id);
    if (!record) throw new Error('Record not found');

    this.records.delete(args.where.id);
    return record;
  }
}
```

### 2. Mini Object-Relational Mapper II (Rich Filtering & Sorting)

> **Problem Statement:** Implement a simplified Prisma-like in-memory object-relational mapper (ORM) with richer filtering and sorting.

```typescript
type FilterOperator<V> = {
  equals?: V;
  in?: V[];
  notIn?: V[];
  lt?: number;
  lte?: number;
  gt?: number;
  gte?: number;
  contains?: string;
  startsWith?: string;
  endsWith?: string;
};

type ComplexWhere<T> = {
  [K in keyof T]?: T[K] | FilterOperator<T[K]>;
} & {
  AND?: ComplexWhere<T>[];
  OR?: ComplexWhere<T>[];
  NOT?: ComplexWhere<T>;
};

type OrderBy<T> = {
  [K in keyof T]?: 'asc' | 'desc';
};

export class AdvancedModelDelegate<T extends { id: string | number }> {
  private records: Map<string | number, T> = new Map();

  async create(args: { data: T }): Promise<T> {
    const record = { ...args.data };
    this.records.set(record.id, record);
    return record;
  }

  private matchesCondition(val: any, cond: any): boolean {
    if (cond === undefined) return true;
    if (typeof cond !== 'object' || cond === null || Array.isArray(cond)) {
      return val === cond;
    }

    const ops = cond as FilterOperator<any>;
    if (ops.equals !== undefined && val !== ops.equals) return false;
    if (ops.in !== undefined && !ops.in.includes(val)) return false;
    if (ops.notIn !== undefined && ops.notIn.includes(val)) return false;
    if (ops.gt !== undefined && val <= ops.gt) return false;
    if (ops.gte !== undefined && val < ops.gte) return false;
    if (ops.lt !== undefined && val >= ops.lt) return false;
    if (ops.lte !== undefined && val > ops.lte) return false;
    if (ops.contains !== undefined && !String(val).includes(ops.contains)) return false;
    if (ops.startsWith !== undefined && !String(val).startsWith(ops.startsWith)) return false;
    if (ops.endsWith !== undefined && !String(val).endsWith(ops.endsWith)) return false;

    return true;
  }

  private evaluateWhere(item: T, where?: ComplexWhere<T>): boolean {
    if (!where) return true;

    if (where.AND && !where.AND.every((w) => this.evaluateWhere(item, w))) return false;
    if (where.OR && !where.OR.some((w) => this.evaluateWhere(item, w))) return false;
    if (where.NOT && this.evaluateWhere(item, where.NOT)) return false;

    for (const [key, cond] of Object.entries(where)) {
      if (['AND', 'OR', 'NOT'].includes(key)) continue;
      if (!this.matchesCondition(item[key as keyof T], cond)) return false;
    }

    return true;
  }

  async findMany(args?: {
    where?: ComplexWhere<T>;
    orderBy?: OrderBy<T>;
    skip?: number;
    take?: number;
  }): Promise<T[]> {
    let list = Array.from(this.records.values()).filter((item) =>
      this.evaluateWhere(item, args?.where)
    );

    if (args?.orderBy) {
      const [field, direction] = Object.entries(args.orderBy)[0] as [keyof T, 'asc' | 'desc'];
      list.sort((a, b) => {
        if (a[field] < b[field]) return direction === 'asc' ? -1 : 1;
        if (a[field] > b[field]) return direction === 'asc' ? 1 : -1;
        return 0;
      });
    }

    if (args?.skip) list = list.slice(args.skip);
    if (args?.take !== undefined) list = list.slice(0, args.take);

    return list;
  }
}
```

### 3. Mini Object-Relational Mapper III (Field Selection & Relational Includes)

> **Problem Statement:** Implement a simplified Prisma-like in-memory object-relational mapper (ORM) with field selection and relation includes.

```typescript
export interface SchemaConfig {
  models: {
    [modelName: string]: {
      relations?: {
        [relationName: string]: {
          type: 'hasMany' | 'belongsTo';
          targetModel: string;
          foreignKey: string;
          referenceKey?: string; // default 'id'
        };
      };
    };
  };
}

export class RelationalORM {
  private tables = new Map<string, Map<string | number, any>>();
  private schema: SchemaConfig;

  constructor(schema: SchemaConfig) {
    this.schema = schema;
    for (const model of Object.keys(schema.models)) {
      this.tables.set(model, new Map());
      (this as any)[model] = this.createDelegate(model);
    }
  }

  private createDelegate(modelName: string) {
    const table = this.tables.get(modelName)!;
    const modelDef = this.schema.models[modelName];

    return {
      create: async (args: { data: any }) => {
        table.set(args.data.id, { ...args.data });
        return { ...args.data };
      },

      findUnique: async (args: {
        where: { id: string | number };
        select?: Record<string, boolean>;
        include?: Record<string, boolean>;
      }) => {
        const row = table.get(args.where.id);
        if (!row) return null;

        let result: any = { ...row };

        // 1. Process Relations (include)
        if (args.include && modelDef.relations) {
          for (const [relName, enabled] of Object.entries(args.include)) {
            if (!enabled || !modelDef.relations[relName]) continue;

            const rel = modelDef.relations[relName];
            const targetTable = this.tables.get(rel.targetModel)!;
            const targetRows = Array.from(targetTable.values());

            if (rel.type === 'belongsTo') {
              const refId = row[rel.foreignKey];
              result[relName] = targetTable.get(refId) ?? null;
            } else if (rel.type === 'hasMany') {
              const ownKey = rel.referenceKey || 'id';
              result[relName] = targetRows.filter(
                (tr) => tr[rel.foreignKey] === row[ownKey]
              );
            }
          }
        }

        // 2. Process Field Selection (select)
        if (args.select) {
          const selected: any = {};
          for (const [k, v] of Object.entries(args.select)) {
            if (v && result[k] !== undefined) {
              selected[k] = result[k];
            }
          }
          return selected;
        }

        return result;
      },
    };
  }
}
```

### 4. Rich Text to HTML

> **Problem Statement:** Implement a function that converts styled text ranges into a canonical HTML string.

```typescript
export interface StyleRange {
  start: number;
  end: number;
  style: 'b' | 'i' | 'u' | 'em' | 'strong' | 's' | string;
}

export function richTextToHTML(text: string, ranges: StyleRange[]): string {
  // Generate boundary tags for every index
  const tagOpen = new Map<number, string[]>();
  const tagClose = new Map<number, string[]>();

  for (const { start, end, style } of ranges) {
    const clampedStart = Math.max(0, Math.min(start, text.length));
    const clampedEnd = Math.max(0, Math.min(end, text.length));

    if (clampedStart >= clampedEnd) continue;

    if (!tagOpen.has(clampedStart)) tagOpen.set(clampedStart, []);
    if (!tagClose.has(clampedEnd)) tagClose.set(clampedEnd, []);

    tagOpen.get(clampedStart)!.push(style);
    tagClose.get(clampedEnd)!.push(style);
  }

  let html = '';
  const activeStack: string[] = [];

  for (let i = 0; i <= text.length; i++) {
    // 1. Close tags terminating at this position
    if (tagClose.has(i)) {
      const closing = tagClose.get(i)!;
      for (const style of closing) {
        const lastIdx = activeStack.lastIndexOf(style);
        if (lastIdx !== -1) {
          // Re-nest to maintain clean balanced HTML
          const toReopen = activeStack.slice(lastIdx + 1);
          for (let j = activeStack.length - 1; j >= lastIdx; j--) {
            html += `</${activeStack[j]}>`;
          }
          activeStack.splice(lastIdx, 1);
          for (const s of toReopen) {
            html += `<${s}>`;
          }
        }
      }
    }

    // 2. Open tags starting at this position
    if (tagOpen.has(i)) {
      const opening = tagOpen.get(i)!;
      for (const style of opening) {
        html += `<${style}>`;
        activeStack.push(style);
      }
    }

    // 3. Append current character
    if (i < text.length) {
      const char = text[i];
      if (char === '&') html += '&amp;';
      else if (char === '<') html += '&lt;';
      else if (char === '>') html += '&gt;';
      else html += char;
    }
  }

  // Close remaining unclosed tags
  while (activeStack.length) {
    html += `</${activeStack.pop()}>`;
  }

  return html;
}
```

### 2. Shortest Substring Containing Characters (Minimum Window Substring)

> **Problem Statement:** Implement a function to return the smallest substring of a string containing all characters from another string.

```typescript
function minWindow(s: string, t: string): string {
  if (s.length === 0 || t.length === 0) return '';

  const targetCounts = new Map<string, number>();
  for (const char of t) {
    targetCounts.set(char, (targetCounts.get(char) || 0) + 1);
  }

  const windowCounts = new Map<string, number>();
  let have = 0;
  const need = targetCounts.size;

  let minLen = Infinity;
  let minWindowIndices: [number, number] = [-1, -1];
  let left = 0;

  for (let right = 0; right < s.length; right++) {
    const char = s[right];
    windowCounts.set(char, (windowCounts.get(char) || 0) + 1);

    if (targetCounts.has(char) && windowCounts.get(char) === targetCounts.get(char)) {
      have++;
    }

    while (have === need) {
      if (right - left + 1 < minLen) {
        minLen = right - left + 1;
        minWindowIndices = [left, right];
      }

      const leftChar = s[left];
      windowCounts.set(leftChar, windowCounts.get(leftChar)! - 1);
      if (targetCounts.has(leftChar) && windowCounts.get(leftChar)! < targetCounts.get(leftChar)!) {
        have--;
      }
      left++;
    }
  }

  return minLen === Infinity ? '' : s.slice(minWindowIndices[0], minWindowIndices[1] + 1);
}
```

### 5. Find Word in Grid (Word Search / Backtracking)

> **Problem Statement:** Implement a function to check the existence of a word in a grid.

```typescript
function exist(board: string[][], word: string): boolean {
  const rows = board.length;
  const cols = board[0].length;

  const backtrack = (r: number, c: number, index: number): boolean => {
    if (index === word.length) return true;
    if (
      r < 0 ||
      c < 0 ||
      r >= rows ||
      c >= cols ||
      board[r][c] !== word[index]
    ) {
      return false;
    }

    const temp = board[r][c];
    board[r][c] = '#'; // Mark cell as visited

    const found =
      backtrack(r + 1, c, index + 1) ||
      backtrack(r - 1, c, index + 1) ||
      backtrack(r, c + 1, index + 1) ||
      backtrack(r, c - 1, index + 1);

    board[r][c] = temp; // Restore cell
    return found;
  };

  for (let r = 0; r < rows; r++) {
    for (let c = 0; c < cols; c++) {
      if (board[r][c] === word[0] && backtrack(r, c, 0)) {
        return true;
      }
    }
  }

  return false;
}
```
