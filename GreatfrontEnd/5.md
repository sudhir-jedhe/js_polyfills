* **Example:**

```javascript
function promisify(fn) {
  return function (...args) {
    return new Promise((resolve, reject) => {
      fn.call(this, ...args, (err, result) => {
        if (err) return reject(err);
        resolve(result);

  };
```

**2. Turtle**

* **Prompt:** Implement a `Turtle` class that simulates movement on a 2D plane with chainable methods (`forward`, `backward`, `left`, `right`, `position`).
* **Example:**

```javascript
class Turtle {
  constructor() {
    this.x = 0;
    this.y = 0;
    this.angle = 0; // facing +X (East)
  }

  forward(dist) {
    const rad = (this.angle * Math.PI) / 180;
    this.x += Math.round(dist * Math.cos(rad));
    this.y += Math.round(dist * Math.sin(rad));
    return this;
  }

  backward(dist) {
    return this.forward(-dist);
  }

  left(deg) {
    this.angle = (this.angle + deg) % 360;
    return this;
  }

  right(deg) {
    this.angle = (this.angle - deg + 360) % 360;
    return this;
  }

  position() {
    return [this.x, this.y];
  }
}

// Usage:
// const t = new Turtle();
// t.forward(10).left(90).forward(5).position(); // [10, 5]
```

```javascript
function promiseTimeout(promise, duration) {
  return new Promise((resolve, reject) => {
    const timer = setTimeout(() => {
      reject(new Error('Promise timed out'));
    }, duration);

      .then((val) => {
        clearTimeout(timer);
        resolve(val);
      })

  });
```

* **Example:**

```jsx
import React, { useState } from 'react';

const INITIAL_CARDS = ['🍎', '🍌', '🍇', '🍎', '🍌', '🍇'];

function MemoryGame() {
  const [cards] = useState(() => INITIAL_CARDS.sort(() => Math.random() - 0.5));
  const [flipped, setFlipped] = useState([]);
  const [matched, setMatched] = useState([]);

  const handleCardClick = (index) => {
    if (flipped.length === 2 || flipped.includes(index) || matched.includes(index)) return;

    const newFlipped = [...flipped, index];
    setFlipped(newFlipped);

    if (newFlipped.length === 2) {
      const [first, second] = newFlipped;
      if (cards[first] === cards[second]) {
        setMatched((prev) => [...prev, first, second]);
      }
      setTimeout(() => setFlipped([]), 800);
    }
  };

  return (
    <div style={{ display: 'grid', gridTemplateColumns: 'repeat(3, 60px)', gap: 8 }}>
      {cards.map((card, idx) => {
        const isRevealed = flipped.includes(idx) || matched.includes(idx);
        return (
          <button
            key={idx}
            onClick={() => handleCardClick(idx)}
            style={{ width: 60, height: 60, fontSize: 20 }}
          >
            {isRevealed ? card : '❓'}
          </button>
        );
      })}
    </div>
  );
}
```

```javascript
async function promiseMerge(p1, p2) {
  const [res1, res2] = await Promise.all([p1, p2]);

  if (typeof res1 === 'number' && typeof res2 === 'number') return res1 + res2;
  if (typeof res1 === 'string' && typeof res2 === 'string') return res1 + res2;
```

```jsx
function MortgageCalculator() {

  const [rate, setRate] = useState(3.5);
  const [years, setYears] = useState(30);

    const monthlyRate = rate / 100 / 12;
    const totalPayments = years * 12;
```

* **Prompt:** Build a numerical counter that records history steps and supports undo and redo operations.
* **Example:**

```jsx
import React, { useState } from 'react';

function UndoableCounter() {
  const [history, setHistory] = useState([0]);
  const [index, setIndex] = useState(0);

  const update = (delta) => {
    const nextVal = history[index] + delta;
    const nextHistory = [...history.slice(0, index + 1), nextVal];
    setHistory(nextHistory);
    setIndex(nextHistory.length - 1);
  };

  return (
    <div>
      <h2>Count: {history[index]}</h2>
      <button onClick={() => update(1)}>+1</button>
      <button onClick={() => update(-1)}>-1</button>
      <button onClick={() => setIndex((i) => Math.max(i - 1, 0))} disabled={index === 0}>Undo</button>
      <button onClick={() => setIndex((i) => Math.min(i + 1, history.length - 1))} disabled={index === history.length - 1}>Redo</button>
    </div>
  );
}
```

**33. Breadth-First Search (BFS)**

* **Prompt:** Implement a BFS algorithm to traverse a directed graph from a start node and return visited nodes in traversal order.
* **Example:**

```javascript
function bfs(graph, start) {
  const visited = new Set([start]);
  const queue = [start];
  const order = [];

  while (queue.length > 0) {
    const node = queue.shift();
    order.push(node);

    for (const neighbor of (graph[node] || [])) {
      if (!visited.has(neighbor)) {
        visited.add(neighbor);
        queue.push(neighbor);
      }
    }
  }

  return order;
}

// Usage:
// bfs({ A: ['B', 'C'], B: ['D'], C: ['E'], D: [], E: [] }, 'A'); // ['A', 'B', 'C', 'D', 'E']

```

**35. Heap Sort**

* **Prompt:** Implement heap sort by constructing a max-heap and iteratively extracting the maximum element.
* **Example:**

```javascript
function heapSort(arr) {
  const n = arr.length;

  function heapify(size, rootIdx) {
    let largest = rootIdx;
    const left = 2 * rootIdx + 1;
    const right = 2 * rootIdx + 2;

    if (left < size && arr[left] > arr[largest]) largest = left;
    if (right < size && arr[right] > arr[largest]) largest = right;

    if (largest !== rootIdx) {
      [arr[rootIdx], arr[largest]] = [arr[largest], arr[rootIdx]];
      heapify(size, largest);
    }
  }

  for (let i = Math.floor(n / 2) - 1; i >= 0; i--) heapify(n, i);
  for (let i = n - 1; i > 0; i--) {
    [arr[0], arr[i]] = [arr[i], arr[0]];
    heapify(i, 0);
  }

  return arr;
}

// Usage:
// heapSort([12, 11, 13, 5, 6, 7]); // [5, 6, 7, 11, 12, 13]

```

**36. Topological Sort**

* **Prompt:** Implement topological sort on a Directed Acyclic Graph (DAG) using Kahn's algorithm or DFS.
* **Example:**

```javascript
function topologicalSort(numNodes, edges) {
  const inDegree = Array(numNodes).fill(0);
  const adj = Array.from({ length: numNodes }, () => []);

  for (const [src, dst] of edges) {
    adj[src].push(dst);
    inDegree[dst]++;
  }

  const queue = [];
  for (let i = 0; i < numNodes; i++) {
    if (inDegree[i] === 0) queue.push(i);
  }

  const order = [];
  while (queue.length > 0) {
    const node = queue.shift();
    order.push(node);

    for (const neighbor of adj[node]) {
      inDegree[neighbor]--;
      if (inDegree[neighbor] === 0) queue.push(neighbor);
    }
  }

  return order.length === numNodes ? order : []; // Empty if cycle exists
}

```

**37. Binary Search**

* **Prompt:** Implement binary search on a sorted array returning the target index or `-1` if not found.
* **Example:**

```javascript
function binarySearch(arr, target) {
  let left = 0;
  let right = arr.length - 1;

  while (left <= right) {
    const mid = Math.floor((left + right) / 2);
    if (arr[mid] === target) return mid;
    if (arr[mid] < target) left = mid + 1;
    else right = mid - 1;
  }

  return -1;
}

// Usage:
// binarySearch([1, 3, 5, 7, 9], 5); // 2
```

**38. Depth-First Search (DFS)**

* **Prompt:** Implement a DFS graph traversal algorithm visiting all connected nodes.
* **Example:**

```javascript
function dfs(graph, start, visited = new Set(), order = []) {
  visited.add(start);
  order.push(start);

  for (const neighbor of (graph[start] || [])) {
    if (!visited.has(neighbor)) {
      dfs(graph, neighbor, visited, order);
    }
  }

  return order;
}
```

**39. Insertion Sort**

* **Prompt:** Implement in-place insertion sort by placing each element into its proper position in the sorted prefix.
* **Example:**

```javascript
function insertionSort(arr) {
  for (let i = 1; i < arr.length; i++) {
    const current = arr[i];
    let j = i - 1;
    while (j >= 0 && arr[j] > current) {
      arr[j + 1] = arr[j];
      j--;
    }
    arr[j + 1] = current;
  }
  return arr;
}

// Usage:
// insertionSort([12, 11, 13, 5, 6]); // [5, 6, 11, 12, 13]
```

**40. Merge Sort**

* **Prompt:** Implement a recursive divide-and-conquer merge sort.
* **Example:**

```javascript
function mergeSort(arr) {
  if (arr.length <= 1) return arr;

  const mid = Math.floor(arr.length / 2);
  const left = mergeSort(arr.slice(0, mid));
  const right = mergeSort(arr.slice(mid));

  return merge(left, right);
}

function merge(left, right) {
  const result = [];
  let l = 0;
  let r = 0;

  while (l < left.length && r < right.length) {
    if (left[l] < right[r]) result.push(left[l++]);
    else result.push(right[r++]);
  }

  return result.concat(left.slice(l)).concat(right.slice(r));
}

// Usage:
// mergeSort([38, 27, 43, 3, 9, 82, 10]); // [3, 9, 10, 27, 38, 43, 82]
```
