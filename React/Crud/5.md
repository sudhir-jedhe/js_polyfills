import React, { useReducer, useState, useEffect, useMemo } from 'react';

// --- MOCK DATABASE & ASYNC API SERVICE ---
let mockDatabase = [
  { id: '1', title: 'Implement OAuth Flow', priority: 'High', status: 'Pending' },
  { id: '2', title: 'Optimize SQL Queries', priority: 'Medium', status: 'Completed' },
  { id: '3', title: 'Draft API Documentation', priority: 'Low', status: 'Pending' }
];

const api = {
  fetchTasks: async () => {
    await new Promise((r) => setTimeout(r, 600));
    return [...mockDatabase];
  },
  createTask: async (task) => {
    await new Promise((r) => setTimeout(r, 400));
    const created = { ...task, id: Math.random().toString(36).substring(2, 9) };
    mockDatabase.push(created);
    return created;
  },
  updateTask: async (id, updates) => {
    await new Promise((r) => setTimeout(r, 400));
    mockDatabase = mockDatabase.map((t) => (t.id === id ? { ...t, ...updates } : t));
    return mockDatabase.find((t) => t.id === id);
  },
  deleteTask: async (id) => {
    await new Promise((r) => setTimeout(r, 400));
    mockDatabase = mockDatabase.filter((t) => t.id !== id);
    return id;
  }
};

// --- REDUCER & ACTIONS ---
const initialState = {
  tasks: [],
  loading: true,
  error: null,
  notification: null
};

function taskReducer(state, action) {
  switch (action.type) {
    case 'FETCH_SUCCESS':
      return { ...state, tasks: action.payload, loading: false };
    case 'SET_LOADING':
      return { ...state, loading: action.payload };
    case 'SET_ERROR':
      return { ...state, error: action.payload, loading: false };
    case 'CREATE_TASK':
      return { ...state, tasks: [action.payload, ...state.tasks] };
    case 'UPDATE_TASK':
      return {
        ...state,
        tasks: state.tasks.map((t) => (t.id === action.payload.id ? action.payload : t))
      };
    case 'DELETE_TASK':
      return {
        ...state,
        tasks: state.tasks.filter((t) => t.id !== action.payload)
      };
    case 'SET_NOTIFICATION':
      return { ...state, notification: action.payload };
    default:
      return state;
  }
}

// --- MAIN COMPONENT ---
export default function AdvancedTaskCrud() {
  const [state, dispatch] = useReducer(taskReducer, initialState);
  const { tasks, loading, error, notification } = state;

  // Search, Filter, and Pagination/Modal state
  const [searchQuery, setSearchQuery] = useState('');
  const [filterPriority, setFilterPriority] = useState('All');
  const [editingTask, setEditingTask] = useState(null);
  const [formData, setFormData] = useState({ title: '', priority: 'Medium', status: 'Pending' });

  // Load initial tasks (READ)
  useEffect(() => {
    let isMounted = true;
    dispatch({ type: 'SET_LOADING', payload: true });
    api
      .fetchTasks()
      .then((data) => {
        if (isMounted) dispatch({ type: 'FETCH_SUCCESS', payload: data });
      })
      .catch((err) => {
        if (isMounted) dispatch({ type: 'SET_ERROR', payload: 'Failed to fetch tasks.' });
      });
    return () => {
      isMounted = false;
    };
  }, []);

  // Show temporary toast message
  const triggerNotification = (msg) => {
    dispatch({ type: 'SET_NOTIFICATION', payload: msg });
    setTimeout(() => dispatch({ type: 'SET_NOTIFICATION', payload: null }), 3000);
  };

  // Form submission (CREATE / UPDATE)
  const handleSubmit = async (e) => {
    e.preventDefault();
    if (!formData.title.trim()) return;

    try {
      if (editingTask) {
        const updated = await api.updateTask(editingTask.id, formData);
        dispatch({ type: 'UPDATE_TASK', payload: updated });
        triggerNotification('Task updated successfully');
        setEditingTask(null);
      } else {
        const created = await api.createTask(formData);
        dispatch({ type: 'CREATE_TASK', payload: created });
        triggerNotification('Task created successfully');
      }
      setFormData({ title: '', priority: 'Medium', status: 'Pending' });
    } catch {
      triggerNotification('An error occurred. Please try again.');
    }
  };

  // Fast status toggle (Inline UPDATE)
  const handleToggleStatus = async (task) => {
    const nextStatus = task.status === 'Completed' ? 'Pending' : 'Completed';
    try {
      const updated = await api.updateTask(task.id, { status: nextStatus });
      dispatch({ type: 'UPDATE_TASK', payload: updated });
    } catch {
      triggerNotification('Could not update status');
    }
  };

  // DELETE operation
  const handleDelete = async (id) => {
    try {
      await api.deleteTask(id);
      dispatch({ type: 'DELETE_TASK', payload: id });
      triggerNotification('Task deleted');
      if (editingTask?.id === id) {
        setEditingTask(null);
        setFormData({ title: '', priority: 'Medium', status: 'Pending' });
      }
    } catch {
      triggerNotification('Failed to delete task');
    }
  };

  const handleEditInit = (task) => {
    setEditingTask(task);
    setFormData({ title: task.title, priority: task.priority, status: task.status });
  };

  // Memoized client-side filtering and search
  const filteredTasks = useMemo(() => {
    return tasks.filter((t) => {
      const matchesSearch = t.title.toLowerCase().includes(searchQuery.toLowerCase());
      const matchesPriority = filterPriority === 'All' || t.priority === filterPriority;
      return matchesSearch && matchesPriority;
    });
  }, [tasks, searchQuery, filterPriority]);

  return (
    <div style={{ maxWidth: '720px', margin: '2rem auto', fontFamily: 'system-ui, sans-serif' }}>
      <h2>Project Task Manager</h2>

      {notification && (
        <div style={{ padding: '8px 12px', background: '#e0f2fe', color: '#0369a1', borderRadius: '4px', marginBottom: '1rem' }}>
          {notification}
        </div>
      )}

      {/* FORM: CREATE & UPDATE */}
      <form onSubmit={handleSubmit} style={{ display: 'grid', gridTemplateColumns: '2fr 1fr 1fr auto', gap: '8px', marginBottom: '1.5rem' }}>
        <input
          type="text"
          placeholder="Task title..."
          value={formData.title}
          onChange={(e) => setFormData({ ...formData, title: e.target.value })}
          style={{ padding: '8px' }}
        />
        <select
          value={formData.priority}
          onChange={(e) => setFormData({ ...formData, priority: e.target.value })}
          style={{ padding: '8px' }}
        >
          <option value="Low">Low</option>
          <option value="Medium">Medium</option>
          <option value="High">High</option>
        </select>
        <select
          value={formData.status}
          onChange={(e) => setFormData({ ...formData, status: e.target.value })}
          style={{ padding: '8px' }}
        >
          <option value="Pending">Pending</option>
          <option value="Completed">Completed</option>
        </select>
        <button type="submit" style={{ padding: '8px 14px' }}>
          {editingTask ? 'Save' : 'Create'}
        </button>
      </form>

      {/* QUERY CONTROLS: FILTER & SEARCH */}
      <div style={{ display: 'flex', gap: '8px', marginBottom: '1rem' }}>
        <input
          type="text"
          placeholder="Filter by title..."
          value={searchQuery}
          onChange={(e) => setSearchQuery(e.target.value)}
          style={{ flex: 1, padding: '6px' }}
        />
        <select
          value={filterPriority}
          onChange={(e) => setFilterPriority(e.target.value)}
          style={{ padding: '6px' }}
        >
          <option value="All">All Priorities</option>
          <option value="High">High</option>
          <option value="Medium">Medium</option>
          <option value="Low">Low</option>
        </select>
      </div>

      {/* LIST DATA */}
      {loading ? (
        <p>Loading tasks from server...</p>
      ) : error ? (
        <p style={{ color: 'red' }}>{error}</p>
      ) : filteredTasks.length === 0 ? (
        <p style={{ color: '#666' }}>No matching tasks found.</p>
      ) : (
        <table style={{ width: '100%', borderCollapse: 'collapse', textAlign: 'left' }}>
          <thead>
            <tr style={{ borderBottom: '2px solid #ccc' }}>
              <th style={{ padding: '8px' }}>Status</th>
              <th style={{ padding: '8px' }}>Title</th>
              <th style={{ padding: '8px' }}>Priority</th>
              <th style={{ padding: '8px', textAlign: 'right' }}>Actions</th>
            </tr>
          </thead>
          <tbody>
            {filteredTasks.map((task) => (
              <tr key={task.id} style={{ borderBottom: '1px solid #eee' }}>
                <td style={{ padding: '8px' }}>
                  <button
                    onClick={() => handleToggleStatus(task)}
                    style={{
                      cursor: 'pointer',
                      border: 'none',
                      background: task.status === 'Completed' ? '#dcfce7' : '#fef9c3',
                      color: task.status === 'Completed' ? '#15803d' : '#854d0e',
                      padding: '4px 8px',
                      borderRadius: '4px'
                    }}
                  >
                    {task.status}
                  </button>
                </td>
                <td style={{ padding: '8px', textDecoration: task.status === 'Completed' ? 'line-through' : 'none' }}>
                  {task.title}
                </td>
                <td style={{ padding: '8px' }}>{task.priority}</td>
                <td style={{ padding: '8px', textAlign: 'right' }}>
                  <button onClick={() => handleEditInit(task)} style={{ marginRight: '6px' }}>
                    Edit
                  </button>
                  <button onClick={() => handleDelete(task.id)} style={{ color: 'red' }}>
                    Delete
                  </button>
                </td>
              </tr>
            ))}
          </tbody>
        </table>
      )}
    </div>
  );
}
