Here is a complete, lightweight CRUD (Create, Read, Update, Delete) implementation in React using modern hooks (`useState`) to manage local state.

```jsx
import React, { useState } from 'react';

export default function CrudApp() {
  const [items, setItems] = useState([
    { id: 1, text: 'Learn React State' },
    { id: 2, text: 'Build a CRUD app' }
  ]);
  const [input, setInput] = useState('');
  const [editingId, setEditingId] = useState(null);
  const [editText, setEditText] = useState('');

  // CREATE
  const handleCreate = (e) => {
    e.preventDefault();
    if (!input.trim()) return;
    const newItem = { id: Date.now(), text: input.trim() };
    setItems([...items, newItem]);
    setInput('');
  };

  // UPDATE (Save)
  const handleUpdate = (id) => {
    if (!editText.trim()) return;
    setItems(items.map((item) => (item.id === id ? { ...item, text: editText.trim() } : item)));
    setEditingId(null);
    setEditText('');
  };

  // DELETE
  const handleDelete = (id) => {
    setItems(items.filter((item) => item.id !== id));
  };

  // Set row into edit mode
  const startEditing = (item) => {
    setEditingId(item.id);
    setEditText(item.text);
  };

  return (
    <div style={{ maxWidth: '480px', margin: '2rem auto', fontFamily: 'sans-serif' }}>
      <h2>CRUD App</h2>

      {/* CREATE FORM */}
      <form onSubmit={handleCreate} style={{ display: 'flex', gap: '8px', marginBottom: '1.5rem' }}>
        <input
          type="text"
          placeholder="Add a new item..."
          value={input}
          onChange={(e) => setInput(e.target.value)}
          style={{ flex: 1, padding: '8px' }}
        />
        <button type="submit" style={{ padding: '8px 16px' }}>Add</button>
      </form>

      {/* READ, UPDATE, DELETE LIST */}
      <ul style={{ listStyle: 'none', padding: 0 }}>
        {items.map((item) => (
          <li
            key={item.id}
            style={{
              display: 'flex',
              alignItems: 'center',
              justifyContent: 'space-between',
              padding: '8px',
              borderBottom: '1px solid #ccc'
            }}
          >
            {editingId === item.id ? (
              <>
                <input
                  type="text"
                  value={editText}
                  onChange={(e) => setEditText(e.target.value)}
                  style={{ flex: 1, marginRight: '8px' }}
                />
                <button onClick={() => handleUpdate(item.id)}>Save</button>
                <button onClick={() => setEditingId(null)} style={{ marginLeft: '4px' }}>Cancel</button>
              </>
            ) : (
              <>
                <span>{item.text}</span>
                <div>
                  <button onClick={() => startEditing(item)} style={{ marginRight: '6px' }}>Edit</button>
                  <button onClick={() => handleDelete(item.id)} style={{ color: 'red' }}>Delete</button>
                </div>
              </>
            )}
          </li>
        ))}
      </ul>
    </div>
  );
}

```

**Core CRUD Operations Used**

* **Create:** `setItems([...items, newItem])` appends a new item using object spread syntax.
* **Read:** `items.map(...)` renders the collection into list elements.
* **Update:** `items.map((item) => item.id === id ? { ...item, text: editText } : item)` replaces the matching entry while leaving other entries untouched.
* **Delete:** `items.filter((item) => item.id !== id)` removes the matching item by excluding its `id`.
