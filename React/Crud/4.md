Here is an example of a **User Management CRUD app** that handles multi-field objects (`name` and `role`) and uses `localStorage` so data persists across page refreshes.

```jsx
import React, { useState, useEffect } from 'react';

const INITIAL_USERS = [
  { id: 1, name: 'Alice Smith', role: 'Admin' },
  { id: 2, name: 'Bob Jones', role: 'Developer' }
];

export default function UserCrudApp() {
  // READ (Load initial state from localStorage or fallback)
  const [users, setUsers] = useState(() => {
    const saved = localStorage.getItem('crud_users');
    return saved ? JSON.parse(saved) : INITIAL_USERS;
  });

  const [formData, setFormData] = useState({ name: '', role: 'User' });
  const [editId, setEditId] = useState(null);

  // Sync state changes with localStorage
  useEffect(() => {
    localStorage.setItem('crud_users', JSON.stringify(users));
  }, [users]);

  // Handle generic input change
  const handleChange = (e) => {
    const { name, value } = e.target;
    setFormData((prev) => ({ ...prev, [name]: value }));
  };

  // CREATE or UPDATE submission handler
  const handleSubmit = (e) => {
    e.preventDefault();
    if (!formData.name.trim()) return;

    if (editId !== null) {
      // UPDATE: Replace existing record
      setUsers(users.map((u) => (u.id === editId ? { ...u, ...formData } : u)));
      setEditId(null);
    } else {
      // CREATE: Append new user
      const newUser = { id: Date.now(), ...formData };
      setUsers([...users, newUser]);
    }

    setFormData({ name: '', role: 'User' });
  };

  // Prep form for editing
  const startEdit = (user) => {
    setEditId(user.id);
    setFormData({ name: user.name, role: user.role });
  };

  const cancelEdit = () => {
    setEditId(null);
    setFormData({ name: '', role: 'User' });
  };

  // DELETE
  const handleDelete = (id) => {
    setUsers(users.filter((user) => user.id !== id));
    if (editId === id) cancelEdit();
  };

  return (
    <div style={{ maxWidth: '600px', margin: '2rem auto', fontFamily: 'sans-serif' }}>
      <h2>User Directory</h2>

      {/* CREATE / EDIT FORM */}
      <form onSubmit={handleSubmit} style={{ display: 'flex', gap: '8px', marginBottom: '1.5rem' }}>
        <input
          name="name"
          placeholder="Full Name"
          value={formData.name}
          onChange={handleChange}
          style={{ flex: 2, padding: '8px' }}
        />
        <select
          name="role"
          value={formData.role}
          onChange={handleChange}
          style={{ flex: 1, padding: '8px' }}
        >
          <option value="User">User</option>
          <option value="Developer">Developer</option>
          <option value="Admin">Admin</option>
        </select>
        <button type="submit" style={{ padding: '8px 16px' }}>
          {editId !== null ? 'Save' : 'Add'}
        </button>
        {editId !== null && (
          <button type="button" onClick={cancelEdit} style={{ padding: '8px' }}>
            Cancel
          </button>
        )}
      </form>

      {/* READ / DELETE TABLE */}
      <table style={{ width: '100%', borderCollapse: 'collapse', textAlign: 'left' }}>
        <thead>
          <tr style={{ borderBottom: '2px solid #ccc' }}>
            <th style={{ padding: '8px' }}>Name</th>
            <th style={{ padding: '8px' }}>Role</th>
            <th style={{ padding: '8px', textAlign: 'right' }}>Actions</th>
          </tr>
        </thead>
        <tbody>
          {users.length === 0 ? (
            <tr>
              <td colSpan="3" style={{ textAlign: 'center', padding: '16px', color: '#666' }}>
                No users found.
              </td>
            </tr>
          ) : (
            users.map((user) => (
              <tr key={user.id} style={{ borderBottom: '1px solid #eee' }}>
                <td style={{ padding: '8px' }}>{user.name}</td>
                <td style={{ padding: '8px' }}>{user.role}</td>
                <td style={{ padding: '8px', textAlign: 'right' }}>
                  <button onClick={() => startEdit(user)} style={{ marginRight: '6px' }}>
                    Edit
                  </button>
                  <button onClick={() => handleDelete(user.id)} style={{ color: 'red' }}>
                    Delete
                  </button>
                </td>
              </tr>
            ))
          )}
        </tbody>
      </table>
    </div>
  );
}

```

**Key Patterns Added**

* **Unified Form Handler:** A single `formData` object and dynamic key access `[e.target.name]: value` handles multiple fields cleanly without duplicating state variables.
* **Shared Form for Add & Edit:** The form switches between **Create** and **Update** modes based on whether `editId` is populated.
* **Persistent Storage:** `useEffect` and `localStorage` sync data so entries remain available across browser refreshes.
