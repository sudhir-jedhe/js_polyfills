### 1. Custom Hooks

#### `hooks/useLocalStorage.js`

A reusable hook synchronizing React state directly with browser storage.

```javascript
import { useState, useEffect } from 'react';

export function useLocalStorage(key, initialValue) {
  const [value, setValue] = useState(() => {
    try {
      const stored = localStorage.getItem(key);
      return stored !== null ? JSON.parse(stored) : initialValue;
    } catch {
      return initialValue;
    }
  });

  useEffect(() => {
    try {
      localStorage.setItem(key, JSON.stringify(value));
    } catch (err) {
      console.error(`Error saving ${key} to localStorage:`, err);
    }
  }, [key, value]);

  return [value, setValue];
}

```

#### `hooks/useDebounce.js`

Uses `useEffect` and `useRef` to throttle rapid user input (e.g., search queries).

```javascript
import { useState, useEffect } from 'react';

export function useDebounce(value, delay = 300) {
  const [debouncedValue, setDebouncedValue] = useState(value);

  useEffect(() => {
    const handler = setTimeout(() => setDebouncedValue(value), delay);
    return () => clearTimeout(handler);
  }, [value, delay]);

  return debouncedValue;
}

```

#### `hooks/useAuth.js`

Encapsulates authentication state, role verification, and dispatch methods.

```javascript
import { useSelector, useDispatch } from 'react-redux';
import { login, logout } from '../store/authSlice';
import { useCallback } from 'react';

export function useAuth() {
  const dispatch = useDispatch();
  const { user, token, isAuthenticated } = useSelector((state) => state.auth);

  const handleLogin = useCallback(
    (credentials) => dispatch(login(credentials)),
    [dispatch]
  );

  const handleLogout = useCallback(
    () => dispatch(logout()),
    [dispatch]
  );

  // Authorization helper check
  const can = useCallback(
    (requiredRole) => {
      if (!isAuthenticated || !user) return false;
      if (user.role === 'admin') return true; // Superuser bypass
      return user.role === requiredRole;
    },
    [isAuthenticated, user]
  );

  return { user, token, isAuthenticated, handleLogin, handleLogout, can };
}

```

#### `hooks/useRecordForm.js`

Manages multi-field CRUD form inputs, validation, and submission lifecycles.

```javascript
import { useState, useCallback } from 'react';

export function useRecordForm(initialValues, onSubmit) {
  const [values, setValues] = useState(initialValues);
  const [errors, setErrors] = useState({});

  const handleChange = useCallback((e) => {
    const { name, value } = e.target;
    setValues((prev) => ({ ...prev, [name]: value }));
  }, []);

  const resetForm = useCallback(() => {
    setValues(initialValues);
    setErrors({});
  }, [initialValues]);

  const handleSubmit = useCallback(
    (e) => {
      e.preventDefault();
      const currentErrors = {};
      if (!values.title?.trim()) currentErrors.title = 'Title is required';
      if (!values.category?.trim()) currentErrors.category = 'Category is required';

      if (Object.keys(currentErrors).length > 0) {
        setErrors(currentErrors);
        return;
      }
      onSubmit(values);
      resetForm();
    },
    [values, onSubmit, resetForm]
  );

  return { values, setValues, errors, handleChange, handleSubmit, resetForm };
}

```

### 2. Redux Toolkit Store

#### `store/authSlice.js`

Handles user login credentials, session persistence, and RBAC token states.

```javascript
import { createSlice } from '@reduxjs/toolkit';

const initialAuth = JSON.parse(localStorage.getItem('auth_session')) || {
  user: null,
  token: null,
  isAuthenticated: false
};

const authSlice = createSlice({
  name: 'auth',
  initialState: initialAuth,
  reducers: {
    login: (state, action) => {
      const { username, role } = action.payload;
      state.user = { username, role };
      state.token = `jwt_mock_${Date.now()}`;
      state.isAuthenticated = true;
      localStorage.setItem('auth_session', JSON.stringify(state));
    },
    logout: (state) => {
      state.user = null;
      state.token = null;
      state.isAuthenticated = false;
      localStorage.removeItem('auth_session');
    }
  }
});

export const { login, logout } = authSlice.actions;
export default authSlice.reducer;

```

#### `store/recordsSlice.js`

Handles CRUD actions for records and keeps data synced to `localStorage`.

```javascript
import { createSlice } from '@reduxjs/toolkit';

const loadSavedRecords = () => {
  try {
    const data = localStorage.getItem('app_records');
    return data ? JSON.parse(data) : [
      { id: '1', title: 'Server Migration Plan', category: 'DevOps', author: 'admin' },
      { id: '2', title: 'Q3 Brand Assets', category: 'Design', author: 'john_editor' }
    ];
  } catch {
    return [];
  }
};

const recordsSlice = createSlice({
  name: 'records',
  initialState: { items: loadSavedRecords() },
  reducers: {
    addRecord: (state, action) => {
      state.items.unshift(action.payload);
      localStorage.setItem('app_records', JSON.stringify(state.items));
    },
    updateRecord: (state, action) => {
      const index = state.items.findIndex((item) => item.id === action.payload.id);
      if (index !== -1) {
        state.items[index] = action.payload;
        localStorage.setItem('app_records', JSON.stringify(state.items));
      }
    },
    deleteRecord: (state, action) => {
      state.items = state.items.filter((item) => item.id !== action.payload);
      localStorage.setItem('app_records', JSON.stringify(state.items));
    }
  }
});

export const { addRecord, updateRecord, deleteRecord } = recordsSlice.actions;
export default recordsSlice.reducer;

```

### 3. Application Component (`App.jsx`)

Demonstrating the CRUD UI, search filtering, authentication screen, and role authorization.

```jsx
import React, { useState, useMemo, useRef, useEffect, useId } from 'react';
import { useSelector, useDispatch } from 'react-redux';
import { addRecord, updateRecord, deleteRecord } from './store/recordsSlice';
import { useAuth } from './hooks/useAuth';
import { useRecordForm } from './hooks/useRecordForm';
import { useDebounce } from './hooks/useDebounce';

export default function App() {
  const dispatch = useDispatch();
  const items = useSelector((state) => state.records.items);
  const { user, isAuthenticated, handleLogin, handleLogout, can } = useAuth();

  // Component state hooks
  const [editingItem, setEditingItem] = useState(null);
  const [search, setSearch] = useState('');
  const debouncedSearch = useDebounce(search, 250);

  // Accessible ID generation
  const titleInputId = useId();
  const searchInputId = useId();

  // DOM node reference
  const titleFocusRef = useRef(null);

  // Authentication Mock Form State
  const [loginUsername, setLoginUsername] = useState('Sarah');
  const [loginRole, setLoginRole] = useState('editor');

  // Multi-field CRUD form with custom hook
  const { values, setValues, errors, handleChange, handleSubmit, resetForm } =
    useRecordForm(
      { title: '', category: 'Development' },
      (formData) => {
        if (editingItem) {
          dispatch(
            updateRecord({
              ...editingItem,
              ...formData
            })
          );
          setEditingItem(null);
        } else {
          dispatch(
            addRecord({
              id: Date.now().toString(),
              ...formData,
              author: user.username
            })
          );
        }
      }
    );

  // Sync editing item with form values
  useEffect(() => {
    if (editingItem) {
      setValues({ title: editingItem.title, category: editingItem.category });
      titleFocusRef.current?.focus();
    }
  }, [editingItem, setValues]);

  // Authorization: Roles allowed to perform mutations
  const canCreate = can('admin') || can('editor');
  const canDelete = can('admin'); // Only admins can delete

  // Derived filtered items using useMemo
  const filteredRecords = useMemo(() => {
    return items.filter((item) =>
      item.title.toLowerCase().includes(debouncedSearch.toLowerCase()) ||
      item.category.toLowerCase().includes(debouncedSearch.toLowerCase())
    );
  }, [items, debouncedSearch]);

  // Render Login state if not authenticated
  if (!isAuthenticated) {
    return (
      <main style={{ maxWidth: '420px', margin: '4rem auto', fontFamily: 'sans-serif' }}>
        <h2>Sign In</h2>
        <form
          onSubmit={(e) => {
            e.preventDefault();
            handleLogin({ username: loginUsername, role: loginRole });
          }}
          style={{ display: 'flex', flexDirection: 'column', gap: '12px' }}
        >
          <label>Username</label>
          <input
            type="text"
            value={loginUsername}
            onChange={(e) => setLoginUsername(e.target.value)}
            style={{ padding: '8px' }}
          />

          <label>Select Role (Authorization Mode)</label>
          <select
            value={loginRole}
            onChange={(e) => setLoginRole(e.target.value)}
            style={{ padding: '8px' }}
          >
            <option value="viewer">Viewer (Read Only)</option>
            <option value="editor">Editor (Create, Read, Edit)</option>
            <option value="admin">Admin (Full CRUD)</option>
          </select>

          <button type="submit" style={{ padding: '10px', background: '#0284c7', color: '#fff', border: 'none' }}>
            Enter Workspace
          </button>
        </form>
      </main>
    );
  }

  return (
    <div style={{ maxWidth: '780px', margin: '2rem auto', fontFamily: 'sans-serif' }}>
      {/* Session header */}
      <header style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'center', borderBottom: '1px solid #ccc', paddingBottom: '1rem' }}>
        <div>
          <h2 style={{ margin: 0 }}>Resource Registry</h2>
          <small>
            Logged in as <strong>{user.username}</strong> | Role: <span style={{ textTransform: 'uppercase' }}>{user.role}</span>
          </small>
        </div>
        <button onClick={handleLogout} style={{ padding: '6px 12px' }}>
          Logout
        </button>
      </header>

      {/* CREATE & UPDATE Panel (Restricted to Editor & Admin) */}
      {canCreate ? (
        <section style={{ margin: '1.5rem 0', padding: '1rem', background: '#f8fafc', border: '1px solid #e2e8f0' }}>
          <h3>{editingItem ? 'Update Record' : 'Create Record'}</h3>
          <form onSubmit={handleSubmit} style={{ display: 'flex', gap: '8px' }}>
            <div style={{ flex: 2 }}>
              <label htmlFor={titleInputId} style={{ display: 'none' }}>Title</label>
              <input
                id={titleInputId}
                ref={titleFocusRef}
                name="title"
                placeholder="Title"
                value={values.title}
                onChange={handleChange}
                style={{ width: '100%', padding: '8px', boxSizing: 'border-box' }}
              />
              {errors.title && <small style={{ color: 'red' }}>{errors.title}</small>}
            </div>

            <div style={{ flex: 1 }}>
              <select
                name="category"
                value={values.category}
                onChange={handleChange}
                style={{ width: '100%', padding: '8px' }}
              >
                <option value="Development">Development</option>
                <option value="DevOps">DevOps</option>
                <option value="Design">Design</option>
              </select>
            </div>

            <button type="submit" style={{ padding: '8px 16px' }}>
              {editingItem ? 'Save' : 'Add'}
            </button>
            {editingItem && (
              <button
                type="button"
                onClick={() => {
                  setEditingItem(null);
                  resetForm();
                }}
              >
                Cancel
              </button>
            )}
          </form>
        </section>
      ) : (
        <p style={{ fontStyle: 'italic', color: '#64748b' }}>
          Notice: Viewers have read-only permissions. Sign in as Editor or Admin to add or modify records.
        </p>
      )}

      {/* Search Bar */}
      <div style={{ marginBottom: '1rem' }}>
        <label htmlFor={searchInputId} style={{ display: 'block', marginBottom: '4px' }}>
          Filter Database:
        </label>
        <input
          id={searchInputId}
          type="text"
          placeholder="Type to query (debounced)..."
          value={search}
          onChange={(e) => setSearch(e.target.value)}
          style={{ width: '100%', padding: '8px', boxSizing: 'border-box' }}
        />
      </div>

      {/* READ / DELETE Data View */}
      <table style={{ width: '100%', borderCollapse: 'collapse' }}>
        <thead>
          <tr style={{ textAlign: 'left', borderBottom: '2px solid #cbd5e1' }}>
            <th style={{ padding: '8px' }}>Title</th>
            <th style={{ padding: '8px' }}>Category</th>
            <th style={{ padding: '8px' }}>Author</th>
            <th style={{ padding: '8px', textAlign: 'right' }}>Actions</th>
          </tr>
        </thead>
        <tbody>
          {filteredRecords.length === 0 ? (
            <tr>
              <td colSpan="4" style={{ padding: '16px', textAlign: 'center', color: '#64748b' }}>
                No records matching your search.
              </td>
            </tr>
          ) : (
            filteredRecords.map((item) => (
              <tr key={item.id} style={{ borderBottom: '1px solid #e2e8f0' }}>
                <td style={{ padding: '8px' }}>{item.title}</td>
                <td style={{ padding: '8px' }}>{item.category}</td>
                <td style={{ padding: '8px' }}>{item.author}</td>
                <td style={{ padding: '8px', textAlign: 'right' }}>
                  {canCreate && (
                    <button
                      onClick={() => setEditingItem(item)}
                      style={{ marginRight: '6px' }}
                    >
                      Edit
                    </button>
                  )}
                  {canDelete && (
                    <button
                      onClick={() => dispatch(deleteRecord(item.id))}
                      style={{ color: '#dc2626' }}
                    >
                      Delete
                    </button>
                  )}
                </td>
              </tr>
            ))
          )}
        </tbody>
      </table>
    </div>
  );
}

```

### Key Architectural Concepts Applied

| Domain                   | Implementation Details                                                                                                            |
| ------------------------ | --------------------------------------------------------------------------------------------------------------------------------- |
| **Authentication**       | Session token stored in `authSlice` and synchronized directly with `localStorage.auth_session`.                                   |
| **Authorization (RBAC)** | `can(role)` helper provides strict access control: **Viewer** (Read), **Editor** (Create/Read/Update), and **Admin** (Full CRUD). |
| **Redux State**          | Redux Toolkit (`configureStore`, `createSlice`) manages records and auth state globally.                                          |
| **Persistence**          | Changes to records automatically commit to browser `localStorage` on each action dispatched.                                      |
| **Built-in Hooks**       | `useState`, `useEffect`, `useCallback`, `useMemo`, `useRef` (focus management), and `useId` (form accessibility).                 |
| **Custom Hooks**         | `useAuth`, `useLocalStorage`, `useDebounce`, and `useRecordForm` abstract logic away from UI rendering.                           |
