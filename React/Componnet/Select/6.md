### 1. Canvas Text-Measurement & Layout Persistence Hook

This hook provides:

1. **Accurate Off-DOM Text Measuring:** Uses an in-memory 2D Canvas context with matching font styles and cell padding to calculate exact text widths for headers and visible cells.
2. **`localStorage` Persistence:** Automatically reads and writes column widths, order, and visibility to retain user table states across reloads.

```typescript
// utils/columnPersistence.ts
import { useState, useEffect, useCallback } from 'react';

// Reusable 2D Canvas Context for high-performance off-DOM text measurement
let canvasContext: CanvasRenderingContext2D | null = null;

function getTextWidth(text: string, font = '13px Inter, system-ui, sans-serif'): number {
  if (!canvasContext) {
    const canvas = document.createElement('canvas');
    canvasContext = canvas.getContext('2d');
  }
  if (!canvasContext) return text.length * 8; // Fallback estimate
  canvasContext.font = font;
  return canvasContext.measureText(text).width;
}

export function calculateAutoFitWidth<T extends Record<string, any>>(
  columnId: string,
  headerText: string,
  data: T[],
  padding = 32, // Padding + borders + sort/kebab icon space
  minWidth = 60,
  maxWidth = 450
): number {
  let maxContentWidth = getTextWidth(headerText, '600 13px Inter, system-ui, sans-serif');

  // Sample data up to 100 rows for high-speed measurement
  const sampleSize = Math.min(data.length, 100);
  for (let i = 0; i < sampleSize; i++) {
    const rawVal = data[i]?.[columnId];
    if (rawVal !== undefined && rawVal !== null) {
      const cellText = String(rawVal);
      const textWidth = getTextWidth(cellText, '400 13px Inter, system-ui, sans-serif');
      if (textWidth > maxContentWidth) {
        maxContentWidth = textWidth;
      }
    }
  }

  const computedWidth = Math.ceil(maxContentWidth + padding);
  return Math.max(minWidth, Math.min(maxWidth, computedWidth));
}

export function useTableLayoutPersistence(
  tableKey: string,
  initialWidths: Record<string, number>
) {
  const [columnWidths, setColumnWidths] = useState<Record<string, number>>(() => {
    try {
      const saved = localStorage.getItem(`tbl_widths_${tableKey}`);
      return saved ? JSON.parse(saved) : initialWidths;
    } catch {
      return initialWidths;
    }
  });

  const updateWidth = useCallback(
    (colId: string, width: number) => {
      setColumnWidths((prev) => {
        const next = { ...prev, [colId]: width };
        try {
          localStorage.setItem(`tbl_widths_${tableKey}`, JSON.stringify(next));
        } catch (e) {
          console.error('Failed to save table layout to localStorage', e);
        }
        return next;
      });
    },
    [tableKey]
  );

  const resetLayout = useCallback(() => {
    localStorage.removeItem(`tbl_widths_${tableKey}`);
    setColumnWidths(initialWidths);
  }, [tableKey, initialWidths]);

  return { columnWidths, updateWidth, setColumnWidths, resetLayout };
}
```

### 2. Live Aggregation Footer Engine

This module computes sticky table footers that dynamically recalculate functions (`SUM`, `AVG`, `COUNT`, `MIN`, `MAX`, `MEDIAN`, `DISTINCT_COUNT`) over filtered data sets.

```typescript
// utils/aggregations.ts
export type AggregationType = 'SUM' | 'AVG' | 'COUNT' | 'MIN' | 'MAX' | 'MEDIAN' | 'DISTINCT_COUNT';

export function calculateAggregation<T extends Record<string, any>>(
  data: T[],
  accessorKey: string,
  type: AggregationType
): string | number {
  if (!data.length) return 0;

  const rawValues = data.map((row) => row[accessorKey]).filter((val) => val !== undefined && val !== null && val !== '');

  switch (type) {
    case 'COUNT':
      return rawValues.length;

    case 'DISTINCT_COUNT':
      return new Set(rawValues).size;

    case 'SUM': {
      const nums = rawValues.map(Number).filter((n) => !isNaN(n));
      const sum = nums.reduce((acc, v) => acc + v, 0);
      return Math.round(sum * 100) / 100;
    }

    case 'AVG': {
      const nums = rawValues.map(Number).filter((n) => !isNaN(n));
      if (!nums.length) return 0;
      const avg = nums.reduce((acc, v) => acc + v, 0) / nums.length;
      return Math.round(avg * 100) / 100;
    }

    case 'MIN': {
      const nums = rawValues.map(Number).filter((n) => !isNaN(n));
      return nums.length ? Math.min(...nums) : 0;
    }

    case 'MAX': {
      const nums = rawValues.map(Number).filter((n) => !isNaN(n));
      return nums.length ? Math.max(...nums) : 0;
    }

    case 'MEDIAN': {
      const nums = rawValues.map(Number).filter((n) => !isNaN(n)).sort((a, b) => a - b);
      if (!nums.length) return 0;
      const mid = Math.floor(nums.length / 2);
      const median = nums.length % 2 !== 0 ? nums[mid] : (nums[mid - 1] + nums[mid]) / 2;
      return Math.round(median * 100) / 100;
    }

    default:
      return '';
  }
}
```

### 3. WAI-ARIA 1.2 TreeView Keyboard-Accessible Select

This component implements keyboard interactions:

* `ArrowDown` / `ArrowUp`: Navigate visible tree nodes (skips children of collapsed groups).
* `ArrowRight`: Expand collapsed group node, or move to first child if already open.
* `ArrowLeft`: Collapse open group node, or navigate to parent node if already closed.
* `Space` / `Enter`: Toggle selection of current node (cascades to all descendants).
* `Home` / `End`: Jump to first / last visible item.
* `Type-ahead`: Focus moves to matching node on alphanumeric keypress.

```tsx
// components/AriaAccessibleTreeSelect.tsx
import React, { useState, useRef, useEffect, useMemo, useId } from 'react';
import { GroupedOptionNode, getNodeSelectionState, getAllLeafIds, filterGroupTree } from '../types/groupedSelect';
import { flattenTreeForVirtualization } from '../utils/virtualTreeEngine';

interface AriaAccessibleTreeSelectProps {
  data: GroupedOptionNode[];
  selectedLeafIds: string[];
  onChange: (selectedIds: string[]) => void;
  placeholder?: string;
  itemHeight?: number;
  viewportHeight?: number;
}

export const AriaAccessibleTreeSelect: React.FC<AriaAccessibleTreeSelectProps> = ({
  data,
  selectedLeafIds,
  onChange,
  placeholder = 'Select column options...',
  itemHeight = 32,
  viewportHeight = 260,
}) => {
  const [isOpen, setIsOpen] = useState(false);
  const [searchQuery, setSearchQuery] = useState('');
  const [collapsedSet, setCollapsedSet] = useState<Set<string>>(new Set());
  const [activeIndex, setActiveIndex] = useState(0);
  const [scrollTop, setScrollTop] = useState(0);

  const containerRef = useRef<HTMLDivElement>(null);
  const listboxRef = useRef<HTMLDivElement>(null);
  const treeId = useId();

  const selectedSet = useMemo(() => new Set(selectedLeafIds), [selectedLeafIds]);
  const filteredTree = useMemo(() => filterGroupTree(data, searchQuery), [data, searchQuery]);
  const flatNodes = useMemo(
    () => flattenTreeForVirtualization(filteredTree, collapsedSet),
    [filteredTree, collapsedSet]
  );

  // Synchronize Scroll on Arrow Navigation
  useEffect(() => {
    if (!isOpen || !listboxRef.current) return;
    const targetTop = activeIndex * itemHeight;
    const targetBottom = targetTop + itemHeight;

    if (targetTop < listboxRef.current.scrollTop) {
      listboxRef.current.scrollTop = targetTop;
    } else if (targetBottom > listboxRef.current.scrollTop + viewportHeight) {
      listboxRef.current.scrollTop = targetBottom - viewportHeight;
    }
  }, [activeIndex, isOpen, itemHeight, viewportHeight]);

  const toggleSelect = (node: GroupedOptionNode) => {
    const leafIds = getAllLeafIds(node);
    const allSelected = leafIds.every((id) => selectedSet.has(id));

    if (allSelected) {
      onChange(selectedLeafIds.filter((id) => !leafIds.includes(id)));
    } else {
      onChange(Array.from(new Set([...selectedLeafIds, ...leafIds])));
    }
  };

  // WAI-ARIA 1.2 Tree Keyboard Controller
  const handleKeyDown = (e: React.KeyboardEvent) => {
    if (!isOpen) {
      if (['Enter', ' ', 'ArrowDown', 'ArrowUp'].includes(e.key)) {
        e.preventDefault();
        setIsOpen(true);
      }
      return;
    }

    const current = flatNodes[activeIndex];

    switch (e.key) {
      case 'ArrowDown':
        e.preventDefault();
        setActiveIndex((prev) => (prev < flatNodes.length - 1 ? prev + 1 : prev));
        break;

      case 'ArrowUp':
        e.preventDefault();
        setActiveIndex((prev) => (prev > 0 ? prev - 1 : 0));
        break;

      case 'ArrowRight':
        e.preventDefault();
        if (current && current.hasChildren) {
          if (collapsedSet.has(current.id)) {
            // Expand group
            setCollapsedSet((prev) => {
              const next = new Set(prev);
              next.delete(current.id);
              return next;
            });
          } else {
            // Move into first child
            setActiveIndex((prev) => Math.min(flatNodes.length - 1, prev + 1));
          }
        }
        break;

      case 'ArrowLeft':
        e.preventDefault();
        if (current) {
          if (current.hasChildren && !collapsedSet.has(current.id)) {
            // Collapse open group
            setCollapsedSet((prev) => new Set(prev).add(current.id));
          } else if (current.parentId) {
            // Jump focus to parent
            const parentIdx = flatNodes.findIndex((n) => n.id === current.parentId);
            if (parentIdx !== -1) setActiveIndex(parentIdx);
          }
        }
        break;

      case 'Home':
        e.preventDefault();
        setActiveIndex(0);
        break;

      case 'End':
        e.preventDefault();
        setActiveIndex(flatNodes.length - 1);
        break;

      case ' ':
      case 'Enter':
        e.preventDefault();
        if (current && !current.node.disabled) {
          toggleSelect(current.node);
        }
        break;

      case 'Escape':
        e.preventDefault();
        setIsOpen(false);
        break;

      default:
        // Single alphanumeric type-ahead jump
        if (e.key.length === 1 && /[a-zA-Z0-9]/.test(e.key)) {
          const matchIdx = flatNodes.findIndex(
            (n, idx) =>
              idx > activeIndex &&
              n.node.label.toLowerCase().startsWith(e.key.toLowerCase())
          );
          if (matchIdx !== -1) {
            setActiveIndex(matchIdx);
          } else {
            const fallbackIdx = flatNodes.findIndex((n) =>
              n.node.label.toLowerCase().startsWith(e.key.toLowerCase())
            );
            if (fallbackIdx !== -1) setActiveIndex(fallbackIdx);
          }
        }
        break;
    }
  };

  // Virtual Window Calculation
  const totalHeight = flatNodes.length * itemHeight;
  const startIndex = Math.max(0, Math.floor(scrollTop / itemHeight) - 3);
  const endIndex = Math.min(
    flatNodes.length - 1,
    Math.floor((scrollTop + viewportHeight) / itemHeight) + 3
  );

  const visibleNodes = [];
  for (let i = startIndex; i <= endIndex; i++) {
    if (flatNodes[i]) {
      visibleNodes.push({ item: flatNodes[i], index: i, offsetTop: i * itemHeight });
    }
  }

  return (
    <div
      ref={containerRef}
      onKeyDown={handleKeyDown}
      style={{ position: 'relative', width: 320, fontFamily: 'Inter, system-ui, sans-serif' }}
    >
      {/* Combobox Anchor */}
      <div
        role="combobox"
        aria-haspopup="tree"
        aria-expanded={isOpen}
        aria-controls={treeId}
        tabIndex={0}
        onClick={() => setIsOpen((prev) => !prev)}
        style={{
          display: 'flex',
          alignItems: 'center',
          justifyContent: 'space-between',
          padding: '7px 12px',
          background: '#ffffff',
          border: `1px solid ${isOpen ? '#2563eb' : '#cbd5e1'}`,
          borderRadius: 6,
          cursor: 'pointer',
          boxShadow: isOpen ? '0 0 0 2px rgba(37, 99, 235, 0.2)' : 'none',
          outline: 'none',
          fontSize: 13,
        }}
      >
        <span style={{ color: selectedLeafIds.length ? '#0f172a' : '#94a3b8' }}>
          {selectedLeafIds.length ? `${selectedLeafIds.length} column(s) active` : placeholder}
        </span>
        <span style={{ fontSize: 10, color: '#64748b', transform: isOpen ? 'rotate(180deg)' : 'none' }}>
          ▼
        </span>
      </div>

      {/* ARIA TreeView Popup */}
      {isOpen && (
        <div
          id={treeId}
          role="tree"
          aria-label="Table Column Preferences"
          style={{
            position: 'absolute',
            top: 'calc(100% + 4px)',
            left: 0,
            right: 0,
            zIndex: 1000,
            background: '#ffffff',
            border: '1px solid #e2e8f0',
            borderRadius: 6,
            boxShadow: '0 10px 15px -3px rgba(0, 0, 0, 0.1)',
            overflow: 'hidden',
          }}
        >
          <div style={{ padding: 6, background: '#f8fafc', borderBottom: '1px solid #e2e8f0' }}>
            <input
              type="text"
              placeholder="Search or use arrow keys..."
              value={searchQuery}
              onChange={(e) => setSearchQuery(e.target.value)}
              onClick={(e) => e.stopPropagation()}
              style={{ width: '100%', padding: '5px 8px', fontSize: 12, border: '1px solid #cbd5e1', borderRadius: 4, boxSizing: 'border-box' }}
            />
          </div>

          <div
            ref={listboxRef}
            onScroll={(e) => setScrollTop(e.currentTarget.scrollTop)}
            style={{ height: viewportHeight, overflowY: 'auto', position: 'relative' }}
          >
            <div style={{ height: totalHeight, width: '100%', position: 'relative' }}>
              {visibleNodes.map(({ item, index, offsetTop }) => {
                const isSelected = selectedSet.has(item.id);
                const selectionState = getNodeSelectionState(item.node, selectedSet);
                const isFocused = activeIndex === index;

                return (
                  <div
                    key={item.id}
                    role="treeitem"
                    aria-selected={isSelected}
                    aria-expanded={item.hasChildren ? !collapsedSet.has(item.id) : undefined}
                    aria-level={item.depth + 1}
                    onClick={() => {
                      setActiveIndex(index);
                      toggleSelect(item.node);
                    }}
                    style={{
                      position: 'absolute',
                      top: 0,
                      left: 0,
                      width: '100%',
                      height: itemHeight,
                      transform: `translateY(${offsetTop}px)`,
                      display: 'flex',
                      alignItems: 'center',
                      paddingLeft: `${item.depth * 16 + 8}px`,
                      paddingRight: 8,
                      boxSizing: 'border-box',
                      fontSize: 12,
                      cursor: 'pointer',
                      backgroundColor: isFocused ? '#eff6ff' : 'transparent',
                      color: item.node.disabled ? '#94a3b8' : '#1e293b',
                    }}
                  >
                    {item.hasChildren ? (
                      <span style={{ fontSize: 9, width: 14, color: '#64748b' }}>
                        {collapsedSet.has(item.id) ? '▶' : '▼'}
                      </span>
                    ) : (
                      <span style={{ width: 14 }} />
                    )}

                    <input
                      type="checkbox"
                      checked={selectionState === 'checked'}
                      ref={(el) => {
                        if (el) el.indeterminate = selectionState === 'indeterminate';
                      }}
                      onChange={() => {}}
                      style={{ marginRight: 6, cursor: 'pointer' }}
                    />

                    <span style={{ fontWeight: item.hasChildren ? 600 : 400, flex: 1, overflow: 'hidden', textOverflow: 'ellipsis', whiteSpace: 'nowrap' }}>
                      {item.node.label}
                    </span>
                  </div>
                );
              })}
            </div>
          </div>
        </div>
      )}
    </div>
  );
};
```

```tsx
                      padding: '10px 14px',
                      fontSize: 13,
                      color: '#0f172a',
                      borderRight: '1px solid #e2e8f0',
                    }}
                  >
                    {displayVal && (
                      <div>
                        <span style={{ fontSize: 10, color: '#64748b', textTransform: 'uppercase', display: 'block' }}>
                          {config?.type}
                        </span>
                        <span>{displayVal}</span>
                      </div>
                    )}
                  </td>
                );
              })}
            </tr>
          </tfoot>
        </table>
      </div>
    </div>
  );
};
```
