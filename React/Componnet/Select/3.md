Here are the key aspects of **Multi-Select** components across design, user experience, and technical implementation.

---

### 1. Common UI Patterns

| Pattern                          | How It Works                                                                                 | Best Used For                                                                     |
| -------------------------------- | -------------------------------------------------------------------------------------------- | --------------------------------------------------------------------------------- |
| **Chips / Tags in Input**        | Selected items appear as removable pill badges inside or below the input box.                | Short lists with 1–10 selections (e.g., tags, recipients, assignees).             |
| **Checkbox Dropdown**            | Standard select menu where each option has an explicit checkbox and stays open on selection. | Long lists with searchable options, filters, and bulk tables.                     |
| **Dual Listbox (Transfer List)** | Two side-by-side boxes ("Available" and "Selected") with move buttons (`>` / `<`).           | High-stakes selections (e.g., user permissions, assigning roles, column pickers). |
| **Select All / Bulk Table Bar**  | Master checkbox in the header controlling row-level selection with a counter.                | Enterprise data tables, batch actions (archive, delete, export).                  |

---

### 2. Core UX & Accessibility (a11y) Essentials

* **Keyboard Navigation:**
* `ArrowUp` / `ArrowDown`: Move active focus between list items.
* `Space` / `Enter`: Toggle selection state for the focused item.
* `Backspace`: Remove the last selected tag when the search input is empty.
* `Escape`: Dismiss the menu immediately.

* **ARIA Attributes:**
* Container role: `role="combobox"` or `role="listbox"` with `aria-multiselectable="true"`.
* Items: `role="option"` with dynamic `aria-selected="true|false"`.

* **Overflow Handling:** When many tags are selected, collapse the excess behind a badge (e.g., `+4 more`) to avoid breaking the layout.
* **Bulk Operations:** Provide a "Select All", "Clear All", or invert selection shortcut for lists with more than 10 items.

```tsx
// components/AccessibleMultiSelect.tsx
import React, { useState, useRef, useEffect, useMemo, useId } from 'react';
import { SelectOptionItem, OptionGroup } from '../types/multiSelect';

interface AccessibleMultiSelectProps<T = string | number> {
  items: SelectOptionItem<T>[];
  value: T[];
  onChange: (value: T[]) => void;
  placeholder?: string;
  maxDisplayTags?: number;
  disabled?: boolean;
}

export function AccessibleMultiSelect<T extends string | number>({
  items,
  value,
  onChange,
  placeholder = 'Select options...',
  maxDisplayTags = 3,
  disabled = false,
}: AccessibleMultiSelectProps<T>) {
  const [isOpen, setIsOpen] = useState(false);
  const [searchQuery, setSearchQuery] = useState('');
  const [highlightedIndex, setHighlightedIndex] = useState(0);

  const containerRef = useRef<HTMLDivElement>(null);
  const inputRef = useRef<HTMLInputElement>(null);
  const listboxId = useId();

  const selectedSet = useMemo(() => new Set(value), [value]);

  // Group items by category if provided
  const groupedData = useMemo(() => {
    const query = searchQuery.toLowerCase().trim();
    const filtered = items.filter(
      (item) =>
        item.label.toLowerCase().includes(query) ||
        item.subLabel?.toLowerCase().includes(query) ||
        item.group?.toLowerCase().includes(query)
    );

    const groups: Record<string, SelectOptionItem<T>[]> = {};
    filtered.forEach((item) => {
      const g = item.group || 'General';
      if (!groups[g]) groups[g] = [];
      groups[g].push(item);
    });

    return Object.entries(groups).map(([name, groupItems]) => ({
      name,
      items: groupItems,
    }));
  }, [items, searchQuery]);

  // Flatten for keyboard navigation indices
  const flatVisibleItems = useMemo(
    () => groupedData.flatMap((g) => g.items),
    [groupedData]
  );

  // Close when clicking outside
  useEffect(() => {
    const handleOutsideClick = (e: MouseEvent) => {
      if (containerRef.current && !containerRef.current.contains(e.target as Node)) {
        setIsOpen(false);
      }
    };
    document.addEventListener('mousedown', handleOutsideClick);
    return () => document.removeEventListener('mousedown', handleOutsideClick);
  }, []);

  const toggleItem = (id: T) => {
    if (selectedSet.has(id)) {
      onChange(value.filter((v) => v !== id));
    } else {
      onChange([...value, id]);
    }
  };

  const toggleGroup = (group: OptionGroup<T>) => {
    const activeGroupIds = group.items.filter((i) => !i.disabled).map((i) => i.id);
    const allSelected = activeGroupIds.every((id) => selectedSet.has(id));

    if (allSelected) {
      onChange(value.filter((v) => !activeGroupIds.includes(v)));
    } else {
      const combined = Array.from(new Set([...value, ...activeGroupIds]));
      onChange(combined);
    }
  };

  // Keyboard navigation
  const handleKeyDown = (e: React.KeyboardEvent) => {
    if (disabled) return;

    if (!isOpen) {
      if (['ArrowDown', 'ArrowUp', 'Enter', ' '].includes(e.key)) {
        e.preventDefault();
        setIsOpen(true);
      }
      return;
    }

    switch (e.key) {
      case 'ArrowDown':
        e.preventDefault();
        setHighlightedIndex((prev) => (prev < flatVisibleItems.length - 1 ? prev + 1 : 0));
        break;
      case 'ArrowUp':
        e.preventDefault();
        setHighlightedIndex((prev) => (prev > 0 ? prev - 1 : flatVisibleItems.length - 1));
        break;
      case 'Enter':
      case ' ':
        e.preventDefault();
        if (flatVisibleItems[highlightedIndex] && !flatVisibleItems[highlightedIndex].disabled) {
          toggleItem(flatVisibleItems[highlightedIndex].id);
        }
        break;
      case 'Backspace':
        if (searchQuery === '' && value.length > 0) {
          onChange(value.slice(0, -1)); // Remove the last selected tag
        }
        break;
      case 'Escape':
        e.preventDefault();
        setIsOpen(false);
        break;
    }
  };

  const itemLookup = useMemo(() => {
    const map = new Map<T, SelectOptionItem<T>>();
    items.forEach((item) => map.set(item.id, item));
    return map;
  }, [items]);

  const visibleTags = value.slice(0, maxDisplayTags);
  const hiddenTagCount = value.length - maxDisplayTags;

  let runningIndex = -1;

  return (
    <div
      ref={containerRef}
      onKeyDown={handleKeyDown}
      style={{
        position: 'relative',
        width: '100%',
        maxWidth: 440,
        fontFamily: 'Inter, system-ui, sans-serif',
      }}
    >
      {/* Combobox Trigger Box */}
      <div
        role="combobox"
        aria-expanded={isOpen}
        aria-haspopup="listbox"
        aria-controls={listboxId}
        onClick={() => {
          if (!disabled) {
            setIsOpen(true);
            inputRef.current?.focus();
          }
        }}
        style={{
          display: 'flex',
          flexWrap: 'wrap',
          alignItems: 'center',
          gap: 6,
          minHeight: 40,
          padding: '4px 8px',
          background: disabled ? '#f8fafc' : '#ffffff',
          border: `1px solid ${isOpen ? '#2563eb' : '#cbd5e1'}`,
          borderRadius: 6,
          boxShadow: isOpen ? '0 0 0 2px rgba(37, 99, 235, 0.15)' : 'none',
          cursor: disabled ? 'not-allowed' : 'text',
        }}
      >
        {/* Render Active Tag Badges */}
        {visibleTags.map((id) => {
          const item = itemLookup.get(id);
          return (
            <span
              key={String(id)}
              style={{
                display: 'inline-flex',
                alignItems: 'center',
                gap: 4,
                backgroundColor: '#eff6ff',
                color: '#1d4ed8',
                border: '1px solid #bfdbfe',
                borderRadius: 4,
                padding: '2px 6px',
                fontSize: 12,
                fontWeight: 500,
              }}
            >
              {item ? item.label : String(id)}
              <span
                role="button"
                aria-label={`Remove ${item?.label || id}`}
                onClick={(e) => {
                  e.stopPropagation();
                  onChange(value.filter((v) => v !== id));
                }}
                style={{ cursor: 'pointer', fontSize: 12, fontWeight: 700, marginLeft: 2 }}
              >
                ✕
              </span>
            </span>
          );
        })}

        {/* Collapsible Tag Overflow */}
        {hiddenTagCount > 0 && (
          <span
            style={{
              backgroundColor: '#f1f5f9',
              color: '#475569',
              borderRadius: 4,
              padding: '2px 6px',
              fontSize: 11,
              fontWeight: 600,
            }}
          >
            +{hiddenTagCount} more
          </span>
        )}

        {/* Inline Search Input */}
        <input
          ref={inputRef}
          type="text"
          value={searchQuery}
          disabled={disabled}
          placeholder={value.length === 0 ? placeholder : ''}
          onChange={(e) => {
            setSearchQuery(e.target.value);
            if (!isOpen) setIsOpen(true);
            setHighlightedIndex(0);
          }}
          style={{
            flex: 1,
            minWidth: 80,
            border: 'none',
            outline: 'none',
            fontSize: 13,
            padding: '4px 0',
            background: 'transparent',
          }}
        />

        {/* Controls */}
        <div style={{ display: 'flex', alignItems: 'center', gap: 6, marginLeft: 'auto' }}>
          {value.length > 0 && !disabled && (
            <button
              type="button"
              aria-label="Clear all selections"
              onClick={(e) => {
                e.stopPropagation();
                onChange([]);
              }}
              style={{ background: 'none', border: 'none', cursor: 'pointer', color: '#94a3b8', fontSize: 13 }}
            >
              ✕
            </button>
          )}
          <span style={{ fontSize: 10, color: '#64748b', transform: isOpen ? 'rotate(180deg)' : 'none' }}>
            ▼
          </span>
        </div>
      </div>

      {/* Accessible Listbox Dropdown */}
      {isOpen && (
        <div
          id={listboxId}
          role="listbox"
          aria-multiselectable="true"
          style={{
            position: 'absolute',
            top: 'calc(100% + 4px)',
            left: 0,
            right: 0,
            zIndex: 1000,
            maxHeight: 280,
            overflowY: 'auto',
            background: '#ffffff',
            border: '1px solid #e2e8f0',
            borderRadius: 6,
            boxShadow: '0 10px 15px -3px rgba(0, 0, 0, 0.1)',
            padding: 4,
          }}
        >
          {flatVisibleItems.length === 0 ? (
            <div style={{ padding: '10px 12px', fontSize: 13, color: '#94a3b8', textAlign: 'center' }}>
              No matching options found
            </div>
          ) : (
            groupedData.map((group) => {
              const groupIds = group.items.filter((i) => !i.disabled).map((i) => i.id);
              const groupSelectedCount = groupIds.filter((id) => selectedSet.has(id)).length;
              const isGroupAllSelected = groupSelectedCount === groupIds.length && groupIds.length > 0;
              const isGroupIndeterminate = groupSelectedCount > 0 && !isGroupAllSelected;

              return (
                <div key={group.name} style={{ marginBottom: 6 }}>
                  {/* Group Header with Bulk Toggle */}
                  <div
                    onClick={() => toggleGroup(group)}
                    style={{
                      display: 'flex',
                      alignItems: 'center',
                      justifyContent: 'space-between',
                      padding: '6px 8px',
                      fontSize: 11,
                      fontWeight: 700,
                      textTransform: 'uppercase',
                      letterSpacing: '0.05em',
                      color: '#64748b',
                      backgroundColor: '#f8fafc',
                      borderRadius: 4,
                      cursor: 'pointer',
                    }}
                  >
                    <span>{group.name}</span>
                    <input
                      type="checkbox"
                      checked={isGroupAllSelected}
                      ref={(el) => {
                        if (el) el.indeterminate = isGroupIndeterminate;
                      }}
                      readOnly
                      style={{ cursor: 'pointer' }}
                    />
                  </div>

                  {/* Group Child Items */}
                  {group.items.map((item) => {
                    runningIndex++;
                    const itemIdx = runningIndex;
                    const isSelected = selectedSet.has(item.id);
                    const isHighlighted = highlightedIndex === itemIdx;

                    return (
                      <div
                        key={String(item.id)}
                        role="option"
                        aria-selected={isSelected}
                        aria-disabled={item.disabled}
                        onClick={() => !item.disabled && toggleItem(item.id)}
                        onMouseEnter={() => !item.disabled && setHighlightedIndex(itemIdx)}
                        style={{
                          display: 'flex',
                          alignItems: 'center',
                          justifyContent: 'space-between',
                          padding: '7px 10px',
                          fontSize: 13,
                          borderRadius: 4,
                          cursor: item.disabled ? 'not-allowed' : 'pointer',
                          color: item.disabled ? '#cbd5e1' : '#1e293b',
                          backgroundColor: isHighlighted
                            ? '#eff6ff'
                            : isSelected
                            ? '#f8fafc'
                            : 'transparent',
                        }}
                      >
                        <div style={{ display: 'flex', alignItems: 'center', gap: 8 }}>
                          <input
                            type="checkbox"
                            checked={isSelected}
                            disabled={item.disabled}
                            readOnly
                            style={{ cursor: 'pointer' }}
                          />
                          <div>
                            <span style={{ fontWeight: isSelected ? 600 : 400 }}>{item.label}</span>
                            {item.subLabel && (
                              <span style={{ fontSize: 11, color: '#94a3b8', marginLeft: 6 }}>
                                ({item.subLabel})
                              </span>
                            )}
                          </div>
                        </div>
                        {isSelected && <span style={{ color: '#2563eb', fontWeight: 700 }}>✓</span>}
                      </div>
                    );
                  })}
                </div>
              );
            })
          )}
        </div>
      )}
    </div>
  );
}
```

```tsx
// components/DualTransferList.tsx
import React, { useState, useMemo } from 'react';
import { SelectOptionItem } from '../types/multiSelect';

interface DualTransferListProps<T = string | number> {
  items: SelectOptionItem<T>[];
  value: T[];
  onChange: (value: T[]) => void;
  leftTitle?: string;
  rightTitle?: string;
}

export function DualTransferList<T extends string | number>({
  items,
  value,
  onChange,
  leftTitle = 'Available Options',
  rightTitle = 'Selected Items',
}: DualTransferListProps<T>) {
  const [leftSearch, setLeftSearch] = useState('');
  const [rightSearch, setRightSearch] = useState('');
  const [selectedLeft, setSelectedLeft] = useState<Set<T>>(new Set());
  const [selectedRight, setSelectedRight] = useState<Set<T>>(new Set());

  const selectedSet = useMemo(() => new Set(value), [value]);

  const availableItems = useMemo(
    () =>
      items
        .filter((i) => !selectedSet.has(i.id))
        .filter((i) => i.label.toLowerCase().includes(leftSearch.toLowerCase())),
    [items, selectedSet, leftSearch]
  );

  const chosenItems = useMemo(
    () =>
      items
        .filter((i) => selectedSet.has(i.id))
        .filter((i) => i.label.toLowerCase().includes(rightSearch.toLowerCase())),
    [items, selectedSet, rightSearch]
  );

  const transferToRight = () => {
    onChange([...value, ...Array.from(selectedLeft)]);
    setSelectedLeft(new Set());
  };

  const transferToLeft = () => {
    onChange(value.filter((id) => !selectedRight.has(id)));
    setSelectedRight(new Set());
  };

  const transferAllToRight = () => {
    onChange(items.filter((i) => !i.disabled).map((i) => i.id));
    setSelectedLeft(new Set());
  };

  const transferAllToLeft = () => {
    onChange([]);
    setSelectedRight(new Set());
  };

  return (
    <div
      style={{
        display: 'flex',
        alignItems: 'center',
        gap: 16,
        fontFamily: 'Inter, system-ui, sans-serif',
      }}
    >
      {/* Left Listbox Panel */}
      <div style={{ flex: 1, border: '1px solid #cbd5e1', borderRadius: 6, overflow: 'hidden' }}>
        <div style={{ padding: '8px 12px', background: '#f8fafc', borderBottom: '1px solid #e2e8f0', fontWeight: 600, fontSize: 13 }}>
          {leftTitle} ({availableItems.length})
        </div>
        <div style={{ padding: 6, borderBottom: '1px solid #e2e8f0' }}>
          <input
            placeholder="Search available..."
            value={leftSearch}
            onChange={(e) => setLeftSearch(e.target.value)}
            style={{ width: '100%', padding: '4px 8px', fontSize: 12, border: '1px solid #cbd5e1', borderRadius: 4, boxSizing: 'border-box' }}
          />
        </div>
        <div style={{ height: 200, overflowY: 'auto', padding: 4 }}>
          {availableItems.map((item) => (
            <div
              key={String(item.id)}
              onClick={() => {
                const next = new Set(selectedLeft);
                next.has(item.id) ? next.delete(item.id) : next.add(item.id);
                setSelectedLeft(next);
              }}
              style={{
                padding: '6px 8px',
                fontSize: 13,
                borderRadius: 4,
                cursor: 'pointer',
                backgroundColor: selectedLeft.has(item.id) ? '#eff6ff' : 'transparent',
              }}
            >
              <input type="checkbox" checked={selectedLeft.has(item.id)} readOnly style={{ marginRight: 8 }} />
              {item.label}
            </div>
          ))}
        </div>
      </div>

      {/* Action Buttons */}
      <div style={{ display: 'flex', flexDirection: 'column', gap: 6 }}>
        <button
          disabled={availableItems.length === 0}
          onClick={transferAllToRight}
          style={{ padding: '6px 12px', fontSize: 12, cursor: 'pointer', borderRadius: 4, border: '1px solid #cbd5e1' }}
        >
          ≫
        </button>
        <button
          disabled={selectedLeft.size === 0}
          onClick={transferToRight}
          style={{ padding: '6px 12px', fontSize: 12, cursor: 'pointer', borderRadius: 4, border: '1px solid #cbd5e1', backgroundColor: '#eff6ff', color: '#2563eb', fontWeight: 700 }}
        >
          ＞
        </button>
        <button
          disabled={selectedRight.size === 0}
          onClick={transferToLeft}
          style={{ padding: '6px 12px', fontSize: 12, cursor: 'pointer', borderRadius: 4, border: '1px solid #cbd5e1', backgroundColor: '#eff6ff', color: '#2563eb', fontWeight: 700 }}
        >
          ＜
        </button>
        <button
          disabled={value.length === 0}
          onClick={transferAllToLeft}
          style={{ padding: '6px 12px', fontSize: 12, cursor: 'pointer', borderRadius: 4, border: '1px solid #cbd5e1' }}
        >
          ≪
        </button>
      </div>

      {/* Right Listbox Panel */}
      <div style={{ flex: 1, border: '1px solid #cbd5e1', borderRadius: 6, overflow: 'hidden' }}>
        <div style={{ padding: '8px 12px', background: '#f8fafc', borderBottom: '1px solid #e2e8f0', fontWeight: 600, fontSize: 13 }}>
          {rightTitle} ({chosenItems.length})
        </div>
        <div style={{ padding: 6, borderBottom: '1px solid #e2e8f0' }}>
          <input
            placeholder="Search selected..."
            value={rightSearch}
            onChange={(e) => setRightSearch(e.target.value)}
            style={{ width: '100%', padding: '4px 8px', fontSize: 12, border: '1px solid #cbd5e1', borderRadius: 4, boxSizing: 'border-box' }}

                const next = new Set(selectedRight);
```

```tsx
import { DualTransferList } from './components/DualTransferList';
import { TableBatchSelectionBar } from './components/TableBatchSelectionBar';
import { SelectOptionItem } from './types/multiSelect';

const ROLE_PERMISSIONS: SelectOptionItem<string>[] = [
  { id: 'p_read', label: 'View Analytics', group: 'Analytics', subLabel: 'read-only' },
  { id: 'p_export', label: 'Export Reports', group: 'Analytics', subLabel: 'csv/pdf' },
  { id: 'p_user_create', label: 'Create Users', group: 'User Management' },
  { id: 'p_user_delete', label: 'Delete Users', group: 'User Management', disabled: true },
  { id: 'p_billing', label: 'Manage Billing', group: 'Finance' },
];

interface EmployeeRow {
  id: number;
  name: string;
  role: string;
  status: string;
}

const INITIAL_ROWS: EmployeeRow[] = [
  { id: 1, name: 'Alice Cooper', role: 'Developer', status: 'Active' },
  { id: 2, name: 'Bob Smith', role: 'Manager', status: 'Pending' },
  { id: 3, name: 'Charlie Kim', role: 'Security Architect', status: 'Active' },
  { id: 4, name: 'Diana Ross', role: 'Designer', status: 'Archived' },
];

export default function App() {
  const [selectedPermissions, setSelectedPermissions] = useState<string[]>(['p_read', 'p_export']);
  const [assignedRoles, setAssignedRoles] = useState<string[]>(['p_read']);
  const [tableData, setTableData] = useState<EmployeeRow[]>(INITIAL_ROWS);
  const [selectedRowIds, setSelectedRowIds] = useState<Set<string | number>>(new Set([1, 2]));

  // Batch Handlers
  const handleBatchDelete = (ids: (string | number)[]) => {
    setTableData((prev) => prev.filter((r) => !ids.includes(r.id)));
    setSelectedRowIds(new Set());
  };

  const handleBatchStatusUpdate = (ids: (string | number)[], status: string) => {
    setTableData((prev) =>
      prev.map((r) => (ids.includes(r.id) ? { ...r, status } : r))
    );
    setSelectedRowIds(new Set());
  };

  return (
    <div style={{ padding: 32, display: 'flex', flexDirection: 'column', gap: 40, maxWidth: 900 }}>
      {/* 1. Grouped Combobox */}
      <section>
        <h3 style={{ margin: '0 0 12px 0' }}>1. Grouped Accessible Multi-Select (Combobox)</h3>
        <AccessibleMultiSelect
          items={ROLE_PERMISSIONS}
          value={selectedPermissions}
          onChange={setSelectedPermissions}
          placeholder="Assign system permissions..."
        />
      </section>

      {/* 2. Transfer List */}
      <section>
        <h3 style={{ margin: '0 0 12px 0' }}>2. Dual Transfer Listbox</h3>
        <DualTransferList
          items={ROLE_PERMISSIONS}
          value={assignedRoles}
          onChange={setAssignedRoles}
          leftTitle="Available Permissions"
          rightTitle="Assigned to Role"
        />
      </section>

      {/* 3. Table Row Selection & Batch Action Bar */}
      <section>
        <h3 style={{ margin: '0 0 12px 0' }}>3. Table Multi-Selection with Batch Actions</h3>
        <TableBatchSelectionBar
          allRows={tableData}
          selectedIds={selectedRowIds}
          onClearSelection={() => setSelectedRowIds(new Set())}
          onBatchDelete={handleBatchDelete}
          onBatchStatusUpdate={handleBatchStatusUpdate}
        />

        <table style={{ width: '100%', borderCollapse: 'collapse', border: '1px solid #cbd5e1' }}>
          <thead>
            <tr style={{ background: '#f8fafc' }}>
              <th style={{ width: 40, padding: 8, border: '1px solid #e2e8f0' }}>
                <input
                  type="checkbox"
                  checked={selectedRowIds.size === tableData.length && tableData.length > 0}
                  onChange={() => {
                    setSelectedRowIds(
                      selectedRowIds.size === tableData.length
                        ? new Set()
                        : new Set(tableData.map((r) => r.id))
                    );
                  }}
                />
              </th>
              <th style={{ padding: 8, textAlign: 'left', border: '1px solid #e2e8f0' }}>Employee</th>
              <th style={{ padding: 8, textAlign: 'left', border: '1px solid #e2e8f0' }}>Role</th>
              <th style={{ padding: 8, textAlign: 'left', border: '1px solid #e2e8f0' }}>Status</th>
            </tr>
          </thead>
          <tbody>
            {tableData.map((row) => (
              <tr
                key={row.id}
                style={{
                  backgroundColor: selectedRowIds.has(row.id) ? '#eff6ff' : 'transparent',
                }}
              >
                <td style={{ textAlign: 'center', padding: 8, border: '1px solid #e2e8f0' }}>
                  <input
                    type="checkbox"
                    checked={selectedRowIds.has(row.id)}
                    onChange={() => {
                      const next = new Set(selectedRowIds);
                      next.has(row.id) ? next.delete(row.id) : next.add(row.id);
                      setSelectedRowIds(next);
                    }}
                  />
                </td>
                <td style={{ padding: 8, border: '1px solid #e2e8f0' }}>{row.name}</td>
                <td style={{ padding: 8, border: '1px solid #e2e8f0' }}>{row.role}</td>
                <td style={{ padding: 8, border: '1px solid #e2e8f0' }}>
                  <span
                    style={{
                      padding: '2px 8px',
                      borderRadius: 12,
                      fontSize: 11,
                      fontWeight: 600,
                      backgroundColor:
                        row.status === 'Active'
                          ? '#dcfce7'
                          : row.status === 'Pending'
                          ? '#fef9c3'
                          : '#f1f5f9',
                      color:
                        row.status === 'Active'
                          ? '#15803d'
                          : row.status === 'Pending'
                          ? '#854d0e'
                          : '#475569',
                    }}
                  >
                    {row.status}
                  </span>
                </td>
              </tr>
            ))}
          </tbody>
        </table>
      </section>
    </div>
  );
}
```
