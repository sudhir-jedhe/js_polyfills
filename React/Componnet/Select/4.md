Here is the complete implementation for a **Grouped Select Component with Multi-Level / Nested Hierarchy** built from scratch in React and TypeScript.

This covers:

1. **Collapsible / Accordion Groups:** Expand and collapse individual group headers.
2. **Multi-Level / Nested Sub-Groups:** Infinite or multi-tier nested parent-child options.
3. **Tri-State / Indeterminate Checkboxes:** Parent headers dynamically calculate checked, unchecked, or indeterminate states and cascade selections to all descendants.
4. **Group-Aware Search:** Filtering an option automatically keeps its full parent hierarchy visible and expanded.

### 1. Types & Tree Utilities

```typescript
// types/groupedSelect.ts
export interface GroupedOptionNode {
  id: string;
  label: string;
  disabled?: boolean;
  children?: GroupedOptionNode[];
}

export type SelectionState = 'checked' | 'unchecked' | 'indeterminate';

/**
 * Recursively collects all selectable (leaf) IDs under a given node.
 */
export function getAllLeafIds(node: GroupedOptionNode): string[] {
  if (!node.children || node.children.length === 0) {
    return node.disabled ? [] : [node.id];
  }
  return node.children.flatMap((child) => getAllLeafIds(child));
}

/**
 * Calculates whether a node is checked, unchecked, or indeterminate based on active selections.
 */
export function getNodeSelectionState(
  node: GroupedOptionNode,
  selectedSet: Set<string>
): SelectionState {
  const leafIds = getAllLeafIds(node);
  if (leafIds.length === 0) return 'unchecked';

  const selectedCount = leafIds.filter((id) => selectedSet.has(id)).length;

  if (selectedCount === leafIds.length) return 'checked';
  if (selectedCount > 0) return 'indeterminate';
  return 'unchecked';
}

/**
 * Filters a hierarchical tree while keeping ancestors of matching nodes.
 */
export function filterGroupTree(
  nodes: GroupedOptionNode[],
  query: string
): GroupedOptionNode[] {
  const cleanQuery = query.toLowerCase().trim();
  if (!cleanQuery) return nodes;

  function filterNode(node: GroupedOptionNode): GroupedOptionNode | null {
    const isSelfMatch = node.label.toLowerCase().includes(cleanQuery);
    
    if (node.children && node.children.length > 0) {
      const filteredChildren = node.children
        .map((child) => filterNode(child))
        .filter((child): child is GroupedOptionNode => child !== null);

      if (filteredChildren.length > 0 || isSelfMatch) {
        return { ...node, children: filteredChildren };
      }
    } else if (isSelfMatch) {
      return node;
    }

    return null;
  }

  return nodes
    .map((node) => filterNode(node))
    .filter((node): node is GroupedOptionNode => node !== null);
}

```

### 2. Recursive Tree Option Item

```tsx
// components/GroupedOptionItem.tsx
import React, { useEffect, useRef } from 'react';
import { GroupedOptionNode, getNodeSelectionState } from '../types/groupedSelect';

interface GroupedOptionItemProps {
  node: GroupedOptionNode;
  depth?: number;
  selectedSet: Set<string>;
  collapsedSet: Set<string>;
  onToggleSelect: (node: GroupedOptionNode) => void;
  onToggleCollapse: (nodeId: string) => void;
}

export const GroupedOptionItem: React.FC<GroupedOptionItemProps> = ({
  node,
  depth = 0,
  selectedSet,
  collapsedSet,
  onToggleSelect,
  onToggleCollapse,
}) => {
  const checkboxRef = useRef<HTMLInputElement>(null);
  const isBranch = Boolean(node.children && node.children.length > 0);
  const isCollapsed = collapsedSet.has(node.id);
  const selectionState = getNodeSelectionState(node, selectedSet);

  // Sync indeterminate property with native DOM element
  useEffect(() => {
    if (checkboxRef.current) {
      checkboxRef.current.indeterminate = selectionState === 'indeterminate';
    }
  }, [selectionState]);

  return (
    <div style={{ display: 'flex', flexDirection: 'column' }}>
      {/* Option Row */}
      <div
        onClick={() => {
          if (!node.disabled) {
            onToggleSelect(node);
          }
        }}
        style={{
          display: 'flex',
          alignItems: 'center',
          gap: 8,
          padding: '6px 8px',
          paddingLeft: `${depth * 20 + 8}px`,
          fontSize: 13,
          color: node.disabled ? '#94a3b8' : '#1e293b',
          backgroundColor: isBranch ? '#f8fafc' : 'transparent',
          cursor: node.disabled ? 'not-allowed' : 'pointer',
          borderRadius: 4,
          transition: 'background-color 0.15s ease',
        }}
        onMouseEnter={(e) => {
          if (!node.disabled) e.currentTarget.style.backgroundColor = '#f1f5f9';
        }}
        onMouseLeave={(e) => {
          if (!node.disabled) {
            e.currentTarget.style.backgroundColor = isBranch ? '#f8fafc' : 'transparent';
          }
        }}
      >
        {/* Accordion Expand / Collapse Trigger */}
        {isBranch ? (
          <button
            type="button"
            onClick={(e) => {
              e.stopPropagation();
              onToggleCollapse(node.id);
            }}
            style={{
              background: 'none',
              border: 'none',
              cursor: 'pointer',
              fontSize: 10,
              width: 16,
              height: 16,
              display: 'flex',
              alignItems: 'center',
              justifyContent: 'center',
              color: '#64748b',
              padding: 0,
            }}
          >
            {isCollapsed ? '▶' : '▼'}
          </button>
        ) : (
          <span style={{ width: 16 }} />
        )}

        {/* Checkbox with Tri-State Support */}
        <input
          ref={checkboxRef}
          type="checkbox"
          checked={selectionState === 'checked'}
          disabled={node.disabled}
          onChange={() => {}} // Controlled by parent row click
          style={{ cursor: node.disabled ? 'not-allowed' : 'pointer' }}
        />

        {/* Label */}
        <span
          style={{
            fontWeight: isBranch ? 600 : 400,
            flex: 1,
            overflow: 'hidden',
            textOverflow: 'ellipsis',
            whiteSpace: 'nowrap',
          }}
        >
          {node.label}
        </span>
      </div>

      {/* Recursive Render for Child Branches */}
      {isBranch && !isCollapsed && node.children && (
        <div style={{ display: 'flex', flexDirection: 'column' }}>
          {node.children.map((child) => (
            <GroupedOptionItem
              key={child.id}
              node={child}
              depth={depth + 1}
              selectedSet={selectedSet}
              collapsedSet={collapsedSet}
              onToggleSelect={onToggleSelect}
              onToggleCollapse={onToggleCollapse}
            />
          ))}
        </div>
      )}
    </div>
  );
};
```

### 3. Main Hierarchical Grouped Select Component

```tsx
// components/HierarchicalGroupedSelect.tsx
import React, { useState, useRef, useEffect, useMemo } from 'react';
import { GroupedOptionNode, getAllLeafIds, filterGroupTree } from '../types/groupedSelect';
import { GroupedOptionItem } from './GroupedOptionItem';

interface HierarchicalGroupedSelectProps {
  data: GroupedOptionNode[];
  value: string[];
  onChange: (value: string[]) => void;
  placeholder?: string;
  maxDisplayTags?: number;
  disabled?: boolean;
}

export const HierarchicalGroupedSelect: React.FC<HierarchicalGroupedSelectProps> = ({
  data,
  value,
  onChange,
  placeholder = 'Select from categories...',
  maxDisplayTags = 3,
  disabled = false,
}) => {
  const [isOpen, setIsOpen] = useState(false);
  const [searchQuery, setSearchQuery] = useState('');
  const [collapsedSet, setCollapsedSet] = useState<Set<string>>(new Set());

  const containerRef = useRef<HTMLDivElement>(null);
  const selectedSet = useMemo(() => new Set(value), [value]);

  // Filtered Tree View
  const filteredTree = useMemo(() => filterGroupTree(data, searchQuery), [data, searchQuery]);

  // Flatten map of leaf nodes for label lookups in tag pills
  const leafMap = useMemo(() => {
    const map = new Map<string, string>();
    function traverse(nodes: GroupedOptionNode[]) {
      for (const n of nodes) {
        if (!n.children || n.children.length === 0) {
          map.set(n.id, n.label);
        } else {
          traverse(n.children);
        }
      }
    }
    traverse(data);
    return map;
  }, [data]);

  // Click Outside Listener
  useEffect(() => {
    const handleOutsideClick = (e: MouseEvent) => {
      if (containerRef.current && !containerRef.current.contains(e.target as Node)) {
        setIsOpen(false);
      }
    };
    document.addEventListener('mousedown', handleOutsideClick);
    return () => document.removeEventListener('mousedown', handleOutsideClick);
  }, []);

  // Selection toggle (cascades through all leaf descendants)
  const handleToggleSelect = (node: GroupedOptionNode) => {
    const targetLeafIds = getAllLeafIds(node);
    const allSelected = targetLeafIds.every((id) => selectedSet.has(id));

    if (allSelected) {
      // Uncheck all descendants
      onChange(value.filter((id) => !targetLeafIds.includes(id)));
    } else {
      // Check all descendants
      const next = Array.from(new Set([...value, ...targetLeafIds]));
      onChange(next);
    }
  };

  const handleToggleCollapse = (nodeId: string) => {
    setCollapsedSet((prev) => {
      const next = new Set(prev);
      next.has(nodeId) ? next.delete(nodeId) : next.add(nodeId);
      return next;
    });
  };

  const removeTag = (id: string, e: React.MouseEvent) => {
    e.stopPropagation();
    onChange(value.filter((v) => v !== id));
  };

  const clearAll = (e: React.MouseEvent) => {
    e.stopPropagation();
    onChange([]);
  };

  const visibleTags = value.slice(0, maxDisplayTags);
  const hiddenCount = value.length - maxDisplayTags;

  return (
    <div
      ref={containerRef}
      style={{
        position: 'relative',
        width: '100%',
        maxWidth: 420,
        fontFamily: 'Inter, system-ui, sans-serif',
        userSelect: 'none',
      }}
    >
      {/* Trigger Box with Selected Tags */}
      <div
        onClick={() => !disabled && setIsOpen((prev) => !prev)}
        style={{
          display: 'flex',
          flexWrap: 'wrap',
          alignItems: 'center',
          gap: 6,
          minHeight: 40,
          padding: '4px 8px',
          background: disabled ? '#f8fafc' : '#ffffff',
          border: `1px solid ${isOpen ? '#2563eb' : '#cbd5e1'}`,
          borderRadius: 6,
          cursor: disabled ? 'not-allowed' : 'pointer',
          boxShadow: isOpen ? '0 0 0 2px rgba(37, 99, 235, 0.15)' : 'none',
        }}
      >
        {value.length === 0 && (
          <span style={{ fontSize: 13, color: '#94a3b8', paddingLeft: 4 }}>{placeholder}</span>
        )}

        {/* Selected Tags */}
        {visibleTags.map((id) => (
          <span
            key={id}
            style={{
              display: 'inline-flex',
              alignItems: 'center',
              gap: 4,
              backgroundColor: '#eff6ff',
              color: '#1d4ed8',
              border: '1px solid #bfdbfe',
              borderRadius: 4,
              padding: '2px 6px',
              fontSize: 12,
              fontWeight: 500,
            }}
          >
            {leafMap.get(id) || id}
            <span
              onClick={(e) => removeTag(id, e)}
              style={{ cursor: 'pointer', fontSize: 12, fontWeight: 700, marginLeft: 2 }}
            >
              ✕
            </span>
          </span>
        ))}

        {hiddenCount > 0 && (
          <span
            style={{
              backgroundColor: '#f1f5f9',
              color: '#475569',
              borderRadius: 4,
              padding: '2px 6px',
              fontSize: 11,
              fontWeight: 600,
            }}
          >
            +{hiddenCount} more
          </span>
        )}

        {/* Right Actions */}
        <div style={{ marginLeft: 'auto', display: 'flex', alignItems: 'center', gap: 6 }}>
          {value.length > 0 && !disabled && (
            <button
              type="button"
              onClick={clearAll}
              style={{
                background: 'none',
                border: 'none',
                cursor: 'pointer',
                color: '#94a3b8',
                fontSize: 13,
              }}
            >
              ✕
            </button>
          )}
          <span
            style={{
              fontSize: 10,
              color: '#64748b',
              transform: isOpen ? 'rotate(180deg)' : 'none',
              transition: 'transform 0.15s ease',
            }}
          >
            ▼
          </span>
        </div>
      </div>

      {/* Dropdown Menu */}
      {isOpen && (
        <div
          style={{
            position: 'absolute',
            top: 'calc(100% + 4px)',
            left: 0,
            right: 0,
            zIndex: 1000,
            background: '#ffffff',
            border: '1px solid #e2e8f0',
            borderRadius: 6,
            boxShadow: '0 10px 15px -3px rgba(0, 0, 0, 0.1)',
            overflow: 'hidden',
          }}
        >
          {/* Search Filter */}
          <div style={{ padding: 6, borderBottom: '1px solid #f1f5f9', background: '#f8fafc' }}>
            <input
              type="text"
              placeholder="Search groups & options..."
              value={searchQuery}
              onChange={(e) => setSearchQuery(e.target.value)}
              onClick={(e) => e.stopPropagation()}
              style={{
                width: '100%',
                padding: '6px 8px',
                fontSize: 13,
                border: '1px solid #cbd5e1',
                borderRadius: 4,
                outline: 'none',
                boxSizing: 'border-box',
              }}
            />
          </div>

          {/* Hierarchical Tree Container */}
          <div style={{ maxHeight: 260, overflowY: 'auto', padding: 6 }}>
            {filteredTree.length === 0 ? (
              <div style={{ padding: 12, fontSize: 13, color: '#94a3b8', textAlign: 'center' }}>
                No matching results
              </div>
            ) : (
              filteredTree.map((node) => (
                <GroupedOptionItem
                  key={node.id}
                  node={node}
                  selectedSet={selectedSet}
                  collapsedSet={collapsedSet}
                  onToggleSelect={handleToggleSelect}
                  onToggleCollapse={handleToggleCollapse}
                />
              ))
            )}
          </div>
        </div>
      )}
    </div>
  );
};

```
