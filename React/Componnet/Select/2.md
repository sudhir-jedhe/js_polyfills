Here is the complete zero-dependency implementation of an enterprise-grade **Multi-Select Component** built from scratch in React and TypeScript.

This component brings together:

1. **Multi-Selection & Chip/Tag Badges:** Interactive badge pills with removal handlers (`✕`), tag overflow count (`+N more`), and a "Clear All" action.
2. **Debounced Async Data Fetching:** A custom debounce engine with cancellation tokens to eliminate race conditions during remote queries, complete with loading spinners.
3. **High-Performance List Virtualization:** A windowing algorithm calculating dynamic slice offsets so dropdowns with 10,000+ items run at 60 FPS.

### 1. Debounce Hook & Dropdown Virtualizer

```typescript
// hooks/useSelectVirtualizer.ts
import { useState, useEffect, useRef, useCallback } from 'react';

// --- 1. Generic Debounce Hook ---
export function useDebounce<T>(value: T, delayMs = 300): T {
  const [debouncedValue, setDebouncedValue] = useState<T>(value);

  useEffect(() => {
    const handler = setTimeout(() => setDebouncedValue(value), delayMs);
    return () => clearTimeout(handler);
  }, [value, delayMs]);

  return debouncedValue;
}

// --- 2. Lightweight Dropdown Windowing Virtualizer ---
interface DropdownVirtualizerProps {
  totalCount: number;
  itemHeight: number;
  containerHeight: number;
  overscan?: number;
}

export function useDropdownVirtualizer({
  totalCount,
  itemHeight,
  containerHeight,
  overscan = 4,
}: DropdownVirtualizerProps) {
  const [scrollTop, setScrollTop] = useState(0);
  const scrollContainerRef = useRef<HTMLDivElement | null>(null);

  const onScroll = useCallback((e: React.UIEvent<HTMLDivElement>) => {
    setScrollTop(e.currentTarget.scrollTop);
  }, []);

  const totalContentHeight = totalCount * itemHeight;
  const startIndex = Math.max(0, Math.floor(scrollTop / itemHeight) - overscan);
  const endIndex = Math.min(
    totalCount - 1,
    Math.floor((scrollTop + containerHeight) / itemHeight) + overscan
  );

  const virtualRows = [];
  for (let i = startIndex; i <= endIndex; i++) {
    virtualRows.push({
      index: i,
      offsetTop: i * itemHeight,
      height: itemHeight,
    });
  }

  return {
    scrollContainerRef,
    onScroll,
    virtualRows,
    totalContentHeight,
    startIndex,
  };
}
```

### 2. Multi-Select Component Implementation

```tsx
// components/EnterpriseMultiSelect.tsx
import React, { useState, useRef, useEffect, useMemo } from 'react';
import { useDebounce, useDropdownVirtualizer } from '../hooks/useSelectVirtualizer';

export interface MultiSelectOption<T = string | number> {
  value: T;
  label: string;
  subLabel?: string;
  disabled?: boolean;
}

interface EnterpriseMultiSelectProps<T = string | number> {
  // Static options or async loader
  options?: MultiSelectOption<T>[];
  fetchOptions?: (query: string) => Promise<MultiSelectOption<T>[]>;
  value: T[];
  onChange: (value: T[]) => void;
  placeholder?: string;
  maxDisplayTags?: number;
  disabled?: boolean;
  itemHeight?: number;
  dropdownHeight?: number;
}

export function EnterpriseMultiSelect<T extends string | number>({
  options = [],
  fetchOptions,
  value = [],
  onChange,
  placeholder = 'Select items...',
  maxDisplayTags = 3,
  disabled = false,
  itemHeight = 36,
  dropdownHeight = 220,
}: EnterpriseMultiSelectProps<T>) {
  const [isOpen, setIsOpen] = useState(false);
  const [searchQuery, setSearchQuery] = useState('');
  const [asyncItems, setAsyncItems] = useState<MultiSelectOption<T>[]>(options);
  const [isLoading, setIsLoading] = useState(false);
  const [highlightedIndex, setHighlightedIndex] = useState(0);

  const containerRef = useRef<HTMLDivElement>(null);
  const searchInputRef = useRef<HTMLInputElement>(null);
  const debouncedQuery = useDebounce(searchQuery, 280);

  // Selected map for O(1) membership checks
  const selectedSet = useMemo(() => new Set(value), [value]);

  // -------------------------------------------------------------
  // Async Data Fetching Pipeline
  // -------------------------------------------------------------
  useEffect(() => {
    let isCancelled = false;

    if (fetchOptions) {
      setIsLoading(true);
      fetchOptions(debouncedQuery)
        .then((data) => {
          if (!isCancelled) {
            setAsyncItems(data);
            setIsLoading(false);
            setHighlightedIndex(0);
          }
        })
        .catch((err) => {
          if (!isCancelled) {
            console.error('Failed to load options:', err);
            setIsLoading(false);
          }
        });
    } else {
      // Local sync filtering
      const filtered = options.filter(
        (opt) =>
          opt.label.toLowerCase().includes(debouncedQuery.toLowerCase()) ||
          opt.subLabel?.toLowerCase().includes(debouncedQuery.toLowerCase())
      );
      setAsyncItems(filtered);
      setHighlightedIndex(0);
    }

    return () => {
      isCancelled = true;
    };
  }, [debouncedQuery, fetchOptions, options]);

  // -------------------------------------------------------------
  // Virtualizer Configuration
  // -------------------------------------------------------------
  const { scrollContainerRef, onScroll, virtualRows, totalContentHeight } = useDropdownVirtualizer({
    totalCount: asyncItems.length,
    itemHeight,
    containerHeight: dropdownHeight,
    overscan: 4,
  });

  // Click Outside Listener
  useEffect(() => {
    const handleOutsideClick = (e: MouseEvent) => {
      if (containerRef.current && !containerRef.current.contains(e.target as Node)) {
        setIsOpen(false);
      }
    };
    document.addEventListener('mousedown', handleOutsideClick);
    return () => document.removeEventListener('mousedown', handleOutsideClick);
  }, []);

  // Keyboard Navigation
  const handleKeyDown = (e: React.KeyboardEvent) => {
    if (disabled) return;

    if (!isOpen) {
      if (['ArrowDown', 'ArrowUp', 'Enter', ' '].includes(e.key)) {
        e.preventDefault();
        setIsOpen(true);
      }
      return;
    }

    switch (e.key) {
      case 'ArrowDown':
        e.preventDefault();
        setHighlightedIndex((prev) => (prev < asyncItems.length - 1 ? prev + 1 : 0));
        break;
      case 'ArrowUp':
        e.preventDefault();
        setHighlightedIndex((prev) => (prev > 0 ? prev - 1 : asyncItems.length - 1));
        break;
      case 'Enter':
        e.preventDefault();
        if (asyncItems[highlightedIndex] && !asyncItems[highlightedIndex].disabled) {
          toggleOption(asyncItems[highlightedIndex].value);
        }
        break;
      case 'Escape':
        e.preventDefault();
        setIsOpen(false);
        break;
    }
  };

  // Selection Toggles
  const toggleOption = (val: T) => {
    if (selectedSet.has(val)) {
      onChange(value.filter((v) => v !== val));
    } else {
      onChange([...value, val]);
    }
  };

  const removeTag = (val: T, e: React.MouseEvent) => {
    e.stopPropagation();
    onChange(value.filter((v) => v !== val));
  };

  const clearAll = (e: React.MouseEvent) => {
    e.stopPropagation();
    onChange([]);
  };

  // Lookup map for fast label resolution of active tag badges
  const selectedOptionsMap = useMemo(() => {
    const map = new Map<T, MultiSelectOption<T>>();
    [...options, ...asyncItems].forEach((opt) => map.set(opt.value, opt));
    return map;
  }, [options, asyncItems]);

  const visibleSelectedValues = value.slice(0, maxDisplayTags);
  const hiddenCount = value.length - maxDisplayTags;

  return (
    <div
      ref={containerRef}
      onKeyDown={handleKeyDown}
      style={{
        position: 'relative',
        width: '100%',
        maxWidth: 420,
        fontFamily: 'Inter, system-ui, sans-serif',
        userSelect: 'none',
      }}
    >
      {/* Control Input Trigger with Tag Badges */}
      <div
        onClick={() => {
          if (!disabled) {
            setIsOpen((prev) => !prev);
            setTimeout(() => searchInputRef.current?.focus(), 50);
          }
        }}
        style={{
          display: 'flex',
          flexWrap: 'wrap',
          alignItems: 'center',
          gap: 6,
          minHeight: 40,
          padding: '4px 8px',
          background: disabled ? '#f8fafc' : '#ffffff',
          border: `1px solid ${isOpen ? '#2563eb' : '#cbd5e1'}`,
          borderRadius: 6,
          cursor: disabled ? 'not-allowed' : 'pointer',
          boxShadow: isOpen ? '0 0 0 2px rgba(37, 99, 235, 0.15)' : 'none',
          boxSizing: 'border-box',
        }}
      >
        {value.length === 0 && (
          <span style={{ fontSize: 13, color: '#94a3b8', paddingLeft: 4 }}>{placeholder}</span>
        )}

        {/* Tag Badges */}
        {visibleSelectedValues.map((val) => {
          const opt = selectedOptionsMap.get(val);
          const label = opt ? opt.label : String(val);

          return (
            <span
              key={String(val)}
              style={{
                display: 'inline-flex',
                alignItems: 'center',
                gap: 4,
                backgroundColor: '#eff6ff',
                color: '#1d4ed8',
                border: '1px solid #bfdbfe',
                borderRadius: 4,
                padding: '2px 6px',
                fontSize: 12,
                fontWeight: 500,
              }}
            >
              {label}
              <span
                onClick={(e) => removeTag(val, e)}
                style={{
                  cursor: 'pointer',
                  fontSize: 12,
                  fontWeight: 700,
                  color: '#3b82f6',
                  marginLeft: 2,
                }}
              >
                ✕
              </span>
            </span>
          );
        })}

        {/* Overflow Pill */}
        {hiddenCount > 0 && (
          <span
            style={{
              backgroundColor: '#f1f5f9',
              color: '#475569',
              borderRadius: 4,
              padding: '2px 6px',
              fontSize: 11,
              fontWeight: 600,
            }}
          >
            +{hiddenCount} more
          </span>
        )}

        {/* Right Actions: Clear Button + Spinner + Chevron */}
        <div style={{ marginLeft: 'auto', display: 'flex', alignItems: 'center', gap: 6 }}>
          {value.length > 0 && !disabled && (
            <button
              type="button"
              onClick={clearAll}
              style={{
                background: 'none',
                border: 'none',
                cursor: 'pointer',
                color: '#94a3b8',
                fontSize: 13,
                padding: '2px 4px',
              }}
            >
              ✕
            </button>
          )}

          {isLoading ? (
            <div
              style={{
                width: 12,
                height: 12,
                border: '2px solid #cbd5e1',
                borderTopColor: '#2563eb',
                borderRadius: '50%',
                animation: 'spin 0.6s linear infinite',
              }}
            />
          ) : (
            <span
              style={{
                fontSize: 10,
                color: '#64748b',
                transform: isOpen ? 'rotate(180deg)' : 'none',
                transition: 'transform 0.15s ease',
              }}
            >
              ▼
            </span>
          )}
        </div>
      </div>

      {/* Floating Virtualized Dropdown */}
      {isOpen && (
        <div
          style={{
            position: 'absolute',
            top: 'calc(100% + 4px)',
            left: 0,
            right: 0,
            zIndex: 1000,
            background: '#ffffff',
            border: '1px solid #e2e8f0',
            borderRadius: 6,
            boxShadow: '0 10px 15px -3px rgba(0, 0, 0, 0.1)',
            overflow: 'hidden',
          }}
        >
          {/* Search Input Bar */}
          <div style={{ padding: 6, borderBottom: '1px solid #f1f5f9', background: '#f8fafc' }}>
            <input
              ref={searchInputRef}
              type="text"
              placeholder="Filter items..."
              value={searchQuery}
              onChange={(e) => setSearchQuery(e.target.value)}
              onClick={(e) => e.stopPropagation()}
              style={{
                width: '100%',
                padding: '6px 8px',
                fontSize: 13,
                border: '1px solid #cbd5e1',
                borderRadius: 4,
                outline: 'none',
                boxSizing: 'border-box',
              }}
            />
          </div>

          {/* Virtual Window Viewport */}
          <div
            ref={scrollContainerRef}
            onScroll={onScroll}
            style={{
              height: dropdownHeight,
              overflowY: 'auto',
              position: 'relative',
            }}
          >
            {asyncItems.length === 0 ? (
              <div style={{ padding: 16, fontSize: 13, color: '#94a3b8', textAlign: 'center' }}>
                {isLoading ? 'Searching...' : 'No matching results'}
              </div>
            ) : (
              <div style={{ height: totalContentHeight, width: '100%', position: 'relative' }}>
                {virtualRows.map(({ index, offsetTop, height }) => {
                  const opt = asyncItems[index];
                  if (!opt) return null;

                  const isSelected = selectedSet.has(opt.value);
                  const isHighlighted = highlightedIndex === index;

                  return (
                    <div
                      key={String(opt.value)}
                      onClick={() => !opt.disabled && toggleOption(opt.value)}
                      onMouseEnter={() => !opt.disabled && setHighlightedIndex(index)}
                      style={{
                        position: 'absolute',
                        top: 0,
                        left: 0,
                        width: '100%',
                        height: `${height}px`,
                        transform: `translateY(${offsetTop}px)`,
                        display: 'flex',
                        alignItems: 'center',
                        justifyContent: 'space-between',
                        padding: '0 10px',
                        boxSizing: 'border-box',
                        fontSize: 13,
                        cursor: opt.disabled ? 'not-allowed' : 'pointer',
                        color: opt.disabled ? '#cbd5e1' : isSelected ? '#1d4ed8' : '#1e293b',
                        backgroundColor: isHighlighted
                          ? '#eff6ff'
                          : isSelected
                          ? '#f8fafc'
                          : '#ffffff',
                      }}
                    >
                      <div style={{ display: 'flex', alignItems: 'center', gap: 8, overflow: 'hidden' }}>
                        <input
                          type="checkbox"
                          checked={isSelected}
                          disabled={opt.disabled}
                          readOnly
                          style={{ cursor: 'pointer' }}
                        />
                        <span style={{ textOverflow: 'ellipsis', overflow: 'hidden', whiteSpace: 'nowrap' }}>
                          {opt.label}
                        </span>
                        {opt.subLabel && (
                          <span style={{ fontSize: 11, color: '#94a3b8' }}>({opt.subLabel})</span>
                        )}
                      </div>

                      {isSelected && <span style={{ fontSize: 12, fontWeight: 700 }}>✓</span>}
                    </div>
                  );
                })}
              </div>
            )}
          </div>
        </div>
      )}
    </div>
  );
}
```
