Here is the complete zero-dependency implementation of a **Button Group System built from scratch** in React with TypeScript.

It includes:

1. **Basic Button Group** (Shared borders, unified radius, focus ring layering).
2. **Variants & Colors** (`contained`, `outlined`, `text` paired with `primary`, `secondary`, `success`, `error`, `warning`).
3. **Orientation** (`horizontal` vs `vertical`).
4. **Elevation Control** (`disableElevation` toggle).
5. **Split Button Group** (Primary action with attached popover caret).
6. **Polymorphism & Context Inheritance** (Children inherit `size`, `color`, `variant`, and `disabled` automatically).

### 1. `ButtonGroupContext` & Component Engine

```tsx
// components/ButtonGroup.tsx
import React, { createContext, useContext } from 'react';
import { ButtonVariant, ButtonColor, ButtonSize, COLOR_PALETTES } from '../utils/buttonUtils';

export interface ButtonGroupContextValue {
  variant?: ButtonVariant;
  color?: ButtonColor;
  size?: ButtonSize;
  disabled?: boolean;
  disableElevation?: boolean;
  orientation?: 'horizontal' | 'vertical';
}

export const ButtonGroupContext = createContext<ButtonGroupContextValue | null>(null);

export interface ButtonGroupProps extends React.HTMLAttributes<HTMLDivElement> {
  variant?: ButtonVariant;
  color?: ButtonColor;
  size?: ButtonSize;
  orientation?: 'horizontal' | 'vertical';
  disabled?: boolean;
  disableElevation?: boolean;
  fullWidth?: boolean;
  children: React.ReactNode;
}

export const ButtonGroup: React.FC<ButtonGroupProps> = ({
  variant = 'outlined',
  color = 'primary',
  size = 'md',
  orientation = 'horizontal',
  disabled = false,
  disableElevation = false,
  fullWidth = false,
  children,
  style,
  className,
  ...restProps
}) => {
  const isVertical = orientation === 'vertical';
  const palette = COLOR_PALETTES[color];

  // Context value passed to standard child <Button /> components
  const contextValue: ButtonGroupContextValue = {
    variant,
    color,
    size,
    disabled,
    disableElevation,
    orientation,
  };

  return (
    <ButtonGroupContext.Provider value={contextValue}>
      <div
        role="group"
        className={`custom-button-group ${className || ''}`}
        style={{
          display: fullWidth ? 'flex' : 'inline-flex',
          flexDirection: isVertical ? 'column' : 'row',
          width: fullWidth ? '100%' : 'auto',
          borderRadius: 6,
          position: 'relative',
          boxShadow:
            variant === 'contained' && !disableElevation && !disabled
              ? '0 1px 3px rgba(0,0,0,0.12), 0 1px 2px rgba(0,0,0,0.06)'
              : 'none',
          ...style,
        }}
        {...restProps}
      >
        {children}

        {/* Global CSS rules for contiguous border handling and radius clipping */}
        <style>{`
          .custom-button-group > button,
          .custom-button-group > a {
            position: relative;
          }

          /* --- Horizontal Layout Rules --- */
          .custom-button-group[style*="flex-direction: row"] > button:not(:first-of-type),
          .custom-button-group[style*="flex-direction: row"] > a:not(:first-of-type) {
            border-top-left-radius: 0 !important;
            border-bottom-left-radius: 0 !important;
            margin-left: -1px;
          }

          .custom-button-group[style*="flex-direction: row"] > button:not(:last-of-type),
          .custom-button-group[style*="flex-direction: row"] > a:not(:last-of-type) {
            border-top-right-radius: 0 !important;
            border-bottom-right-radius: 0 !important;
          }

          /* Contained Separator Divider (Horizontal) */
          .custom-button-group[style*="flex-direction: row"] > button[data-variant="contained"]:not(:last-of-type)::after {
            content: "";
            position: absolute;
            right: 0;
            top: 20%;
            bottom: 20%;
            width: 1px;
            background-color: rgba(255, 255, 255, 0.35);
          }

          /* --- Vertical Layout Rules --- */
          .custom-button-group[style*="flex-direction: column"] > button:not(:first-of-type),
          .custom-button-group[style*="flex-direction: column"] > a:not(:first-of-type) {
            border-top-left-radius: 0 !important;
            border-top-right-radius: 0 !important;
            margin-top: -1px;
          }

          .custom-button-group[style*="flex-direction: column"] > button:not(:last-of-type),
          .custom-button-group[style*="flex-direction: column"] > a:not(:last-of-type) {
            border-bottom-left-radius: 0 !important;
            border-bottom-right-radius: 0 !important;
          }

          /* Contained Separator Divider (Vertical) */
          .custom-button-group[style*="flex-direction: column"] > button[data-variant="contained"]:not(:last-of-type)::after {
            content: "";
            position: absolute;
            bottom: 0;
            left: 15%;
            right: 15%;
            height: 1px;
            background-color: rgba(255, 255, 255, 0.35);
          }

          /* Elevate active or focused buttons above siblings to prevent clipping */
          .custom-button-group > button:hover,
          .custom-button-group > button:focus-visible,
          .custom-button-group > a:hover,
          .custom-button-group > a:focus-visible {
            z-index: 2;
          }
        `}</style>
      </div>
    </ButtonGroupContext.Provider>
  );
};

```

### 2. Button Component Context-Integration

Update our `Button` component to automatically inherit settings from `ButtonGroup` when nested inside one.

```tsx
// components/Button.tsx (Context Adapter Extension)
import React, { useContext } from 'react';
import { ButtonGroupContext } from './ButtonGroup';
// [Uses existing core button setup]

export const useButtonGroupContext = () => useContext(ButtonGroupContext);

// Inside the Button component implementation:
export const ButtonGroupAwareProps = (props: any) => {
  const group = useButtonGroupContext();

  return {
    variant: props.variant || group?.variant || 'contained',
    color: props.color || group?.color || 'primary',
    size: props.size || group?.size || 'md',
    disabled: props.disabled !== undefined ? props.disabled : group?.disabled,
    disableElevation: props.disableElevation !== undefined ? props.disableElevation : group?.disableElevation,
  };
};
```

### 3. Advanced Split Button Component

A split button composed of a primary action and an adjacent dropdown toggle within a unified `ButtonGroup`.

```tsx
// components/SplitButtonGroup.tsx
import React, { useState, useRef, useEffect } from 'react';
import { ButtonGroup } from './ButtonGroup';
import { Button } from './Button';
import { ButtonColor, ButtonVariant, ButtonSize } from '../utils/buttonUtils';

export interface SplitOption {
  id: string;
  label: string;
  description?: string;
}

interface SplitButtonGroupProps {
  options: SplitOption[];
  onSelect: (option: SplitOption, index: number) => void;
  variant?: ButtonVariant;
  color?: ButtonColor;
  size?: ButtonSize;
  disabled?: boolean;
  disableElevation?: boolean;
}

export const SplitButtonGroup: React.FC<SplitButtonGroupProps> = ({
  options,
  onSelect,
  variant = 'contained',
  color = 'primary',
  size = 'md',
  disabled = false,
  disableElevation = false,
}) => {
  const [selectedIndex, setSelectedIndex] = useState(0);
  const [isOpen, setIsOpen] = useState(false);
  const anchorRef = useRef<HTMLDivElement>(null);

  const activeOption = options[selectedIndex] || options[0];

  useEffect(() => {
    const handleOutsideClick = (e: MouseEvent) => {
      if (anchorRef.current && !anchorRef.current.contains(e.target as Node)) {
        setIsOpen(false);
      }
    };
    document.addEventListener('mousedown', handleOutsideClick);
    return () => document.removeEventListener('mousedown', handleOutsideClick);
  }, []);

  return (
    <div ref={anchorRef} style={{ position: 'relative', display: 'inline-block' }}>
      <ButtonGroup
        variant={variant}
        color={color}
        size={size}
        disabled={disabled}
        disableElevation={disableElevation}
      >
        {/* Main Action */}
        <Button
          data-variant={variant}
          onClick={() => onSelect(activeOption, selectedIndex)}
        >
          {activeOption.label}
        </Button>

        {/* Caret Action */}
        <Button
          data-variant={variant}
          aria-haspopup="menu"
          aria-expanded={isOpen}
          aria-label="Select alternate option"
          onClick={() => setIsOpen((prev) => !prev)}
          style={{ padding: size === 'sm' ? '0 6px' : size === 'lg' ? '0 12px' : '0 8px' }}
        >
          <span style={{ fontSize: 10, transform: isOpen ? 'rotate(180deg)' : 'none', transition: 'transform 0.15s' }}>
            ▼
          </span>
        </Button>
      </ButtonGroup>

      {/* Popover Menu */}
      {isOpen && (
        <div
          role="menu"
          style={{
            position: 'absolute',
            top: 'calc(100% + 4px)',
            right: 0,
            zIndex: 1000,
            minWidth: 200,
            background: '#ffffff',
            border: '1px solid #e2e8f0',
            borderRadius: 6,
            boxShadow: '0 10px 15px -3px rgba(0, 0, 0, 0.1)',
            padding: '4px 0',
            fontFamily: 'Inter, system-ui, sans-serif',
          }}
        >
          {options.map((option, idx) => {
            const isSelected = idx === selectedIndex;
            return (
              <div
                key={option.id}
                role="menuitem"
                onClick={() => {
                  setSelectedIndex(idx);
                  setIsOpen(false);
                  onSelect(option, idx);
                }}
                style={{
                  padding: '8px 14px',
                  fontSize: 13,
                  cursor: 'pointer',
                  backgroundColor: isSelected ? '#eff6ff' : 'transparent',
                  color: isSelected ? '#2563eb' : '#1e293b',
                  fontWeight: isSelected ? 600 : 400,
                  display: 'flex',
                  flexDirection: 'column',
                }}
              >
                <span>{option.label}</span>
                {option.description && (
                  <span style={{ fontSize: 11, color: '#64748b', marginTop: 2 }}>
                    {option.description}
                  </span>
                )}
              </div>
            );
          })}
        </div>
      )}
    </div>
  );
};

```

### Key Technical Details

| Feature                    | Implementation Mechanism                                                                                                                                            |
| -------------------------- | ------------------------------------------------------------------------------------------------------------------------------------------------------------------- |
| **Seamless Borders**       | Applied using CSS sibling selectors (`:not(:first-of-type) { margin-left: -1px; border-radius: 0; }`). Contiguous borders merge cleanly without doubling thickness. |
| **Focus Z-Index Stacking** | Active, hovered, or focused child buttons receive `z-index: 2`, ensuring their focus rings or hover outlines overlap adjacent siblings rather than getting clipped. |
| **Vertical Stacking**      | `orientation="vertical"` sets `flex-direction: column` and shifts collapsing borders from horizontal (`margin-left: -1px`) to vertical (`margin-top: -1px`).        |
| **Elevation Toggle**       | `disableElevation={true}` removes the container-level drop shadow (`boxShadow: 'none'`) for a flat UI look.                                                         |
| **Context Propagation**    | `ButtonGroupContext` propagates `variant`, `color`, `size`, and `disabled` down to any arbitrary child `<Button/>` without requiring manual prop drilling.          |
