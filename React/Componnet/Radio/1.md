***  1.md ***

Here is the complete zero-dependency implementation of a **Radio & Radio Group Component System built from scratch** in React with TypeScript.

It includes:

1. **Headless Context & Hook (`useRadioGroup`)**: Manages mutual exclusion, keyboard focus, and state across child radio controls.
2. **`RadioGroup` Component**: Direction control (`row` vs `column`), error states, and accessible `<fieldset>` semantics.
3. **Core `Radio` Component**: Built on hidden accessible `<input type="radio">`, custom dual-ring SVG radio animations, ripple effect, and focus ring.
4. **Standalone Radio Buttons**: Can operate outside a `RadioGroup` using independent `checked`/`onChange` props.
5. **Sizes, Colors & Label Placements**: `sm`/`md`/`lg` sizing, semantic theme palettes (`primary`, `secondary`, `success`, `error`, `warning`), and label alignments (`start`, `end`, `top`, `bottom`).
6. **Card-Style Custom Radio Selection**: Custom interactive payment/plan cards powered by `useRadioGroup`.

```tsx
import React, { createContext, useContext, useId } from 'react';
import { RadioSize, RadioColor, LabelPlacement } from '../utils/radioUtils';

export interface RadioGroupContextValue {
  name: string;
  value?: string | number;
  onChange?: (event: React.ChangeEvent<HTMLInputElement>, value: string) => void;
  size?: RadioSize;
  color?: RadioColor;
  labelPlacement?: LabelPlacement;

export const RadioGroupContext = createContext<RadioGroupContextValue | null>(null);
```

### 3. Core `Radio` Component

Works either as a child of `RadioGroup` (inheriting `name`, `value`, `onChange`, `size`, `color`) or as a standalone controlled/uncontrolled radio button.

```tsx
// components/Radio.tsx
import React, { useId } from 'react';
import { useRadioGroup } from '../context/RadioGroupContext';
import {
  RadioSize,
  RadioColor,
  LabelPlacement,
  RADIO_COLORS,
  RADIO_SIZES,
} from '../utils/radioUtils';

export interface RadioProps
  extends Omit<React.InputHTMLAttributes<HTMLInputElement>, 'size' | 'color'> {
  value?: string | number;
  label?: React.ReactNode;
  description?: React.ReactNode;
  size?: RadioSize;
  color?: RadioColor;
  labelPlacement?: LabelPlacement;
}

export const Radio: React.FC<RadioProps> = ({
  value,
  checked: controlledChecked,
  defaultChecked,
  onChange: customOnChange,
  disabled: customDisabled,
  name: customName,
  label,
  description,
  size: customSize,
  color: customColor,
  labelPlacement: customLabelPlacement,
  className,
  style,
  id: explicitId,
  ...restProps
}) => {
  const group = useRadioGroup();
  const generatedId = useId();
  const inputId = explicitId || generatedId;
  const descriptionId = `${inputId}-desc`;

  // Merge group context with individual props (individual props override group)
  const name = customName || group?.name;
  const size = customSize || group?.size || 'md';
  const color = customColor || group?.color || 'primary';
  const labelPlacement = customLabelPlacement || group?.labelPlacement || 'end';
  const disabled = customDisabled !== undefined ? customDisabled : group?.disabled || false;

  // Determine checked state
  const isChecked =
    group && value !== undefined
      ? String(group.value) === String(value)
      : controlledChecked;

  const handleChange = (e: React.ChangeEvent<HTMLInputElement>) => {
    if (disabled) return;
    customOnChange?.(e);
    if (group && value !== undefined) {
      group.onChange?.(e, String(value));
    }
  };

  const palette = RADIO_COLORS[color];
  const sizeSpec = RADIO_SIZES[size];

  const isVertical = labelPlacement === 'top' || labelPlacement === 'bottom';
  const isReversed = labelPlacement === 'start' || labelPlacement === 'top';

  return (
    <label
      htmlFor={inputId}
      className={className}
      style={{
        display: 'inline-flex',
        flexDirection: isVertical
          ? isReversed
            ? 'column-reverse'
            : 'column'
          : isReversed
          ? 'row-reverse'
          : 'row',
        alignItems: isVertical ? 'flex-start' : 'center',
        justifyContent: isReversed && !isVertical ? 'space-between' : 'flex-start',
        gap: isVertical ? 4 : 8,
        cursor: disabled ? 'not-allowed' : 'pointer',
        opacity: disabled ? 0.45 : 1,
        userSelect: 'none',
        position: 'relative',
        fontFamily: 'Inter, system-ui, -apple-system, sans-serif',
        ...style,
      }}
    >
      {/* Hidden Native Radio Input */}
      <input
        type="radio"
        id={inputId}
        name={name}
        value={value}
        checked={isChecked}
        defaultChecked={defaultChecked}
        onChange={handleChange}
        disabled={disabled}
        aria-describedby={description ? descriptionId : undefined}
        style={{
          position: 'absolute',
          width: 1,
          height: 1,
          padding: 0,
          margin: -1,
          overflow: 'hidden',
          clip: 'rect(0, 0, 0, 0)',
          whiteSpace: 'nowrap',
          border: 0,
        }}
        {...restProps}
      />

      {/* Visual Radio Circle */}
      <div
        className="radio-circle"
        style={{
          width: sizeSpec.outerDiameter,
          height: sizeSpec.outerDiameter,
          borderRadius: '50%',
          display: 'flex',
          alignItems: 'center',
          justifyContent: 'center',
          flexShrink: 0,
          transition: 'border-color 0.15s ease, background-color 0.15s ease',
          boxSizing: 'border-box',
        }}
      >
        {/* Animated Inner Dot */}
        <div
          className="radio-dot"
          style={{
            width: sizeSpec.innerDot,
            height: sizeSpec.innerDot,
            borderRadius: '50%',
            backgroundColor: palette.main,
            transition: 'transform 0.15s cubic-bezier(0.4, 0, 0.2, 1), opacity 0.15s ease',
          }}
        />
      </div>

      {/* Text Label & Description */}
      {(label || description) && (
        <div style={{ display: 'flex', flexDirection: 'column' }}>
          {label && (
            <span
              style={{
                fontSize: sizeSpec.fontSize,
                fontWeight: 500,
                color: '#1e293b',
                lineHeight: 1.3,
              }}
            >
              {label}
            </span>
          )}
          {description && (
            <span
              id={descriptionId}
              style={{
                fontSize: sizeSpec.descSize,
                color: '#64748b',
                lineHeight: 1.3,
                marginTop: 2,
              }}
            >
              {description}
            </span>
          )}
        </div>
      )}

      {/* Dynamic CSS for hover, checked state, and focus-visible */}
      <style>{`
        /* Unchecked Default */
        input[type="radio"] + .radio-circle {
          border: 2px solid #cbd5e1;
          background-color: #ffffff;
        }
        input[type="radio"]:not(:disabled) + .radio-circle:hover {
          border-color: ${palette.main};
          background-color: ${palette.hoverBg};
        }

        /* Checked State */
        input[type="radio"]:checked + .radio-circle {
          border-color: ${palette.main};
        }

        /* Dot Scaling */
        input[type="radio"]:not(:checked) + .radio-circle .radio-dot {
          opacity: 0;
          transform: scale(0);
        }
        input[type="radio"]:checked + .radio-circle .radio-dot {
          opacity: 1;
          transform: scale(1);
        }

        /* Focus Visible Ring */
        input[type="radio"]:focus-visible + .radio-circle {
          outline: 2px solid ${palette.main};
          outline-offset: 2px;
        }
      `}</style>
    </label>
  );
};
```

### 4. `RadioGroup` Component

```tsx
// components/RadioGroup.tsx
import React, { useId } from 'react';
import { RadioGroupContext } from '../context/RadioGroupContext';
import { RadioSize, RadioColor, LabelPlacement } from '../utils/radioUtils';

export interface RadioGroupProps {
  name?: string;
  value?: string | number;
  defaultValue?: string | number;
  onChange?: (event: React.ChangeEvent<HTMLInputElement>, value: string) => void;
  direction?: 'row' | 'column';
  label?: React.ReactNode;
  description?: React.ReactNode;
  error?: React.ReactNode;
  showError?: boolean;
  size?: RadioSize;
  color?: RadioColor;
  labelPlacement?: LabelPlacement;
  disabled?: boolean;
  children: React.ReactNode;
  style?: React.CSSProperties;
  className?: string;
}

export const RadioGroup: React.FC<RadioGroupProps> = ({
  name: explicitName,
  value: controlledValue,
  defaultValue,
  onChange,
  direction = 'column',
  label,
  description,
  error,
  showError = false,
  size = 'md',
  color = 'primary',
  labelPlacement = 'end',
  disabled = false,
  children,
  style,
  className,
}) => {
  const generatedName = useId();
  const name = explicitName || generatedName;
  const labelId = `${name}-label`;
  const errorId = `${name}-error`;

  const [uncontrolledValue, setUncontrolledValue] = React.useState<string | number | undefined>(
    defaultValue
  );

  const isControlled = controlledValue !== undefined;
  const activeValue = isControlled ? controlledValue : uncontrolledValue;

  const handleRadioChange = (event: React.ChangeEvent<HTMLInputElement>, newValue: string) => {
    if (!isControlled) {
      setUncontrolledValue(newValue);
    }
    onChange?.(event, newValue);
  };

  const hasError = showError && Boolean(error);

  return (
    <RadioGroupContext.Provider
      value={{
        name,
        value: activeValue,
        onChange: handleRadioChange,
        size,
        color: hasError ? 'error' : color,
        labelPlacement,
        disabled,
      }}
    >
      <fieldset
        role="radiogroup"
        aria-labelledby={label ? labelId : undefined}
        aria-describedby={hasError ? errorId : undefined}
        aria-invalid={hasError}
        className={className}
        style={{
          border: 'none',
          padding: 0,
          margin: 0,
          fontFamily: 'Inter, system-ui, -apple-system, sans-serif',
          ...style,
        }}
      >
        {label && (
          <legend
            id={labelId}
            style={{
              fontSize: '14px',
              fontWeight: 600,
              color: hasError ? '#dc2626' : '#0f172a',
              marginBottom: description ? 2 : 8,
              padding: 0,
            }}
          >
            {label}
          </legend>
        )}

        {description && (
          <p style={{ margin: '0 0 10px 0', fontSize: '12px', color: '#64748b' }}>
            {description}
          </p>
        )}

        {/* Radio List Layout */}
        <div
          style={{
            display: 'flex',
            flexDirection: direction,
            gap: direction === 'row' ? 20 : 10,
            flexWrap: 'wrap',
          }}
        >
          {children}
        </div>

        {/* Error Feedback Message */}
        {hasError && (
          <span
            id={errorId}
            role="alert"
            style={{
              display: 'flex',
              alignItems: 'center',
              gap: 4,
              marginTop: 6,
              fontSize: '12px',
              fontWeight: 500,
              color: '#dc2626',
            }}
          >
            <span>⚠️</span> {error}
          </span>
        )}
      </fieldset>
    </RadioGroupContext.Provider>
  );
};

```

### 5. Custom Card Radio Selector (`useRadioGroup` Customization)

Shows how to build rich custom UI options (such as plan or subscription pickers) that plug into `useRadioGroup` for keyboard navigation and state sharing.

```tsx
// components/CustomPlanCardRadio.tsx
import React from 'react';
import { useRadioGroup } from '../context/RadioGroupContext';

export interface PlanOption {
  value: string;
  title: string;
  price: string;
  description: string;
  badge?: string;
}

export const CustomPlanCardRadio: React.FC<PlanOption> = ({
  value,
  title,
  price,
  description,
  badge,
}) => {
  const group = useRadioGroup();
  const isSelected = String(group?.value) === String(value);
  const inputId = `${group?.name}-${value}`;

  return (
    <label
      htmlFor={inputId}
      style={{
        flex: 1,
        minWidth: 160,
        padding: '14px 16px',
        borderRadius: 8,
        border: `2px solid ${isSelected ? '#2563eb' : '#e2e8f0'}`,
        backgroundColor: isSelected ? '#eff6ff' : '#ffffff',
        cursor: group?.disabled ? 'not-allowed' : 'pointer',
        display: 'flex',
        flexDirection: 'column',
        gap: 6,
        position: 'relative',
        transition: 'all 0.15s ease',
        userSelect: 'none',
      }}
    >
      <input
        type="radio"
        id={inputId}
        name={group?.name}
        value={value}
        checked={isSelected}
        disabled={group?.disabled}
        onChange={(e) => group?.onChange?.(e, value)}
        style={{
          position: 'absolute',
          opacity: 0,
          pointerEvents: 'none',
        }}
      />

      <div style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'center' }}>
        <span style={{ fontWeight: 700, fontSize: '14px', color: isSelected ? '#1e40af' : '#0f172a' }}>
          {title}
        </span>
        {badge && (
          <span
            style={{
              fontSize: '10px',
              fontWeight: 700,
              backgroundColor: '#dbeafe',
              color: '#1d4ed8',
              padding: '2px 6px',
              borderRadius: 12,
            }}
          >
            {badge}
          </span>
        )}
      </div>

      <div style={{ fontSize: '18px', fontWeight: 800, color: '#0f172a' }}>{price}</div>
      <div style={{ fontSize: '12px', color: '#64748b' }}>{description}</div>
    </label>
  );
};
```

### Key Technical Details

| Feature                                 | Implementation Mechanism                                                                                                                                                                              |
| --------------------------------------- | ----------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------- |
| **Context Propagation**                 | `RadioGroupContext` passes `name`, `value`, `onChange`, `size`, `color`, `labelPlacement`, and `disabled` down to nested `<Radio/>` components without manual prop drilling.                          |
| **Mutual Exclusion & Focus Management** | Relies on the browser's native `<input type="radio" name={...} />` mechanism, preserving native keyboard arrow-key navigation (`ArrowUp`/`ArrowDown`/`ArrowLeft`/`ArrowRight`) and tab-stop grouping. |
| **Semantic Accessibility**              | Wraps groups in `<fieldset role="radiogroup">`, connects labels via `aria-labelledby`, and binds error messages using `aria-describedby` with `aria-invalid={true}`.                                  |
| **Standalone Flexibility**              | If `<Radio/>` is used without an enclosing `<RadioGroup>`, it falls back gracefully to independent `name`, `checked`, and `onChange` attributes.                                                      |
| **Custom Extensibility**                | Custom components like `CustomPlanCardRadio` consume `useRadioGroup()` to build custom styled cards while remaining accessible form inputs.                                                           |
