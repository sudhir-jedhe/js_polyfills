### 1. Mathematics & Headless Hook (`usePagination`)

This algorithm calculates visible page ranges (`boundaryCount`, `siblingCount`), handles ellipsis logic (`start-ellipsis`, `end-ellipsis`), and formats keyboard events for navigation.

```typescript
// hooks/usePagination.ts
import { useState, useMemo } from 'react';

export type PaginationItemType =
  | 'page'
  | 'first'
  | 'last'
  | 'next'
  | 'previous'
  | 'start-ellipsis'
  | 'end-ellipsis';

export interface UsePaginationProps {
  count?: number; // Total number of pages
  page?: number; // Current active page (controlled)
  defaultPage?: number; // Initial page (uncontrolled)
  boundaryCount?: number; // Number of always-visible boundary pages at start/end
  siblingCount?: number; // Number of visible pages flanking the current page
  hideNextButton?: boolean;
  hidePrevButton?: boolean;
  showFirstButton?: boolean;
  showLastButton?: boolean;
  disabled?: boolean;
  onChange?: (event: React.SyntheticEvent, page: number) => void;
}

export interface PaginationItemInfo {
  type: PaginationItemType;
  page: number | null;
  selected: boolean;
  disabled: boolean;
  onClick: (event: React.SyntheticEvent) => void;
  'aria-current'?: 'page' | undefined;
  'aria-label': string;
}

export function usePagination(props: UsePaginationProps) {
  const {
    count = 1,
    page: controlledPage,
    defaultPage = 1,
    boundaryCount = 1,
    siblingCount = 1,
    hideNextButton = false,
    hidePrevButton = false,
    showFirstButton = false,
    showLastButton = false,
    disabled = false,
    onChange,
  } = props;

  const [pageState, setPageState] = useState(defaultPage);
  const isControlled = controlledPage !== undefined;
  const currentPage = isControlled ? controlledPage : pageState;

  const handleClick = (event: React.SyntheticEvent, value: number) => {
    if (!isControlled) {
      setPageState(value);
    }
    onChange?.(event, value);
  };

  const range = (start: number, end: number) => {
    const length = end - start + 1;
    return Array.from({ length }, (_, i) => start + i);
  };

  const itemsList = useMemo(() => {
    const startPages = range(1, Math.min(boundaryCount, count));
    const endPages = range(Math.max(count - boundaryCount + 1, boundaryCount + 1), count);

    const siblingsStart = Math.max(
      Math.min(
        currentPage - siblingCount,
        count - boundaryCount - siblingCount * 2 - 1
      ),
      boundaryCount + 2
    );

    const siblingsEnd = Math.min(
      Math.max(
        currentPage + siblingCount,
        boundaryCount + siblingCount * 2 + 2
      ),
      endPages.length > 0 ? endPages[0] - 2 : count - 1
    );

    const itemList: (number | PaginationItemType)[] = [];

    // First button
    if (showFirstButton) itemList.push('first');
    // Previous button
    if (!hidePrevButton) itemList.push('previous');

    // Start boundaries
    itemList.push(...startPages);

    // Start Ellipsis
    if (siblingsStart > boundaryCount + 2) {
      itemList.push('start-ellipsis');
    } else if (boundaryCount + 1 < count - boundaryCount) {
      itemList.push(boundaryCount + 1);
    }

    // Main Siblings
    itemList.push(...range(siblingsStart, siblingsEnd));

    // End Ellipsis
    if (siblingsEnd < count - boundaryCount - 1) {
      itemList.push('end-ellipsis');
    } else if (count - boundaryCount > boundaryCount) {
      itemList.push(count - boundaryCount);
    }

    // End boundaries
    itemList.push(...endPages);

    // Next button
    if (!hideNextButton) itemList.push('next');
    // Last button
    if (showLastButton) itemList.push('last');

    return itemList;
  }, [count, currentPage, boundaryCount, siblingCount, showFirstButton, showLastButton, hidePrevButton, hideNextButton]);

  const items: PaginationItemInfo[] = itemsList.map((item) => {
    if (typeof item === 'number') {
      return {
        type: 'page',
        page: item,
        selected: item === currentPage,
        disabled: disabled,
        'aria-current': item === currentPage ? 'page' : undefined,
        'aria-label': `Go to page ${item}`,
        onClick: (e: React.SyntheticEvent) => handleClick(e, item),
      };
    }

    const mapTypeToPage: Record<PaginationItemType, number | null> = {
      first: 1,
      previous: Math.max(currentPage - 1, 1),
      next: Math.min(currentPage + 1, count),
      last: count,
      'start-ellipsis': null,
      'end-ellipsis': null,
      page: null,
    };

    const targetPage = mapTypeToPage[item];
    const isNavigationDisabled =
      disabled ||
      (item === 'first' && currentPage <= 1) ||
      (item === 'previous' && currentPage <= 1) ||
      (item === 'next' && currentPage >= count) ||
      (item === 'last' && currentPage >= count) ||
      item.includes('ellipsis');

    const labels: Record<PaginationItemType, string> = {
      first: 'Go to first page',
      previous: 'Go to previous page',
      next: 'Go to next page',
      last: 'Go to last page',
      'start-ellipsis': 'Previous pages ellipsis',
      'end-ellipsis': 'Next pages ellipsis',
      page: '',
    };

    return {
      type: item,
      page: targetPage,
      selected: false,
      disabled: isNavigationDisabled,
      'aria-label': labels[item],
      onClick: (e: React.SyntheticEvent) => {
        if (!isNavigationDisabled && targetPage !== null) {
          handleClick(e, targetPage);
        }
      },
    };
  });

  return {
    items,
    currentPage,
  };
}
```

### 2. Core Pagination Item & master `Pagination` Component

```tsx
// components/Pagination.tsx
import React, { forwardRef } from 'react';
import { usePagination, UsePaginationProps, PaginationItemInfo } from '../hooks/usePagination';

export type PaginationVariant = 'text' | 'outlined';
export type PaginationShape = 'circular' | 'rounded';
export type PaginationSize = 'sm' | 'md' | 'lg';
export type PaginationColor = 'primary' | 'secondary' | 'standard';

export interface PaginationItemProps {
  item: PaginationItemInfo;
  variant?: PaginationVariant;
  shape?: PaginationShape;
  size?: PaginationSize;
  color?: PaginationColor;
  renderItem?: (item: PaginationItemInfo) => React.ReactNode;
  firstIcon?: React.ReactNode;
  prevIcon?: React.ReactNode;
  nextIcon?: React.ReactNode;
  lastIcon?: React.ReactNode;
}

export const PaginationItem: React.FC<PaginationItemProps> = ({
  item,
  variant = 'text',
  shape = 'circular',
  size = 'md',
  color = 'primary',
  renderItem,
  firstIcon = '«',
  prevIcon = '‹',
  nextIcon = '›',
  lastIcon = '»',
}) => {
  if (renderItem) {
    return <>{renderItem(item)}</>;
  }

  const { type, page, selected, disabled, onClick, ...ariaProps } = item;

  if (type === 'start-ellipsis' || type === 'end-ellipsis') {
    return (
      <span
        style={{
          display: 'inline-flex',
          alignItems: 'center',
          justifyContent: 'center',
          padding: '0 4px',
          color: '#94a3b8',
          fontSize: 14,
          userSelect: 'none',
        }}
      >
        …
      </span>
    );
  }

  const sizeMetrics = {
    sm: { diameter: 26, fontSize: 12 },
    md: { diameter: 32, fontSize: 13 },
    lg: { diameter: 40, fontSize: 15 },
  }[size];

  const colorPalettes = {
    standard: { bg: '#f1f5f9', text: '#0f172a', border: '#cbd5e1' },
    primary: { bg: '#2563eb', text: '#ffffff', border: '#2563eb' },
    secondary: { bg: '#475569', text: '#ffffff', border: '#475569' },
  }[color];

  const iconMap: Record<string, React.ReactNode> = {
    first: firstIcon,
    previous: prevIcon,
    next: nextIcon,
    last: lastIcon,
    page: page,
  };

  const isOutlined = variant === 'outlined';

  return (
    <button
      type="button"
      disabled={disabled}
      onClick={onClick}
      {...ariaProps}
      style={{
        minWidth: sizeMetrics.diameter,
        height: sizeMetrics.diameter,
        padding: '0 6px',
        borderRadius: shape === 'circular' ? '50%' : 6,
        border: isOutlined
          ? `1px solid ${selected ? colorPalettes.border : '#cbd5e1'}`
          : '1px solid transparent',
        backgroundColor: selected
          ? isOutlined && color === 'standard'
            ? '#f8fafc'
            : colorPalettes.bg
          : 'transparent',
        color: selected
          ? isOutlined && color === 'standard'
            ? '#0f172a'
            : colorPalettes.text
          : '#334155',
        fontSize: sizeMetrics.fontSize,
        fontWeight: selected ? 700 : 500,
        fontFamily: 'Inter, system-ui, sans-serif',
        cursor: disabled ? 'not-allowed' : 'pointer',
        opacity: disabled ? 0.38 : 1,
        display: 'inline-flex',
        alignItems: 'center',
        justifyContent: 'center',
        outline: 'none',
        userSelect: 'none',
        transition: 'all 0.15s ease-in-out',
      }}
      onMouseEnter={(e) => {
        if (!disabled && !selected) {
          e.currentTarget.style.backgroundColor = '#f1f5f9';
        }
      }}
      onMouseLeave={(e) => {
        if (!disabled && !selected) {
          e.currentTarget.style.backgroundColor = 'transparent';
        }
      }}
    >
      {iconMap[type]}
    </button>
  );
};

export interface PaginationProps extends UsePaginationProps {
  variant?: PaginationVariant;
  shape?: PaginationShape;
  size?: PaginationSize;
  color?: PaginationColor;
  className?: string;
  style?: React.CSSProperties;
  renderItem?: (item: PaginationItemInfo) => React.ReactNode;
}

export const Pagination: React.FC<PaginationProps> = ({
  variant = 'text',
  shape = 'circular',
  size = 'md',
  color = 'primary',
  className,
  style,
  renderItem,
  ...hookProps
}) => {
  const { items } = usePagination(hookProps);

  return (
    <nav
      aria-label="pagination navigation"
      className={`pagination-root ${className || ''}`}
      style={{ display: 'inline-block', ...style }}
    >
      <ul
        style={{
          display: 'flex',
          alignItems: 'center',
          gap: 4,
          margin: 0,
          padding: 0,
          listStyle: 'none',
        }}
      >
        {items.map((item, idx) => (
          <li key={idx}>
            <PaginationItem
              item={item}
              variant={variant}
              shape={shape}
              size={size}
              color={color}
              renderItem={renderItem}
            />
          </li>
        ))}
      </ul>
    </nav>
  );
};
```

### 3. Integrated Table Pagination Component

```tsx
// components/TablePagination.tsx
import React from 'react';

export interface TablePaginationProps {
  count: number; // Total row count
  page: number; // 0-based or 1-based page index
  rowsPerPage: number;
  rowsPerPageOptions?: number[];
  onPageChange: (event: React.MouseEvent<HTMLButtonElement> | null, newPage: number) => void;
  onRowsPerPageChange: (event: React.ChangeEvent<HTMLSelectElement>) => void;
  labelRowsPerPage?: string;
  showFirstLastButtons?: boolean;
}

export const TablePagination: React.FC<TablePaginationProps> = ({
  count,
  page,
  rowsPerPage,
  rowsPerPageOptions = [10, 25, 50, 100],
  onPageChange,
  onRowsPerPageChange,
  labelRowsPerPage = 'Rows per page:',
  showFirstLastButtons = true,
}) => {
  // Assuming 0-based page indexing for table APIs
  const from = count === 0 ? 0 : page * rowsPerPage + 1;
  const to = Math.min(count, (page + 1) * rowsPerPage);
  const totalPages = Math.ceil(count / rowsPerPage);

  return (
    <div
      style={{
        display: 'flex',
        alignItems: 'center',
        justifyContent: 'flex-end',
        gap: 20,
        padding: '10px 16px',
        fontSize: 13,
        color: '#475569',
        fontFamily: 'Inter, system-ui, sans-serif',
        borderTop: '1px solid #e2e8f0',
        backgroundColor: '#ffffff',
        userSelect: 'none',
      }}
    >
      {/* Rows per page selector */}
      <div style={{ display: 'flex', alignItems: 'center', gap: 8 }}>
        <span>{labelRowsPerPage}</span>
        <select
          value={rowsPerPage}
          onChange={onRowsPerPageChange}
          style={{
            padding: '4px 6px',
            fontSize: 13,
            border: '1px solid #cbd5e1',
            borderRadius: 4,
            outline: 'none',
            cursor: 'pointer',
            backgroundColor: '#ffffff',
          }}
        >
          {rowsPerPageOptions.map((opt) => (
            <option key={opt} value={opt}>
              {opt}
            </option>
          ))}
        </select>
      </div>

      {/* Row range tracker */}
      <div>
        <span>
          {from}–{to} of {count}
        </span>
      </div>

      {/* Navigation Arrow Actions */}
      <div style={{ display: 'flex', alignItems: 'center', gap: 2 }}>
        {showFirstLastButtons && (
          <button
            type="button"
            aria-label="First page"
            disabled={page <= 0}
            onClick={(e) => onPageChange(e, 0)}
            style={tableNavBtnStyle(page <= 0)}
          >
            «
          </button>
        )}
        <button
          type="button"
          aria-label="Previous page"
          disabled={page <= 0}
          onClick={(e) => onPageChange(e, page - 1)}
          style={tableNavBtnStyle(page <= 0)}
        >
          ‹
        </button>
        <button
          type="button"
          aria-label="Next page"
          disabled={page >= totalPages - 1}
          onClick={(e) => onPageChange(e, page + 1)}
          style={tableNavBtnStyle(page >= totalPages - 1)}
        >
          ›
        </button>
        {showFirstLastButtons && (
          <button
            type="button"
            aria-label="Last page"
            disabled={page >= totalPages - 1}
            onClick={(e) => onPageChange(e, totalPages - 1)}
            style={tableNavBtnStyle(page >= totalPages - 1)}
          >
            »
          </button>
        )}
      </div>
    </div>
  );
};

const tableNavBtnStyle = (disabled: boolean): React.CSSProperties => ({
  width: 28,
  height: 28,
  borderRadius: '50%',
  border: 'none',
  backgroundColor: 'transparent',
  color: '#334155',
  fontSize: 16,
  cursor: disabled ? 'not-allowed' : 'pointer',
  opacity: disabled ? 0.35 : 1,
  display: 'inline-flex',
  alignItems: 'center',
  justifyContent: 'center',
  outline: 'none',
});
```

### 4. Router Integration Adapter

Passes router query params (`?page=N`) to render real link anchors for SEO while retaining single-page app behavior.

```tsx
// adapters/RouterPagination.tsx
import React from 'react';
import { Pagination, PaginationProps } from '../components/Pagination';
import { PaginationItemInfo } from '../hooks/usePagination';

// Router Link Mock
export const RouterPagination: React.FC<{
  count: number;
  currentPage: number;
  baseUrl: string;
}> = ({ count, currentPage, baseUrl }) => {
  return (
    <Pagination
      count={count}
      page={currentPage}
      renderItem={(item: PaginationItemInfo) => {
        if (item.type.includes('ellipsis')) {
          return <span style={{ padding: '0 6px', color: '#94a3b8' }}>…</span>;
        }

        const href = `${baseUrl}?page=${item.page}`;

        return (
          <a
            href={href}
            aria-label={item['aria-label']}
            aria-current={item['aria-current']}
            onClick={(e) => {
              e.preventDefault(); // SPA navigate
              item.onClick(e);
            }}
            style={{
              minWidth: 32,
              height: 32,
              borderRadius: '50%',
              backgroundColor: item.selected ? '#2563eb' : 'transparent',
              color: item.selected ? '#ffffff' : '#1e293b',
              display: 'inline-flex',
              alignItems: 'center',
              justifyContent: 'center',
              textDecoration: 'none',
              fontSize: 13,
              fontWeight: item.selected ? 700 : 500,
              cursor: item.disabled ? 'not-allowed' : 'pointer',
              opacity: item.disabled ? 0.4 : 1,
            }}
          >
            {item.type === 'previous'
              ? '‹'
              : item.type === 'next'
              ? '›'
              : item.page}
          </a>
        );
      }}
    />
  );
};
```

### Key Technical Details

| Feature                         | Implementation Mechanism                                                                                                                                                                                         |
| ------------------------------- | ---------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------- |
| **Sibling & Boundary Math**     | Calculates `siblingsStart` and `siblingsEnd` relative to `boundaryCount` and clamps them inside `[1, count]`, dynamically generating `'start-ellipsis'` and `'end-ellipsis'` slots.                              |
| **WAI-ARIA Accessibility**      | Wraps the list in `<nav aria-label="pagination navigation">` and unordered list items (`<ul>`/`<li>`), applying `aria-current="page"` to the active button and descriptive `aria-label` tags for screen readers. |
| **Zero-Based Table Pagination** | Bridges standard 1-based page interfaces with 0-based data table indexes (`page * rowsPerPage + 1` to `min(count, (page + 1) * rowsPerPage)`).                                                                   |
| **Custom SEO Routing**          | Through the `renderItem` prop, individual items can be mapped to semantic `<a>` tags with valid `href` search params for server-side rendering and search engine indexing.                                       |
