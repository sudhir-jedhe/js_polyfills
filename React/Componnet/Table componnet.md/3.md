In data grid and table UI design, action menus vary significantly depending on the target scope:

| Scope                  | Trigger Location                            | Primary Actions       | Typical Use Case |
| ---------------------- | ------------------------------------------- | --------------------- | ---------------- |
| **Column Action Menu** | Header cell (kebab icon `⋮` or right-click) | • Sort (Asc/Desc)<br> |

<br>• Filter / Search values<br>

<br>• Pin/Freeze column<br>

<br>• Hide/Show columns<br>

<br>• Group by column<br>

<br>• Auto-fit / Resize | Structural table customization and aggregate data manipulation. |
| **Row Action Menu** | End of row (`⋮`), checkbox selection bar, or right-click | • View details / Open record<br>

<br>• Edit row<br>

<br>• Duplicate / Clone<br>

<br>• Change status<br>

<br>• Delete / Archive<br>

<br>• Export selected | Performing CRUD operations and state changes on an entire entity. |
| **Cell Action Menu** | Inside a focused cell or on right-click | • Copy cell value / Copy formatted<br>

<br>• Inline edit / Clear value<br>

<br>• Drill-down / Filter by this value<br>

<br>• Add comment / Note<br>

<br>• View revision history | Targeted inspection, micro-editing, or contextual filtering based on a single data point. |

We could design the component specifications/wireframes, map keyboard shortcuts for power users, or write frontend implementation patterns (React, AG Grid, TanStack Table). Tell me where you'd like to take this next.

Here is the complete, integrated update to our zero-dependency table engine. We'll introduce a **unified contextual Action Menu system** (Column, Row, and Cell action menus with submenus, keyboard escape listeners, and click-outside dismissal) integrated directly into our core state and rendering engine.

### 1. Action Menu Architecture & Types

```typescript
// types/menus.ts
export type MenuTargetType = 'column' | 'row' | 'cell';

export interface BaseMenuCoords {
  x: number;
  y: number;
}

export interface ColumnMenuPayload {
  type: 'column';
  columnId: string;
}

export interface RowMenuPayload {
  type: 'row';
  rowId: string | number;
}

export interface CellMenuPayload {
  type: 'cell';
  rowId: string | number;
  columnId: string;
  value: any;
}

export type ActiveMenuState = (
  | ColumnMenuPayload
  | RowMenuPayload
  | CellMenuPayload
) & BaseMenuCoords;

export interface MenuItem {
  id: string;
  label: string;
  icon?: string;
  danger?: boolean;
  disabled?: boolean;
  divider?: boolean;
  onClick?: () => void;
}
```

### 2. Context Menu Portal Component

A clean, floating menu handler that closes on outside clicks or `Escape` keypresses, with support for divider lines and danger actions.

```tsx
// components/TableContextMenu.tsx
import React, { useEffect, useRef } from 'react';
import { MenuItem } from '../types/menus';

interface TableContextMenuProps {
  x: number;
  y: number;
  items: MenuItem[];
  onClose: () => void;
  title?: string;
}

export const TableContextMenu: React.FC<TableContextMenuProps> = ({
  x,
  y,
  items,
  onClose,
  title,
}) => {
  const menuRef = useRef<HTMLDivElement>(null);

  useEffect(() => {
    const handleOutsideClick = (e: MouseEvent) => {
      if (menuRef.current && !menuRef.current.contains(e.target as Node)) {
        onClose();
      }
    };
    const handleKeyDown = (e: KeyboardEvent) => {
      if (e.key === 'Escape') onClose();
    };

    document.addEventListener('mousedown', handleOutsideClick);
    document.addEventListener('keydown', handleKeyDown);
    return () => {
      document.removeEventListener('mousedown', handleOutsideClick);
      document.removeEventListener('keydown', handleKeyDown);
    };
  }, [onClose]);

  // Viewport edge collision prevention
  const adjustedX = Math.min(x, window.innerWidth - 220);
  const adjustedY = Math.min(y, window.innerHeight - 300);

  return (
    <div
      ref={menuRef}
      style={{
        position: 'fixed',
        top: adjustedY,
        left: adjustedX,
        zIndex: 9999,
        background: '#ffffff',
        border: '1px solid #e2e8f0',
        borderRadius: '6px',
        boxShadow: '0 10px 15px -3px rgba(0, 0, 0, 0.1), 0 4px 6px -4px rgba(0, 0, 0, 0.1)',
        minWidth: '190px',
        padding: '4px 0',
        fontFamily: 'Inter, system-ui, sans-serif',
      }}
    >
      {title && (
        <div
          style={{
            padding: '6px 12px',
            fontSize: '11px',
            fontWeight: 700,
            textTransform: 'uppercase',
            letterSpacing: '0.05em',
            color: '#94a3b8',
            borderBottom: '1px solid #f1f5f9',
            marginBottom: '4px',
          }}
        >
          {title}
        </div>
      )}
      {items.map((item, idx) => (
        <React.Fragment key={item.id || idx}>
          {item.divider ? (
            <div style={{ height: 1, background: '#f1f5f9', margin: '4px 0' }} />
          ) : (
            <button
              disabled={item.disabled}
              onClick={() => {
                if (!item.disabled) {
                  item.onClick?.();
                  onClose();
                }
              }}
              style={{
                display: 'flex',
                alignItems: 'center',
                justifyContent: 'space-between',
                width: '100%',
                padding: '7px 12px',
                fontSize: '13px',
                textAlign: 'left',
                border: 'none',
                background: 'transparent',
                cursor: item.disabled ? 'not-allowed' : 'pointer',
                color: item.disabled ? '#cbd5e1' : item.danger ? '#ef4444' : '#1e293b',
                transition: 'background 0.15s ease',
              }}
              onMouseEnter={(e) => {
                if (!item.disabled) {
                  e.currentTarget.style.background = item.danger ? '#fef2f2' : '#f8fafc';
                }
              }}
              onMouseLeave={(e) => {
                e.currentTarget.style.background = 'transparent';
              }}
            >
              <span>{item.label}</span>
              {item.icon && <span style={{ marginLeft: 8, fontSize: 12 }}>{item.icon}</span>}
            </button>
          )}
        </React.Fragment>
      ))}
    </div>
  );
};

```

### 3. Integrated Table Engine with Action Handlers

Here is the updated table component integrating:

1. **Column Actions**: Pin Left/Right, Clear Pin, Sort ASC/DESC, Hide Column, Group By Column.
2. **Row Actions**: Quick Edit Row, Duplicate Row, Select/Deselect, Delete Row, Export Row CSV.
3. **Cell Actions**: Copy Cell Value, Quick Filter by Value, Clear Cell, Edit Cell.

```tsx
// components/EnterpriseTableWithMenus.tsx
import React, { useState } from 'react';
import { useTableEngine } from '../hooks/useTableEngine';
import { ColumnDef } from '../types/table';
import { TableContextMenu } from './TableContextMenu';
import { ActiveMenuState, MenuItem } from '../types/menus';
import { exportToCSV } from '../utils/export';

interface DemoUser {
  id: number;
  name: string;
  role: string;
  department: string;
  salary: number;
}

const INITIAL_DATA: DemoUser[] = [
  { id: 1, name: 'Alice Walker', role: 'Staff Engineer', department: 'Engineering', salary: 145000 },
  { id: 2, name: 'Bob Miller', role: 'Product Manager', department: 'Product', salary: 130000 },
  { id: 3, name: 'Charlie Kim', role: 'Security Architect', department: 'Security', salary: 160000 },
  { id: 4, name: 'Diana Ross', role: 'UX Designer', department: 'Design', salary: 115000 },
];

const INITIAL_COLUMNS: ColumnDef<DemoUser>[] = [
  { id: 'id', header: 'ID', accessorKey: 'id', width: 70 },
  { id: 'name', header: 'Employee Name', accessorKey: 'name' },
  { id: 'role', header: 'Role Title', accessorKey: 'role' },
  { id: 'department', header: 'Department', accessorKey: 'department' },
  { id: 'salary', header: 'Salary ($)', accessorKey: 'salary' },
];

export const EnterpriseTableWithMenus: React.FC = () => {
  const table = useTableEngine(INITIAL_DATA, INITIAL_COLUMNS);
  const [activeMenu, setActiveMenu] = useState<ActiveMenuState | null>(null);
  const [editingCell, setEditingCell] = useState<{ rowId: number | string; colId: string } | null>(null);

  // -------------------------------------------------------------
  // Context Menu Item Builders
  // -------------------------------------------------------------

  const getColumnMenuItems = (columnId: string): MenuItem[] => {
    const isPinnedLeft = table.columnPinning[columnId] === 'left';
    const isPinnedRight = table.columnPinning[columnId] === 'right';

    return [
      {
        id: 'sort-asc',
        label: 'Sort Ascending',
        icon: '▲',
        onClick: () => table.setSorting([{ columnId, desc: false }]),
      },
      {
        id: 'sort-desc',
        label: 'Sort Descending',
        icon: '▼',
        onClick: () => table.setSorting([{ columnId, desc: true }]),
      },
      { id: 'd1', label: '', divider: true },
      {
        id: 'pin-left',
        label: isPinnedLeft ? 'Unpin (Left)' : 'Pin to Left',
        icon: '⇚',
        onClick: () =>
          table.setColumnPinning((p) => ({ ...p, [columnId]: isPinnedLeft ? false : 'left' })),
      },
      {
        id: 'pin-right',
        label: isPinnedRight ? 'Unpin (Right)' : 'Pin to Right',
        icon: '⇛',
        onClick: () =>
          table.setColumnPinning((p) => ({ ...p, [columnId]: isPinnedRight ? false : 'right' })),
      },
      {
        id: 'group-by',
        label: 'Group By This Column',
        icon: '⊞',
        onClick: () => table.setGrouping([columnId]),
      },
      { id: 'd2', label: '', divider: true },
      {
        id: 'hide-column',
        label: 'Hide Column',
        danger: true,
        onClick: () => table.setColumnVisibility((v) => ({ ...v, [columnId]: false })),
      },
    ];
  };

  const getRowMenuItems = (rowId: string | number): MenuItem[] => {
    const targetRow = table.rawSortedData.find((r) => r.id === rowId);

    return [
      {
        id: 'select-row',
        label: table.selectedRowIds.has(rowId) ? 'Deselect Row' : 'Select Row',
        icon: '✓',
        onClick: () => table.toggleRowSelection(rowId),
      },
      {
        id: 'duplicate-row',
        label: 'Duplicate Row',
        icon: '⎘',
        onClick: () => {
          if (targetRow) {
            table.addRow({ ...targetRow, id: Date.now() });
          }
        },
      },
      {
        id: 'export-row',
        label: 'Export Row (CSV)',
        icon: '⤓',
        onClick: () => targetRow && exportToCSV([targetRow], `row-${rowId}.csv`),
      },
      { id: 'd1', label: '', divider: true },
      {
        id: 'delete-row',
        label: 'Delete Row',
        danger: true,
        icon: '🗑',
        onClick: () => table.deleteRow(rowId),
      },
    ];
  };

  const getCellMenuItems = (
    rowId: string | number,
    columnId: string,
    value: any
  ): MenuItem[] => {
    return [
      {
        id: 'copy-value',
        label: 'Copy Cell Value',
        icon: '📋',
        onClick: () => navigator.clipboard.writeText(String(value ?? '')),
      },
      {
        id: 'edit-cell',
        label: 'Edit Cell',
        icon: '✎',
        onClick: () => setEditingCell({ rowId, colId: columnId }),
      },
      {
        id: 'filter-by-val',
        label: `Filter by "${String(value).slice(0, 10)}"`,
        icon: '🔍',
        onClick: () =>
          table.setFilters((f) => ({
            ...f,
            [columnId]: { operator: 'equals', value },
          })),
      },
      { id: 'd1', label: '', divider: true },
      {
        id: 'clear-cell',
        label: 'Clear Value',
        danger: true,
        onClick: () => table.updateRow(rowId, { [columnId]: '' } as any),
      },
    ];
  };

  // -------------------------------------------------------------
  // Render
  // -------------------------------------------------------------

  const visibleColumns = table.columns.filter(
    (c) => table.columnVisibility[c.id] !== false
  );

  return (
    <div style={{ padding: 16, fontFamily: 'Inter, system-ui, sans-serif' }}>
      <div style={{ marginBottom: 12, display: 'flex', justifyContent: 'space-between' }}>
        <h3 style={{ margin: 0 }}>Enterprise Table (Right-Click Enabled)</h3>
        {Object.values(table.columnVisibility).some((v) => v === false) && (
          <button
            onClick={() => table.setColumnVisibility({})}
            style={{ fontSize: 12, padding: '4px 8px', cursor: 'pointer' }}
          >
            Reset Hidden Columns
          </button>
        )}
      </div>

      <div style={{ overflowX: 'auto', border: '1px solid #e2e8f0', borderRadius: 6 }}>
        <table style={{ width: '100%', borderCollapse: 'collapse', textAlign: 'left' }}>
          {/* Header with Column Menu */}
          <thead>
            <tr style={{ background: '#f8fafc', borderBottom: '2px solid #e2e8f0' }}>
              <th style={{ padding: '8px 12px', width: 40 }}>
                <input
                  type="checkbox"
                  checked={table.selectedRowIds.size === table.data.length && table.data.length > 0}
                  onChange={table.toggleSelectAll}
                />
              </th>
              {visibleColumns.map((col) => {
                const pin = table.columnPinning[col.id];
                return (
                  <th
                    key={col.id}
                    onContextMenu={(e) => {
                      e.preventDefault();
                      setActiveMenu({ type: 'column', columnId: col.id, x: e.clientX, y: e.clientY });
                    }}
                    style={{
                      padding: '8px 12px',
                      fontSize: 13,
                      fontWeight: 600,
                      color: '#475569',
                      borderRight: '1px solid #e2e8f0',
                      position: pin ? 'sticky' : undefined,
                      left: pin === 'left' ? 0 : undefined,
                      right: pin === 'right' ? 0 : undefined,
                      background: pin ? '#f1f5f9' : undefined,
                    }}
                  >
                    <div style={{ display: 'flex', alignItems: 'center', justifyContent: 'space-between' }}>
                      <span>{col.header}</span>
                      <button
                        onClick={(e) => {
                          const rect = e.currentTarget.getBoundingClientRect();
                          setActiveMenu({ type: 'column', columnId: col.id, x: rect.left, y: rect.bottom + 4 });
                        }}
                        style={{ border: 'none', background: 'transparent', cursor: 'pointer', color: '#94a3b8' }}
                      >
                        ⋮
                      </button>
                    </div>
                  </th>
                );
              })}
              <th style={{ padding: '8px 12px', width: 50, textAlign: 'center' }}>•••</th>
            </tr>
          </thead>

          {/* Rows with Cell and Row Actions */}
          <tbody>
            {table.data.map((row) => (
              <tr
                key={row.id}
                onContextMenu={(e) => {
                  e.preventDefault();
                  setActiveMenu({ type: 'row', rowId: row.id, x: e.clientX, y: e.clientY });
                }}
                style={{
                  borderBottom: '1px solid #f1f5f9',
                  background: table.selectedRowIds.has(row.id) ? '#f0f9ff' : '#ffffff',
                }}
              >
                <td style={{ padding: '8px 12px' }}>
                  <input
                    type="checkbox"
                    checked={table.selectedRowIds.has(row.id)}
                    onChange={() => table.toggleRowSelection(row.id)}
                  />
                </td>

                {visibleColumns.map((col) => {
                  const val = col.accessorKey ? row[col.accessorKey] : '';
                  const isEditing = editingCell?.rowId === row.id && editingCell?.colId === col.id;

                  return (
                    <td
                      key={col.id}
                      onContextMenu={(e) => {
                        e.preventDefault();
                        e.stopPropagation(); // Prevents row menu from overriding cell menu
                        setActiveMenu({
                          type: 'cell',
                          rowId: row.id,
                          columnId: col.id,
                          value: val,
                          x: e.clientX,
                          y: e.clientY,
                        });
                      }}
                      onDoubleClick={() => setEditingCell({ rowId: row.id, colId: col.id })}
                      style={{
                        padding: '6px 12px',
                        fontSize: 13,
                        color: '#1e293b',
                        borderRight: '1px solid #f1f5f9',
                      }}
                    >
                      {isEditing && col.accessorKey ? (
                        <input
                          autoFocus
                          defaultValue={String(val)}
                          onBlur={(e) => {
                            table.updateRow(row.id, { [col.accessorKey!]: e.target.value } as any);
                            setEditingCell(null);
                          }}
                          onKeyDown={(e) => {
                            if (e.key === 'Enter') {
                              table.updateRow(row.id, { [col.accessorKey!]: e.currentTarget.value } as any);
                              setEditingCell(null);
                            }
                          }}
                          style={{ width: '100%', padding: '2px 4px', fontSize: 13 }}
                        />
                      ) : (
                        String(val)
                      )}
                    </td>
                  );
                })}

                {/* Explicit Row Action Trigger */}
                <td style={{ padding: '8px 12px', textAlign: 'center' }}>
                  <button
                    onClick={(e) => {
                      const rect = e.currentTarget.getBoundingClientRect();
                      setActiveMenu({ type: 'row', rowId: row.id, x: rect.left - 150, y: rect.bottom + 4 });
                    }}
                    style={{ border: 'none', background: 'transparent', cursor: 'pointer', color: '#94a3b8' }}
                  >
                    ⋮
                  </button>
                </td>
              </tr>
            ))}
          </tbody>
        </table>
      </div>

      {/* Render Active Context Menu */}
      {activeMenu && (
        <TableContextMenu
          x={activeMenu.x}
          y={activeMenu.y}
          title={
            activeMenu.type === 'column'
              ? 'Column Options'
              : activeMenu.type === 'row'
              ? 'Row Actions'
              : 'Cell Actions'
          }
          items={
            activeMenu.type === 'column'
              ? getColumnMenuItems(activeMenu.columnId)
              : activeMenu.type === 'row'
              ? getRowMenuItems(activeMenu.rowId)
              : getCellMenuItems(activeMenu.rowId, activeMenu.columnId, activeMenu.value)
          }
          onClose={() => setActiveMenu(null)}
        />
      )}
    </div>
  );
};
```
