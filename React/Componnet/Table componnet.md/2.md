## 1. Windowed Row Virtualizer From Scratch

This hook computes scroll offsets, dynamically slices visible rows based on container height, and adds top/bottom buffer space via CSS translateY transforms.

```typescript
// hooks/useRowVirtualizer.ts
import { useState, useEffect, useRef, useCallback } from 'react';

interface VirtualizerOptions {
  count: number;
  itemHeight: number;
  overscan?: number;
}

export function useRowVirtualizer({ count, itemHeight, overscan = 5 }: VirtualizerOptions) {
  const containerRef = useRef<HTMLDivElement | null>(null);
  const [scrollTop, setScrollTop] = useState(0);
  const [containerHeight, setContainerHeight] = useState(600);

  useEffect(() => {
    const element = containerRef.current;
    if (!element) return;

    const handleScroll = () => {
      setScrollTop(element.scrollTop);
    };

    const resizeObserver = new ResizeObserver((entries) => {
      for (const entry of entries) {
        setContainerHeight(entry.contentRect.height);
      }
    });

    element.addEventListener('scroll', handleScroll, { passive: true });
    resizeObserver.observe(element);

    return () => {
      element.removeEventListener('scroll', handleScroll);
      resizeObserver.disconnect();
    };
  }, []);

  const totalHeight = count * itemHeight;
  const startIndex = Math.max(0, Math.floor(scrollTop / itemHeight) - overscan);
  const endIndex = Math.min(
    count - 1,
    Math.floor((scrollTop + containerHeight) / itemHeight) + overscan
  );

  const virtualItems = [];
  for (let i = startIndex; i <= endIndex; i++) {
    virtualItems.push({
      index: i,
      offsetTop: i * itemHeight,
      height: itemHeight,
    });
  }

  const scrollToIndex = useCallback(
    (index: number) => {
      if (containerRef.current) {
        containerRef.current.scrollTop = index * itemHeight;
      }
    },
    [itemHeight]
  );

  return {
    containerRef,
    virtualItems,
    totalHeight,
    startIndex,
    endIndex,
    scrollToIndex,
  };
}
```

```typescript
// types/treeAndGrouping.ts
export interface HierarchicalColumnDef<T> {
  id: string;
  header: string;
  accessorKey?: keyof T;
  columns?: HierarchicalColumnDef<T>[];
  width?: number;
  pin?: 'left' | 'right' | false;
}

export interface TreeRowNode<T> {
  id: string | number;
  data: T;
  children?: TreeRowNode<T>[];
  expanded?: boolean;
}

export interface FlattenedTreeRow<T> {
  id: string | number;
  data: T;
  depth: number;
  hasChildren: boolean;
  expanded: boolean;
  parentId?: string | number | null;
}

export interface ColumnHeaderCell<T> {
  column: HierarchicalColumnDef<T>;
  rowSpan: number;
  colSpan: number;
  depth: number;
  isLeaf: boolean;
}
```

```typescript
// utils/treeAndColumns.ts
import { HierarchicalColumnDef, ColumnHeaderCell, TreeRowNode, FlattenedTreeRow } from '../types/treeAndGrouping';

export function getHeaderDepth<T>(columns: HierarchicalColumnDef<T>[]): number {
  let maxDepth = 0;
  function traverse(cols: HierarchicalColumnDef<T>[], currentDepth: number) {
    for (const col of cols) {
      if (col.columns && col.columns.length > 0) {
        traverse(col.columns, currentDepth + 1);
      } else {
        if (currentDepth > maxDepth) maxDepth = currentDepth;
      }
    }
  }
  traverse(columns, 1);
  return maxDepth;
}

export function getLeafColumns<T>(columns: HierarchicalColumnDef<T>[]): HierarchicalColumnDef<T>[] {
  const leaves: HierarchicalColumnDef<T>[] = [];
  function traverse(cols: HierarchicalColumnDef<T>[]) {
    for (const col of cols) {
      if (col.columns && col.columns.length > 0) {
        traverse(col.columns);
      } else {
        leaves.push(col);
      }
    }
  }
  traverse(columns);
  return leaves;
}

export function buildHeaderMatrix<T>(columns: HierarchicalColumnDef<T>[]): ColumnHeaderCell<T>[][] {
  const maxDepth = getHeaderDepth(columns);
  const matrix: ColumnHeaderCell<T>[][] = Array.from({ length: maxDepth }, () => []);

  function calculateColSpan(col: HierarchicalColumnDef<T>): number {
    if (!col.columns || col.columns.length === 0) return 1;
    return col.columns.reduce((sum, child) => sum + calculateColSpan(child), 0);
  }

  function traverse(cols: HierarchicalColumnDef<T>[], currentDepth: number) {
    for (const col of cols) {
      const isLeaf = !col.columns || col.columns.length === 0;
      const colSpan = calculateColSpan(col);
      const rowSpan = isLeaf ? maxDepth - currentDepth + 1 : 1;

      matrix[currentDepth - 1].push({
        column: col,
        rowSpan,
        colSpan,
        depth: currentDepth,
        isLeaf,
      });

      if (col.columns && col.columns.length > 0) {
        traverse(col.columns, currentDepth + 1);
      }
    }
  }

  traverse(columns, 1);
  return matrix;
}

export function flattenTreeRows<T>(
  nodes: TreeRowNode<T>[],
  expandedMap: Set<string | number>,
  depth = 0,
  parentId: string | number | null = null
): FlattenedTreeRow<T>[] {
  const result: FlattenedTreeRow<T>[] = [];

  for (const node of nodes) {
    const isExpanded = expandedMap.has(node.id);
    const hasChildren = Boolean(node.children && node.children.length > 0);

    result.push({
      id: node.id,
      data: node.data,
      depth,
      hasChildren,
      expanded: isExpanded,
      parentId,
    });

    if (hasChildren && isExpanded && node.children) {
      result.push(...flattenTreeRows(node.children, expandedMap, depth + 1, node.id));
    }
  }

  return result;
}
```

## 3. Integrated Virtualized Tree Table Component

This component handles multi-level nested headers, sub-row collapsing/expanding, dynamic cell rendering, and high-performance row virtualization.

```tsx
// components/VirtualizedTreeTable.tsx
import React, { useState, useMemo } from 'react';
import { useRowVirtualizer } from '../hooks/useRowVirtualizer';
import { HierarchicalColumnDef, TreeRowNode } from '../types/treeAndGrouping';
import { buildHeaderMatrix, getLeafColumns, flattenTreeRows } from '../utils/treeAndColumns';

interface VirtualizedTreeTableProps<T> {
  data: TreeRowNode<T>[];
  columns: HierarchicalColumnDef<T>[];
  rowHeight?: number;
  viewportHeight?: number;
  onRowClick?: (row: T) => void;
}

export function VirtualizedTreeTable<T extends Record<string, any>>({
  data,
  columns,
  rowHeight = 40,
  viewportHeight = 500,
  onRowClick,
}: VirtualizedTreeTableProps<T>) {
  const [expandedIds, setExpandedIds] = useState<Set<string | number>>(new Set());

  const toggleExpand = (id: string | number) => {
    setExpandedIds((prev) => {
      const next = new Set(prev);
      next.has(id) ? next.delete(id) : next.add(id);
      return next;
    });
  };

  const headerMatrix = useMemo(() => buildHeaderMatrix(columns), [columns]);
  const leafColumns = useMemo(() => getLeafColumns(columns), [columns]);
  const flatRows = useMemo(() => flattenTreeRows(data, expandedIds), [data, expandedIds]);

  const { containerRef, virtualItems, totalHeight } = useRowVirtualizer({
    count: flatRows.length,
    itemHeight: rowHeight,
    overscan: 6,
  });

  return (
    <div
      style={{
        border: '1px solid #d1d5db',
        borderRadius: 8,
        overflow: 'hidden',
        fontFamily: 'Inter, system-ui, sans-serif',
        display: 'flex',
        flexDirection: 'column',
      }}
    >
      {/* Scrollable Container */}
      <div
        ref={containerRef}
        style={{
          height: viewportHeight,
          overflowY: 'auto',
          overflowX: 'auto',
          position: 'relative',
        }}
      >
        <table
          style={{
            width: '100%',
            borderCollapse: 'collapse',
            tableLayout: 'fixed',
          }}
        >
          {/* Header */}
          <thead
            style={{
              position: 'sticky',
              top: 0,
              zIndex: 10,
              backgroundColor: '#f8fafc',
              boxShadow: '0 1px 2px rgba(0,0,0,0.05)',
            }}
          >
            {headerMatrix.map((row, rIdx) => (
              <tr key={rIdx}>
                {row.map((cell, cIdx) => (
                  <th
                    key={`${rIdx}-${cIdx}`}
                    rowSpan={cell.rowSpan}
                    colSpan={cell.colSpan}
                    style={{
                      border: '1px solid #e2e8f0',
                      padding: '8px 12px',
                      fontSize: 13,
                      fontWeight: 600,
                      color: '#334155',
                      textAlign: 'left',
                      backgroundColor: '#f8fafc',
                      width: cell.column.width,
                    }}
                  >
                    {cell.column.header}
                  </th>
                ))}
              </tr>
            ))}
          </thead>

          {/* Virtual Body Spacer */}
          <tbody
            style={{
              height: `${totalHeight}px`,
              position: 'relative',
              width: '100%',
            }}
          >
            {virtualItems.map(({ index, offsetTop, height }) => {
              const row = flatRows[index];
              if (!row) return null;

              return (
                <tr
                  key={row.id}
                  onClick={() => onRowClick?.(row.data)}
                  style={{
                    position: 'absolute',
                    top: 0,
                    left: 0,
                    width: '100%',
                    height: `${height}px`,
                    transform: `translateY(${offsetTop}px)`,
                    display: 'table',
                    tableLayout: 'fixed',
                    backgroundColor: index % 2 === 0 ? '#ffffff' : '#fcfcfd',
                    borderBottom: '1px solid #f1f5f9',
                    cursor: onRowClick ? 'pointer' : 'default',
                  }}
                >
                  {leafColumns.map((col, cIdx) => {
                    const isFirstCell = cIdx === 0;
                    const cellValue = col.accessorKey ? row.data[col.accessorKey] : '';

                    return (
                      <td
                        key={col.id}
                        style={{
                          padding: '0 12px',
                          height: `${height}px`,
                          borderRight: '1px solid #f1f5f9',
                          fontSize: 13,
                          color: '#0f172a',
                          whiteSpace: 'nowrap',
                          overflow: 'hidden',
                          textOverflow: 'ellipsis',
                          verticalAlign: 'middle',
                          width: col.width,
                        }}
                      >
                        {isFirstCell ? (
                          <div
                            style={{
                              display: 'flex',
                              alignItems: 'center',
                              paddingLeft: `${row.depth * 20}px`,
                            }}
                          >
                            {row.hasChildren ? (
                              <button
                                onClick={(e) => {
                                  e.stopPropagation();
                                  toggleExpand(row.id);
                                }}
                                style={{
                                  background: 'none',
                                  border: 'none',
                                  cursor: 'pointer',
                                  padding: '0 6px 0 0',
                                  fontSize: 12,
                                  color: '#64748b',
                                  lineHeight: 1,
                                }}
                              >
                                {row.expanded ? '▼' : '►'}
                              </button>
                            ) : (
                              <span style={{ display: 'inline-block', width: 14 }} />
                            )}
                            <span>{cellValue}</span>
                          </div>
                        ) : (
                          cellValue
                        )}
                      </td>
                    );
                  })}
                </tr>
              );
            })}
          </tbody>
        </table>
      </div>
    </div>
  );
}
```
