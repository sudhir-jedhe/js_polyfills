### 1. FLIP Animation & Geometry Utilities

This hook captures row bounding rectangles before state updates (`First`), tracks position shifts after render (`Last`), applies an inverted CSS transform (`Invert`), and animates them cleanly to zero (`Play`).

```typescript
// utils/flipAndGeometry.ts
import { useLayoutEffect, useRef } from 'react';

export interface BoxRect {
  startX: number;
  startY: number;
  currentX: number;
  currentY: number;
}

export function getNormalizedRect(box: BoxRect) {
  const left = Math.min(box.startX, box.currentX);
  const top = Math.min(box.startY, box.currentY);
  const width = Math.abs(box.currentX - box.startX);
  const height = Math.abs(box.currentY - box.startY);
  return { left, top, right: left + width, bottom: top + height, width, height };
}

export function isIntersecting(r1: { left: number; top: number; right: number; bottom: number }, r2: DOMRect) {
  return !(
    r2.left > r1.right ||
    r2.right < r1.left ||
    r2.top > r1.bottom ||
    r2.bottom < r1.top
  );
}

export function useFlipAnimation<T extends { id: string | number }>(items: T[], duration = 300) {
  const positions = useRef<Map<string | number, DOMRect>>(new Map());
  const elements = useRef<Map<string | number, HTMLElement>>(new Map());

  // Capture 'First' positions right before paint
  useLayoutEffect(() => {
    elements.current.forEach((el, id) => {
      const firstRect = positions.current.get(id);
      const lastRect = el.getBoundingClientRect();

      if (firstRect) {
        const deltaY = firstRect.top - lastRect.top;
        if (deltaY !== 0) {
          // Invert: Snap to old position with zero transition
          el.style.transform = `translate3d(0, ${deltaY}px, 0)`;
          el.style.transition = 'transform 0s';

          // Force reflow
          el.getBoundingClientRect();

          // Play: Animate cleanly to final position
          requestAnimationFrame(() => {
            el.style.transform = 'translate3d(0, 0, 0)';
            el.style.transition = `transform ${duration}ms cubic-bezier(0.2, 0, 0, 1)`;
          });
        }
      }
      positions.current.set(id, lastRect);
    });
  }, [items, duration]);

  const registerElement = (id: string | number, el: HTMLElement | null) => {
    if (el) {
      elements.current.set(id, el);
      if (!positions.current.has(id)) {
        positions.current.set(id, el.getBoundingClientRect());
      }
    } else {
      elements.current.delete(id);
      positions.current.delete(id);
    }
  };

  return { registerElement };
}
```

### 1. Range Selection & Clipboard Matrix Engine

This utility file handles:

* Normalizing 2D coordinate selections `(startRow, startCol) -> (endRow, endCol)`.
* Determining which cells lie inside the selection and detecting outer border edges (`top`, `bottom`, `left`, `right`).
* Serialization to Tab-Separated Values (`\t` columns, `\r\n` lines) for OS clipboard compatibility (Excel, Google Sheets, Apple Numbers).
* 2D matrix clipboard deserialization and in-place grid mutation.

```typescript
// utils/spreadsheetEngine.ts
export interface CellCoord {
  rowIndex: number;
  colIndex: number;
}

export interface CellRange {
  start: CellCoord;
  end: CellCoord;
}

export interface NormalizedRange {
  minRow: number;
  maxRow: number;
  minCol: number;
  maxCol: number;
}

export function normalizeRange(range: CellRange): NormalizedRange {
  return {
    minRow: Math.min(range.start.rowIndex, range.end.rowIndex),
    maxRow: Math.max(range.start.rowIndex, range.end.rowIndex),
    minCol: Math.min(range.start.colIndex, range.end.colIndex),
    maxCol: Math.max(range.start.colIndex, range.end.colIndex),
  };
}

export function isCellInRange(r: number, c: number, range: NormalizedRange | null): boolean {
  if (!range) return false;
  return r >= range.minRow && r <= range.maxRow && c >= range.minCol && c <= range.maxCol;
}

export interface CellBorderStyles {
  isTop: boolean;
  isBottom: boolean;
  isLeft: boolean;
  isRight: boolean;
}

export function getCellBorderStyles(r: number, c: number, range: NormalizedRange | null): CellBorderStyles {
  if (!range || !isCellInRange(r, c, range)) {
    return { isTop: false, isBottom: false, isLeft: false, isRight: false };
  }
  return {
    isTop: r === range.minRow,
    isBottom: r === range.maxRow,
    isLeft: c === range.minCol,
    isRight: c === range.maxCol,
  };
}

/**
 * Serializes a 2D cell slice into tab-delimited text (`\t` and `\n`)
 */
export function serializeRangeToTSV<T>(
  data: T[][],
  range: NormalizedRange
): string {
  const rows: string[] = [];
  for (let r = range.minRow; r <= range.maxRow; r++) {
    const rowCells: string[] = [];
    for (let c = range.minCol; c <= range.maxCol; c++) {
      const val = data[r]?.[c] ?? '';
      // Escape inner tabs or line breaks if necessary
      rowCells.push(String(val).replace(/\t/g, ' '));
    }
    rows.push(rowCells.join('\t'));
  }
  return rows.join('\r\n');
}

/**
 * Parses tab-delimited clipboard text into a 2D matrix
 */
export function parseTSV(clipboardText: string): string[][] {
  const cleanText = clipboardText.replace(/\r\n/g, '\n').replace(/\r/g, '\n');
  const lines = cleanText.split('\n');
  // Strip trailing empty lines generated by clipboard buffers
  if (lines.length > 0 && lines[lines.length - 1] === '') {
    lines.pop();
  }
  return lines.map((line) => line.split('\t'));
}
```

### 1. Reactive Calculation Graph & Formula Engine

This module converts Excel notation (`A1`, `B2`, `SUM(A1:B3)`, `AVG`, `MIN`, `MAX`) into computed values, tracks dependencies between cells, detects circular references, and propagates updates using a Directed Acyclic Graph (DAG).

```typescript
// utils/formulaGraph.ts
export type CellKey = string; // e.g., "A1", "C4"

export function toCellKey(row: number, col: number): CellKey {
  return `${String.fromCharCode(65 + col)}${row + 1}`;
}

export function fromCellKey(key: CellKey): { row: number; col: number } {
  const match = key.match(/^([A-Z]+)(\d+)$/);
  if (!match) throw new Error(`Invalid cell key: ${key}`);
  const colStr = match[1];
  const rowStr = match[2];

  let col = 0;
  for (let i = 0; i < colStr.length; i++) {
    col = col * 26 + (colStr.charCodeAt(i) - 64);
  }
  return { row: parseInt(rowStr, 10) - 1, col: col - 1 };
}

export class CalculationGraph {
  private rawFormulas: Map<CellKey, string> = new Map();
  private computedValues: Map<CellKey, any> = new Map();
  private dependencies: Map<CellKey, Set<CellKey>> = new Map(); // Key depends on Set<Keys>
  private dependents: Map<CellKey, Set<CellKey>> = new Map(); // Key is referenced by Set<Keys>

  public setCellValue(key: CellKey, rawValue: string) {
    this.rawFormulas.set(key, rawValue);
    this.updateGraph(key);
  }

  public getRawValue(key: CellKey): string {
    return this.rawFormulas.get(key) ?? '';
  }

  public getComputedValue(key: CellKey): any {
    return this.computedValues.get(key) ?? '';
  }

  private extractReferences(formula: string): CellKey[] {
    const singleCellRegex = /[A-Z]+\d+/g;
    const rangeRegex = /([A-Z]+\d+):([A-Z]+\d+)/g;
    const refs: Set<CellKey> = new Set();

    // Parse ranges like A1:B3
    let match;
    while ((match = rangeRegex.exec(formula)) !== null) {
      const start = fromCellKey(match[1]);
      const end = fromCellKey(match[2]);
      const minR = Math.min(start.row, end.row);
      const maxR = Math.max(start.row, end.row);
      const minC = Math.min(start.col, end.col);
      const maxC = Math.max(start.col, end.col);

      for (let r = minR; r <= maxR; r++) {
        for (let c = minC; c <= maxC; c++) {
          refs.add(toCellKey(r, c));
        }
      }
    }

    // Parse single keys like A1
    const stripped = formula.replace(rangeRegex, '');
    while ((match = singleCellRegex.exec(stripped)) !== null) {
      refs.add(match[0]);
    }

    return Array.from(refs);
  }

  private updateGraph(key: CellKey) {
    const raw = this.rawFormulas.get(key) ?? '';

    // Clear old dependencies
    const oldDeps = this.dependencies.get(key) || new Set();
    oldDeps.forEach((dep) => this.dependents.get(dep)?.delete(key));

    if (raw.startsWith('=')) {
      const refs = this.extractReferences(raw.toUpperCase());
      this.dependencies.set(key, new Set(refs));
      refs.forEach((ref) => {
        if (!this.dependents.has(ref)) this.dependents.set(ref, new Set());
        this.dependents.get(ref)!.add(key);
      });
    } else {
      this.dependencies.delete(key);
    }

    this.recalculate(key);
  }

  private hasCycle(node: CellKey, visited = new Set<CellKey>(), stack = new Set<CellKey>()): boolean {
    visited.add(node);
    stack.add(node);

    const neighbors = this.dependencies.get(node) || new Set();
    for (const neighbor of neighbors) {
      if (!visited.has(neighbor) && this.hasCycle(neighbor, visited, stack)) return true;
      if (stack.has(neighbor)) return true;
    }

    stack.delete(node);
    return false;
  }

  public recalculate(changedKey: CellKey) {
    if (this.hasCycle(changedKey)) {
      this.computedValues.set(changedKey, '#CIRCULAR!');
      return;
    }

    this.evaluateCell(changedKey);

    // Recalculate all dependents in topological order
    const queue = Array.from(this.dependents.get(changedKey) || []);
    const visited = new Set<CellKey>();

    while (queue.length > 0) {
      const current = queue.shift()!;
      if (!visited.has(current)) {
        visited.add(current);
        this.evaluateCell(current);
        const next = this.dependents.get(current);
        if (next) queue.push(...Array.from(next));
      }
    }
  }

  private evaluateCell(key: CellKey) {
    const raw = this.rawFormulas.get(key) ?? '';
    if (!raw.startsWith('=')) {
      this.computedValues.set(key, raw);
      return;
    }

    try {
      let expression = raw.slice(1).toUpperCase();

      // Built-in Aggregate Functions
      expression = expression.replace(/SUM\(([A-Z]+\d+):([A-Z]+\d+)\)/g, (_, start, end) => {
        const values = this.getRangeNumericValues(start, end);
        return String(values.reduce((acc, v) => acc + v, 0));
      });

      expression = expression.replace(/AVG\(([A-Z]+\d+):([A-Z]+\d+)\)/g, (_, start, end) => {
        const values = this.getRangeNumericValues(start, end);
        return values.length ? String(values.reduce((acc, v) => acc + v, 0) / values.length) : '0';
      });

      expression = expression.replace(/MIN\(([A-Z]+\d+):([A-Z]+\d+)\)/g, (_, start, end) => {
        const values = this.getRangeNumericValues(start, end);
        return values.length ? String(Math.min(...values)) : '0';
      });

      expression = expression.replace(/MAX\(([A-Z]+\d+):([A-Z]+\d+)\)/g, (_, start, end) => {
        const values = this.getRangeNumericValues(start, end);
        return values.length ? String(Math.max(...values)) : '0';
      });

      // Replace single cell tokens with numeric or literal value
      expression = expression.replace(/[A-Z]+\d+/g, (cellRef) => {
        const val = this.computedValues.get(cellRef) ?? 0;
        const num = Number(val);
        return isNaN(num) ? `"${val}"` : String(num);
      });

      // Safe arithmetic evaluator
      const result = Function(`"use strict"; return (${expression})`)();
      this.computedValues.set(key, typeof result === 'number' ? Math.round(result * 100) / 100 : result);
    } catch {
      this.computedValues.set(key, '#ERROR!');
    }
  }

  private getRangeNumericValues(startKey: CellKey, endKey: CellKey): number[] {
    const start = fromCellKey(startKey);
    const end = fromCellKey(endKey);
    const values: number[] = [];

    const minR = Math.min(start.row, end.row);
    const maxR = Math.max(start.row, end.row);
    const minC = Math.min(start.col, end.col);
    const maxC = Math.max(start.col, end.col);

    for (let r = minR; r <= maxR; r++) {
      for (let c = minC; c <= maxC; c++) {
        const val = Number(this.computedValues.get(toCellKey(r, c)));
        if (!isNaN(val)) values.push(val);
      }
    }
    return values;
  }
}
```

### 2. Auto-Fill Series Predictor

This engine calculates continuous series intervals (numbers, step series, dates, sequential names like "Q1, Q2" or "Item 1, Item 2") when dragging the auto-fill handle.

```typescript
// utils/autoFill.ts
export function predictNextSeriesValue(sequence: string[], targetOffset: number): string {
  if (sequence.length === 0) return '';
  if (sequence.length === 1) {
    const val = sequence[0];
    const numMatch = val.match(/^(.*?)(\d+)$/);
    if (numMatch) {
      const prefix = numMatch[1];
      const baseNum = parseInt(numMatch[2], 10);
      return `${prefix}${baseNum + targetOffset}`;
    }
    const num = Number(val);
    return isNaN(num) ? val : String(num + targetOffset);
  }

  // Linear progression detection: 2+ items
  const numbers = sequence.map((v) => Number(v));
  const isPureNumbers = numbers.every((n) => !isNaN(n));

  if (isPureNumbers) {
    const step = (numbers[numbers.length - 1] - numbers[0]) / (numbers.length - 1);
    return String(numbers[numbers.length - 1] + step * targetOffset);
  }

  // Fallback pattern replication
  const patternIndex = (targetOffset - 1) % sequence.length;
  return sequence[patternIndex];
}
```

### 1. Transactional Undo/Redo Command Manager

This module implements the Command Pattern with inverse operations, allowing atomic cell mutations, range replacements, and auto-fills to be grouped into transactional batches that execute cleanly with `Ctrl+Z` and `Ctrl+Y` / `Ctrl+Shift+Z`.

```typescript
// utils/commandManager.ts
export interface CellDiff {
  key: string;
  previousValue: string;
  nextValue: string;
}

export interface Command {
  execute: () => void;
  undo: () => void;
  description?: string;
}

export class BatchCellEditCommand implements Command {
  constructor(
    private diffs: CellDiff[],
    private applyDiffs: (diffs: { key: string; value: string }[]) => void,
    public description = 'Batch Cell Edit'
  ) {}

  execute(): void {
    this.applyDiffs(this.diffs.map((d) => ({ key: d.key, value: d.nextValue })));
  }

  undo(): void {
    this.applyDiffs(this.diffs.map((d) => ({ key: d.key, value: d.previousValue })));
  }
}

export class UndoRedoHistory {
  private undoStack: Command[] = [];
  private redoStack: Command[] = [];
  private maxHistory: number;

  constructor(maxHistory = 50) {
    this.maxHistory = maxHistory;
  }

  public execute(command: Command): void {
    command.execute();
    this.undoStack.push(command);
    this.redoStack = []; // Clear redo chain on new user action
    if (this.undoStack.length > this.maxHistory) {
      this.undoStack.shift();
    }
  }

  public undo(): boolean {
    const command = this.undoStack.pop();
    if (!command) return false;
    command.undo();
    this.redoStack.push(command);
    return true;
  }

  public redo(): boolean {
    const command = this.redoStack.pop();
    if (!command) return false;
    command.execute();
    this.undoStack.push(command);
    return true;
  }

  public canUndo(): boolean {
    return this.undoStack.length > 0;
  }

  public canRedo(): boolean {
    return this.redoStack.length > 0;
  }
}
```

### 2. Extensible Conditional Formatting Engine

This engine handles value-based highlights, continuous 3-color gradients (min/mid/max interpolation), and data bars based on range calculations.

```typescript
// utils/conditionalFormatting.ts
export type RuleType = 'threshold' | 'colorScale' | 'dataBar';

export interface ThresholdRule {
  type: 'threshold';
  operator: 'gt' | 'lt' | 'eq' | 'between';
  value: number;
  endValue?: number;
  style: React.CSSProperties;
}

export interface ColorScaleRule {
  type: 'colorScale';
  minColor: [number, number, number]; // RGB
  midColor?: [number, number, number];
  maxColor: [number, number, number];
}

export interface DataBarRule {
  type: 'dataBar';
  barColor: string;
}

export type FormattingRule = (ThresholdRule | ColorScaleRule | DataBarRule) & {
  id: string;
  targetCol: number;
};

function interpolateRGB(
  c1: [number, number, number],
  c2: [number, number, number],
  factor: number
): string {
  const r = Math.round(c1[0] + factor * (c2[0] - c1[0]));
  const g = Math.round(c1[1] + factor * (c2[1] - c1[1]));
  const b = Math.round(c1[2] + factor * (c2[2] - c1[2]));
  return `rgb(${r}, ${g}, ${b})`;
}

export function evaluateConditionalStyle(
  val: any,
  rule: FormattingRule,
  columnMin: number,
  columnMax: number
): { style: React.CSSProperties; dataBarWidth?: number } {
  const num = Number(val);
  const isNum = !isNaN(num);

  if (rule.type === 'threshold') {
    if (!isNum) return { style: {} };
    let match = false;
    if (rule.operator === 'gt') match = num > rule.value;
    if (rule.operator === 'lt') match = num < rule.value;
    if (rule.operator === 'eq') match = num === rule.value;
    if (rule.operator === 'between' && rule.endValue !== undefined) {
      match = num >= rule.value && num <= rule.endValue;
    }
    return { style: match ? rule.style : {} };
  }

  if (rule.type === 'colorScale') {
    if (!isNum || columnMax === columnMin) return { style: {} };
    const ratio = Math.max(0, Math.min(1, (num - columnMin) / (columnMax - columnMin)));

    let bg = '';
    if (!rule.midColor) {
      bg = interpolateRGB(rule.minColor, rule.maxColor, ratio);
    } else {
      if (ratio < 0.5) {
        bg = interpolateRGB(rule.minColor, rule.midColor, ratio * 2);
      } else {
        bg = interpolateRGB(rule.midColor, rule.maxColor, (ratio - 0.5) * 2);
      }
    }
    return { style: { backgroundColor: bg } };
  }

  if (rule.type === 'dataBar') {
    if (!isNum || columnMax === columnMin) return { style: {} };
    const percent = Math.max(0, Math.min(100, ((num - columnMin) / (columnMax - columnMin)) * 100));
    return {
      style: { position: 'relative' },
      dataBarWidth: percent,
    };
  }

  return { style: {} };
}
```

### 3. Bidirectional (2D) Virtualizer Hook

This custom hook tracks both horizontal (`scrollLeft`) and vertical (`scrollTop`) coordinates to compute dual-axis viewport windows, rendering only the intersecting cells while managing sticky header offsets.

```typescript
// hooks/useBidirectionalVirtualizer.ts
import { useState, useEffect, useRef } from 'react';

interface Virtualizer2DOptions {
  rowCount: number;
  colCount: number;
  rowHeight: number;
  colWidth: number;
  overscan?: number;
}

export function useBidirectionalVirtualizer({
  rowCount,
  colCount,
  rowHeight,
  colWidth,
  overscan = 3,
}: Virtualizer2DOptions) {
  const containerRef = useRef<HTMLDivElement | null>(null);
  const [scroll, setScroll] = useState({ top: 0, left: 0 });
  const [viewport, setViewport] = useState({ width: 800, height: 500 });

  useEffect(() => {
    const el = containerRef.current;
    if (!el) return;

    const onScroll = () => {
      setScroll({ top: el.scrollTop, left: el.scrollLeft });
    };

    const resizeObserver = new ResizeObserver((entries) => {
      for (const entry of entries) {
        setViewport({
          width: entry.contentRect.width,
          height: entry.contentRect.height,
        });
      }
    });

    el.addEventListener('scroll', onScroll, { passive: true });
    resizeObserver.observe(el);

    return () => {
      el.removeEventListener('scroll', onScroll);
      resizeObserver.disconnect();
    };
  }, []);

  const totalHeight = rowCount * rowHeight;
  const totalWidth = colCount * colWidth;

  const startRow = Math.max(0, Math.floor(scroll.top / rowHeight) - overscan);
  const endRow = Math.min(rowCount - 1, Math.floor((scroll.top + viewport.height) / rowHeight) + overscan);

  const startCol = Math.max(0, Math.floor(scroll.left / colWidth) - overscan);
  const endCol = Math.min(colCount - 1, Math.floor((scroll.left + viewport.width) / colWidth) + overscan);

  const visibleRows = [];
  for (let r = startRow; r <= endRow; r++) {
    visibleRows.push({ index: r, top: r * rowHeight });
  }

  const visibleCols = [];
  for (let c = startCol; c <= endCol; c++) {
    visibleCols.push({ index: c, left: c * colWidth });
  }

  return {
    containerRef,
    totalHeight,
    totalWidth,
    visibleRows,
    visibleCols,
    startRow,
    endRow,
    startCol,
    endCol,
  };
}
```
