Here is the implementation for **Column Reordering (Drag & Drop)** and **Row Reordering (Drag & Drop)** built completely from scratch using the native HTML5 Drag and Drop API with zero external libraries.

---

### 1. Drag & Drop State and Reordering Utilities

```typescript
// utils/dnd.ts
export type DragType = 'column' | 'row';

export interface DragState {
  type: DragType | null;
  draggedIndex: number | null;
  targetIndex: number | null;
  dropPosition: 'top' | 'bottom' | 'left' | 'right' | null;
}

/**
 * Immutably moves an item from one index to another in an array
 */
export function reorderArray<T>(list: T[], startIndex: number, endIndex: number): T[] {
  const result = Array.from(list);
  const [removed] = result.splice(startIndex, 1);
  result.splice(endIndex, 0, removed);
  return result;
}

```

### 2. Full Table Component with Column & Row Dragging

This component includes:

1. **Column Dragging:** Drag any column header left or right to reorder. It displays an active vertical insertion indicator border.
2. **Row Dragging:** Drag a dedicated grab handle (`⠿`) on each row. It displays a horizontal drop indicator.
3. **Optimized Event Prevention:** Native event lifecycles (`onDragStart`, `onDragOver`, `onDragLeave`, `onDrop`, `onDragEnd`) with state isolation so row and column drags do not conflict.

```tsx
// components/DraggableEnterpriseTable.tsx
import React, { useState, useMemo } from 'react';
import { useTableEngine } from '../hooks/useTableEngine';
import { ColumnDef } from '../types/table';
import { reorderArray } from '../utils/dnd';

interface RecordItem {
  id: number;
  code: string;
  name: string;
  category: string;
  priority: 'Low' | 'Medium' | 'High';
  stock: number;
}

const SAMPLE_RECORDS: RecordItem[] = [
  { id: 101, code: 'SKU-001', name: 'Ultra Wireless Mouse', category: 'Peripherals', priority: 'High', stock: 45 },
  { id: 102, code: 'SKU-002', name: 'Mechanical RGB Keyboard', category: 'Peripherals', priority: 'Medium', stock: 120 },
  { id: 103, code: 'SKU-003', name: '4K IPS Curved Monitor', category: 'Displays', priority: 'Critical' as any, stock: 18 },
  { id: 104, code: 'SKU-004', name: 'USB-C Thunderbolt Dock', category: 'Accessories', priority: 'Low', stock: 85 },
  { id: 105, code: 'SKU-005', name: 'Ergonomic Desk Mat', category: 'Accessories', priority: 'Medium', stock: 210 },
];

const INITIAL_COLS: ColumnDef<RecordItem>[] = [
  { id: 'code', header: 'Product Code', accessorKey: 'code' },
  { id: 'name', header: 'Item Name', accessorKey: 'name' },
  { id: 'category', header: 'Category', accessorKey: 'category' },
  { id: 'priority', header: 'Priority Level', accessorKey: 'priority' },
  { id: 'stock', header: 'Stock Units', accessorKey: 'stock' },
];

export const DraggableEnterpriseTable: React.FC = () => {
  const table = useTableEngine(SAMPLE_RECORDS, INITIAL_COLS);

  // Drag States
  const [colDragIndex, setColDragIndex] = useState<number | null>(null);
  const [colDropTarget, setColDropTarget] = useState<number | null>(null);

  const [rowDragIndex, setRowDragIndex] = useState<number | null>(null);
  const [rowDropTarget, setRowDropTarget] = useState<number | null>(null);

  // Column order resolution
  const orderedColumns = useMemo(() => {
    return table.columnOrder
      .map((colId) => table.columns.find((c) => c.id === colId))
      .filter((c): c is ColumnDef<RecordItem> => Boolean(c) && table.columnVisibility[c.id] !== false);
  }, [table.columnOrder, table.columns, table.columnVisibility]);

  // -------------------------------------------------------------
  // Column Drag Handlers
  // -------------------------------------------------------------
  const handleColDragStart = (e: React.DragEvent, index: number) => {
    setColDragIndex(index);
    e.dataTransfer.effectAllowed = 'move';
    e.dataTransfer.setData('text/plain', `col-${index}`);
  };

  const handleColDragOver = (e: React.DragEvent, index: number) => {
    e.preventDefault();
    if (colDragIndex === null || colDragIndex === index) return;
    setColDropTarget(index);
  };

  const handleColDrop = (e: React.DragEvent, targetIndex: number) => {
    e.preventDefault();
    if (colDragIndex !== null && colDragIndex !== targetIndex) {
      const newOrder = reorderArray(table.columnOrder, colDragIndex, targetIndex);
      table.setColumnOrder(newOrder);
    }
    setColDragIndex(null);
    setColDropTarget(null);
  };

  const handleColDragEnd = () => {
    setColDragIndex(null);
    setColDropTarget(null);
  };

  // -------------------------------------------------------------
  // Row Drag Handlers
  // -------------------------------------------------------------
  const handleRowDragStart = (e: React.DragEvent, index: number) => {
    setRowDragIndex(index);
    e.dataTransfer.effectAllowed = 'move';
    e.dataTransfer.setData('text/plain', `row-${index}`);
  };

  const handleRowDragOver = (e: React.DragEvent, index: number) => {
    e.preventDefault();
    if (rowDragIndex === null || rowDragIndex === index) return;
    setRowDropTarget(index);
  };

  const handleRowDrop = (e: React.DragEvent, targetIndex: number) => {
    e.preventDefault();
    if (rowDragIndex !== null && rowDragIndex !== targetIndex) {
      // Reordering paginated/rendered rows
      const targetSourceRow = table.data[rowDragIndex];
      const targetDestRow = table.data[targetIndex];

      const fromMasterIdx = table.rawSortedData.findIndex((r) => r.id === targetSourceRow.id);
      const toMasterIdx = table.rawSortedData.findIndex((r) => r.id === targetDestRow.id);

      if (fromMasterIdx !== -1 && toMasterIdx !== -1) {
        const reordered = reorderArray(table.rawSortedData, fromMasterIdx, toMasterIdx);
        // Sync back into main data source
        reordered.forEach((r, idx) => {
          table.updateRow(r.id, r);
        });
      }
    }
    setRowDragIndex(null);
    setRowDropTarget(null);
  };

  const handleRowDragEnd = () => {
    setRowDragIndex(null);
    setRowDropTarget(null);
  };

  return (
    <div style={{ padding: 16, fontFamily: 'Inter, system-ui, sans-serif' }}>
      <div style={{ marginBottom: 12, display: 'flex', alignItems: 'center', justifyContent: 'space-between' }}>
        <div>
          <h3 style={{ margin: 0, fontSize: 16, color: '#0f172a' }}>Reorderable Grid (Columns & Rows)</h3>
          <p style={{ margin: '4px 0 0', fontSize: 12, color: '#64748b' }}>
            Drag column headers horizontally or click & hold the <code>⠿</code> handle to reorder rows vertically.
          </p>
        </div>
      </div>

      <div style={{ border: '1px solid #e2e8f0', borderRadius: 8, overflow: 'hidden' }}>
        <table style={{ width: '100%', borderCollapse: 'collapse', textAlign: 'left' }}>
          {/* Header Row with Column Drag and Drop */}
          <thead>
            <tr style={{ background: '#f8fafc', borderBottom: '2px solid #e2e8f0' }}>
              {/* Row Handle Blank Header */}
              <th style={{ width: 40, padding: '10px 8px', borderRight: '1px solid #e2e8f0', textAlign: 'center' }}>
                ↕
              </th>

              {/* Selection Header */}
              <th style={{ width: 40, padding: '10px 8px', borderRight: '1px solid #e2e8f0', textAlign: 'center' }}>
                <input
                  type="checkbox"
                  checked={table.selectedRowIds.size === table.data.length && table.data.length > 0}
                  onChange={table.toggleSelectAll}
                />
              </th>

              {/* Draggable Column Headers */}
              {orderedColumns.map((col, idx) => {
                const isDragging = colDragIndex === idx;
                const isOver = colDropTarget === idx;

                return (
                  <th
                    key={col.id}
                    draggable
                    onDragStart={(e) => handleColDragStart(e, idx)}
                    onDragOver={(e) => handleColDragOver(e, idx)}
                    onDrop={(e) => handleColDrop(e, idx)}
                    onDragEnd={handleColDragEnd}
                    style={{
                      padding: '10px 14px',
                      fontSize: 13,
                      fontWeight: 600,
                      color: '#334155',
                      cursor: 'grab',
                      userSelect: 'none',
                      position: 'relative',
                      borderRight: '1px solid #e2e8f0',
                      backgroundColor: isDragging ? '#e2e8f0' : isOver ? '#e0e7ff' : '#f8fafc',
                      borderLeft: isOver ? '3px solid #6366f1' : undefined,
                      transition: 'background-color 0.15s',
                    }}
                  >
                    <div style={{ display: 'flex', alignItems: 'center', justifyContent: 'space-between' }}>
                      <span>{col.header}</span>
                      <span style={{ fontSize: 11, color: '#94a3b8', marginLeft: 6 }}>⋮⋮</span>
                    </div>
                  </th>
                );
              })}
            </tr>
          </thead>

          {/* Body Rows with Row Drag and Drop */}
          <tbody>
            {table.data.map((row, rIdx) => {
              const isRowDragging = rowDragIndex === rIdx;
              const isRowOver = rowDropTarget === rIdx;

              return (
                <tr
                  key={row.id}
                  onDragOver={(e) => handleRowDragOver(e, rIdx)}
                  onDrop={(e) => handleRowDrop(e, rIdx)}
                  style={{
                    borderBottom: '1px solid #e2e8f0',
                    backgroundColor: isRowDragging ? '#f1f5f9' : table.selectedRowIds.has(row.id) ? '#f0f9ff' : '#ffffff',
                    borderTop: isRowOver ? '2px solid #6366f1' : undefined,
                    opacity: isRowDragging ? 0.4 : 1,
                    transition: 'background-color 0.1s',
                  }}
                >
                  {/* Row Drag Handle */}
                  <td
                    draggable
                    onDragStart={(e) => handleRowDragStart(e, rIdx)}
                    onDragEnd={handleRowDragEnd}
                    style={{
                      width: 40,
                      padding: '8px 4px',
                      textAlign: 'center',
                      cursor: 'grab',
                      userSelect: 'none',
                      color: '#94a3b8',
                      fontSize: 16,
                      borderRight: '1px solid #f1f5f9',
                    }}
                  >
                    ⠿
                  </td>

                  {/* Checkbox Cell */}
                  <td style={{ width: 40, padding: '8px 8px', textAlign: 'center', borderRight: '1px solid #f1f5f9' }}>
                    <input
                      type="checkbox"
                      checked={table.selectedRowIds.has(row.id)}
                      onChange={() => table.toggleRowSelection(row.id)}
                    />
                  </td>

                  {/* Data Cells in dynamic column order */}
                  {orderedColumns.map((col) => {
                    const value = col.accessorKey ? row[col.accessorKey] : '';
                    return (
                      <td
                        key={col.id}
                        style={{
                          padding: '10px 14px',
                          fontSize: 13,
                          color: '#1e293b',
                          borderRight: '1px solid #f1f5f9',
                        }}
                      >
                        {String(value)}
                      </td>
                    );
                  })}
                </tr>
              );
            })}
          </tbody>
        </table>
      </div>
    </div>
  );
};

```
