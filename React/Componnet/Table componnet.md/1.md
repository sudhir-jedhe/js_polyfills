## 1. Core Architecture & Types

```typescript
// types/table.ts
export type ColumnPin = 'left' | 'right' | false;

export type FilterOperator =
  | 'fuzzy' | 'contains' | 'startsWith' | 'endsWith'
  | 'equals' | 'notEquals' | 'between' | 'betweenInclusive'
  | 'gt' | 'gte' | 'lt' | 'lte' | 'empty' | 'notEmpty';

export interface FilterCondition {
  operator: FilterOperator;
  value: any;
  endValue?: any; // For between/betweenInclusive
}

export interface ColumnDef<T> {
  id: string;
  header: string;
  accessorKey?: keyof T;
  accessorFn?: (row: T) => any;
  pin?: ColumnPin;
  visible?: boolean;
  order?: number;
  enableSorting?: boolean;
  enableFiltering?: boolean;
  enableGrouping?: boolean;
  cell?: (info: { value: any; row: T; rowIndex: number }) => any;
  subColumns?: ColumnDef<T>[];
}

export interface TableState<T> {
  data: T[];
  columns: ColumnDef<T>[];
  selectedRowIds: Set<string | number>;
  sorting: { columnId: string; desc: boolean }[];
  filters: Record<string, FilterCondition>;
  globalSearch: string;
  columnOrder: string[];
  columnVisibility: Record<string, boolean>;
  columnPinning: Record<string, ColumnPin>;
  grouping: string[];
  theme: 'light' | 'dark' | 'corporate';
  pageIndex: number;
  pageSize: number;
}
```

## 2. Advanced Multi-Operator Filter Engine

```typescript
// utils/filters.ts
import { FilterOperator, FilterCondition } from '../types/table';

export function fuzzyMatch(pattern: string, text: string): boolean {
  pattern = pattern.toLowerCase();
  text = text.toLowerCase();
  let pIdx = 0;
  let tIdx = 0;
  while (pIdx < pattern.length && tIdx < text.length) {
    if (pattern[pIdx] === text[tIdx]) pIdx++;
    tIdx++;
  }
  return pIdx === pattern.length;
}

export function evaluateFilter(cellValue: any, filter: FilterCondition): boolean {
  const { operator, value, endValue } = filter;
  const isNil = cellValue === null || cellValue === undefined || cellValue === '';

  switch (operator) {
    case 'empty':
      return isNil;
    case 'notEmpty':
      return !isNil;
    case 'fuzzy':
      return isNil ? false : fuzzyMatch(String(value), String(cellValue));
    case 'contains':
      return isNil ? false : String(cellValue).toLowerCase().includes(String(value).toLowerCase());
    case 'startsWith':
      return isNil ? false : String(cellValue).toLowerCase().startsWith(String(value).toLowerCase());
    case 'endsWith':
      return isNil ? false : String(cellValue).toLowerCase().endsWith(String(value).toLowerCase());
    case 'equals':
      return cellValue === value || String(cellValue).toLowerCase() === String(value).toLowerCase();
    case 'notEquals':
      return cellValue !== value && String(cellValue).toLowerCase() !== String(value).toLowerCase();
    case 'between': {
      const v = Number(cellValue), s = Number(value), e = Number(endValue);
      return !isNaN(v) && v > s && v < e;
    }
    case 'betweenInclusive': {
      const v = Number(cellValue), s = Number(value), e = Number(endValue);
      return !isNaN(v) && v >= s && v <= e;
    }
    case 'gt':
      return Number(cellValue) > Number(value);
    case 'gte':
      return Number(cellValue) >= Number(value);
    case 'lt':
      return Number(cellValue) < Number(value);
    case 'lte':
      return Number(cellValue) <= Number(value);
    default:
      return true;
  }
}
```

## 3. Data Pipeline: Sorting, Grouping & Row Selection Hook

```typescript
// hooks/useTableEngine.ts
import { useState, useMemo, useCallback } from 'react';
import { ColumnDef, TableState, FilterCondition } from '../types/table';
import { evaluateFilter } from '../utils/filters';

export function useTableEngine<T extends { id: string | number }>(
  initialData: T[],
  initialColumns: ColumnDef<T>[]
) {
  const [data, setData] = useState<T[]>(initialData);
  const [columns, setColumns] = useState<ColumnDef<T>[]>(initialColumns);
  const [selectedRowIds, setSelectedRowIds] = useState<Set<string | number>>(new Set());
  const [sorting, setSorting] = useState<{ columnId: string; desc: boolean }[]>([]);
  const [filters, setFilters] = useState<Record<string, FilterCondition>>({});
  const [globalSearch, setGlobalSearch] = useState('');
  const [columnOrder, setColumnOrder] = useState<string[]>(initialColumns.map((c) => c.id));
  const [columnVisibility, setColumnVisibility] = useState<Record<string, boolean>>({});
  const [columnPinning, setColumnPinning] = useState<Record<string, 'left' | 'right' | false>>({});
  const [grouping, setGrouping] = useState<string[]>([]);
  const [pageIndex, setPageIndex] = useState(0);
  const [pageSize, setPageSize] = useState(10);
  const [theme, setTheme] = useState<'light' | 'dark' | 'corporate'>('light');

  // CRUD Operations
  const addRow = useCallback((row: T) => setData((prev) => [row, ...prev]), []);
  const updateRow = useCallback((id: string | number, updated: Partial<T>) => {
    setData((prev) => prev.map((r) => (r.id === id ? { ...r, ...updated } : r)));
  }, []);
  const deleteRow = useCallback((id: string | number) => {
    setData((prev) => prev.filter((r) => r.id !== id));
    setSelectedRowIds((prev) => {
      const next = new Set(prev);
      next.delete(id);
      return next;
    });
  }, []);
  const addColumn = useCallback((col: ColumnDef<T>) => {
    setColumns((prev) => [...prev, col]);
    setColumnOrder((prev) => [...prev, col.id]);
  }, []);
  const deleteColumn = useCallback((colId: string) => {
    setColumns((prev) => prev.filter((c) => c.id !== colId));
    setColumnOrder((prev) => prev.filter((id) => id !== colId));
  }, []);

  // Filter & Global Search Pipeline
  const filteredData = useMemo(() => {
    return data.filter((row) => {
      if (globalSearch) {
        const matchesGlobal = Object.values(row).some((val) =>
          String(val).toLowerCase().includes(globalSearch.toLowerCase())
        );
        if (!matchesGlobal) return false;
      }
      for (const [colId, condition] of Object.entries(filters)) {
        const col = columns.find((c) => c.id === colId);
        const cellValue = col?.accessorKey ? row[col.accessorKey] : col?.accessorFn?.(row);
        if (!evaluateFilter(cellValue, condition)) return false;
      }
      return true;
    });
  }, [data, columns, filters, globalSearch]);

  // Sort Pipeline
  const sortedData = useMemo(() => {
    if (!sorting.length) return filteredData;
    return [...filteredData].sort((a, b) => {
      for (const sort of sorting) {
        const col = columns.find((c) => c.id === sort.columnId);
        const aVal = col?.accessorKey ? a[col.accessorKey] : col?.accessorFn?.(a);
        const bVal = col?.accessorKey ? b[col.accessorKey] : col?.accessorFn?.(b);
        if (aVal < bVal) return sort.desc ? 1 : -1;
        if (aVal > bVal) return sort.desc ? -1 : 1;
      }
      return 0;
    });
  }, [filteredData, sorting, columns]);

  // Pagination
  const paginatedData = useMemo(() => {
    const start = pageIndex * pageSize;
    return sortedData.slice(start, start + pageSize);
  }, [sortedData, pageIndex, pageSize]);

  // Selection Handlers
  const toggleRowSelection = (id: string | number) => {
    setSelectedRowIds((prev) => {
      const next = new Set(prev);
      next.has(id) ? next.delete(id) : next.add(id);
      return next;
    });
  };

  const toggleSelectAll = () => {
    if (selectedRowIds.size === paginatedData.length) {
      setSelectedRowIds(new Set());
    } else {
      setSelectedRowIds(new Set(paginatedData.map((r) => r.id)));
    }
  };

  return {
    data: paginatedData,
    rawSortedData: sortedData,
    columns,
    selectedRowIds,
    sorting,
    filters,
    globalSearch,
    columnOrder,
    columnVisibility,
    columnPinning,
    grouping,
    theme,
    pageIndex,
    pageSize,
    setPageIndex,
    setPageSize,
    setTheme,
    setSorting,
    setFilters,
    setGlobalSearch,
    setColumnOrder,
    setColumnVisibility,
    setColumnPinning,
    setGrouping,
    toggleRowSelection,
    toggleSelectAll,
    addRow,
    updateRow,
    deleteRow,
    addColumn,
    deleteColumn,
  };
}
```

## 4. Export Pipelines (CSV & PDF)

```typescript
// utils/export.ts
export function exportToCSV<T>(rows: T[], filename = 'export.csv') {
  if (!rows.length) return;
  const headers = Object.keys(rows[0] as Record<string, any>);
  const csvContent = [
    headers.join(','),
    ...rows.map((row) =>
      headers
        .map((header) => {
          const val = (row as Record<string, any>)[header];
          return `"${String(val ?? '').replace(/"/g, '""')}"`;
        })
        .join(',')
    ),
  ].join('\r\n');

  const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
  const url = URL.createObjectURL(blob);
  const link = document.createElement('a');
  link.setAttribute('href', url);
  link.setAttribute('download', filename);
  document.body.appendChild(link);
  link.click();
  document.body.removeChild(link);
}

export function exportToPDF<T>(rows: T[], title = 'Data Export') {
  const printWindow = window.open('', '_blank');
  if (!printWindow || !rows.length) return;
  const headers = Object.keys(rows[0] as Record<string, any>);
  const html = `
    <html>
      <head>
        <title>${title}</title>
        <style>
          table { width: 100%; border-collapse: collapse; font-family: sans-serif; }
          th, td { border: 1px solid #ddd; padding: 8px; text-align: left; }
          th { background-color: #f2f2f2; }
        </style>
      </head>
      <body>
        <h2>${title}</h2>
        <table>
          <thead><tr>${headers.map((h) => `<th>${h}</th>`).join('')}</tr></thead>
          <tbody>
            ${rows
              .map(
                (r) =>
                  `<tr>${headers.map((h) => `<td>${(r as any)[h] ?? ''}</td>`).join('')}</tr>`
              )
              .join('')}
          </tbody>
        </table>
      </body>
    </html>
  `;
  printWindow.document.write(html);
  printWindow.document.close();
  printWindow.print();
}
```

## 5. Architectural Implementation Guide for Key Use Cases

| Category                   | Example Focus                             | Implementation Blueprint                                                                                                                                                                                                           |
| -------------------------- | ----------------------------------------- | ---------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------- |
| **Foundations**            | *Basic / Minimal / Custom Headless*       | Decouple state management into a custom React hook `useTableEngine`. Use standard semantic HTML `<table>`, `<thead>`, `<tbody>` wrapped with accessors.                                                                            |
| **Exports**                | *CSV, PDF, Selective Export*              | Pass either `data` (all), `paginatedData` (page-only), or `data.filter(r => selectedRowIds.has(r.id))` to CSV/PDF helpers.                                                                                                         |
| **Ordering & Dragging**    | *Column / Row Reordering*                 | Use HTML5 Drag & Drop API (`draggable`, `onDragStart`, `onDragOver`, `onDrop`) to swap indices in `columnOrder` or `data` arrays.                                                                                                  |
| **Editing Modes**          | *Inline Cell, Row, Modal, Tree*           | Maintain an `editingRowId` or `editingCellKey` state. Render text/select inputs on active edit targets and commit via `updateRow()`.                                                                                               |
| **Grouping & Aggregation** | *Tree Rows, Column Groups, Aggregations*  | Group data recursively by a key (e.g. `groupKey`), calculate aggregate sums/averages for parent nodes, and track `expandedRowIds: Set<string>`.                                                                                    |
| **Filtering & Pinning**    | *Faceted, Popover, Pin Left/Right*        | Pin columns using CSS `position: sticky` with calculated horizontal offsets (`left: 0`, `right: 0`). Attach popovers on column headers updating the `filters` state.                                                               |
| **Virtualization & Async** | *Row/Col Virtualization, Infinite Scroll* | Compute dynamic offsets based on `scrollTop` and fixed `rowHeight` (e.g. `Math.floor(scrollTop / rowHeight)`), rendering only the viewport slice. Fetch async data using `useEffect` or `@tanstack/react-query` cursor pagination. |

## 6. Full Interactive Implementation Example

```tsx
// components/EnterpriseTable.tsx
import React, { useState } from 'react';
import { useTableEngine } from '../hooks/useTableEngine';
import { ColumnDef, FilterOperator } from '../types/table';
import { exportToCSV, exportToPDF } from '../utils/export';

interface UserData {
  id: number;
  name: string;
  age: number;
  role: string;
  status: string;
}

const SAMPLE_DATA: UserData[] = [
  { id: 1, name: 'Alice Cooper', age: 29, role: 'Developer', status: 'Active' },
  { id: 2, name: 'Bob Smith', age: 42, role: 'Manager', status: 'Inactive' },
  { id: 3, name: 'Jon Doe', age: 35, role: 'Designer', status: 'Active' },
];

const COLUMNS: ColumnDef<UserData>[] = [
  { id: 'id', header: 'ID', accessorKey: 'id', pin: 'left' },
  { id: 'name', header: 'Name', accessorKey: 'name' },
  { id: 'age', header: 'Age', accessorKey: 'age' },
  { id: 'role', header: 'Role', accessorKey: 'role' },
  { id: 'status', header: 'Status', accessorKey: 'status' },
];

export const EnterpriseTable: React.FC = () => {
  const {
    data,
    rawSortedData,
    columns,
    selectedRowIds,
    sorting,
    filters,
    globalSearch,
    theme,
    setTheme,
    setSorting,
    setFilters,
    setGlobalSearch,
    toggleRowSelection,
    toggleSelectAll,
    addRow,
    updateRow,
    deleteRow,
    addColumn,
    deleteColumn,
  } = useTableEngine(SAMPLE_DATA, COLUMNS);

  const [editingId, setEditingId] = useState<number | null>(null);
  const [filterCol, setFilterCol] = useState<string>('name');
  const [filterOp, setFilterOp] = useState<FilterOperator>('fuzzy');
  const [filterVal, setFilterVal] = useState<string>('');

  const applyColumnFilter = () => {
    setFilters((prev) => ({
      ...prev,
      [filterCol]: { operator: filterOp, value: filterVal },
    }));
  };

  const getExportData = (type: 'all' | 'selected' | 'page') => {
    if (type === 'selected') return rawSortedData.filter((r) => selectedRowIds.has(r.id));
    if (type === 'page') return data;
    return rawSortedData;
  };

  return (
    <div style={{ padding: 16, fontFamily: 'sans-serif', background: theme === 'dark' ? '#1e1e1e' : '#fff', color: theme === 'dark' ? '#fff' : '#000' }}>
      {/* Controls Bar */}
      <div style={{ display: 'flex', gap: 8, flexWrap: 'wrap', marginBottom: 12 }}>
        <input
          placeholder="Global Search..."
          value={globalSearch}
          onChange={(e) => setGlobalSearch(e.target.value)}
          style={{ padding: 6 }}
        />

        {/* Filter Builder */}
        <select value={filterCol} onChange={(e) => setFilterCol(e.target.value)}>
          {columns.map((c) => <option key={c.id} value={c.id}>{c.header}</option>)}
        </select>
        <select value={filterOp} onChange={(e) => setFilterOp(e.target.value as FilterOperator)}>
          <option value="fuzzy">Fuzzy (≈)</option>
          <option value="contains">Contains (*)</option>
          <option value="startsWith">Starts With (a)</option>
          <option value="endsWith">Ends With (z)</option>
          <option value="equals">Equals (=)</option>
          <option value="notEquals">Not Equals (≠)</option>
          <option value="gt">Greater Than (&gt;)</option>
          <option value="lt">Less Than (&lt;)</option>
          <option value="empty">Empty (Ø)</option>
          <option value="notEmpty">Not Empty (!Ø)</option>
        </select>
        <input
          placeholder="Value"
          value={filterVal}
          onChange={(e) => setFilterVal(e.target.value)}
          style={{ padding: 6 }}
        />
        <button onClick={applyColumnFilter}>Apply Filter</button>
        <button onClick={() => setFilters({})}>Reset Filters</button>

        {/* Export Triggers */}
        <button onClick={() => exportToCSV(getExportData('all'))}>Export All (CSV)</button>
        <button onClick={() => exportToCSV(getExportData('selected'))}>Export Selected (CSV)</button>
        <button onClick={() => exportToPDF(getExportData('page'))}>Export Page (PDF)</button>

        {/* Theme Switcher */}
        <select value={theme} onChange={(e) => setTheme(e.target.value as any)}>
          <option value="light">Theme: Light</option>
          <option value="dark">Theme: Dark</option>
          <option value="corporate">Theme: Corporate</option>
        </select>
      </div>

      {/* Table Render */}
      <table style={{ width: '100%', borderCollapse: 'collapse', border: '1px solid #ccc' }}>
        <thead>
          <tr style={{ background: theme === 'dark' ? '#333' : '#f0f0f0' }}>
            <th style={{ padding: 8, border: '1px solid #ccc' }}>
              <input
                type="checkbox"
                checked={selectedRowIds.size === data.length && data.length > 0}
                onChange={toggleSelectAll}
              />
            </th>
            {columns.map((col) => (
              <th
                key={col.id}
                style={{
                  padding: 8,
                  border: '1px solid #ccc',
                  cursor: 'pointer',
                  position: col.pin ? 'sticky' : undefined,
                  left: col.pin === 'left' ? 0 : undefined,
                  right: col.pin === 'right' ? 0 : undefined,
                }}
                onClick={() => {
                  const isDesc = sorting.find((s) => s.columnId === col.id)?.desc;
                  setSorting([{ columnId: col.id, desc: !isDesc }]);
                }}
              >
                {col.header} {sorting.find((s) => s.columnId === col.id) ? (sorting.find((s) => s.columnId === col.id)?.desc ? '▼' : '▲') : ''}
              </th>
            ))}
            <th style={{ padding: 8, border: '1px solid #ccc' }}>Actions</th>
          </tr>
        </thead>
        <tbody>
          {data.map((row) => (
            <tr key={row.id} style={{ background: selectedRowIds.has(row.id) ? '#e6f7ff' : 'transparent' }}>
              <td style={{ padding: 8, border: '1px solid #ccc', textAlign: 'center' }}>
                <input
                  type="checkbox"
                  checked={selectedRowIds.has(row.id)}
                  onChange={() => toggleRowSelection(row.id)}
                />
              </td>
              {columns.map((col) => (
                <td key={col.id} style={{ padding: 8, border: '1px solid #ccc' }}>
                  {editingId === row.id && col.accessorKey ? (
                    <input
                      defaultValue={String(row[col.accessorKey])}
                      onBlur={(e) => updateRow(row.id, { [col.accessorKey!]: e.target.value } as any)}
                    />
                  ) : (
                    String(row[col.accessorKey as keyof UserData] ?? '')
                  )}
                </td>
              ))}
              <td style={{ padding: 8, border: '1px solid #ccc' }}>
                <button onClick={() => setEditingId(editingId === row.id ? null : row.id)}>
                  {editingId === row.id ? 'Done' : 'Edit'}
                </button>
                <button onClick={() => deleteRow(row.id)} style={{ marginLeft: 4 }}>Delete</button>
              </td>
            </tr>
          ))}
        </tbody>
      </table>
    </div>
  );
};
```
