***  1.md ***

Here is the complete, zero-dependency, production-ready **Tabs Component System built from scratch** in React and TypeScript.

This covers:

1. **Core Architecture**: `Tabs`, `Tab`, `TabPanel`, `TabContext`, and `TabList`.
2. **Layout & Alignment**: Fixed, Full-Width, Centered, Vertical, and Wrapped Multi-Line Labels.
3. **Scrollable Tabs**: Smooth scroll container with Automatic, Forced, and Prevented scroll chevron buttons.
4. **Icons & Positioning**: Top, Bottom, Start, and End icon placements.
5. **Theme Colors, Disabled States, and Custom Indicators**: Dynamic sliding underline/pill indicator computed via DOM rects.
6. **Third-Party Router Integration**: Polymorphic `as` prop supporting React Router / TanStack Router links.
7. **WAI-ARIA 1.2 Tabs Spec**: `role="tablist"`, `role="tab"`, `role="tabpanel"`, `aria-selected`, `aria-controls`, and circular roving keyboard traversal (`ArrowLeft`/`ArrowRight` or `ArrowUp`/`ArrowDown`, `Home`, `End`).

### 2. Core Tabs & Sliding Indicator Engine

```tsx
// components/Tabs.tsx
import React, { useState, useRef, useEffect, useCallback, useId } from 'react';
import { TabsContext } from '../context/TabsContext';
import {
  TabsOrientation,
  TabsVariant,
  ScrollButtonsSetting,
  TabColor,
} from '../types/tabs';

const COLOR_MAP: Record<TabColor, { text: string; indicator: string }> = {
  primary: { text: '#2563eb', indicator: '#2563eb' },
  secondary: { text: '#475569', indicator: '#475569' },
  success: { text: '#16a34a', indicator: '#16a34a' },
  error: { text: '#dc2626', indicator: '#dc2626' },
  warning: { text: '#d97706', indicator: '#d97706' },
  purple: { text: '#7c3aed', indicator: '#7c3aed' },
};

export interface TabsProps extends Omit<React.HTMLAttributes<HTMLDivElement>, 'onChange'> {
  value: any;
  onChange: (event: React.SyntheticEvent, value: any) => void;
  orientation?: TabsOrientation;
  variant?: TabsVariant;
  centered?: boolean;
  scrollButtons?: ScrollButtonsSetting;
  color?: TabColor;
  customTextColor?: string;
  customIndicatorColor?: string;
  children: React.ReactNode;
}

export const Tabs: React.FC<TabsProps> = ({
  value,
  onChange,
  orientation = 'horizontal',
  variant = 'standard',
  centered = false,
  scrollButtons = 'auto',
  color = 'primary',
  customTextColor,
  customIndicatorColor,
  children,
  style,
  className,
  ...restProps
}) => {
  const scrollContainerRef = useRef<HTMLDivElement>(null);
  const tabElementsRef = useRef<Map<any, HTMLButtonElement>>(new Map());
  const [indicatorStyle, setIndicatorStyle] = useState<React.CSSProperties>({ opacity: 0 });
  const [canScrollLeft, setCanScrollLeft] = useState(false);
  const [canScrollRight, setCanScrollRight] = useState(false);

  const isVertical = orientation === 'vertical';
  const isScrollable = variant === 'scrollable';
  const resolvedColor = COLOR_MAP[color] || COLOR_MAP.primary;
  const textColor = customTextColor || resolvedColor.text;
  const indicatorColor = customIndicatorColor || resolvedColor.indicator;

  const registerTab = useCallback((val: any, el: HTMLButtonElement | null) => {
    if (el) tabElementsRef.current.set(val, el);
    else tabElementsRef.current.delete(val);
  }, []);

  // Compute Active Tab Indicator Position
  const updateIndicator = useCallback(() => {
    const activeTabNode = tabElementsRef.current.get(value);
    const containerNode = scrollContainerRef.current;
    if (!activeTabNode || !containerNode) {
      setIndicatorStyle({ opacity: 0 });
      return;
    }

    const containerRect = containerNode.getBoundingClientRect();
    const tabRect = activeTabNode.getBoundingClientRect();

    if (isVertical) {
      const top = tabRect.top - containerRect.top + containerNode.scrollTop;
      setIndicatorStyle({
        top,
        height: tabRect.height,
        right: 0,
        width: 2.5,
        opacity: 1,
      });
    } else {
      const left = tabRect.left - containerRect.left + containerNode.scrollLeft;
      setIndicatorStyle({
        left,
        width: tabRect.width,
        bottom: 0,
        height: 2.5,
        opacity: 1,
      });
    }
  }, [value, isVertical]);

  // Update Scroll Button Visibilities
  const checkScrollability = useCallback(() => {
    if (!isScrollable || !scrollContainerRef.current || isVertical) return;
    const el = scrollContainerRef.current;
    setCanScrollLeft(el.scrollLeft > 1);
    setCanScrollRight(el.scrollLeft < el.scrollWidth - el.clientWidth - 1);
  }, [isScrollable, isVertical]);

  useEffect(() => {
    updateIndicator();
    checkScrollability();
    const handleResize = () => {
      updateIndicator();
      checkScrollability();
    };
    window.addEventListener('resize', handleResize);
    return () => window.removeEventListener('resize', handleResize);
  }, [updateIndicator, checkScrollability]);

  // Scroll active tab into view
  useEffect(() => {
    const activeTab = tabElementsRef.current.get(value);
    if (activeTab && isScrollable && scrollContainerRef.current) {
      const container = scrollContainerRef.current;
      const cRect = container.getBoundingClientRect();
      const tRect = activeTab.getBoundingClientRect();

      if (tRect.left < cRect.left) {
        container.scrollLeft -= cRect.left - tRect.left + 24;
      } else if (tRect.right > cRect.right) {
        container.scrollLeft += tRect.right - cRect.right + 24;
      }
    }
  }, [value, isScrollable]);

  const handleScroll = (dir: 'left' | 'right') => {
    if (!scrollContainerRef.current) return;
    const scrollAmount = 180;
    scrollContainerRef.current.scrollBy({
      left: dir === 'left' ? -scrollAmount : scrollAmount,
      behavior: 'smooth',
    });
  };

  // Keyboard navigation across tabs
  const handleKeyDown = (e: React.KeyboardEvent) => {
    const tabs = Array.from(tabElementsRef.current.values()).filter((t) => !t.disabled);
    const activeIndex = tabs.findIndex((t) => t === document.activeElement);
    if (activeIndex === -1) return;

    let nextIndex = activeIndex;

    if (isVertical) {
      if (e.key === 'ArrowDown') nextIndex = (activeIndex + 1) % tabs.length;
      else if (e.key === 'ArrowUp') nextIndex = (activeIndex - 1 + tabs.length) % tabs.length;
    } else {
      if (e.key === 'ArrowRight') nextIndex = (activeIndex + 1) % tabs.length;
      else if (e.key === 'ArrowLeft') nextIndex = (activeIndex - 1 + tabs.length) % tabs.length;
    }

    if (e.key === 'Home') nextIndex = 0;
    else if (e.key === 'End') nextIndex = tabs.length - 1;

    if (nextIndex !== activeIndex) {
      e.preventDefault();
      tabs[nextIndex].focus();
      tabs[nextIndex].click();
    }
  };

  const showLeftButton =
    isScrollable && !isVertical && (scrollButtons === true || (scrollButtons === 'auto' && canScrollLeft));
  const showRightButton =
    isScrollable && !isVertical && (scrollButtons === true || (scrollButtons === 'auto' && canScrollRight));

  return (
    <TabsContext.Provider
      value={{
        value,
        onChange,
        orientation,
        textColor,
        indicatorColor,
        registerTab,
        tabElements: tabElementsRef.current,
      }}
    >
      <div
        className={`tabs-root ${className || ''}`}
        style={{
          display: 'flex',
          alignItems: 'center',
          flexDirection: isVertical ? 'column' : 'row',
          position: 'relative',
          borderBottom: isVertical ? 'none' : '1px solid #e2e8f0',
          borderRight: isVertical ? '1px solid #e2e8f0' : 'none',
          fontFamily: 'Inter, system-ui, sans-serif',
          ...style,
        }}
        {...restProps}
      >
        {/* Left Scroll Button */}
        {showLeftButton && (
          <button
            type="button"
            aria-label="Scroll tabs left"
            disabled={!canScrollLeft}
            onClick={() => handleScroll('left')}
            style={scrollBtnStyle(!canScrollLeft)}
          >
            ‹
          </button>
        )}

        {/* Tab List Scroll Viewport */}
        <div
          ref={scrollContainerRef}
          role="tablist"
          aria-orientation={orientation}
          onKeyDown={handleKeyDown}
          onScroll={checkScrollability}
          style={{
            display: 'flex',
            flexDirection: isVertical ? 'column' : 'row',
            justifyContent: centered && !isVertical ? 'center' : 'flex-start',
            width: isVertical ? 'auto' : '100%',
            overflowX: isScrollable && !isVertical ? 'auto' : 'hidden',
            overflowY: isScrollable && isVertical ? 'auto' : 'hidden',
            position: 'relative',
            scrollbarWidth: 'none',
            msOverflowStyle: 'none',
          }}
        >
          {children}

          {/* Animated Active Indicator */}
          <div
            className="tabs-indicator"
            style={{
              position: 'absolute',
              backgroundColor: indicatorColor,
              transition: 'all 0.22s cubic-bezier(0.4, 0, 0.2, 1)',
              zIndex: 1,
              ...indicatorStyle,
            }}
          />
        </div>

        {/* Right Scroll Button */}
        {showRightButton && (
          <button
            type="button"
            aria-label="Scroll tabs right"
            disabled={!canScrollRight}
            onClick={() => handleScroll('right')}
            style={scrollBtnStyle(!canScrollRight)}
          >
            ›
          </button>
        )}
      </div>
    </TabsContext.Provider>
  );
};

const scrollBtnStyle = (disabled: boolean): React.CSSProperties => ({
  background: 'none',
  border: 'none',
  fontSize: 20,
  fontWeight: 700,
  color: disabled ? '#cbd5e1' : '#475569',
  cursor: disabled ? 'not-allowed' : 'pointer',
  padding: '0 8px',
  height: '100%',
  display: 'inline-flex',
  alignItems: 'center',
  justifyContent: 'center',
  outline: 'none',
  userSelect: 'none',
});

```

### 3. Core `Tab` Component (Polymorphic, Icons, Wrapped Labels)

```tsx
// components/Tab.tsx
import React, { forwardRef, useEffect, useRef } from 'react';
import { useTabsContext } from '../context/TabsContext';
import { TabIconPosition } from '../types/tabs';

type AsProp<C extends React.ElementType> = {
  as?: C;
};

type PropsToOmit<C extends React.ElementType, P> = keyof (AsProp<C> & P);

type PolymorphicProps<C extends React.ElementType, Props = {}> = React.PropsWithChildren<
  Props & AsProp<C>
> &
  Omit<React.ComponentPropsWithoutRef<C>, PropsToOmit<C, Props>>;

export interface BaseTabProps {
  value: any;
  label?: React.ReactNode;
  icon?: React.ReactNode;
  iconPosition?: TabIconPosition;
  disabled?: boolean;
  wrapped?: boolean;
  fullWidth?: boolean;
}

export type TabProps<C extends React.ElementType = 'button'> = PolymorphicProps<C, BaseTabProps>;

export const Tab = forwardRef(
  <C extends React.ElementType = 'button'>(
    {
      as,
      value,
      label,
      icon,
      iconPosition = 'top',
      disabled = false,
      wrapped = false,
      fullWidth = false,
      onClick,
      style,
      className,
      ...restProps
    }: TabProps<C>,
    ref: React.Ref<any>
  ) => {
    const Component = as || 'button';
    const { value: selectedValue, onChange, textColor, registerTab, orientation } = useTabsContext();
    const internalBtnRef = useRef<HTMLButtonElement | null>(null);

    const isSelected = selectedValue === value;
    const isVertical = orientation === 'vertical';

    const setRefs = (el: any) => {
      internalBtnRef.current = el;
      registerTab(value, el);
      if (typeof ref === 'function') ref(el);
      else if (ref) (ref as any).current = el;
    };

    const handleClick = (e: React.SyntheticEvent) => {
      if (disabled) return;
      onClick?.(e as any);
      onChange(e, value);
    };

    // Calculate Flex Direction for Icon Position
    const getFlexDirection = () => {
      switch (iconPosition) {
        case 'bottom':
          return 'column-reverse';
        case 'start':
          return 'row';
        case 'end':
          return 'row-reverse';
        case 'top':
        default:
          return 'column';
      }
    };

    return (
      <Component
        ref={setRefs}
        role="tab"
        id={`tab-${value}`}
        aria-selected={isSelected}
        aria-controls={`tabpanel-${value}`}
        tabIndex={isSelected ? 0 : -1}
        disabled={Component === 'button' ? disabled : undefined}
        aria-disabled={disabled}
        onClick={handleClick}
        className={`custom-tab ${isSelected ? 'active' : ''} ${className || ''}`}
        style={{
          display: 'inline-flex',
          alignItems: 'center',
          justifyContent: 'center',
          flexDirection: getFlexDirection(),
          gap: 6,
          flex: fullWidth && !isVertical ? '1 1 0px' : 'none',
          minWidth: isVertical ? 120 : 80,
          maxWidth: wrapped ? 160 : 'none',
          padding: '10px 16px',
          fontSize: '13px',
          fontWeight: isSelected ? 600 : 500,
          color: disabled ? '#94a3b8' : isSelected ? textColor : '#64748b',
          backgroundColor: 'transparent',
          border: 'none',
          cursor: disabled ? 'not-allowed' : 'pointer',
          outline: 'none',
          userSelect: 'none',
          textDecoration: 'none',
          whiteSpace: wrapped ? 'normal' : 'nowrap',
          textAlign: 'center',
          lineHeight: 1.3,
          boxSizing: 'border-box',
          transition: 'color 0.15s ease',
          ...style,
        }}
        {...restProps}
      >
        {icon && <span style={{ display: 'inline-flex', fontSize: 18 }}>{icon}</span>}
        {label && <span>{label}</span>}
      </Component>
    );
  }
);

Tab.displayName = 'Tab';

```

### 4. `TabPanel` & `TabContext` (Experimental Compound Pattern)

```tsx
// components/TabPanel.tsx
import React, { createContext, useContext } from 'react';

// --- Tab Context for Container Providers ---
interface StandaloneTabContextValue {
  value: any;
}
const StandaloneTabContext = createContext<StandaloneTabContextValue | null>(null);

export const TabContext: React.FC<{ value: any; children: React.ReactNode }> = ({
  value,
  children,
}) => {
  return (
    <StandaloneTabContext.Provider value={{ value }}>
      {children}
    </StandaloneTabContext.Provider>
  );
};

// --- TabPanel ---
export interface TabPanelProps extends React.HTMLAttributes<HTMLDivElement> {
  value: any;
  currentValue?: any;
  children: React.ReactNode;
}

export const TabPanel: React.FC<TabPanelProps> = ({
  value,
  currentValue,
  children,
  style,
  className,
  ...restProps
}) => {
  const standaloneContext = useContext(StandaloneTabContext);
  const activeValue = currentValue !== undefined ? currentValue : standaloneContext?.value;
  const isSelected = activeValue === value;

  if (!isSelected) return null;

  return (
    <div
      role="tabpanel"
      id={`tabpanel-${value}`}
      aria-labelledby={`tab-${value}`}
      tabIndex={0}
      className={`tab-panel ${className || ''}`}
      style={{
        padding: '16px 4px',
        outline: 'none',
        fontFamily: 'Inter, system-ui, sans-serif',
        ...style,
      }}
      {...restProps}
    >
      {children}
    </div>
  );
};
```

### Architectural & Accessibility Highlights

| Feature                     | Implementation Technique                                                                                                                                                   |
| --------------------------- | -------------------------------------------------------------------------------------------------------------------------------------------------------------------------- |
| **Sliding Rect Indicator**  | Binds to the DOM bounding rect of the active `<Tab/>` relative to the scroll container's `scrollLeft`/`scrollTop`, applying transitions without CSS calculation hardcodes. |
| **WAI-ARIA 1.2 Tabs**       | `role="tablist"` on container, `role="tab"` with `aria-selected` and `aria-controls` on items, and `role="tabpanel"` linked via `aria-labelledby`.                         |
| **Keyboard Traversal Loop** | `onKeyDown` tracks `ArrowLeft`/`ArrowRight` (horizontal) or `ArrowUp`/`ArrowDown` (vertical) with `Home` and `End` support, focusing and selecting tabs seamlessly.        |
| **Polymorphic Routing**     | `Tab` uses generic `as` prop typing with `forwardRef` to render router `<Link>` or `<a>` elements while preserving indicator calculations and ARIA attributes.             |
