*** copy 1.md ***

Here is the complete zero-dependency implementation of a **Breadcrumbs Component System built from scratch** in React with TypeScript.

It includes:

1. **Semantic HTML & Accessibility**: Built with `<nav aria-label="Breadcrumb">` and `<ol>/<li>` hierarchy, applying `aria-current="page"` to the active last breadcrumb.
2. **Custom Separators**: Text strings, slashes, or custom SVG/React icon nodes.
3. **Icons & Avatars**: Start icons aligned with text labels.
4. **Collapsed Breadcrumbs (`maxItems`, `itemsBeforeCollapse`, `itemsAfterCollapse`)**: Automatic middle-item truncation with an expansion ellipsis button (`...`).
5. **Condensed Dropdown Menu**: Accessible dropdown menu nested inside the collapsed ellipsis button to jump directly to hidden middle paths.
6. **React Router / Third-Party Integration**: Polymorphic `as` prop supporting router `<Link>` elements and dynamic route matching.

### 2. Collapsed Dropdown Menu & Core Breadcrumb Link

```tsx
// components/BreadcrumbLink.tsx
import React, { forwardRef } from 'react';

// Polymorphic Type Definitions
type AsProp<C extends React.ElementType> = {
  as?: C;
};

type PropsToOmit<C extends React.ElementType, P> = keyof (AsProp<C> & P);

type PolymorphicProps<C extends React.ElementType, Props = {}> = React.PropsWithChildren<
  Props & AsProp<C>
> &
  Omit<React.ComponentPropsWithoutRef<C>, PropsToOmit<C, Props>>;

export interface BaseBreadcrumbLinkProps {
  icon?: React.ReactNode;
  isCurrent?: boolean;
  disabled?: boolean;
  color?: string;
}

export type BreadcrumbLinkProps<C extends React.ElementType = 'a'> = PolymorphicProps<
  C,
  BaseBreadcrumbLinkProps
>;

export const BreadcrumbLink = forwardRef(
  <C extends React.ElementType = 'a'>(
    {
      as,
      children,
      icon,
      isCurrent = false,
      disabled = false,
      color,
      style,
      className,
      ...restProps
    }: BreadcrumbLinkProps<C>,
    ref: React.Ref<any>
  ) => {
    const Component = isCurrent ? 'span' : as || 'a';

    return (
      <Component
        ref={ref}
        aria-current={isCurrent ? 'page' : undefined}
        className={`breadcrumb-link ${isCurrent ? 'current' : ''} ${className || ''}`}
        style={{
          display: 'inline-flex',
          alignItems: 'center',
          gap: 6,
          color: isCurrent ? '#0f172a' : color || '#64748b',
          textDecoration: 'none',
          fontSize: '13px',
          fontWeight: isCurrent ? 600 : 500,
          cursor: isCurrent || disabled ? 'default' : 'pointer',
          lineHeight: 1.4,
          transition: 'color 0.15s ease',
          outline: 'none',
          ...style,
        }}
        onMouseEnter={(e: React.MouseEvent<HTMLElement>) => {
          if (!isCurrent && !disabled) e.currentTarget.style.color = '#2563eb';
        }}
        onMouseLeave={(e: React.MouseEvent<HTMLElement>) => {
          if (!isCurrent && !disabled) e.currentTarget.style.color = color || '#64748b';
        }}
        {...restProps}
      >
        {icon && (
          <span style={{ display: 'inline-flex', alignItems: 'center', fontSize: 14 }}>
            {icon}
          </span>
        )}
        <span>{children}</span>
      </Component>
    );
  }
);

BreadcrumbLink.displayName = 'BreadcrumbLink';

```

### 3. Collapsed Ellipsis Menu Component

An interactive dropdown rendered inside the ellipsis button to jump to truncated paths.

```tsx
// components/CondensedBreadcrumbMenu.tsx
import React, { useState, useRef, useEffect } from 'react';

interface CondensedBreadcrumbMenuProps {
  hiddenItems: React.ReactNode[];
}

export const CondensedBreadcrumbMenu: React.FC<CondensedBreadcrumbMenuProps> = ({
  hiddenItems,
}) => {
  const [isOpen, setIsOpen] = useState(false);
  const containerRef = useRef<HTMLDivElement>(null);

  useEffect(() => {
    const handleOutsideClick = (e: MouseEvent) => {
      if (containerRef.current && !containerRef.current.contains(e.target as Node)) {
        setIsOpen(false);
      }
    };
    const handleKeyDown = (e: KeyboardEvent) => {
      if (e.key === 'Escape') setIsOpen(false);
    };

    document.addEventListener('mousedown', handleOutsideClick);
    document.addEventListener('keydown', handleKeyDown);
    return () => {
      document.removeEventListener('mousedown', handleOutsideClick);
      document.removeEventListener('keydown', handleKeyDown);
    };
  }, []);

  return (
    <div ref={containerRef} style={{ position: 'relative', display: 'inline-flex' }}>
      <button
        type="button"
        aria-label="Show all breadcrumbs"
        aria-haspopup="menu"
        aria-expanded={isOpen}
        onClick={() => setIsOpen((prev) => !prev)}
        style={{
          display: 'inline-flex',
          alignItems: 'center',
          justifyContent: 'center',
          backgroundColor: '#f1f5f9',
          border: '1px solid #cbd5e1',
          borderRadius: 4,
          padding: '0 6px',
          height: 20,
          fontSize: 11,
          fontWeight: 700,
          color: '#475569',
          cursor: 'pointer',
          outline: 'none',
        }}
      >
        ...
      </button>

      {/* Floating Dropdown */}
      {isOpen && (
        <div
          role="menu"
          style={{
            position: 'absolute',
            top: 'calc(100% + 4px)',
            left: 0,
            zIndex: 1000,
            backgroundColor: '#ffffff',
            border: '1px solid #e2e8f0',
            borderRadius: 6,
            boxShadow: '0 10px 15px -3px rgba(0, 0, 0, 0.1)',
            minWidth: 160,
            padding: '4px 0',
          }}
        >
          {hiddenItems.map((item, idx) => (
            <div
              key={idx}
              role="menuitem"
              onClick={() => setIsOpen(false)}
              style={{
                padding: '6px 12px',
                fontSize: 13,
                cursor: 'pointer',
                display: 'flex',
                alignItems: 'center',
              }}
              onMouseEnter={(e) => (e.currentTarget.style.backgroundColor = '#f8fafc')}
              onMouseLeave={(e) => (e.currentTarget.style.backgroundColor = 'transparent')}
            >
              {item}
            </div>
          ))}
        </div>
      )}
    </div>
  );
};

```

### 4. Master `Breadcrumbs` Component

```tsx
// components/Breadcrumbs.tsx
import React, { useState, Children, isValidElement, cloneElement } from 'react';
import { BreadcrumbsProps } from '../types/breadcrumbs';
import { CondensedBreadcrumbMenu } from './CondensedBreadcrumbMenu';

export const Breadcrumbs: React.FC<BreadcrumbsProps> = ({
  separator = '/',
  maxItems = 8,
  itemsBeforeCollapse = 1,
  itemsAfterCollapse = 1,
  children,
  ariaLabel = 'Breadcrumb',
  condensedMenu = false,
  style,
  className,
  ...restProps
}) => {
  const [isExpanded, setIsExpanded] = useState(false);

  // Flatten valid child elements
  const allItems = Children.toArray(children).filter(isValidElement);
  const totalCount = allItems.length;

  let renderedItems = allItems;
  let hiddenItems: React.ReactNode[] = [];

  // Collapse calculation
  if (!isExpanded && maxItems < totalCount && totalCount > itemsBeforeCollapse + itemsAfterCollapse) {
    const startItems = allItems.slice(0, itemsBeforeCollapse);
    const endItems = allItems.slice(totalCount - itemsAfterCollapse);
    hiddenItems = allItems.slice(itemsBeforeCollapse, totalCount - itemsAfterCollapse);

    const collapseTrigger = condensedMenu ? (
      <CondensedBreadcrumbMenu hiddenItems={hiddenItems} />
    ) : (
      <button
        type="button"
        aria-label="Expand hidden breadcrumbs"
        onClick={() => setIsExpanded(true)}
        style={{
          display: 'inline-flex',
          alignItems: 'center',
          justifyContent: 'center',
          backgroundColor: '#f1f5f9',
          border: '1px solid #cbd5e1',
          borderRadius: 4,
          padding: '0 6px',
          height: 20,
          fontSize: 11,
          fontWeight: 700,
          color: '#475569',
          cursor: 'pointer',
          outline: 'none',
        }}
      >
        ...
      </button>
    );

    renderedItems = [...startItems, collapseTrigger as React.ReactElement, ...endItems];
  }

  return (
    <nav
      aria-label={ariaLabel}
      className={`breadcrumbs ${className || ''}`}
      style={{
        fontFamily: 'Inter, system-ui, sans-serif',
        ...style,
      }}
      {...restProps}
    >
      <ol
        style={{
          display: 'flex',
          alignItems: 'center',
          flexWrap: 'wrap',
          margin: 0,
          padding: 0,
          listStyle: 'none',
          gap: 6,
        }}
      >
        {renderedItems.map((item, idx) => {
          const isLast = idx === renderedItems.length - 1;

          // Clone last child to apply isCurrent=true automatically if it's a BreadcrumbLink
          const enhancedItem =
            isLast && isValidElement(item) && (item.type as any).displayName === 'BreadcrumbLink'
              ? cloneElement(item, { isCurrent: true } as any)
              : item;

          return (
            <React.Fragment key={idx}>
              <li style={{ display: 'inline-flex', alignItems: 'center' }}>
                {enhancedItem}
              </li>

              {/* Separator */}
              {!isLast && (
                <li
                  aria-hidden="true"
                  style={{
                    display: 'inline-flex',
                    alignItems: 'center',
                    color: '#94a3b8',
                    fontSize: '12px',
                    userSelect: 'none',
                  }}
                >
                  {separator}
                </li>
              )}
            </React.Fragment>
          );
        })}
      </ol>
    </nav>
  );
};

```

### 5. Integration with React Router & Dynamic Path Generators

This helper takes current URL pathnames (e.g. `/projects/cloud-native/settings`) and constructs an accessible Breadcrumbs stack using router links.

```tsx
// adapters/ReactRouterBreadcrumbs.tsx
import React, { forwardRef } from 'react';
import { Breadcrumbs } from '../components/Breadcrumbs';
import { BreadcrumbLink } from '../components/BreadcrumbLink';

// Simulated React Router Link Mock
export const RouterLink = forwardRef<HTMLAnchorElement, any>(
  ({ to, children, ...props }, ref) => (
    <a
      ref={ref}
      href={to}
      onClick={(e) => e.preventDefault()} // SPA navigation mock
      {...props}
    >
      {children}
    </a>
  )
);
RouterLink.displayName = 'RouterLink';

// Route Name Formatter Helper
const ROUTE_NAME_MAP: Record<string, string> = {
  projects: 'Projects',
  'cloud-native': 'Cloud Infrastructure',
  deployments: 'Deployments',
  settings: 'Cluster Settings',
};

export const DynamicRouteBreadcrumbs: React.FC<{ pathname: string }> = ({ pathname }) => {
  const pathnames = pathname.split('/').filter((x) => x);

  return (
    <Breadcrumbs separator="›">
      <BreadcrumbLink as={RouterLink} to="/" icon={<span>🏠</span>}>
        Home
      </BreadcrumbLink>

      {pathnames.map((value, index) => {
        const to = `/${pathnames.slice(0, index + 1).join('/')}`;
        const isLast = index === pathnames.length - 1;
        const label = ROUTE_NAME_MAP[value] || decodeURIComponent(value);

        return (
          <BreadcrumbLink
            key={to}
            as={RouterLink}
            to={to}
            isCurrent={isLast}
          >
            {label}
          </BreadcrumbLink>
        );
      })}
    </Breadcrumbs>
  );
};

```

### Key Technical Details

| Feature                         | Implementation Mechanism                                                                                                                                                                                             |
| ------------------------------- | -------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------- |
| **WAI-ARIA Breadcrumb Pattern** | Uses `<nav aria-label="Breadcrumb">` containing an ordered list (`<ol>`), identifying active destinations via `aria-current="page"` and hiding decorative separators from screen readers using `aria-hidden="true"`. |
| **Middle-Item Slicing**         | Computes array bounds based on `maxItems`, `itemsBeforeCollapse`, and `itemsAfterCollapse` to isolate middle items while preserving root and target leaves.                                                          |
| **Condensed Menu Portal**       | In condensed mode, truncated items render into a lightweight, accessible floating dropdown menu (`role="menu"` / `role="menuitem"`).                                                                                 |
| **Polymorphic Routing**         | `BreadcrumbLink` supports an `as` prop with `forwardRef` to render router `<Link>` components seamlessly.                                                                                                            |
