### 1. Positioning & Collision Engine

Handles anchor alignment, collision detection near viewport boundaries, context menu coordinates, and scrolling position logic.

```typescript
// utils/menuPositioning.ts
export type MenuPlacement =
  | 'bottom-start' | 'bottom-end' | 'bottom-center'
  | 'top-start' | 'top-end' | 'top-center'
  | 'right-start' | 'left-start';

export interface VirtualAnchor {
  getBoundingClientRect: () => DOMRect;
}

export function computeMenuPosition(
  anchor: HTMLElement | VirtualAnchor,
  menu: HTMLElement,
  placement: MenuPlacement = 'bottom-start',
  offset = 4
): { top: number; left: number; transformOrigin: string } {
  const anchorRect = anchor.getBoundingClientRect();
  const menuRect = menu.getBoundingClientRect();
  const viewportW = window.innerWidth;
  const viewportH = window.innerHeight;

  let [vertical, horizontal] = placement.split('-') as ['bottom' | 'top' | 'right' | 'left', string];

  // 1. Collision Check
  if (vertical === 'bottom' && anchorRect.bottom + menuRect.height + offset > viewportH) {
    vertical = 'top';
  } else if (vertical === 'top' && anchorRect.top - menuRect.height - offset < 0) {
    vertical = 'bottom';
  }

  let top = 0;
  let left = 0;

  // 2. Compute Vertical Anchor Coordinates
  if (vertical === 'bottom') {
    top = anchorRect.bottom + offset;
  } else if (vertical === 'top') {
    top = anchorRect.top - menuRect.height - offset;
  } else if (vertical === 'right') {
    left = anchorRect.right + offset;
    top = anchorRect.top;
  } else if (vertical === 'left') {
    left = anchorRect.left - menuRect.width - offset;
    top = anchorRect.top;
  }

  // 3. Compute Horizontal Alignment
  if (vertical === 'bottom' || vertical === 'top') {
    if (horizontal === 'end') {
      left = anchorRect.right - menuRect.width;
    } else if (horizontal === 'center') {
      left = anchorRect.left + (anchorRect.width - menuRect.width) / 2;
    } else {
      left = anchorRect.left;
    }
  }

  // 4. Viewport Clamping
  left = Math.max(8, Math.min(left, viewportW - menuRect.width - 8));
  top = Math.max(8, Math.min(top, viewportH - menuRect.height - 8));

  const originY = vertical === 'bottom' ? 'top' : 'bottom';
  const originX = horizontal === 'end' ? 'right' : horizontal === 'center' ? 'center' : 'left';

  return {
    top: top + window.scrollY,
    left: left + window.scrollX,
    transformOrigin: `${originY} ${originX}`,
  };
}

```

### 2. Core Menu Components

```tsx
// components/Menu.tsx
import React, { createContext, useContext, useEffect, useRef, useState, useId } from 'react';
import { createPortal } from 'react-dom';
import { MenuPlacement, VirtualAnchor, computeMenuPosition } from '../utils/menuPositioning';

interface MenuContextValue {
  dense?: boolean;
  onClose?: () => void;
  focusedIndex: number;
  setFocusedIndex: React.Dispatch<React.SetStateAction<number>>;
}

const MenuContext = createContext<MenuContextValue>({
  dense: false,
  focusedIndex: -1,
  setFocusedIndex: () => {},
});

// ==========================================
// 1. MENU CONTAINER
// ==========================================
export interface MenuProps {
  open: boolean;
  anchorEl?: HTMLElement | null;
  virtualAnchor?: VirtualAnchor | null;
  onClose: () => void;
  placement?: MenuPlacement;
  dense?: boolean;
  maxHeight?: number | string;
  transition?: 'scale' | 'fade' | 'slide';
  children: React.ReactNode;
  className?: string;
  style?: React.CSSProperties;
}

export const Menu: React.FC<MenuProps> = ({
  open,
  anchorEl,
  virtualAnchor,
  onClose,
  placement = 'bottom-start',
  dense = false,
  maxHeight,
  transition = 'scale',
  children,
  className,
  style,
}) => {
  const menuRef = useRef<HTMLDivElement>(null);
  const [pos, setPos] = useState({ top: -9999, left: -9999, transformOrigin: 'top left' });
  const [focusedIndex, setFocusedIndex] = useState(-1);
  const menuId = useId();

  const updatePos = () => {
    const target = virtualAnchor || anchorEl;
    if (!target || !menuRef.current) return;
    const computed = computeMenuPosition(target, menuRef.current, placement);
    setPos(computed);
  };

  useEffect(() => {
    if (open) {
      updatePos();
      setFocusedIndex(0);
      const onScrollOrResize = () => updatePos();
      window.addEventListener('scroll', onScrollOrResize, { passive: true });
      window.addEventListener('resize', onScrollOrResize);
      return () => {
        window.removeEventListener('scroll', onScrollOrResize);
        window.removeEventListener('resize', onScrollOrResize);
      };
    }
  }, [open, anchorEl, virtualAnchor, placement]);

  // Click Outside & Escape dismiss
  useEffect(() => {
    if (!open) return;
    const handleOutside = (e: MouseEvent) => {
      if (
        menuRef.current &&
        !menuRef.current.contains(e.target as Node) &&
        !anchorEl?.contains(e.target as Node)
      ) {
        onClose();
      }
    };
    const handleKeyDown = (e: KeyboardEvent) => {
      if (e.key === 'Escape') {
        e.preventDefault();
        onClose();
      }
    };
    document.addEventListener('mousedown', handleOutside);
    document.addEventListener('keydown', handleKeyDown);
    return () => {
      document.removeEventListener('mousedown', handleOutside);
      document.removeEventListener('keydown', handleKeyDown);
    };
  }, [open, anchorEl, onClose]);

  // Keyboard navigation across items
  const handleKeyDown = (e: React.KeyboardEvent) => {
    if (!menuRef.current) return;
    const items = Array.from(
      menuRef.current.querySelectorAll<HTMLElement>('[role^="menuitem"]:not([aria-disabled="true"])')
    );
    if (!items.length) return;

    if (e.key === 'ArrowDown') {
      e.preventDefault();
      setFocusedIndex((prev) => (prev + 1) % items.length);
    } else if (e.key === 'ArrowUp') {
      e.preventDefault();
      setFocusedIndex((prev) => (prev - 1 + items.length) % items.length);
    } else if (e.key === 'Home') {
      e.preventDefault();
      setFocusedIndex(0);
    } else if (e.key === 'End') {
      e.preventDefault();
      setFocusedIndex(items.length - 1);
    }
  };

  if (!open) return null;

  return createPortal(
    <MenuContext.Provider value={{ dense, onClose, focusedIndex, setFocusedIndex }}>
      <div
        id={menuId}
        ref={menuRef}
        role="menu"
        tabIndex={-1}
        onKeyDown={handleKeyDown}
        className={`custom-menu ${transition} ${className || ''}`}
        style={{
          position: 'absolute',
          top: pos.top,
          left: pos.left,
          transformOrigin: pos.transformOrigin,
          backgroundColor: '#ffffff',
          border: '1px solid #e2e8f0',
          borderRadius: 8,
          boxShadow: '0 10px 15px -3px rgba(0, 0, 0, 0.1), 0 4px 6px -4px rgba(0, 0, 0, 0.1)',
          padding: '4px 0',
          minWidth: 180,
          maxHeight: maxHeight || 'auto',
          overflowY: maxHeight ? 'auto' : 'visible',
          zIndex: 1000,
          fontFamily: 'Inter, system-ui, sans-serif',
          outline: 'none',
          boxSizing: 'border-box',
          ...style,
        }}
      >
        {children}
      </div>

      <style>{`
        .custom-menu.scale {
          animation: menuScale 0.15s cubic-bezier(0, 0, 0.2, 1);
        }
        .custom-menu.fade {
          animation: menuFade 0.15s ease-out;
        }
        .custom-menu.slide {
          animation: menuSlide 0.15s cubic-bezier(0, 0, 0.2, 1);
        }
        @keyframes menuScale {
          from { opacity: 0; transform: scale(0.92); }
          to { opacity: 1; transform: scale(1); }
        }
        @keyframes menuFade {
          from { opacity: 0; }
          to { opacity: 1; }
        }
        @keyframes menuSlide {
          from { opacity: 0; transform: translateY(-8px); }
          to { opacity: 1; transform: translateY(0); }
        }
      `}</style>
    </MenuContext.Provider>,
    document.body
  );
};

// ==========================================
// 2. MENU ITEM
// ==========================================
export interface MenuItemProps extends React.HTMLAttributes<HTMLDivElement> {
  disabled?: boolean;
  selected?: boolean;
  dense?: boolean;
  inset?: boolean;
  icon?: React.ReactNode;
  shortcut?: string;
}

export const MenuItem: React.FC<MenuItemProps> = ({
  children,
  disabled = false,
  selected = false,
  dense: itemDense,
  inset = false,
  icon,
  shortcut,
  onClick,
  style,
  className,
  ...restProps
}) => {
  const { dense: groupDense, onClose } = useContext(MenuContext);
  const dense = itemDense !== undefined ? itemDense : groupDense;

  const handleClick = (e: React.MouseEvent<HTMLDivElement>) => {
    if (disabled) return;
    onClick?.(e);
    onClose?.();
  };

  return (
    <div
      role="menuitem"
      aria-disabled={disabled}
      onClick={handleClick}
      className={`custom-menu-item ${selected ? 'selected' : ''} ${className || ''}`}
      style={{
        display: 'flex',
        alignItems: 'center',
        padding: `${dense ? '5px' : '8px'} 12px`,
        paddingLeft: inset ? 36 : 12,
        fontSize: dense ? '12px' : '13px',
        fontWeight: selected ? 600 : 500,
        color: disabled ? '#94a3b8' : selected ? '#2563eb' : '#1e293b',
        backgroundColor: selected ? '#eff6ff' : 'transparent',
        cursor: disabled ? 'not-allowed' : 'pointer',
        userSelect: 'none',
        outline: 'none',
        transition: 'background-color 0.12s ease',
        ...style,
      }}
      onMouseEnter={(e) => {
        if (!disabled && !selected) e.currentTarget.style.backgroundColor = '#f8fafc';
      }}
      onMouseLeave={(e) => {
        if (!disabled && !selected) e.currentTarget.style.backgroundColor = 'transparent';
      }}
      {...restProps}
    >
      {icon && (
        <span
          style={{
            marginRight: 10,
            display: 'inline-flex',
            alignItems: 'center',
            fontSize: dense ? 14 : 16,
            color: disabled ? '#94a3b8' : '#64748b',
          }}
        >
          {icon}
        </span>
      )}

      <span style={{ flex: 1, whiteSpace: 'nowrap', overflow: 'hidden', textOverflow: 'ellipsis' }}>
        {children}
      </span>

      {shortcut && (
        <span style={{ marginLeft: 16, fontSize: '11px', color: '#94a3b8', letterSpacing: '0.04em' }}>
          {shortcut}
        </span>
      )}
    </div>
  );
};
```

### Architectural & Accessibility Details

* **WAI-ARIA 1.2 Hierarchy**: Root container renders `role="menu"`, items render `role="menuitem"`, checkboxes declare `role="menuitemcheckbox"` with `aria-checked`, and radio buttons use `role="menuitemradio"`.
* **Keyboard Loop & Focus Management**: Arrow keys (`ArrowUp`/`ArrowDown`) traverse items, skipping disabled rows. `Escape` dismisses the menu and restores focus to the trigger anchor.
* **Collision-Aware Flipping**: Automatically flips axes from `bottom` to `top` if menu height collides with viewport bounds.
* **Virtual Positioning**: Supports mouse coordinate anchoring (`clientX`/`clientY`) for custom desktop-grade context menus.
