Here is a complete, production-ready, zero-dependency implementation of the **Stepper Component System** (Horizontal, Vertical, Non-Linear, Alternative Label, Collapse Transitions, Custom Icons, Error States, and Mobile Steppers with Text/Dots/Progress) built from scratch in React with TypeScript.

### 1. Types & Context Engine

```tsx
// types/stepper.ts
import React from 'react';

export type StepperOrientation = 'horizontal' | 'vertical';
export type MobileStepperVariant = 'text' | 'dots' | 'progress';

export interface StepperContextValue {
  activeStep: number;
  orientation: StepperOrientation;
  alternativeLabel: boolean;
  nonLinear: boolean;
  connector?: React.ReactNode;
  isStepOptional?: (step: number) => boolean;
  isStepFailed?: (step: number) => boolean;
  isStepCompleted?: (step: number) => boolean;
}

export interface StepContextValue {
  index: number;
  last: boolean;
  expanded: boolean;
  icon?: React.ReactNode;
  active: boolean;
  completed: boolean;
  disabled: boolean;
  error: boolean;
  optional: boolean;
}

```

```tsx
// context/StepperContext.tsx
import React, { createContext, useContext } from 'react';
import { StepperContextValue, StepContextValue } from '../types/stepper';

export const StepperContext = createContext<StepperContextValue | null>(null);
export const StepContext = createContext<StepContextValue | null>(null);

export const useStepper = () => {
  const context = useContext(StepperContext);
  if (!context) throw new Error('useStepper must be used within a <Stepper />');
  return context;
};

export const useStep = () => {
  const context = useContext(StepContext);
  if (!context) throw new Error('useStep must be used within a <Step />');
  return context;
};
```

#### StepIcon & StepConnector

```tsx
// components/StepPrimitives.tsx
import React from 'react';
import { useStepper, useStep } from '../context/StepperContext';

export const StepIcon: React.FC<{
  icon?: React.ReactNode;
  active?: boolean;
  completed?: boolean;
  error?: boolean;
  customIcon?: React.ReactNode;
}> = ({ icon, active, completed, error, customIcon }) => {
  if (customIcon) return <>{customIcon}</>;

  const getBackgroundColor = () => {
    if (error) return '#dc2626';
    if (completed || active) return '#2563eb';
    return '#cbd5e1';
  };

  return (
    <div
      style={{
        width: 28,
        height: 28,
        borderRadius: '50%',
        backgroundColor: getBackgroundColor(),
        color: '#ffffff',
        display: 'flex',
        alignItems: 'center',
        justifyContent: 'center',
        fontSize: 13,
        fontWeight: 700,
        flexShrink: 0,
        transition: 'all 0.2s ease',
      }}
    >
      {error ? '!' : completed ? '✓' : icon}
    </div>
  );
};

export const StepConnector: React.FC<{ style?: React.CSSProperties }> = ({ style }) => {
  const { orientation, alternativeLabel } = useStepper();
  const { active, completed } = useStep();

  const isVertical = orientation === 'vertical';

  if (isVertical) {
    return (
      <div
        style={{
          marginLeft: 14,
          paddingLeft: 14,
          borderLeft: `2px solid ${completed ? '#2563eb' : '#e2e8f0'}`,
          minHeight: 24,
          transition: 'border-color 0.2s ease',
          ...style,
        }}
      />
    );
  }

  return (
    <div
      style={{
        flex: '1 1 auto',
        borderTop: `2px solid ${completed ? '#2563eb' : '#e2e8f0'}`,
        margin: alternativeLabel ? '0' : '0 8px',
        position: alternativeLabel ? 'absolute' : 'relative',
        top: alternativeLabel ? 14 : 'auto',
        left: alternativeLabel ? 'calc(-50% + 20px)' : 'auto',
        right: alternativeLabel ? 'calc(50% + 20px)' : 'auto',
        transition: 'border-color 0.2s ease',
        ...style,
      }}
    />
  );
};
```

#### StepLabel, StepButton, Step & Stepper

```tsx
// components/Stepper.tsx
import React, { Children, isValidElement, cloneElement } from 'react';
import { StepperContext, StepContext, useStepper, useStep } from '../context/StepperContext';
import { StepperOrientation } from '../types/stepper';
import { StepIcon, StepConnector } from './StepPrimitives';

// ==========================================
// 1. STEPPER ROOT
// ==========================================
export interface StepperProps extends React.HTMLAttributes<HTMLDivElement> {
  activeStep?: number;
  orientation?: StepperOrientation;
  alternativeLabel?: boolean;
  nonLinear?: boolean;
  connector?: React.ReactNode;
  children: React.ReactNode;
}

export const Stepper: React.FC<StepperProps> = ({
  activeStep = 0,
  orientation = 'horizontal',
  alternativeLabel = false,
  nonLinear = false,
  connector = <StepConnector />,
  children,
  style,
  className,
  ...rest
}) => {
  const childrenArray = Children.toArray(children).filter(isValidElement);

  return (
    <StepperContext.Provider
      value={{
        activeStep,
        orientation,
        alternativeLabel: orientation === 'vertical' ? false : alternativeLabel,
        nonLinear,
        connector,
      }}
    >
      <div
        className={`stepper-root ${className || ''}`}
        style={{
          display: 'flex',
          flexDirection: orientation === 'vertical' ? 'column' : 'row',
          alignItems: orientation === 'vertical' || alternativeLabel ? 'stretch' : 'center',
          width: '100%',
          fontFamily: 'Inter, system-ui, sans-serif',
          ...style,
        }}
        {...rest}
      >
        {childrenArray.map((child, index) =>
          cloneElement(child as React.ReactElement<any>, {
            index,
            last: index === childrenArray.length - 1,
          })
        )}
      </div>
    </StepperContext.Provider>
  );
};

// ==========================================
// 2. STEP
// ==========================================
export interface StepProps extends React.HTMLAttributes<HTMLDivElement> {
  index?: number;
  last?: boolean;
  active?: boolean;
  completed?: boolean;
  disabled?: boolean;
  expanded?: boolean;
  children: React.ReactNode;
}

export const Step: React.FC<StepProps> = ({
  index = 0,
  last = false,
  active: activeProp,
  completed: completedProp,
  disabled: disabledProp,
  expanded = false,
  children,
  style,
  ...rest
}) => {
  const { activeStep, orientation, alternativeLabel, connector } = useStepper();

  const active = activeProp !== undefined ? activeProp : activeStep === index;
  const completed = completedProp !== undefined ? completedProp : activeStep > index;
  const disabled = disabledProp !== undefined ? disabledProp : activeStep < index && !active;

  return (
    <StepContext.Provider
      value={{
        index,
        last,
        expanded,
        active,
        completed,
        disabled,
        error: false,
        optional: false,
      }}
    >
      <div
        style={{
          flex: orientation === 'horizontal' ? 1 : 'none',
          position: 'relative',
          display: 'flex',
          flexDirection: orientation === 'vertical' ? 'column' : 'row',
          alignItems: orientation === 'vertical' || alternativeLabel ? 'stretch' : 'center',
          ...style,
        }}
        {...rest}
      >
        {/* Connector for horizontal layout */}
        {index !== 0 && orientation === 'horizontal' && connector}
        <div
          style={{
            display: 'flex',
            flexDirection: alternativeLabel ? 'column' : 'row',
            alignItems: alternativeLabel ? 'center' : 'stretch',
            width: '100%',
          }}
        >
          {children}
        </div>
      </div>
    </StepContext.Provider>
  );
};

// ==========================================
// 3. STEP LABEL
// ==========================================
export interface StepLabelProps extends React.HTMLAttributes<HTMLDivElement> {
  optional?: React.ReactNode;
  error?: boolean;
  icon?: React.ReactNode;
  StepIconComponent?: React.ElementType;
}

export const StepLabel: React.FC<StepLabelProps> = ({
  children,
  optional,
  error = false,
  icon: customIcon,
  StepIconComponent,
  style,
  ...rest
}) => {
  const { alternativeLabel } = useStepper();
  const { index, active, completed } = useStep();

  return (
    <div
      style={{
        display: 'flex',
        alignItems: 'center',
        flexDirection: alternativeLabel ? 'column' : 'row',
        textAlign: alternativeLabel ? 'center' : 'left',
        gap: 10,
        ...style,
      }}
      {...rest}
    >
      {StepIconComponent ? (
        <StepIconComponent
          icon={index + 1}
          active={active}
          completed={completed}
          error={error}
        />
      ) : (
        <StepIcon
          icon={index + 1}
          active={active}
          completed={completed}
          error={error}
          customIcon={customIcon}
        />
      )}

      <div style={{ display: 'flex', flexDirection: 'column' }}>
        <span
          style={{
            fontSize: 14,
            fontWeight: active ? 700 : 500,
            color: error ? '#dc2626' : active ? '#0f172a' : '#64748b',
            lineHeight: 1.3,
          }}
        >
          {children}
        </span>
        {optional && (
          <span style={{ fontSize: 11, color: error ? '#dc2626' : '#94a3b8', marginTop: 2 }}>
            {optional}
          </span>
        )}
      </div>
    </div>
  );
};

// ==========================================
// 4. STEP BUTTON (Non-Linear Steppers)
// ==========================================
export const StepButton: React.FC<
  React.ButtonHTMLAttributes<HTMLButtonElement> & { icon?: React.ReactNode; optional?: React.ReactNode }
> = ({ children, icon, optional, onClick, style, ...rest }) => {
  return (
    <button
      type="button"
      onClick={onClick}
      style={{
        background: 'none',
        border: 'none',
        padding: '6px 8px',
        cursor: 'pointer',
        display: 'flex',
        alignItems: 'center',
        borderRadius: 6,
        outline: 'none',
        transition: 'background-color 0.15s ease',
        ...style,
      }}
      onMouseEnter={(e) => (e.currentTarget.style.backgroundColor = '#f1f5f9')}
      onMouseLeave={(e) => (e.currentTarget.style.backgroundColor = 'transparent')}
      {...rest}
    >
      <StepLabel optional={optional} icon={icon}>
        {children}
      </StepLabel>
    </button>
  );
};
```

### 3. StepContent (Vertical Collapse Transition & Search Performance)

```tsx
// components/StepContent.tsx
import React, { useRef } from 'react';
import { useStep, useStepper } from '../context/StepperContext';

export interface StepContentProps extends React.HTMLAttributes<HTMLDivElement> {
  unmountOnExit?: boolean; // When false, keeps DOM rendered for SEO/crawlers
  children: React.ReactNode;
}

export const StepContent: React.FC<StepContentProps> = ({
  unmountOnExit = true,
  children,
  style,
  ...rest
}) => {
  const { orientation } = useStepper();
  const { active, last } = useStep();
  const contentRef = useRef<HTMLDivElement>(null);

  if (orientation !== 'vertical') return null;

  const isExpanded = active;

  if (!isExpanded && unmountOnExit) {
    return (
      <div
        style={{
          marginLeft: 14,
          paddingLeft: 24,
          borderLeft: last ? 'none' : '2px solid #e2e8f0',
        }}
      />
    );
  }

  return (
    <div
      style={{
        marginLeft: 14,
        paddingLeft: 24,
        borderLeft: last ? 'none' : '2px solid #e2e8f0',
        overflow: 'hidden',
        transition: 'max-height 0.3s cubic-bezier(0.4, 0, 0.2, 1), opacity 0.3s ease',
        maxHeight: isExpanded ? 1000 : 0,
        opacity: isExpanded ? 1 : 0,
        ...style,
      }}
      {...rest}
    >
      <div ref={contentRef} style={{ padding: '8px 0 16px 0' }}>
        {children}
      </div>
    </div>
  );
};
```

### 4. MobileStepper (Text, Dots, Progress)

```tsx
// components/MobileStepper.tsx
import React from 'react';
import { MobileStepperVariant } from '../types/stepper';

export interface MobileStepperProps extends React.HTMLAttributes<HTMLDivElement> {
  steps: number;
  activeStep: number;
  variant?: MobileStepperVariant;
  nextButton: React.ReactNode;
  backButton: React.ReactNode;
  position?: 'bottom' | 'top' | 'static';
}

export const MobileStepper: React.FC<MobileStepperProps> = ({
  steps,
  activeStep,
  variant = 'dots',
  nextButton,
  backButton,
  position = 'static',
  style,
  className,
  ...rest
}) => {
  const progressPercent = Math.min(100, Math.max(0, ((activeStep + 1) / steps) * 100));

  return (
    <div
      className={`mobile-stepper ${className || ''}`}
      style={{
        display: 'flex',
        alignItems: 'center',
        justifyContent: 'space-between',
        padding: '10px 16px',
        backgroundColor: '#ffffff',
        border: '1px solid #e2e8f0',
        borderRadius: position === 'static' ? 8 : 0,
        fontFamily: 'Inter, system-ui, sans-serif',
        ...(position === 'bottom'
          ? { position: 'fixed', bottom: 0, left: 0, right: 0, zIndex: 1000, boxShadow: '0 -2px 10px rgba(0,0,0,0.05)' }
          : position === 'top'
          ? { position: 'fixed', top: 0, left: 0, right: 0, zIndex: 1000 }
          : {}),
        ...style,
      }}
      {...rest}
    >
      {backButton}

      {/* Center Progress Indicator */}
      <div style={{ flex: 1, display: 'flex', justifyContent: 'center', padding: '0 16px' }}>
        {variant === 'text' && (
          <span style={{ fontSize: 13, fontWeight: 600, color: '#475569' }}>
            {activeStep + 1} / {steps}
          </span>
        )}

        {variant === 'dots' && (
          <div style={{ display: 'flex', gap: 6, alignItems: 'center' }}>
            {Array.from({ length: steps }).map((_, index) => (
              <div
                key={index}
                style={{
                  width: index === activeStep ? 16 : 6,
                  height: 6,
                  borderRadius: 3,
                  backgroundColor: index === activeStep ? '#2563eb' : '#cbd5e1',
                  transition: 'all 0.2s cubic-bezier(0.4, 0, 0.2, 1)',
                }}
              />
            ))}
          </div>
        )}

        {variant === 'progress' && (
          <div
            style={{
              width: '100%',
              maxWidth: 200,
              height: 6,
              borderRadius: 3,
              backgroundColor: '#e2e8f0',
              overflow: 'hidden',
            }}
          >
            <div
              style={{
                width: `${progressPercent}%`,
                height: '100%',
                backgroundColor: '#2563eb',
                borderRadius: 3,
                transition: 'width 0.3s ease-in-out',
              }}
            />
          </div>
        )}
      </div>

      {nextButton}
    </div>
  );
};
```

### Architectural Highlights

| Feature                            | Implementation Technique                                                                                                                                        |
| ---------------------------------- | --------------------------------------------------------------------------------------------------------------------------------------------------------------- |
| **Linear & Non-Linear Execution**  | Built on hierarchical React Context (`StepperContext` & `StepContext`). `StepButton` permits random indexing without enforcing sequential progression.          |
| **Alternative Labels**             | When `alternativeLabel={true}`, `StepConnector` is absolutely anchored between sibling nodes (`calc(50% + 20px)`) and labels switch to centered column layout.  |
| **SEO & Performance Optimization** | `StepContent` supports `unmountOnExit={false}`, rendering collapsed step DOM trees (`max-height: 0`, `opacity: 0`) so search engine indexers can crawl content. |
| **Mobile Form Factors**            | `MobileStepper` abstracts progression into 3 compact indicators (`text`, expandable `dots`, and CSS transition width `progress` bars).                          |
