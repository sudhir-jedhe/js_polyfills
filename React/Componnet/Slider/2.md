A **Multi-Slider** (or Multi-Thumb / Range Slider) typically serves one of three common use cases:

| Variant                            | Thumb Count          | How It Works                                                                           | Common Applications                                                                        |
| ---------------------------------- | -------------------- | -------------------------------------------------------------------------------------- | ------------------------------------------------------------------------------------------ |
| **Dual Range Slider**              | 2 Thumbs             | Selects a continuous bounded interval $[min, max]$.                                    | Price ranges, date/time spans, age filters.                                                |
| **N-Thumb Segment Slider**         | 3+ Thumbs            | Divides a track into distinct contiguous segments where each thumb acts as a boundary. | Resource allocation, multi-tier pricing brackets, audio split-points, budget distribution. |
| **Independent Multi-Axis Sliders** | N Thumbs / N Sliders | Multiple synchronized or grouped parallel sliders controlling related vector values.   | Graphic EQ bands, RGB/HSL color mixers, 3D physics weights.                                |

### Implementation: N-Thumb Segment Multi-Slider from Scratch

Here is a zero-dependency React + TypeScript implementation supporting an arbitrary number of thumbs ($N \ge 2$), automatic collision prevention, active segment coloring, and keyboard navigation.

```tsx
import React, { useState, useRef, useCallback } from 'react';

export interface MultiSliderProps {
  values?: number[];
  defaultValue?: number[];
  min?: number;
  max?: number;
  step?: number;
  minDistance?: number;
  colors?: string[]; // Background colors for each segment between thumbs
  onChange?: (values: number[]) => void;
  formatValue?: (val: number) => string;
}

export const MultiSlider: React.FC<MultiSliderProps> = ({
  values: controlledValues,
  defaultValue = [20, 50, 80],
  min = 0,
  max = 100,
  step = 1,
  minDistance = 0,
  colors = ['#3b82f6', '#10b981', '#f59e0b', '#ec4899'],
  onChange,
  formatValue = (v) => String(v),
}) => {
  const [internalValues, setInternalValues] = useState<number[]>(defaultValue);
  const [activeThumbIdx, setActiveThumbIdx] = useState<number | null>(null);
  const trackRef = useRef<HTMLDivElement>(null);

  const activeValues = controlledValues ?? internalValues;

  const toPct = useCallback(
    (val: number) => Math.max(0, Math.min(100, ((val - min) / (max - min)) * 100)),
    [min, max]
  );

  const updateValue = (index: number, rawVal: number) => {
    // Quantize to step
    const stepped = Math.round((rawVal - min) / step) * step + min;

    // Constrain by neighbors to prevent thumb crossover
    const lowerBound = index === 0 ? min : activeValues[index - 1] + minDistance;
    const upperBound = index === activeValues.length - 1 ? max : activeValues[index + 1] - minDistance;
    const clamped = Math.max(lowerBound, Math.min(upperBound, stepped));

    const next = [...activeValues];
    next[index] = clamped;

    if (controlledValues === undefined) setInternalValues(next);
    onChange?.(next);
  };

  const handlePointerDown = (e: React.PointerEvent<HTMLDivElement>) => {
    if (!trackRef.current) return;
    trackRef.current.setPointerCapture(e.pointerId);

    const rect = trackRef.current.getBoundingClientRect();
    const clickPct = Math.max(0, Math.min(1, (e.clientX - rect.left) / rect.width));
    const clickVal = min + clickPct * (max - min);

    // Find closest thumb
    let closestIdx = 0;
    let minDiff = Infinity;
    activeValues.forEach((v, idx) => {
      const diff = Math.abs(v - clickVal);
      if (diff < minDiff) {
        minDiff = diff;
        closestIdx = idx;
      }
    });

    setActiveThumbIdx(closestIdx);
    updateValue(closestIdx, clickVal);
  };

  const handlePointerMove = (e: React.PointerEvent<HTMLDivElement>) => {
    if (activeThumbIdx === null || !trackRef.current) return;
    const rect = trackRef.current.getBoundingClientRect();
    const pct = Math.max(0, Math.min(1, (e.clientX - rect.left) / rect.width));
    const val = min + pct * (max - min);
    updateValue(activeThumbIdx, val);
  };

  const handlePointerUp = (e: React.PointerEvent<HTMLDivElement>) => {
    if (trackRef.current && activeThumbIdx !== null) {
      try {
        trackRef.current.releasePointerCapture(e.pointerId);
      } catch {}
      setActiveThumbIdx(null);
    }
  };

  const handleKeyDown = (e: React.KeyboardEvent, index: number) => {
    const delta = e.shiftKey ? step * 10 : step;
    if (e.key === 'ArrowRight' || e.key === 'ArrowUp') {
      e.preventDefault();
      updateValue(index, activeValues[index] + delta);
    } else if (e.key === 'ArrowLeft' || e.key === 'ArrowDown') {
      e.preventDefault();
      updateValue(index, activeValues[index] - delta);
    }
  };

  // Build segments: [0 to thumb0], [thumb0 to thumb1], ..., [thumbN to 100%]
  const boundaries = [0, ...activeValues.map(toPct), 100];

  return (
    <div style={{ width: '100%', padding: '24px 12px', userSelect: 'none', touchAction: 'none' }}>
      <div
        ref={trackRef}
        onPointerDown={handlePointerDown}
        onPointerMove={handlePointerMove}
        onPointerUp={handlePointerUp}
        onPointerCancel={handlePointerUp}
        style={{
          position: 'relative',
          height: 8,
          borderRadius: 4,
          backgroundColor: '#e2e8f0',
          cursor: 'pointer',
        }}
      >
        {/* Render Segment Bands */}
        {boundaries.slice(0, -1).map((startPct, i) => {
          const endPct = boundaries[i + 1];
          return (
            <div
              key={`segment-${i}`}
              style={{
                position: 'absolute',
                left: `${startPct}%`,
                width: `${endPct - startPct}%`,
                height: '100%',
                backgroundColor: colors[i % colors.length],
                borderRadius: 4,
              }}
            />
          );
        })}

        {/* Render Thumbs */}
        {activeValues.map((val, idx) => {
          const pct = toPct(val);
          const isFocused = activeThumbIdx === idx;

          return (
            <div
              key={`thumb-${idx}`}
              role="slider"
              tabIndex={0}
              aria-label={`Thumb ${idx + 1}`}
              aria-valuenow={val}
              aria-valuemin={idx === 0 ? min : activeValues[idx - 1] + minDistance}
              aria-valuemax={idx === activeValues.length - 1 ? max : activeValues[idx + 1] - minDistance}
              onKeyDown={(e) => handleKeyDown(e, idx)}
              style={{
                position: 'absolute',
                left: `${pct}%`,
                top: '50%',
                transform: 'translate(-50%, -50%)',
                width: 18,
                height: 18,
                borderRadius: '50%',
                backgroundColor: '#ffffff',
                border: `2px solid ${colors[idx % colors.length]}`,
                boxShadow: isFocused ? '0 0 0 4px rgba(59, 130, 246, 0.2)' : '0 2px 5px rgba(0,0,0,0.15)',
                outline: 'none',
                cursor: 'grab',
                zIndex: isFocused ? 10 : 5,
              }}
            >
              {/* Floating Value Indicator */}
              <div
                style={{
                  position: 'absolute',
                  bottom: '100%',
                  left: '50%',
                  transform: 'translate(-50%, -6px)',
                  backgroundColor: '#0f172a',
                  color: '#ffffff',
                  padding: '2px 6px',
                  borderRadius: 4,
                  fontSize: 11,
                  fontWeight: 600,
                  whiteSpace: 'nowrap',
                  pointerEvents: 'none',
                }}
              >
                {formatValue(val)}
              </div>
            </div>
          );
        })}
      </div>
    </div>
  );
};

```
