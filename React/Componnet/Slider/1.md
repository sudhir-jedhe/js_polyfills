### 1. Mathematics & Accessibility Engine

This module contains:

1. **Coordinate Conversion**: Bounding-box projections for horizontal/vertical axes.
2. **Non-Linear Transformations**: Power/Logarithmic mathematical functions (e.g., volume, frequencies, storage tiers).
3. **Step Quantization & Value Restriction**: Decimal precision management and mark snapping.
4. **Collision & Distance Constraints**: Multi-thumb collision avoidance and minimum range separation.

```typescript
// utils/sliderMath.ts

export type NonLinearScale = {
  toValue: (percentage: number, min: number, max: number) => number;
  toPercentage: (value: number, min: number, max: number) => number;
};

// Power curve scale (useful for volume and sensitivity)
export const PowerScale: NonLinearScale = {
  toValue: (pct, min, max) => {
    const normalized = Math.pow(pct, 2);
    return min + normalized * (max - min);
  },
  toPercentage: (val, min, max) => {
    const raw = Math.max(0, Math.min(1, (val - min) / (max - min)));
    return Math.sqrt(raw);
  },
};

// Logarithmic scale (useful for frequencies, storage, byte allocations)
export const LogScale: NonLinearScale = {
  toValue: (pct, min, max) => {
    const minLog = Math.log(Math.max(1, min));
    const maxLog = Math.log(max);
    return Math.exp(minLog + pct * (maxLog - minLog));
  },
  toPercentage: (val, min, max) => {
    const minLog = Math.log(Math.max(1, min));
    const maxLog = Math.log(max);
    const clamped = Math.max(1, val);
    return (Math.log(clamped) - minLog) / (maxLog - minLog);
  },
};

// Clamps value strictly between min and max
export function clamp(val: number, min: number, max: number): number {
  return Math.min(Math.max(val, min), max);
}

// Snaps value to fractional step increments while avoiding floating-point drift
export function quantizeStep(val: number, step: number, min: number): number {
  if (step <= 0) return val;
  const stepDecimals = (step.toString().split('.')[1] || '').length;
  const stepsCount = Math.round((val - min) / step);
  const quantized = min + stepsCount * step;
  return Number(quantized.toFixed(stepDecimals));
}

// Finds the closest discrete valid value (for restricted values/marks)
export function snapToClosest(val: number, allowedValues: number[]): number {
  if (allowedValues.length === 0) return val;
  return allowedValues.reduce((prev, curr) =>
    Math.abs(curr - val) < Math.abs(prev - val) ? curr : prev
  );
}

// Projects mouse/touch screen coordinates to percentage [0, 1]
export function calculatePointerPercentage(
  e: MouseEvent | TouchEvent | React.MouseEvent | React.TouchEvent,
  trackElement: HTMLElement,
  vertical: boolean,
  inverted: boolean
): number {
  const rect = trackElement.getBoundingClientRect();
  const clientX = 'touches' in e ? e.touches[0].clientX : (e as MouseEvent).clientX;
  const clientY = 'touches' in e ? e.touches[0].clientY : (e as MouseEvent).clientY;

  let pct = 0;
  if (vertical) {
    pct = (rect.bottom - clientY) / rect.height; // Default 0 at bottom
  } else {
    pct = (clientX - rect.left) / rect.width; // Default 0 at left
  }

  pct = clamp(pct, 0, 1);
  return inverted ? 1 - pct : pct;
}
```

### 2. Universal Slider Core Component

```tsx
// components/Slider.tsx
import React, { useState, useRef, useEffect, useCallback, useId } from 'react';
import {
  clamp,
  quantizeStep,
  snapToClosest,
  calculatePointerPercentage,
  NonLinearScale,
} from '../utils/sliderMath';

export interface Mark {
  value: number;
  label?: React.ReactNode;
}

export type SliderTrackVariant = 'normal' | 'inverted' | 'removed';
export type SliderSize = 'sm' | 'md' | 'lg';

export interface SliderProps {
  // Value & Mode
  value?: number | [number, number];
  defaultValue?: number | [number, number];
  onChange?: (value: any) => void;
  onChangeCommitted?: (value: any) => void;
  min?: number;
  max?: number;
  step?: number | null; // null disables stepping (pure continuous)

  // Track & Orientation
  vertical?: boolean;
  size?: SliderSize;
  track?: SliderTrackVariant;
  disabled?: boolean;
  color?: string; // CSS color string

  // Discrete & Restrictive Snapping
  marks?: boolean | Mark[];
  restrictToMarks?: boolean;
  marksPlacement?: 'before' | 'after' | 'both';

  // Multi-Thumb Range Behavior
  minDistance?: number;
  pushable?: boolean;

  // Tooltips & Indicators
  alwaysShowValue?: boolean;
  formatValue?: (val: number) => string;

  // Non-linear scales
  scale?: NonLinearScale;

  // Layout & Styling
  style?: React.CSSProperties;
  className?: string;
  'aria-label'?: string;
}

export const Slider: React.FC<SliderProps> = ({
  value: controlledValue,
  defaultValue = 0,
  onChange,
  onChangeCommitted,
  min = 0,
  max = 100,
  step = 1,
  vertical = false,
  size = 'md',
  track = 'normal',
  disabled = false,
  color = '#3b82f6',
  marks = false,
  restrictToMarks = false,
  marksPlacement = 'after',
  minDistance = 0,
  pushable = false,
  alwaysShowValue = false,
  formatValue = (v) => String(v),
  scale,
  style,
  'aria-label': ariaLabel = 'Slider',
}) => {
  const isRange = Array.isArray(controlledValue) || Array.isArray(defaultValue);
  const [internalValue, setInternalValue] = useState<number[]>(() => {
    if (controlledValue !== undefined) {
      return Array.isArray(controlledValue) ? controlledValue : [controlledValue];
    }
    return Array.isArray(defaultValue) ? defaultValue : [defaultValue];
  });

  const activeValues = (controlledValue !== undefined
    ? Array.isArray(controlledValue)
      ? controlledValue
      : [controlledValue]
    : internalValue
  ).map((v) => clamp(v, min, max));

  const [activeThumbIndex, setActiveThumbIndex] = useState<number | null>(null);
  const [hoveredThumb, setHoveredThumb] = useState<number | null>(null);
  const trackRef = useRef<HTMLDivElement>(null);
  const sliderId = useId();

  // Normalize mark definitions
  const parsedMarks: Mark[] = useMemo(() => {
    if (!marks) return [];
    if (Array.isArray(marks)) return marks;
    if (step === null || step <= 0) return [];

    const generated: Mark[] = [];
    for (let val = min; val <= max; val += step) {
      generated.push({ value: val });
    }
    return generated;
  }, [marks, min, max, step]);

  const allowedSnapValues = useMemo(() => {
    return parsedMarks.map((m) => m.value);
  }, [parsedMarks]);

  // Convert absolute value to percentage [0, 100]
  const getPercentage = useCallback(
    (val: number) => {
      if (scale) {
        return scale.toPercentage(val, min, max) * 100;
      }
      return ((val - min) / (max - min)) * 100;
    },
    [min, max, scale]
  );

  // Convert percentage [0, 1] to step/mark aligned absolute value
  const getValueFromPercentage = useCallback(
    (pct: number) => {
      let rawVal = scale ? scale.toValue(pct, min, max) : min + pct * (max - min);

      if (restrictToMarks && allowedSnapValues.length > 0) {
        return snapToClosest(rawVal, allowedSnapValues);
      }
      if (step !== null && step > 0) {
        rawVal = quantizeStep(rawVal, step, min);
      }
      return clamp(rawVal, min, max);
    },
    [scale, min, max, restrictToMarks, allowedSnapValues, step]
  );

  // Update thumb values respecting range collisions and minimum distances
  const updateThumbValue = useCallback(
    (newVal: number, thumbIdx: number, triggerCommitted = false) => {
      if (disabled) return;

      let nextValues = [...activeValues];

      if (!isRange) {
        nextValues = [newVal];
      } else {
        const otherIdx = thumbIdx === 0 ? 1 : 0;
        const otherVal = nextValues[otherIdx];

        if (thumbIdx === 0) {
          if (newVal > otherVal - minDistance) {
            if (pushable) {
              const pushed = Math.min(max, newVal + minDistance);
              nextValues[1] = pushed;
              nextValues[0] = pushed - minDistance;
            } else {
              nextValues[0] = otherVal - minDistance;
            }
          } else {
            nextValues[0] = newVal;
          }
        } else {
          if (newVal < otherVal + minDistance) {
            if (pushable) {
              const pushed = Math.max(min, newVal - minDistance);
              nextValues[0] = pushed;
              nextValues[1] = pushed + minDistance;
            } else {
              nextValues[1] = otherVal + minDistance;
            }
          } else {
            nextValues[1] = newVal;
          }
        }
      }

      if (controlledValue === undefined) {
        setInternalValue(nextValues);
      }

      const out = isRange ? ([nextValues[0], nextValues[1]] as [number, number]) : nextValues[0];
      onChange?.(out);
      if (triggerCommitted) {
        onChangeCommitted?.(out);
      }
    },
    [activeValues, disabled, isRange, min, max, minDistance, pushable, controlledValue, onChange, onChangeCommitted]
  );

  // Pointer drag lifecycle
  const handlePointerDown = (e: React.PointerEvent<HTMLDivElement>) => {
    if (disabled || !trackRef.current) return;
    trackRef.current.setPointerCapture(e.pointerId);

    const pct = calculatePointerPercentage(e, trackRef.current, vertical, false);
    const clickedVal = getValueFromPercentage(pct);

    let targetIdx = 0;
    if (isRange) {
      const dist0 = Math.abs(clickedVal - activeValues[0]);
      const dist1 = Math.abs(clickedVal - activeValues[1]);
      targetIdx = dist0 <= dist1 ? 0 : 1;
    }

    setActiveThumbIndex(targetIdx);
    updateThumbValue(clickedVal, targetIdx);
  };

  const handlePointerMove = (e: React.PointerEvent<HTMLDivElement>) => {
    if (activeThumbIndex === null || !trackRef.current) return;
    const pct = calculatePointerPercentage(e, trackRef.current, vertical, false);
    const calculatedVal = getValueFromPercentage(pct);
    updateThumbValue(calculatedVal, activeThumbIndex);
  };

  const handlePointerUp = (e: React.PointerEvent<HTMLDivElement>) => {
    if (activeThumbIndex !== null && trackRef.current) {
      try {
        trackRef.current.releasePointerCapture(e.pointerId);
      } catch {
        // Fallback for detached pointer instances
      }
      const pct = calculatePointerPercentage(e, trackRef.current, vertical, false);
      const calculatedVal = getValueFromPercentage(pct);
      updateThumbValue(calculatedVal, activeThumbIndex, true);
      setActiveThumbIndex(null);
    }
  };

  // Keyboard navigation per thumb
  const handleKeyDown = (e: React.KeyboardEvent, index: number) => {
    if (disabled) return;

    const stepDelta = step || (max - min) / 100;
    const largeDelta = (max - min) / 10;
    let nextVal = activeValues[index];

    switch (e.key) {
      case 'ArrowRight':
      case 'ArrowUp':
        e.preventDefault();
        nextVal += stepDelta;
        break;
      case 'ArrowLeft':
      case 'ArrowDown':
        e.preventDefault();
        nextVal -= stepDelta;
        break;
      case 'PageUp':
        e.preventDefault();
        nextVal += largeDelta;
        break;
      case 'PageDown':
        e.preventDefault();
        nextVal -= largeDelta;
        break;
      case 'Home':
        e.preventDefault();
        nextVal = min;
        break;
      case 'End':
        e.preventDefault();
        nextVal = max;
        break;
      default:
        return;
    }

    if (restrictToMarks && allowedSnapValues.length > 0) {
      nextVal = snapToClosest(nextVal, allowedSnapValues);
    }

    updateThumbValue(clamp(nextVal, min, max), index, true);
  };

  // Size token mappings
  const sizeMap = {
    sm: { track: 4, thumb: 12, fontSize: 10, markSize: 4 },
    md: { track: 6, thumb: 16, fontSize: 12, markSize: 6 },
    lg: { track: 10, thumb: 22, fontSize: 14, markSize: 8 },
  }[size];

  // Active track boundaries
  const pcts = activeValues.map((v) => getPercentage(v));
  const activeStartPct = isRange ? Math.min(pcts[0], pcts[1]) : 0;
  const activeEndPct = isRange ? Math.max(pcts[0], pcts[1]) : pcts[0];

  return (
    <div
      style={{
        display: 'inline-flex',
        flexDirection: vertical ? 'row' : 'column',
        alignItems: 'center',
        position: 'relative',
        userSelect: 'none',
        touchAction: 'none',
        width: vertical ? 'auto' : '100%',
        height: vertical ? '100%' : 'auto',
        minHeight: vertical ? 180 : 'auto',
        minWidth: vertical ? 'auto' : 160,
        opacity: disabled ? 0.45 : 1,
        cursor: disabled ? 'not-allowed' : 'pointer',
        padding: vertical ? '0 18px' : '18px 0',
        boxSizing: 'border-box',
        ...style,
      }}
    >
      {/* Outer Click/Touch Hit Surface */}
      <div
        ref={trackRef}
        onPointerDown={handlePointerDown}
        onPointerMove={handlePointerMove}
        onPointerUp={handlePointerUp}
        onPointerCancel={handlePointerUp}
        style={{
          position: 'relative',
          width: vertical ? sizeMap.track : '100%',
          height: vertical ? '100%' : sizeMap.track,
          backgroundColor: track === 'removed' ? 'transparent' : '#e2e8f0',
          borderRadius: sizeMap.track / 2,
        }}
      >
        {/* Filled Highlight Track Bar */}
        {track !== 'removed' && (
          <div
            style={{
              position: 'absolute',
              backgroundColor: color,
              borderRadius: sizeMap.track / 2,
              ...(vertical
                ? track === 'inverted'
                  ? {
                      top: 0,
                      bottom: `${activeEndPct}%`,
                      width: '100%',
                    }
                  : {
                      bottom: `${activeStartPct}%`,
                      height: `${activeEndPct - activeStartPct}%`,
                      width: '100%',
                    }
                : track === 'inverted'
                ? {
                    right: 0,
                    left: `${activeEndPct}%`,
                    height: '100%',
                  }
                : {
                    left: `${activeStartPct}%`,
                    width: `${activeEndPct - activeStartPct}%`,
                    height: '100%',
                  }),
            }}
          />
        )}

        {/* Custom Marks and Value Labels */}
        {parsedMarks.map((mark) => {
          const markPct = getPercentage(mark.value);
          const isMarkPassed = markPct >= activeStartPct && markPct <= activeEndPct;

          return (
            <React.Fragment key={mark.value}>
              {/* Dot */}
              <div
                style={{
                  position: 'absolute',
                  width: sizeMap.markSize,
                  height: sizeMap.markSize,
                  borderRadius: '50%',
                  backgroundColor: isMarkPassed && track === 'normal' ? '#ffffff' : '#94a3b8',
                  border: `1px solid ${isMarkPassed && track === 'normal' ? color : '#cbd5e1'}`,
                  zIndex: 2,
                  pointerEvents: 'none',
                  ...(vertical
                    ? {
                        bottom: `${markPct}%`,
                        left: '50%',
                        transform: 'translate(-50%, 50%)',
                      }
                    : {
                        left: `${markPct}%`,
                        top: '50%',
                        transform: 'translate(-50%, -50%)',
                      }),
                }}
              />

              {/* Text Label */}
              {mark.label !== undefined && (
                <div
                  style={{
                    position: 'absolute',
                    fontSize: sizeMap.fontSize,
                    color: '#64748b',
                    whiteSpace: 'nowrap',
                    pointerEvents: 'none',
                    zIndex: 1,
                    ...(vertical
                      ? {
                          bottom: `${markPct}%`,
                          ...(marksPlacement === 'before'
                            ? { right: sizeMap.track + 8, transform: 'translateY(50%)' }
                            : { left: sizeMap.track + 8, transform: 'translateY(50%)' }),
                        }
                      : {
                          left: `${markPct}%`,
                          ...(marksPlacement === 'before'
                            ? { bottom: sizeMap.track + 8, transform: 'translateX(-50%)' }
                            : { top: sizeMap.track + 8, transform: 'translateX(-50%)' }),
                        }),
                  }}
                >
                  {mark.label}
                </div>
              )}
            </React.Fragment>
          );
        })}

        {/* Thumbs with Tooltips & ARIA integration */}
        {activeValues.map((val, idx) => {
          const thumbPct = getPercentage(val);
          const isFocused = activeThumbIndex === idx;
          const isHovered = hoveredThumb === idx;
          const showTooltip = alwaysShowValue || isFocused || isHovered;

          return (
            <div
              key={idx}
              role="slider"
              tabIndex={disabled ? -1 : 0}
              aria-label={isRange ? `${ariaLabel} thumb ${idx + 1}` : ariaLabel}
              aria-valuemin={min}
              aria-valuemax={max}
              aria-valuenow={val}
              aria-valuetext={formatValue(val)}
              aria-orientation={vertical ? 'vertical' : 'horizontal'}
              aria-disabled={disabled}
              onKeyDown={(e) => handleKeyDown(e, idx)}
              onFocus={() => setActiveThumbIndex(idx)}
              onBlur={() => setActiveThumbIndex(null)}
              onMouseEnter={() => setHoveredThumb(idx)}
              onMouseLeave={() => setHoveredThumb(null)}
              style={{
                position: 'absolute',
                width: sizeMap.thumb,
                height: sizeMap.thumb,
                backgroundColor: '#ffffff',
                border: `2px solid ${color}`,
                borderRadius: '50%',
                boxShadow: isFocused
                  ? `0 0 0 4px ${color}33, 0 2px 6px rgba(0,0,0,0.15)`
                  : '0 2px 5px rgba(0,0,0,0.18)',
                outline: 'none',
                zIndex: isFocused ? 5 : 4,
                cursor: disabled ? 'not-allowed' : 'grab',
                transition: isFocused ? 'none' : 'box-shadow 0.15s ease',
                ...(vertical
                  ? {
                      bottom: `${thumbPct}%`,
                      left: '50%',
                      transform: 'translate(-50%, 50%)',
                    }
                  : {
                      left: `${thumbPct}%`,
                      top: '50%',
                      transform: 'translate(-50%, -50%)',
                    }),
              }}
            >
              {/* Floating Value Indicator / Bubble Badge */}
              {showTooltip && (
                <div
                  style={{
                    position: 'absolute',
                    backgroundColor: '#1e293b',
                    color: '#ffffff',
                    padding: '3px 7px',
                    borderRadius: 4,
                    fontSize: 11,
                    fontWeight: 600,
                    pointerEvents: 'none',
                    whiteSpace: 'nowrap',
                    boxShadow: '0 2px 8px rgba(0,0,0,0.2)',
                    ...(vertical
                      ? {
                          left: '100%',
                          top: '50%',
                          transform: 'translate(8px, -50%)',
                        }
                      : {
                          bottom: '100%',
                          left: '50%',
                          transform: 'translate(-50%, -8px)',
                        }),
                  }}
                >
                  {formatValue(val)}
                  {/* Tooltip caret */}
                  <div
                    style={{
                      position: 'absolute',
                      width: 0,
                      height: 0,
                      borderStyle: 'solid',
                      ...(vertical
                        ? {
                            right: '100%',
                            top: '50%',
                            transform: 'translateY(-50%)',
                            borderWidth: '4px 4px 4px 0',
                            borderColor: 'transparent #1e293b transparent transparent',
                          }
                        : {
                            top: '100%',
                            left: '50%',
                            transform: 'translateX(-50%)',
                            borderWidth: '4px 4px 0 4px',
                            borderColor: '#1e293b transparent transparent transparent',
                          }),
                    }}
                  />
                </div>
              )}
            </div>
          );
        })}
      </div>
    </div>
  );
};
```

```tsx
// components/SliderWithInput.tsx
import React from 'react';
import { Slider } from './Slider';

interface SliderWithInputProps {
  value: [number, number];
  onChange: (val: [number, number]) => void;
  min?: number;
  max?: number;
  step?: number;
  color?: string;
  prefix?: string;
  suffix?: string;
}

export const SliderWithInput: React.FC<SliderWithInputProps> = ({
  value,
  onChange,
  min = 0,
  max = 1000,
  step = 1,
  color = '#10b981',
  prefix = '$',
  suffix = '',
}) => {
  const handleInputChange = (index: 0 | 1, raw: string) => {
    const num = Number(raw.replace(/[^0-9.-]+/g, ''));
    if (isNaN(num)) return;
    const next: [number, number] = [...value];
    next[index] = num;
    onChange(next);
  };

  return (
    <div
      style={{
        display: 'flex',
        flexDirection: 'column',
        gap: 12,
        padding: 16,
        background: '#ffffff',
        border: '1px solid #e2e8f0',
        borderRadius: 8,
        fontFamily: 'Inter, system-ui, sans-serif',
      }}
    >
      <div style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'center' }}>
        <div style={{ display: 'flex', alignItems: 'center', gap: 4 }}>
          <span style={{ fontSize: 13, color: '#64748b' }}>{prefix}</span>
          <input
            type="number"
            value={value[0]}
            min={min}
            max={value[1]}
            step={step}
            onChange={(e) => handleInputChange(0, e.target.value)}
            style={{
              width: 80,
              padding: '5px 8px',
              border: '1px solid #cbd5e1',
              borderRadius: 4,
              fontSize: 13,
              fontWeight: 600,
              outline: 'none',
            }}
          />
        </div>

        <span style={{ fontSize: 13, color: '#94a3b8' }}>to</span>

        <div style={{ display: 'flex', alignItems: 'center', gap: 4 }}>
          <span style={{ fontSize: 13, color: '#64748b' }}>{prefix}</span>
          <input
            type="number"
            value={value[1]}
            min={value[0]}
            max={max}
            step={step}
            onChange={(e) => handleInputChange(1, e.target.value)}
            style={{
              width: 80,
              padding: '5px 8px',
              border: '1px solid #cbd5e1',
              borderRadius: 4,
              fontSize: 13,
              fontWeight: 600,
              outline: 'none',
            }}
          />
          {suffix && <span style={{ fontSize: 13, color: '#64748b' }}>{suffix}</span>}
        </div>
      </div>

      <Slider
        value={value}
        onChange={onChange}
        min={min}
        max={max}
        step={step}
        color={color}
        minDistance={step * 2}
        pushable
      />
    </div>
  );
};
```

### Key Technical Details

| Feature                | How It Is Achieved                                                                                                                                                                                                                        |
| ---------------------- | ----------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------- |
| **Precision Stepping** | Calculated via `quantizeStep()` using decimal-place counting on the step string to eliminate JavaScript floating-point errors (e.g. `0.30000000000000004`).                                                                               |
| **Pointer Capture**    | Uses `setPointerCapture(pointerId)` on track down. Drags never detach when the cursor moves off the slider bounds or outside the viewport.                                                                                                |
| **Collision Handling** | Multi-thumb sliders clamp thumb indices dynamically using `minDistance` and allow smooth pushing with `pushable={true}`.                                                                                                                  |
| **Non-Linear Scales**  | Accepts custom projection functions (`toValue` and `toPercentage`), applying math curves (like `Math.log` or exponential power functions) while preserving standard step and mark snapping.                                               |
| **Full WAI-ARIA 1.2**  | Supports `role="slider"`, `aria-orientation`, `aria-valuemin`, `aria-valuemax`, `aria-valuenow`, `aria-valuetext`, and standard keyboard interaction (`ArrowUp`/`ArrowDown`/`ArrowLeft`/`ArrowRight`, `PageUp`/`PageDown`, `Home`/`End`). |
