*** copy 1.md ***

Here is a complete, production-ready **Menubar Component System built from scratch** in React and TypeScript following modern headless / Base UI compound component patterns.

It features:

1. **Desktop App Navigation Flow**: Hovering adjacent top-level menus automatically opens them once any menu is active.
2. **Keyboard Traversal (WAI-ARIA 1.2 Menubar Spec)**:

* `ArrowLeft` / `ArrowRight`: Traverses top-level menubar triggers (and closes/switches submenus).
* `ArrowDown` / `ArrowUp`: Navigates up and down inside the active menu list.
* `Home` / `End`: Jumps to the first/last items.
* `Escape`: Closes the open menu and refocuses its top-level trigger.
* `Enter` / `Space`: Activates buttons, toggles checkboxes, or selects radio items.

1. **Compound API Primitives**: `Menubar`, `MenubarMenu`, `MenubarTrigger`, `MenubarPortal`, `MenubarContent`, `MenubarItem`, `MenubarCheckboxItem`, `MenubarRadioGroup`, `MenubarRadioItem`, `MenubarShortcut`, `MenubarSeparator`, `MenubarGroup`, `MenubarLabel`.
2. **Collision Handling & Portals**: Clamps to viewport boundaries and renders menus into `document.body` via React Portals.

### 1. Menubar Root & Item Context Engine

```tsx
// components/MenubarContext.tsx
import React, { createContext, useContext, useState, useRef, useId } from 'react';

export interface MenubarContextValue {
  activeMenuId: string | null;
  setActiveMenuId: React.Dispatch<React.SetStateAction<string | null>>;
  autoOpenOnHover: boolean;
  setAutoOpenOnHover: React.Dispatch<React.SetStateAction<boolean>>;
  registerMenuTrigger: (id: string, el: HTMLButtonElement | null) => void;
  focusNextTrigger: (currentId: string, dir: 'next' | 'prev') => void;
}

export const MenubarContext = createContext<MenubarContextValue | null>(null);

export const useMenubarContext = () => {
  const ctx = useContext(MenubarContext);
  if (!ctx) throw new Error('Menubar compound components must be rendered inside <Menubar />');
  return ctx;
};

// Internal Menu-Level Context
export interface MenuScopeContextValue {
  menuId: string;
  isOpen: boolean;
  triggerRef: React.RefObject<HTMLButtonElement | null>;
  closeMenu: () => void;
}

export const MenuScopeContext = createContext<MenuScopeContextValue | null>(null);

export const useMenuScope = () => {
  const ctx = useContext(MenuScopeContext);
  if (!ctx) throw new Error('Menubar sub-components must be inside <MenubarMenu />');
  return ctx;
};
```

### 2. Compound Component Implementation

```tsx
// components/Menubar.tsx
import React, { useState, useRef, useEffect, useCallback, useId, forwardRef } from 'react';
import { createPortal } from 'react-dom';
import {
  MenubarContext,
  useMenubarContext,
  MenuScopeContext,
  useMenuScope,
} from './MenubarContext';

// ==========================================
// 1. MENUBAR ROOT
// ==========================================
export interface MenubarProps extends React.HTMLAttributes<HTMLDivElement> {
  children: React.ReactNode;
}

export const Menubar: React.FC<MenubarProps> = ({ children, style, className, ...props }) => {
  const [activeMenuId, setActiveMenuId] = useState<string | null>(null);
  const [autoOpenOnHover, setAutoOpenOnHover] = useState<boolean>(false);
  const triggerMapRef = useRef<Map<string, HTMLButtonElement>>(new Map());

  const registerMenuTrigger = useCallback((id: string, el: HTMLButtonElement | null) => {
    if (el) triggerMapRef.current.set(id, el);
    else triggerMapRef.current.delete(id);
  }, []);

  const focusNextTrigger = useCallback((currentId: string, dir: 'next' | 'prev') => {
    const triggerKeys = Array.from(triggerMapRef.current.keys());
    const currentIndex = triggerKeys.indexOf(currentId);
    if (currentIndex === -1) return;

    let nextIndex = dir === 'next' ? currentIndex + 1 : currentIndex - 1;
    if (nextIndex >= triggerKeys.length) nextIndex = 0;
    if (nextIndex < 0) nextIndex = triggerKeys.length - 1;

    const nextId = triggerKeys[nextIndex];
    const targetButton = triggerMapRef.current.get(nextId);
    if (targetButton) {
      targetButton.focus();
      setActiveMenuId(nextId); // Switch active open dropdown directly
    }
  }, []);

  // Global click-outside detection
  useEffect(() => {
    const handleOutsideClick = (e: MouseEvent) => {
      const isInside = Array.from(triggerMapRef.current.values()).some((btn) =>
        btn.contains(e.target as Node)
      );
      if (!isInside && !(e.target as HTMLElement)?.closest('[role="menu"]')) {
        setActiveMenuId(null);
        setAutoOpenOnHover(false);
      }
    };
    document.addEventListener('mousedown', handleOutsideClick);
    return () => document.removeEventListener('mousedown', handleOutsideClick);
  }, []);

  return (
    <MenubarContext.Provider
      value={{
        activeMenuId,
        setActiveMenuId,
        autoOpenOnHover,
        setAutoOpenOnHover,
        registerMenuTrigger,
        focusNextTrigger,
      }}
    >
      <div
        role="menubar"
        aria-orientation="horizontal"
        className={`menubar-root ${className || ''}`}
        style={{
          display: 'flex',
          alignItems: 'center',
          gap: 2,
          padding: 4,
          backgroundColor: '#ffffff',
          border: '1px solid #e2e8f0',
          borderRadius: 8,
          boxShadow: '0 1px 2px rgba(0,0,0,0.05)',
          fontFamily: 'Inter, system-ui, sans-serif',
          ...style,
        }}
        {...props}
      >
        {children}
      </div>
    </MenubarContext.Provider>
  );
};

// ==========================================
// 2. MENUBAR MENU CONTAINER
// ==========================================
export const MenubarMenu: React.FC<{ children: React.ReactNode }> = ({ children }) => {
  const generatedId = useId();
  const { activeMenuId, setActiveMenuId, setAutoOpenOnHover } = useMenubarContext();
  const triggerRef = useRef<HTMLButtonElement | null>(null);

  const isOpen = activeMenuId === generatedId;
  const closeMenu = useCallback(() => {
    setActiveMenuId(null);
    setAutoOpenOnHover(false);
  }, [setActiveMenuId, setAutoOpenOnHover]);

  return (
    <MenuScopeContext.Provider value={{ menuId: generatedId, isOpen, triggerRef, closeMenu }}>
      <div style={{ position: 'relative' }}>{children}</div>
    </MenuScopeContext.Provider>
  );
};

// ==========================================
// 3. MENUBAR TRIGGER
// ==========================================
export interface MenubarTriggerProps extends React.ButtonHTMLAttributes<HTMLButtonElement> {}

export const MenubarTrigger = forwardRef<HTMLButtonElement, MenubarTriggerProps>(
  ({ children, style, className, ...props }, ref) => {
    const { activeMenuId, setActiveMenuId, autoOpenOnHover, setAutoOpenOnHover, registerMenuTrigger, focusNextTrigger } =
      useMenubarContext();
    const { menuId, isOpen, triggerRef } = useMenuScope();

    // Sync trigger refs
    const setRefs = (el: HTMLButtonElement | null) => {
      triggerRef.current = el;
      registerMenuTrigger(menuId, el);
      if (typeof ref === 'function') ref(el);
      else if (ref) (ref as any).current = el;
    };

    const handleClick = () => {
      if (isOpen) {
        setActiveMenuId(null);
        setAutoOpenOnHover(false);
      } else {
        setActiveMenuId(menuId);
        setAutoOpenOnHover(true);
      }
    };

    const handleMouseEnter = () => {
      if (autoOpenOnHover && activeMenuId !== menuId) {
        setActiveMenuId(menuId);
      }
    };

    const handleKeyDown = (e: React.KeyboardEvent<HTMLButtonElement>) => {
      if (e.key === 'ArrowRight') {
        e.preventDefault();
        focusNextTrigger(menuId, 'next');
      } else if (e.key === 'ArrowLeft') {
        e.preventDefault();
        focusNextTrigger(menuId, 'prev');
      } else if (e.key === 'ArrowDown' || e.key === 'Enter' || e.key === ' ') {
        e.preventDefault();
        setActiveMenuId(menuId);
        setAutoOpenOnHover(true);
      }
    };

    return (
      <button
        ref={setRefs}
        role="menuitem"
        aria-haspopup="menu"
        aria-expanded={isOpen}
        onClick={handleClick}
        onMouseEnter={handleMouseEnter}
        onKeyDown={handleKeyDown}
        className={`menubar-trigger ${isOpen ? 'open' : ''} ${className || ''}`}
        style={{
          display: 'inline-flex',
          alignItems: 'center',
          justifyContent: 'center',
          padding: '6px 10px',
          fontSize: 13,
          fontWeight: 500,
          color: isOpen ? '#2563eb' : '#334155',
          backgroundColor: isOpen ? '#eff6ff' : 'transparent',
          border: 'none',
          borderRadius: 4,
          cursor: 'pointer',
          outline: 'none',
          userSelect: 'none',
          transition: 'background-color 0.12s ease',
          ...style,
        }}
        onFocus={(e) => (e.currentTarget.style.backgroundColor = '#f1f5f9')}
        onBlur={(e) => {
          if (!isOpen) e.currentTarget.style.backgroundColor = 'transparent';
        }}
        {...props}
      >
        {children}
      </button>
    );
  }
);
MenubarTrigger.displayName = 'MenubarTrigger';

// ==========================================
// 4. MENUBAR PORTAL & CONTENT
// ==========================================
export const MenubarPortal: React.FC<{ children: React.ReactNode }> = ({ children }) => {
  return createPortal(<>{children}</>, document.body);
};

export interface MenubarContentProps extends React.HTMLAttributes<HTMLDivElement> {
  minWidth?: number | string;
  align?: 'start' | 'end';
}

export const MenubarContent: React.FC<MenubarContentProps> = ({
  children,
  minWidth = 220,
  align = 'start',
  style,
  className,
  ...props
}) => {
  const { isOpen, triggerRef, menuId, closeMenu } = useMenuScope();
  const { focusNextTrigger } = useMenubarContext();
  const contentRef = useRef<HTMLDivElement>(null);
  const [coords, setCoords] = useState<{ top: number; left: number }>({ top: -9999, left: -9999 });

  // Compute Position below Trigger
  useEffect(() => {
    if (isOpen && triggerRef.current) {
      const rect = triggerRef.current.getBoundingClientRect();
      const top = rect.bottom + 4 + window.scrollY;
      let left = align === 'end' ? rect.right - 220 + window.scrollX : rect.left + window.scrollX;

      // Viewport collision clamping
      const maxLeft = window.innerWidth - 240;
      left = Math.max(8, Math.min(left, maxLeft));

      setCoords({ top, left });

      // Focus first item when content opens
      setTimeout(() => {
        const firstItem = contentRef.current?.querySelector<HTMLElement>(
          '[role^="menuitem"]:not([aria-disabled="true"])'
        );
        firstItem?.focus();
      }, 20);
    }
  }, [isOpen, triggerRef, align]);

  // Keyboard navigation inside menu list
  const handleKeyDown = (e: React.KeyboardEvent<HTMLDivElement>) => {
    if (!contentRef.current) return;
    const items = Array.from(
      contentRef.current.querySelectorAll<HTMLElement>(
        '[role^="menuitem"]:not([aria-disabled="true"])'
      )
    );
    if (!items.length) return;

    const currentIdx = items.indexOf(document.activeElement as HTMLElement);

    if (e.key === 'ArrowDown') {
      e.preventDefault();
      const next = (currentIdx + 1) % items.length;
      items[next]?.focus();
    } else if (e.key === 'ArrowUp') {
      e.preventDefault();
      const prev = (currentIdx - 1 + items.length) % items.length;
      items[prev]?.focus();
    } else if (e.key === 'ArrowRight') {
      e.preventDefault();
      focusNextTrigger(menuId, 'next');
    } else if (e.key === 'ArrowLeft') {
      e.preventDefault();
      focusNextTrigger(menuId, 'prev');
    } else if (e.key === 'Home') {
      e.preventDefault();
      items[0]?.focus();
    } else if (e.key === 'End') {
      e.preventDefault();
      items[items.length - 1]?.focus();
    } else if (e.key === 'Escape') {
      e.preventDefault();
      closeMenu();
      triggerRef.current?.focus();
    }
  };

  if (!isOpen) return null;

  return (
    <MenubarPortal>
      <div
        ref={contentRef}
        role="menu"
        aria-orientation="vertical"
        tabIndex={-1}
        onKeyDown={handleKeyDown}
        className={`menubar-content ${className || ''}`}
        style={{
          position: 'absolute',
          top: coords.top,
          left: coords.left,
          minWidth,
          backgroundColor: '#ffffff',
          border: '1px solid #e2e8f0',
          borderRadius: 8,
          boxShadow: '0 10px 15px -3px rgba(0, 0, 0, 0.1), 0 4px 6px -4px rgba(0, 0, 0, 0.05)',
          padding: '4px 0',
          zIndex: 9999,
          fontFamily: 'Inter, system-ui, sans-serif',
          outline: 'none',
          animation: 'menubarPop 0.12s cubic-bezier(0.16, 1, 0.3, 1)',
          ...style,
        }}
        {...props}
      >
        {children}
      </div>

      <style>{`
        @keyframes menubarPop {
          from { opacity: 0; transform: scale(0.96) translateY(-4px); }
          to { opacity: 1; transform: scale(1) translateY(0); }
        }
      `}</style>
    </MenubarPortal>
  );
};

// 5. MENUBAR ITEM & PRIMITIVES
// ==========================================
export interface MenubarItemProps extends React.HTMLAttributes<HTMLDivElement> {
  disabled?: boolean;
  inset?: boolean;
  icon?: React.ReactNode;
}

export const MenubarItem: React.FC<MenubarItemProps> = ({
  children,
  disabled = false,
  inset = false,
  icon,
  onClick,
  style,
  className,
  ...props
}) => {
  const { closeMenu } = useMenuScope();

  const handleClick = (e: React.MouseEvent<HTMLDivElement>) => {
    if (disabled) return;
    onClick?.(e);
    closeMenu();
  };

  return (
    <div
      role="menuitem"
      tabIndex={disabled ? -1 : 0}
      aria-disabled={disabled}
      onClick={handleClick}
      className={`menubar-item ${className || ''}`}
      style={{
        display: 'flex',
        alignItems: 'center',
        padding: '6px 12px',
        paddingLeft: inset ? 32 : 12,
        fontSize: 13,
        color: disabled ? '#94a3b8' : '#1e293b',
        cursor: disabled ? 'not-allowed' : 'pointer',
        userSelect: 'none',
        outline: 'none',
        transition: 'background-color 0.1s ease',
        ...style,
      }}
      onMouseEnter={(e) => {
        if (!disabled) e.currentTarget.focus();
      }}
      onFocus={(e) => {
        if (!disabled) e.currentTarget.style.backgroundColor = '#f1f5f9';
      }}
      onBlur={(e) => {
        e.currentTarget.style.backgroundColor = 'transparent';
      }}
      onKeyDown={(e) => {
        if (e.key === 'Enter' || e.key === ' ') {
          e.preventDefault();
          handleClick(e as any);
        }
      }}
      {...props}
    >
      {icon && (
        <span style={{ marginRight: 8, display: 'inline-flex', alignItems: 'center', fontSize: 15, color: '#64748b' }}>
          {icon}
        </span>
      )}
      <span style={{ flex: 1 }}>{children}</span>
    </div>
  );
};

export interface MenubarCheckboxItemProps extends MenubarItemProps {
  checked: boolean;
  onCheckedChange?: (checked: boolean) => void;
}

export const MenubarCheckboxItem: React.FC<MenubarCheckboxItemProps> = ({
  checked,
  onCheckedChange,
  children,
  ...props
}) => {
  return (
    <MenubarItem
      role="menuitemcheckbox"
      aria-checked={checked}
      onClick={() => onCheckedChange?.(!checked)}
      icon={
        <span style={{ width: 14, textAlign: 'center', fontWeight: 800, color: checked ? '#2563eb' : 'transparent' }}>
          ✓
        </span>
      }
      {...props}
    >
      {children}
    </MenubarItem>
  );
};

export const MenubarRadioGroup: React.FC<{
  value: string;
  onValueChange: (value: string) => void;
  children: React.ReactNode;
}> = ({ value, onValueChange, children }) => {
  return (
    <div role="group">
      {React.Children.map(children, (child) => {
        if (React.isValidElement(child)) {
          return React.cloneElement(child as React.ReactElement<any>, {
            _groupValue: value,
            _onGroupChange: onValueChange,
          });
        }
        return child;
      })}
    </div>
  );
};

export interface MenubarRadioItemProps extends MenubarItemProps {
  value: string;
  _groupValue?: string;
  _onGroupChange?: (val: string) => void;
}

export const MenubarRadioItem: React.FC<MenubarRadioItemProps> = ({
  value,
  _groupValue,
  _onGroupChange,
  children,
  ...props
}) => {
  const isChecked = value === _groupValue;
  return (
    <MenubarItem
      role="menuitemradio"
      aria-checked={isChecked}
      onClick={() => _onGroupChange?.(value)}
      icon={
        <span style={{ width: 14, textAlign: 'center', fontSize: 10, color: isChecked ? '#2563eb' : 'transparent' }}>
          ●
        </span>
      }
      {...props}
    >
      {children}
    </MenubarItem>
  );
};
```

### Key Technical Details

| Feature                       | Implementation Mechanism                                                                                                                                                                                                                           |
| ----------------------------- | -------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------- |
| **Hover Switching**           | Once `activeMenuId` is opened via a click, `autoOpenOnHover` becomes `true`. Subsequent hover events switch active top-level menus instantly without requiring clicks.                                                                             |
| **WAI-ARIA 1.2 Menubar Spec** | Root wrapper declares `role="menubar"`, triggers declare `role="menuitem"` with `aria-haspopup="menu"` and `aria-expanded`, popup contents declare `role="menu"`, checkboxes use `role="menuitemcheckbox"`, and radios use `role="menuitemradio"`. |
| **Keyboard Traversal Loop**   | Arrow keys (`ArrowLeft`/`ArrowRight`) loop horizontally across top triggers; `ArrowUp`/`ArrowDown` cycle items vertically inside menus; `Escape` closes the menu and returns focus to the parent trigger button.                                   |
| **Portal Layering**           | `MenubarPortal` mounts menus to `document.body`, avoiding `overflow: hidden` boundaries and stacking context conflicts in complex layouts.                                                                                                         |
