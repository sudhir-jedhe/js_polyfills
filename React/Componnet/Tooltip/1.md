### 1. Coordinate & Collision Positioning Engine

This module computes bounding-box anchor positions (`top`, `bottom`, `left`, `right` and alignments), adds offsets/distance gaps, handles flip/collision fallback near viewport edges, and tracks virtual anchors/cursor coordinates.

```typescript
// utils/tooltipPositioning.ts
export type TooltipPlacement =
  | 'top' | 'top-start' | 'top-end'
  | 'bottom' | 'bottom-start' | 'bottom-end'
  | 'left' | 'left-start' | 'left-end'
  | 'right' | 'right-start' | 'right-end';

export interface VirtualElement {
  getBoundingClientRect: () => DOMRect;
}

export interface PositionResult {
  x: number;
  y: number;
  actualPlacement: TooltipPlacement;
  arrowX?: number;
  arrowY?: number;
}

export function computeTooltipPosition(
  anchor: HTMLElement | VirtualElement,
  tooltip: HTMLElement,
  placement: TooltipPlacement,
  offsetDistance = 8,
  arrowSize = 6
): PositionResult {
  const anchorRect = anchor.getBoundingClientRect();
  const tooltipRect = tooltip.getBoundingClientRect();
  const viewportWidth = window.innerWidth;
  const viewportHeight = window.innerHeight;

  const [basePlacement, alignment] = placement.split('-') as [
    'top' | 'bottom' | 'left' | 'right',
    'start' | 'end' | undefined
  ];

  let actualBase = basePlacement;

  // 1. Collision Flip Detection
  if (basePlacement === 'top' && anchorRect.top - tooltipRect.height - offsetDistance < 0) {
    actualBase = 'bottom';
  } else if (
    basePlacement === 'bottom' &&
    anchorRect.bottom + tooltipRect.height + offsetDistance > viewportHeight
  ) {
    actualBase = 'top';
  } else if (basePlacement === 'left' && anchorRect.left - tooltipRect.width - offsetDistance < 0) {
    actualBase = 'right';
  } else if (
    basePlacement === 'right' &&
    anchorRect.right + tooltipRect.width + offsetDistance > viewportWidth
  ) {
    actualBase = 'left';
  }

  let x = 0;
  let y = 0;

  // 2. Primary Axes Coordinates
  switch (actualBase) {
    case 'top':
      y = anchorRect.top - tooltipRect.height - offsetDistance;
      break;
    case 'bottom':
      y = anchorRect.bottom + offsetDistance;
      break;
    case 'left':
      x = anchorRect.left - tooltipRect.width - offsetDistance;
      break;
    case 'right':
      x = anchorRect.right + offsetDistance;
      break;
  }

  // 3. Secondary Alignment Axes
  if (actualBase === 'top' || actualBase === 'bottom') {
    if (alignment === 'start') {
      x = anchorRect.left;
    } else if (alignment === 'end') {
      x = anchorRect.right - tooltipRect.width;
    } else {
      x = anchorRect.left + (anchorRect.width - tooltipRect.width) / 2;
    }
  } else {
    if (alignment === 'start') {
      y = anchorRect.top;
    } else if (alignment === 'end') {
      y = anchorRect.bottom - tooltipRect.height;
    } else {
      y = anchorRect.top + (anchorRect.height - tooltipRect.height) / 2;
    }
  }

  // 4. Viewport Clamping (keep tooltip inside window)
  x = Math.max(8, Math.min(x, viewportWidth - tooltipRect.width - 8));
  y = Math.max(8, Math.min(y, viewportHeight - tooltipRect.height - 8));

  // 5. Arrow Alignment Coordinates
  let arrowX: number | undefined;
  let arrowY: number | undefined;

  if (actualBase === 'top' || actualBase === 'bottom') {
    arrowX = anchorRect.left + anchorRect.width / 2 - x - arrowSize;
    arrowX = Math.max(arrowSize, Math.min(arrowX, tooltipRect.width - arrowSize * 3));
  } else {
    arrowY = anchorRect.top + anchorRect.height / 2 - y - arrowSize;
    arrowY = Math.max(arrowSize, Math.min(arrowY, tooltipRect.height - arrowSize * 3));
  }

  const finalPlacement = alignment ? (`${actualBase}-${alignment}` as TooltipPlacement) : actualBase;

  return {
    x: x + window.scrollX,
    y: y + window.scrollY,
    actualPlacement: finalPlacement,
    arrowX,
    arrowY,
  };
}
```

### 2. Universal Tooltip Component

```tsx
// components/Tooltip.tsx
import React, {
  useState,
  useRef,
  useEffect,
  useCallback,
  cloneElement,
  isValidElement,
  useId,
} from 'react';
import { createPortal } from 'react-dom';
import {
  TooltipPlacement,
  VirtualElement,
  computeTooltipPosition,
  PositionResult,
} from '../utils/tooltipPositioning';

export type TooltipTrigger = 'hover' | 'focus' | 'click' | 'manual';
export type TooltipTransition = 'fade' | 'zoom' | 'slide';

export interface TooltipProps {
  title: React.ReactNode;
  description?: React.ReactNode;
  children: React.ReactElement;
  placement?: TooltipPlacement;
  arrow?: boolean;
  distance?: number;
  triggers?: TooltipTrigger[];
  enterDelay?: number;
  leaveDelay?: number;
  interactive?: boolean;
  followCursor?: boolean;
  virtualElement?: VirtualElement | null;
  open?: boolean;
  onOpenChange?: (open: boolean) => void;
  maxWidth?: number | string;
  transition?: TooltipTransition;
  color?: 'dark' | 'light' | 'primary' | 'error';
  disabled?: boolean;
  className?: string;
  style?: React.CSSProperties;
}

export const Tooltip: React.FC<TooltipProps> = ({
  title,
  description,
  children,
  placement = 'top',
  arrow = false,
  distance = 8,
  triggers = ['hover', 'focus'],
  enterDelay = 100,
  leaveDelay = 150,
  interactive = false,
  followCursor = false,
  virtualElement,
  open: controlledOpen,
  onOpenChange,
  maxWidth = 300,
  transition = 'fade',
  color = 'dark',
  disabled = false,
  className,
  style,
}) => {
  const [internalOpen, setInternalOpen] = useState(false);
  const [coords, setCoords] = useState<PositionResult>({
    x: -9999,
    y: -9999,
    actualPlacement: placement,
  });

  const anchorRef = useRef<HTMLElement | null>(null);
  const tooltipRef = useRef<HTMLDivElement | null>(null);
  const cursorRef = useRef<{ clientX: number; clientY: number } | null>(null);
  const enterTimerRef = useRef<NodeJS.Timeout | null>(null);
  const leaveTimerRef = useRef<NodeJS.Timeout | null>(null);

  const tooltipId = useId();
  const isControlled = controlledOpen !== undefined;
  const isVisible = (isControlled ? controlledOpen : internalOpen) && !disabled && Boolean(title || description);

  const setOpenState = useCallback(
    (nextState: boolean) => {
      if (!isControlled) setInternalOpen(nextState);
      onOpenChange?.(nextState);
    },
    [isControlled, onOpenChange]
  );

  const clearTimers = () => {
    if (enterTimerRef.current) clearTimeout(enterTimerRef.current);
    if (leaveTimerRef.current) clearTimeout(leaveTimerRef.current);
  };

  const handleOpen = useCallback(() => {
    clearTimers();
    enterTimerRef.current = setTimeout(() => {
      setOpenState(true);
    }, enterDelay);
  }, [enterDelay, setOpenState]);

  const handleClose = useCallback(() => {
    clearTimers();
    leaveTimerRef.current = setTimeout(() => {
      setOpenState(false);
    }, leaveDelay);
  }, [leaveDelay, setOpenState]);

  // Recalculate Tooltip Position
  const updatePosition = useCallback(() => {
    if (!tooltipRef.current) return;

    let targetAnchor: HTMLElement | VirtualElement | null = anchorRef.current;

    if (virtualElement) {
      targetAnchor = virtualElement;
    } else if (followCursor && cursorRef.current) {
      const { clientX, clientY } = cursorRef.current;
      targetAnchor = {
        getBoundingClientRect: () =>
          new DOMRect(clientX, clientY, 0, 0),
      };
    }

    if (!targetAnchor) return;

    const computed = computeTooltipPosition(
      targetAnchor,
      tooltipRef.current,
      placement,
      distance,
      6
    );
    setCoords(computed);
  }, [placement, distance, followCursor, virtualElement]);

  useEffect(() => {
    if (isVisible) {
      updatePosition();
      const handleScrollOrResize = () => updatePosition();
      window.addEventListener('scroll', handleScrollOrResize, { passive: true });
      window.addEventListener('resize', handleScrollOrResize);
      return () => {
        window.removeEventListener('scroll', handleScrollOrResize);
        window.removeEventListener('resize', handleScrollOrResize);
      };
    }
  }, [isVisible, updatePosition]);

  // Handle ESC key dismiss
  useEffect(() => {
    const handleKeyDown = (e: KeyboardEvent) => {
      if (e.key === 'Escape' && isVisible) {
        handleClose();
      }
    };
    window.addEventListener('keydown', handleKeyDown);
    return () => window.removeEventListener('keydown', handleKeyDown);
  }, [isVisible, handleClose]);

  // Trigger Events Handlers
  const handleMouseEnter = (e: React.MouseEvent) => {
    if (triggers.includes('hover')) handleOpen();
    if (followCursor) {
      cursorRef.current = { clientX: e.clientX, clientY: e.clientY };
      updatePosition();
    }
  };

  const handleMouseMove = (e: React.MouseEvent) => {
    if (followCursor) {
      cursorRef.current = { clientX: e.clientX, clientY: e.clientY };
      updatePosition();
    }
  };

  const handleMouseLeave = () => {
    if (triggers.includes('hover')) handleClose();
  };

  const handleFocus = () => {
    if (triggers.includes('focus')) handleOpen();
  };

  const handleBlur = () => {
    if (triggers.includes('focus')) handleClose();
  };

  const handleClick = (e: React.MouseEvent) => {
    if (triggers.includes('click')) {
      if (isVisible) handleClose();
      else handleOpen();
    }
  };

  // Color Palette Themes
  const themes = {
    dark: { bg: '#0f172a', text: '#ffffff', border: '#1e293b', arrowBorder: '#0f172a' },
    light: { bg: '#ffffff', text: '#0f172a', border: '#cbd5e1', arrowBorder: '#ffffff' },
    primary: { bg: '#2563eb', text: '#ffffff', border: '#1d4ed8', arrowBorder: '#2563eb' },
    error: { bg: '#dc2626', text: '#ffffff', border: '#b91c1c', arrowBorder: '#dc2626' },
  }[color];

  // Base Placement for Arrow Direction
  const basePlacement = coords.actualPlacement.split('-')[0];

  // Child Element Cloning (Handles disabled child elements via pointer wrapper)
  const isChildDisabled = Boolean(children.props.disabled);

  const triggerNode = isValidElement(children) ? (
    isChildDisabled ? (
      <span
        ref={anchorRef as any}
        onMouseEnter={handleMouseEnter}
        onMouseLeave={handleMouseLeave}
        style={{ display: 'inline-block', cursor: 'not-allowed' }}
        tabIndex={0}
        aria-describedby={isVisible ? tooltipId : undefined}
      >
        {cloneElement(children, {
          style: { ...(children.props.style || {}), pointerEvents: 'none' },
        })}
      </span>
    ) : (
      cloneElement(children, {
        ref: anchorRef,
        onMouseEnter: (e: React.MouseEvent) => {
          children.props.onMouseEnter?.(e);
          handleMouseEnter(e);
        },
        onMouseMove: (e: React.MouseEvent) => {
          children.props.onMouseMove?.(e);
          handleMouseMove(e);
        },
        onMouseLeave: (e: React.MouseEvent) => {
          children.props.onMouseLeave?.(e);
          handleMouseLeave();
        },
        onFocus: (e: React.FocusEvent) => {
          children.props.onFocus?.(e);
          handleFocus();
        },
        onBlur: (e: React.FocusEvent) => {
          children.props.onBlur?.(e);
          handleBlur();
        },
        onClick: (e: React.MouseEvent) => {
          children.props.onClick?.(e);
          handleClick(e);
        },
        'aria-describedby': isVisible ? tooltipId : undefined,
      } as any)
    )
  ) : (
    children
  );

  return (
    <>
      {triggerNode}
      {isVisible &&
        createPortal(
          <div
            id={tooltipId}
            ref={tooltipRef}
            role="tooltip"
            onMouseEnter={interactive ? handleOpen : undefined}
            onMouseLeave={interactive ? handleClose : undefined}
            className={`custom-tooltip ${transition} ${className || ''}`}
            style={{
              position: 'absolute',
              top: 0,
              left: 0,
              transform: `translate3d(${coords.x}px, ${coords.y}px, 0)`,
              backgroundColor: themes.bg,
              color: themes.text,
              border: `1px solid ${themes.border}`,
              borderRadius: 6,
              padding: description ? '8px 12px' : '4px 8px',
              maxWidth,
              width: 'max-content',
              zIndex: 9999,
              pointerEvents: interactive ? 'auto' : 'none',
              boxShadow: '0 10px 15px -3px rgba(0,0,0,0.1), 0 4px 6px -4px rgba(0,0,0,0.1)',
              fontFamily: 'Inter, system-ui, sans-serif',
              boxSizing: 'border-box',
              ...style,
            }}
          >
            {/* Title */}
            <div style={{ fontSize: 12, fontWeight: description ? 600 : 500, lineHeight: 1.4 }}>
              {title}
            </div>

            {/* Optional Rich Description */}
            {description && (
              <div style={{ fontSize: 11, opacity: 0.85, marginTop: 4, lineHeight: 1.35 }}>
                {description}
              </div>
            )}

            {/* Tooltip Arrow */}
            {arrow && (
              <div
                style={{
                  position: 'absolute',
                  width: 0,
                  height: 0,
                  borderStyle: 'solid',
                  ...(basePlacement === 'top'
                    ? {
                        bottom: -6,
                        left: coords.arrowX ?? '50%',
                        borderWidth: '6px 6px 0 6px',
                        borderColor: `${themes.arrowBorder} transparent transparent transparent`,
                      }
                    : basePlacement === 'bottom'
                    ? {
                        top: -6,
                        left: coords.arrowX ?? '50%',
                        borderWidth: '0 6px 6px 6px',
                        borderColor: `transparent transparent ${themes.arrowBorder} transparent`,
                      }
                    : basePlacement === 'left'
                    ? {
                        right: -6,
                        top: coords.arrowY ?? '50%',
                        borderWidth: '6px 0 6px 6px',
                        borderColor: `transparent transparent transparent ${themes.arrowBorder}`,
                      }
                    : {
                        left: -6,
                        top: coords.arrowY ?? '50%',
                        borderWidth: '6px 6px 6px 0',
                        borderColor: `transparent ${themes.arrowBorder} transparent transparent`,
                      }),
                }}
              />
            )}

            {/* Embedded Transition Animations */}
            <style>{`
              .custom-tooltip.fade {
                animation: tooltipFadeIn 0.15s ease-out;
              }
              .custom-tooltip.zoom {
                animation: tooltipZoomIn 0.15s cubic-bezier(0.16, 1, 0.3, 1);
              }
              .custom-tooltip.slide {
                animation: tooltipSlideIn 0.15s cubic-bezier(0.16, 1, 0.3, 1);
              }
              @keyframes tooltipFadeIn {
                from { opacity: 0; }
                to { opacity: 1; }
              }
              @keyframes tooltipZoomIn {
                from { opacity: 0; transform: translate3d(${coords.x}px, ${coords.y}px, 0) scale(0.85); }
                to { opacity: 1; transform: translate3d(${coords.x}px, ${coords.y}px, 0) scale(1); }
              }
              @keyframes tooltipSlideIn {
                from { opacity: 0; transform: translate3d(${coords.x}px, ${coords.y + 4}px, 0); }
                to { opacity: 1; transform: translate3d(${coords.x}px, ${coords.y}px, 0); }
              }
            `}</style>
          </div>,
          document.body
        )}
    </>
  );
};
```

### Key Technical Details

| Feature                             | Implementation Mechanism                                                                                                                                           |
| ----------------------------------- | ------------------------------------------------------------------------------------------------------------------------------------------------------------------ |
| **React Portals**                   | Tooltips render directly into `document.body` via `createPortal`, eliminating `overflow: hidden` parent clipping and `z-index` stacking context traps.             |
| **Collision Flipping**              | `computeTooltipPosition` computes viewport boundaries on scroll/resize and automatically flips `top <-> bottom` or `left <-> right` when approaching window edges. |
| **Disabled Elements Support**       | Wraps native disabled buttons in an inline-block `<span>` with `tabIndex={0}` and pointer-event passthroughs, preserving hover/focus triggers.                     |
| **Follow Cursor & Virtual Anchors** | Accepts an abstract `VirtualElement` interface (`{ getBoundingClientRect: () => DOMRect }`), supporting arbitrary cursor tracking or canvas coordinate anchoring.  |
| **Accessibility (a11y)**            | Implements `role="tooltip"`, dynamically links anchors via `aria-describedby={tooltipId}`, and provides `Escape` key dismissal.                                    |
