***  1.md ***

Here is the complete zero-dependency implementation of a **Toggle Button & Toggle Button Group Component System built from scratch** in React with TypeScript.

It covers:

1. **Selection Modes**: Exclusive (Radio-style, single value) & Multiple (Checkbox-style, array of values).
2. **Enforce Value Set (`enforceValueSet`)**: Prevents deselecting the active item in exclusive mode when at least one selection is mandatory.
3. **Orientation & Spacing**: `horizontal` vs `vertical`, contiguous merged borders or isolated spaced buttons.
4. **Sizes & Colors**: `sm`, `md`, `lg` scales paired with theme palettes (`primary`, `secondary`, `success`, `error`, `warning`).
5. **Standalone Toggle**: Independent controlled or uncontrolled button with `selected` and `onChange` props.
6. **WAI-ARIA 1.2 & Keyboard Navigation**: Arrow keys (`ArrowLeft`/`ArrowRight` or `ArrowUp`/`ArrowDown`) with looping roving `tabIndex`, and `Space`/`Enter` selection.

### 3. Core `ToggleButton` Component

```tsx
// components/ToggleButton.tsx
import React from 'react';
import { useToggleButtonGroup } from '../context/ToggleButtonGroupContext';
import {
  ToggleButtonSize,
  ToggleButtonColor,
  TOGGLE_PALETTES,
} from '../types/toggleButton';

export interface ToggleButtonProps
  extends Omit<React.ButtonHTMLAttributes<HTMLButtonElement>, 'value' | 'color'> {
  value: any;
  selected?: boolean;
  size?: ToggleButtonSize;
  color?: ToggleButtonColor;
  disabled?: boolean;
  fullWidth?: boolean;
  children: React.ReactNode;
}

export const ToggleButton = React.forwardRef<HTMLButtonElement, ToggleButtonProps>(
  (
    {
      value,
      selected: customSelected,
      onChange: customOnChange,
      onClick,
      size: customSize,
      color: customColor,
      disabled: customDisabled,
      fullWidth = false,
      children,
      className,
      style,
      ...restProps
    },
    ref
  ) => {
    const group = useToggleButtonGroup();

    // Resolve context vs individual props
    const size = customSize || group?.size || 'md';
    const color = customColor || group?.color || 'standard';
    const disabled = customDisabled !== undefined ? customDisabled : group?.disabled || false;

    // Determine selection state
    let isSelected = false;
    if (customSelected !== undefined) {
      isSelected = customSelected;
    } else if (group && group.value !== undefined) {
      isSelected = Array.isArray(group.value)
        ? group.value.includes(value)
        : group.value === value;
    }

    const handleClick = (e: React.MouseEvent<HTMLButtonElement>) => {
      if (disabled) return;
      onClick?.(e);
      customOnChange?.(e as any);
      group?.onChange?.(e, value);
    };

    const palette = TOGGLE_PALETTES[color];

    const sizeStyles = {
      sm: { padding: '5px 10px', fontSize: '12px', minHeight: '30px' },
      md: { padding: '8px 14px', fontSize: '13px', minHeight: '38px' },
      lg: { padding: '11px 18px', fontSize: '15px', minHeight: '46px' },
    }[size];

    return (
      <button
        ref={ref}
        type="button"
        role="button"
        aria-pressed={isSelected}
        disabled={disabled}
        onClick={handleClick}
        className={`custom-toggle-btn ${isSelected ? 'selected' : ''} ${className || ''}`}
        style={{
          display: fullWidth ? 'flex' : 'inline-flex',
          width: fullWidth ? '100%' : 'auto',
          alignItems: 'center',
          justifyContent: 'center',
          gap: 6,
          boxSizing: 'border-box',
          padding: sizeStyles.padding,
          fontSize: sizeStyles.fontSize,
          fontWeight: isSelected ? 600 : 500,
          fontFamily: 'Inter, system-ui, sans-serif',
          lineHeight: 1.4,
          borderRadius: 6,
          border: `1px solid ${isSelected && color !== 'standard' ? palette.border : '#cbd5e1'}`,
          backgroundColor: isSelected ? palette.selectedBg : '#ffffff',
          color: isSelected ? palette.selectedText : '#334155',
          cursor: disabled ? 'not-allowed' : 'pointer',
          opacity: disabled ? 0.45 : 1,
          outline: 'none',
          userSelect: 'none',
          position: 'relative',
          transition: 'all 0.15s ease-in-out',
          ...style,
        }}
        onMouseEnter={(e) => {
          if (!disabled) {
            e.currentTarget.style.backgroundColor = isSelected
              ? palette.selectedHoverBg
              : '#f8fafc';
          }
        }}
        onMouseLeave={(e) => {
          if (!disabled) {
            e.currentTarget.style.backgroundColor = isSelected
              ? palette.selectedBg
              : '#ffffff';
          }
        }}
        {...restProps}
      >
        {children}
      </button>
    );
  }
);

ToggleButton.displayName = 'ToggleButton';
```

### 4. `ToggleButtonGroup` (Keyboard Roving TabIndex & Modes)

```tsx
// components/ToggleButtonGroup.tsx
import React, { useRef, useId } from 'react';
import { ToggleButtonGroupContext } from '../context/ToggleButtonGroupContext';
import { ToggleButtonSize, ToggleButtonColor } from '../types/toggleButton';

export interface ToggleButtonGroupProps {
  value?: any;
  onChange?: (event: React.MouseEvent<HTMLElement>, value: any) => void;
  exclusive?: boolean;
  enforceValueSet?: boolean;
  size?: ToggleButtonSize;
  color?: ToggleButtonColor;
  orientation?: 'horizontal' | 'vertical';
  spacing?: number;
  disabled?: boolean;
  fullWidth?: boolean;
  children: React.ReactNode;
  'aria-label'?: string;
  className?: string;
  style?: React.CSSProperties;
}

export const ToggleButtonGroup: React.FC<ToggleButtonGroupProps> = ({
  value,
  onChange,
  exclusive = false,
  enforceValueSet = false,
  size = 'md',
  color = 'primary',
  orientation = 'horizontal',
  spacing = 0,
  disabled = false,
  fullWidth = false,
  children,
  'aria-label': ariaLabel,
  className,
  style,
}) => {
  const containerRef = useRef<HTMLDivElement>(null);
  const groupId = useId();
  const isVertical = orientation === 'vertical';

  const handleToggle = (event: React.MouseEvent<HTMLElement>, buttonValue: any) => {
    if (!onChange) return;

    if (exclusive) {
      const isCurrentlySelected = value === buttonValue;
      if (isCurrentlySelected && enforceValueSet) {
        // Prevent deselecting when at least 1 item is required
        return;
      }
      onChange(event, isCurrentlySelected ? null : buttonValue);
    } else {
      const currentValues: any[] = Array.isArray(value) ? value : [];
      const isCurrentlySelected = currentValues.includes(buttonValue);

      if (isCurrentlySelected && enforceValueSet && currentValues.length === 1) {
        // Enforce at least one value active in multi-mode
        return;
      }

      const nextValues = isCurrentlySelected
        ? currentValues.filter((v) => v !== buttonValue)
        : [...currentValues, buttonValue];

      onChange(event, nextValues);
    }
  };

  // Keyboard Roving Focus Handler (WAI-ARIA 1.2)
  const handleKeyDown = (e: React.KeyboardEvent<HTMLDivElement>) => {
    if (!containerRef.current) return;
    const buttons = Array.from(
      containerRef.current.querySelectorAll<HTMLButtonElement>('button:not(:disabled)')
    );
    if (buttons.length === 0) return;

    const currentIdx = buttons.indexOf(document.activeElement as HTMLButtonElement);
    if (currentIdx === -1) return;

    let nextIdx = currentIdx;

    if (orientation === 'horizontal') {
      if (e.key === 'ArrowRight') nextIdx = (currentIdx + 1) % buttons.length;
      else if (e.key === 'ArrowLeft') nextIdx = (currentIdx - 1 + buttons.length) % buttons.length;
    } else {
      if (e.key === 'ArrowDown') nextIdx = (currentIdx + 1) % buttons.length;
      else if (e.key === 'ArrowUp') nextIdx = (currentIdx - 1 + buttons.length) % buttons.length;
    }

    if (e.key === 'Home') nextIdx = 0;
    else if (e.key === 'End') nextIdx = buttons.length - 1;

    if (nextIdx !== currentIdx) {
      e.preventDefault();
      buttons[nextIdx].focus();
    }
  };

  const isMerged = spacing === 0;

  return (
    <ToggleButtonGroupContext.Provider
      value={{ value, onChange: handleToggle, size, color, disabled, orientation, spacing }}
    >
      <div
        id={groupId}
        ref={containerRef}
        role="group"
        aria-label={ariaLabel}
        onKeyDown={handleKeyDown}
        className={`custom-toggle-button-group ${className || ''}`}
        style={{
          display: fullWidth ? 'flex' : 'inline-flex',
          flexDirection: isVertical ? 'column' : 'row',
          width: fullWidth ? '100%' : 'auto',
          gap: spacing > 0 ? spacing : 0,
          position: 'relative',
          borderRadius: 6,
          ...style,
        }}
      >
        {children}

        {/* CSS rules for contiguous seamless border merging */}
        {isMerged && (
          <style>{`
            /* Horizontal Merged Borders */
            .custom-toggle-button-group[style*="flex-direction: row"] > .custom-toggle-btn:not(:first-of-type) {
              border-top-left-radius: 0 !important;
              border-bottom-left-radius: 0 !important;
              margin-left: -1px;
            }
            .custom-toggle-button-group[style*="flex-direction: row"] > .custom-toggle-btn:not(:last-of-type) {
              border-top-right-radius: 0 !important;
              border-bottom-right-radius: 0 !important;
            }

            /* Vertical Merged Borders */
            .custom-toggle-button-group[style*="flex-direction: column"] > .custom-toggle-btn:not(:first-of-type) {
              border-top-left-radius: 0 !important;
              border-top-right-radius: 0 !important;
              margin-top: -1px;
            }
            .custom-toggle-button-group[style*="flex-direction: column"] > .custom-toggle-btn:not(:last-of-type) {
              border-bottom-left-radius: 0 !important;
              border-bottom-right-radius: 0 !important;
            }

            /* Keep active / focused button border on top */
            .custom-toggle-btn.selected,
            .custom-toggle-btn:focus-visible {
              z-index: 2;
            }
          `}</style>
        )}
      </div>
    </ToggleButtonGroupContext.Provider>
  );
};
```

### Key Technical Details

| Feature                        | Implementation Mechanism                                                                                                                                                                      |
| ------------------------------ | --------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------- |
| **Exclusive vs Multiple**      | `exclusive={true}` treats `value` as a primitive scalar (`string                                                                                                                              | number   | null`), while multiple mode maintains an array of items (`any[]`). |
| **Enforce Value Set**          | Intercepts click events and blocks state changes when the user attempts to deselect the sole remaining active item.                                                                           |
| **Seamless Border Merging**    | Uses CSS sibling selectors (`margin-left: -1px` for horizontal, `margin-top: -1px` for vertical) when `spacing={0}`, elevating active/focused elements with `z-index: 2` to prevent clipping. |
| **Roving Focus Keyboard Loop** | `onKeyDown` on `ToggleButtonGroup` traps `ArrowLeft`/`ArrowRight` (or `ArrowUp`/`ArrowDown`) along with `Home`/`End` to cycle active button focus across enabled siblings.                    |
| **WAI-ARIA 1.2**               | Group declares `role="group"`, while each button declares `role="button"` and mirrors selection via `aria-pressed={true                                                                       | false}`. |
