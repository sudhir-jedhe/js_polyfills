Here is the fully refactored, feature-packed production version of the **Task Tracker**.

It uses a consolidated **`useReducer`** architecture to handle:

* **Core UX & Organization:** Inline editing, Due Dates (Overdue / Due Today / Upcoming badges), 4 Priority Levels (Low, Medium, High, Urgent), Categories/Tags (`#Work`, `#Personal`, `#Shopping`, etc.), Real-time Search, and Bulk Actions (Select All, Mark All Done, Clear Completed).
* **Interactivity & Polish:** 5-second Undo Toast for deleted tasks, Keyboard shortcuts (`/` focuses search, `Esc` cancels edit/unfocuses, `Enter` saves), Subtasks with a live progress bar, and Dark / Light theme toggle with `localStorage` persistence.
* **Architecture:** JSON Data Export & Import backup buttons, persistent `localStorage` synchronization, and clean reducer actions.

---

### `TaskTracker.jsx`

```jsx
import React, { useReducer, useEffect, useRef, useMemo } from 'react';

// --- CONFIGURATIONS ---
const PRIORITIES = {
  urgent: { label: 'Urgent', weight: 4, badge: 'bg-rose-100 text-rose-700 dark:bg-rose-950 dark:text-rose-300 border-rose-200 dark:border-rose-800' },
  high: { label: 'High', weight: 3, badge: 'bg-orange-100 text-orange-700 dark:bg-orange-950 dark:text-orange-300 border-orange-200 dark:border-orange-800' },
  medium: { label: 'Med', weight: 2, badge: 'bg-amber-100 text-amber-700 dark:bg-amber-950 dark:text-amber-300 border-amber-200 dark:border-amber-800' },
  low: { label: 'Low', weight: 1, badge: 'bg-emerald-100 text-emerald-700 dark:bg-emerald-950 dark:text-emerald-300 border-emerald-200 dark:border-emerald-800' },
};

const CATEGORIES = ['All', 'Work', 'Personal', 'Shopping', 'Other'];

const INITIAL_TASKS = [
  {
    id: 't-1',
    text: 'Ship client milestone report',
    category: 'Work',
    priority: 'urgent',
    dueDate: new Date(Date.now() - 86400000).toISOString().split('T')[0], // Yesterday (Overdue)
    completed: false,
    createdAt: Date.now() - 3600000,
    subtasks: [
      { id: 's-1', text: 'Generate performance graphs', completed: true },
      { id: 's-2', text: 'Proofread executive summary', completed: false },
    ],
  },
  {
    id: 't-2',
    text: 'Grocery restock for the week',
    category: 'Shopping',
    priority: 'medium',
    dueDate: new Date().toISOString().split('T')[0], // Today
    completed: false,
    createdAt: Date.now() - 1800000,
    subtasks: [],
  },
];

// --- STATE REDUCER ---
const initialState = {
  tasks: (() => {
    try {
      const saved = localStorage.getItem('pro_tasks_v1');
      return saved ? JSON.parse(saved) : INITIAL_TASKS;
    } catch {
      return INITIAL_TASKS;
    }
  })(),
  darkMode: localStorage.getItem('pro_tasks_theme') === 'dark',
  filterStatus: 'all', // 'all' | 'active' | 'completed'
  filterCategory: 'All',
  searchQuery: '',
  sortBy: 'priority-desc', // 'priority-desc' | 'priority-asc' | 'dueDate' | 'newest'
  toast: null, // { message, deletedTask, timerId }
};

function taskReducer(state, action) {
  switch (action.type) {
    case 'TOGGLE_THEME':
      return { ...state, darkMode: !state.darkMode };

    case 'ADD_TASK':
      return { ...state, tasks: [action.payload, ...state.tasks] };

    case 'TOGGLE_TASK':
      return {
        ...state,
        tasks: state.tasks.map((t) =>
          t.id === action.payload ? { ...t, completed: !t.completed } : t
        ),
      };

    case 'UPDATE_TASK_TEXT':
      return {
        ...state,
        tasks: state.tasks.map((t) =>
          t.id === action.payload.id ? { ...t, text: action.payload.text } : t
        ),
      };

    case 'DELETE_TASK': {
      const target = state.tasks.find((t) => t.id === action.payload.id);
      return {
        ...state,
        tasks: state.tasks.filter((t) => t.id !== action.payload.id),
        toast: {
          message: `Deleted "${target?.text?.slice(0, 20)}..."`,
          deletedTask: target,
          deletedIndex: state.tasks.findIndex((t) => t.id === action.payload.id),
        },
      };
    }

    case 'RESTORE_TASK':
      if (!state.toast?.deletedTask) return state;
      const reinserted = [...state.tasks];
      reinserted.splice(state.toast.deletedIndex ?? 0, 0, state.toast.deletedTask);
      return { ...state, tasks: reinserted, toast: null };

    case 'CLEAR_TOAST':
      return { ...state, toast: null };

    case 'ADD_SUBTASK':
      return {
        ...state,
        tasks: state.tasks.map((t) =>
          t.id === action.payload.taskId
            ? {
                ...t,
                subtasks: [
                  ...t.subtasks,
                  { id: crypto.randomUUID(), text: action.payload.text, completed: false },
                ],
              }
            : t
        ),
      };

    case 'TOGGLE_SUBTASK':
      return {
        ...state,
        tasks: state.tasks.map((t) =>
          t.id === action.payload.taskId
            ? {
                ...t,
                subtasks: t.subtasks.map((st) =>
                  st.id === action.payload.subtaskId
                    ? { ...st, completed: !st.completed }
                    : st
                ),
              }
            : t
        ),
      };

    case 'BULK_MARK_ALL':
      return {
        ...state,
        tasks: state.tasks.map((t) => ({ ...t, completed: action.payload })),
      };

    case 'BULK_CLEAR_COMPLETED':
      return {
        ...state,
        tasks: state.tasks.filter((t) => !t.completed),
      };

    case 'IMPORT_TASKS':
      return { ...state, tasks: action.payload };

    case 'SET_SEARCH':
      return { ...state, searchQuery: action.payload };

    case 'SET_STATUS_FILTER':
      return { ...state, filterStatus: action.payload };

    case 'SET_CATEGORY_FILTER':
      return { ...state, filterCategory: action.payload };

    case 'SET_SORT':
      return { ...state, sortBy: action.payload };

    default:
      return state;
  }
}

// --- MAIN COMPONENT ---
export default function TaskTracker() {
  const [state, dispatch] = useReducer(taskReducer, initialState);
  const { tasks, darkMode, filterStatus, filterCategory, searchQuery, sortBy, toast } = state;

  // Form local state
  const [inputText, setInputText] = useState('');
  const [inputPriority, setInputPriority] = useState('medium');
  const [inputCategory, setInputCategory] = useState('Work');
  const [inputDueDate, setInputDueDate] = useState('');

  // Editing state
  const [editingId, setEditingId] = useState(null);
  const [editText, setEditText] = useState('');
  const [newSubtaskInputs, setNewSubtaskInputs] = useState({});

  // Refs
  const searchInputRef = useRef(null);
  const editInputRef = useRef(null);
  const fileImportRef = useRef(null);

  // Sync tasks & theme with localStorage
  useEffect(() => {
    localStorage.setItem('pro_tasks_v1', JSON.stringify(tasks));
  }, [tasks]);

  useEffect(() => {
    localStorage.setItem('pro_tasks_theme', darkMode ? 'dark' : 'light');
    if (darkMode) {
      document.documentElement.classList.add('dark');
    } else {
      document.documentElement.classList.remove('dark');
    }
  }, [darkMode]);

  // Toast Auto-dismiss Timer (5s)
  useEffect(() => {
    if (!toast) return;
    const timer = setTimeout(() => dispatch({ type: 'CLEAR_TOAST' }), 5000);
    return () => clearTimeout(timer);
  }, [toast]);

  // Global Keyboard Shortcuts
  useEffect(() => {
    const handleKeyDown = (e) => {
      // Focus search on '/' when not actively typing in an input
      if (e.key === '/' && !['INPUT', 'SELECT', 'TEXTAREA'].includes(document.activeElement.tagName)) {
        e.preventDefault();
        searchInputRef.current?.focus();
      }
      if (e.key === 'Escape') {
        setEditingId(null);
        if (document.activeElement === searchInputRef.current) {
          searchInputRef.current.blur();
        }
      }
    };
    window.addEventListener('keydown', handleKeyDown);
    return () => window.removeEventListener('keydown', handleKeyDown);
  }, []);

  // Form Submit Handler
  const handleAddTask = (e) => {
    e.preventDefault();
    if (!inputText.trim()) return;

    dispatch({
      type: 'ADD_TASK',
      payload: {
        id: crypto.randomUUID(),
        text: inputText.trim(),
        priority: inputPriority,
        category: inputCategory,
        dueDate: inputDueDate || null,
        completed: false,
        createdAt: Date.now(),
        subtasks: [],
      },
    });

    setInputText('');
    setInputDueDate('');
  };

  // Due Date Badge Helpers
  const getDueBadge = (dueDateString) => {
    if (!dueDateString) return null;
    const today = new Date().toISOString().split('T')[0];
    if (dueDateString < today) {
      return { label: 'Overdue', color: 'bg-red-50 text-red-600 dark:bg-red-950 dark:text-red-300' };
    }
    if (dueDateString === today) {
      return { label: 'Due Today', color: 'bg-amber-50 text-amber-700 dark:bg-amber-950 dark:text-amber-300' };
    }
    return { label: dueDateString, color: 'bg-slate-100 text-slate-600 dark:bg-slate-800 dark:text-slate-400' };
  };

  // Filter & Sort Pipeline
  const processedTasks = useMemo(() => {
    return tasks
      .filter((task) => {
        // Status Filter
        if (filterStatus === 'active' && task.completed) return false;
        if (filterStatus === 'completed' && !task.completed) return false;

        // Category Filter
        if (filterCategory !== 'All' && task.category !== filterCategory) return false;

        // Search Filter (checks title & category)
        if (searchQuery.trim()) {
          const q = searchQuery.toLowerCase();
          const matchesTitle = task.text.toLowerCase().includes(q);
          const matchesCategory = task.category.toLowerCase().includes(q);
          if (!matchesTitle && !matchesCategory) return false;
        }

        return true;
      })
      .slice()
      .sort((a, b) => {
        if (sortBy === 'priority-desc') return PRIORITIES[b.priority].weight - PRIORITIES[a.priority].weight;
        if (sortBy === 'priority-asc') return PRIORITIES[a.priority].weight - PRIORITIES[b.priority].weight;
        if (sortBy === 'newest') return b.createdAt - a.createdAt;
        if (sortBy === 'dueDate') {
          if (!a.dueDate) return 1;
          if (!b.dueDate) return -1;
          return a.dueDate.localeCompare(b.dueDate);
        }
        return 0;
      });
  }, [tasks, filterStatus, filterCategory, searchQuery, sortBy]);

  // Export / Import
  const handleExportJSON = () => {
    const blob = new Blob([JSON.stringify(tasks, null, 2)], { type: 'application/json' });
    const url = URL.createObjectURL(blob);
    const link = document.createElement('a');
    link.href = url;
    link.download = `tasks_backup_${new Date().toISOString().split('T')[0]}.json`;
    link.click();
    URL.revokeObjectURL(url);
  };

  const handleImportJSON = (e) => {
    const file = e.target.files?.[0];
    if (!file) return;
    const reader = new FileReader();
    reader.onload = (event) => {
      try {
        const parsed = JSON.parse(event.target.result);
        if (Array.isArray(parsed)) {
          dispatch({ type: 'IMPORT_TASKS', payload: parsed });
        }
      } catch {
        alert('Invalid JSON file format.');
      }
    };
    reader.readAsText(file);
  };

  return (
    <div className={`min-h-screen py-10 px-4 transition-colors ${darkMode ? 'bg-slate-950 text-slate-100' : 'bg-slate-50 text-slate-800'}`}>
      <div className="max-w-2xl mx-auto bg-white dark:bg-slate-900 rounded-2xl shadow-xl border border-slate-200 dark:border-slate-800 p-6 md:p-8">
        
        {/* HEADER & CONTROLS */}
        <header className="flex items-center justify-between pb-6 border-b border-slate-100 dark:border-slate-800">
          <div>
            <h1 className="text-2xl font-black tracking-tight">Task Master Pro</h1>
            <p className="text-xs text-slate-400 mt-1">Press <kbd className="px-1.5 py-0.5 text-[10px] bg-slate-100 dark:bg-slate-800 rounded border border-slate-300 dark:border-slate-700 font-mono">/</kbd> anywhere to search</p>
          </div>

          <div className="flex items-center gap-2">
            <button
              onClick={() => dispatch({ type: 'TOGGLE_THEME' })}
              className="p-2 rounded-lg bg-slate-100 dark:bg-slate-800 hover:bg-slate-200 dark:hover:bg-slate-700 transition"
              title="Toggle Theme"
            >
              {darkMode ? '☀️' : '🌙'}
            </button>
            <button
              onClick={handleExportJSON}
              className="p-2 text-xs rounded-lg bg-slate-100 dark:bg-slate-800 hover:bg-slate-200 dark:hover:bg-slate-700 font-medium"
              title="Export JSON"
            >
              Export
            </button>
            <button
              onClick={() => fileImportRef.current?.click()}
              className="p-2 text-xs rounded-lg bg-slate-100 dark:bg-slate-800 hover:bg-slate-200 dark:hover:bg-slate-700 font-medium"
              title="Import JSON"
            >
              Import
            </button>
            <input type="file" ref={fileImportRef} onChange={handleImportJSON} accept=".json" className="hidden" />
          </div>
        </header>

        {/* INPUT TASK FORM */}
        <form onSubmit={handleAddTask} className="mt-6 space-y-3">
          <div className="flex gap-2">
            <input
              type="text"
              placeholder="What needs doing next?"
              value={inputText}
              onChange={(e) => setInputText(e.target.value)}
              className="flex-1 px-4 py-2.5 text-sm bg-slate-50 dark:bg-slate-800 border border-slate-200 dark:border-slate-700 rounded-xl focus:outline-none focus:ring-2 focus:ring-blue-500"
            />
            <button
              type="submit"
              className="px-5 py-2.5 bg-blue-600 hover:bg-blue-700 text-white text-sm font-semibold rounded-xl transition shadow-sm"
            >
              Add
            </button>
          </div>

          {/* Form Meta Controls */}
          <div className="flex flex-wrap gap-2 text-xs">
            <select
              value={inputPriority}
              onChange={(e) => setInputPriority(e.target.value)}
              className="px-3 py-1.5 bg-slate-50 dark:bg-slate-800 border border-slate-200 dark:border-slate-700 rounded-lg cursor-pointer"
            >
              <option value="low">Low Priority</option>
              <option value="medium">Med Priority</option>
              <option value="high">High Priority</option>
              <option value="urgent">Urgent</option>
            </select>

            <select
              value={inputCategory}
              onChange={(e) => setInputCategory(e.target.value)}
              className="px-3 py-1.5 bg-slate-50 dark:bg-slate-800 border border-slate-200 dark:border-slate-700 rounded-lg cursor-pointer"
            >
              {CATEGORIES.filter((c) => c !== 'All').map((c) => (
                <option key={c} value={c}>#{c}</option>
              ))}
            </select>

            <input
              type="date"
              value={inputDueDate}
              onChange={(e) => setInputDueDate(e.target.value)}
              className="px-3 py-1.5 bg-slate-50 dark:bg-slate-800 border border-slate-200 dark:border-slate-700 rounded-lg cursor-pointer"
            />
          </div>
        </form>

        {/* SEARCH BAR */}
        <div className="mt-6 relative">
          <span className="absolute inset-y-0 left-3 flex items-center text-slate-400 text-xs">🔍</span>
          <input
            ref={searchInputRef}
            type="text"
            placeholder="Search by title or category... (Press 'Esc' to exit)"
            value={searchQuery}
            onChange={(e) => dispatch({ type: 'SET_SEARCH', payload: e.target.value })}
            className="w-full pl-9 pr-4 py-2 text-xs bg-slate-50 dark:bg-slate-800 border border-slate-200 dark:border-slate-700 rounded-xl focus:outline-none focus:ring-2 focus:ring-blue-500"
          />
        </div>

        {/* FILTER & SORT TOOLBAR */}
        <div className="mt-4 flex flex-wrap items-center justify-between gap-3 text-xs">
          {/* Status Tabs */}
          <div className="flex bg-slate-100 dark:bg-slate-800 p-1 rounded-lg">
            {['all', 'active', 'completed'].map((status) => (
              <button
                key={status}
                onClick={() => dispatch({ type: 'SET_STATUS_FILTER', payload: status })}
                className={`px-3 py-1 capitalize rounded-md transition font-medium ${
                  filterStatus === status
                    ? 'bg-white dark:bg-slate-700 shadow-sm text-blue-600 dark:text-blue-300'
                    : 'text-slate-500 hover:text-slate-700'
                }`}
              >
                {status}
              </button>
            ))}
          </div>

          {/* Category Pills */}
          <div className="flex items-center gap-1 overflow-x-auto">
            {CATEGORIES.map((cat) => (
              <button
                key={cat}
                onClick={() => dispatch({ type: 'SET_CATEGORY_FILTER', payload: cat })}
                className={`px-2.5 py-1 rounded-lg border text-[11px] transition ${
                  filterCategory === cat
                    ? 'border-blue-500 bg-blue-50 dark:bg-blue-950 text-blue-600 dark:text-blue-300 font-semibold'
                    : 'border-slate-200 dark:border-slate-700 text-slate-500'
                }`}
              >
                {cat === 'All' ? 'All Tags' : `#${cat}`}
              </button>
            ))}
          </div>

          {/* Sort Selector */}
          <select
            value={sortBy}
            onChange={(e) => dispatch({ type: 'SET_SORT', payload: e.target.value })}
            className="px-2 py-1 bg-white dark:bg-slate-800 border border-slate-200 dark:border-slate-700 rounded-lg text-slate-600 dark:text-slate-300"
          >
            <option value="priority-desc">Priority: High → Low</option>
            <option value="priority-asc">Priority: Low → High</option>
            <option value="dueDate">Due Date</option>
            <option value="newest">Newest First</option>
          </select>
        </div>

        {/* BULK ACTIONS */}
        <div className="mt-4 flex items-center justify-between text-xs text-slate-400 py-1">
          <div className="flex gap-3">
            <button
              onClick={() => dispatch({ type: 'BULK_MARK_ALL', payload: true })}
              className="hover:text-blue-500 underline decoration-slate-300"
            >
              Mark All Complete
            </button>
            <button
              onClick={() => dispatch({ type: 'BULK_MARK_ALL', payload: false })}
              className="hover:text-blue-500 underline decoration-slate-300"
            >
              Unmark All
            </button>
          </div>
          <button
            onClick={() => dispatch({ type: 'BULK_CLEAR_COMPLETED' })}
            className="hover:text-rose-500 transition"
          >
            Clear Completed
          </button>
        </div>

        {/* TASK LIST */}
        <ul className="mt-4 divide-y divide-slate-100 dark:divide-slate-800">
          {processedTasks.length === 0 ? (
            <li className="py-12 text-center text-sm text-slate-400">
              No matching tasks found.
            </li>
          ) : (
            processedTasks.map((task) => {
              const priorityInfo = PRIORITIES[task.priority] || PRIORITIES.low;
              const dueBadge = getDueBadge(task.dueDate);
              const isEditing = editingId === task.id;

              // Subtask Progress
              const completedSubtasks = task.subtasks.filter((st) => st.completed).length;
              const totalSubtasks = task.subtasks.length;
              const subtaskProgress = totalSubtasks > 0 ? (completedSubtasks / totalSubtasks) * 100 : 0;

              return (
                <li key={task.id} className="py-4 space-y-2 group">
                  {/* Main Task Item */}
                  <div className="flex items-center justify-between gap-3">
                    <div className="flex items-center gap-3 flex-1 min-w-0">
                      <input
                        type="checkbox"
                        checked={task.completed}
                        onChange={() => dispatch({ type: 'TOGGLE_TASK', payload: task.id })}
                        className="w-4 h-4 rounded text-blue-600 focus:ring-blue-500 cursor-pointer"
                      />

                      {isEditing ? (
                        <input
                          ref={editInputRef}
                          type="text"
                          value={editText}
                          onChange={(e) => setEditText(e.target.value)}
                          onBlur={() => {
                            if (editText.trim()) {
                              dispatch({ type: 'UPDATE_TASK_TEXT', payload: { id: task.id, text: editText.trim() } });
                            }
                            setEditingId(null);
                          }}
                          onKeyDown={(e) => {
                            if (e.key === 'Enter') {
                              if (editText.trim()) {
                                dispatch({ type: 'UPDATE_TASK_TEXT', payload: { id: task.id, text: editText.trim() } });
                              }
                              setEditingId(null);
                            }
                            if (e.key === 'Escape') setEditingId(null);
                          }}
                          className="flex-1 px-2 py-0.5 text-sm bg-blue-50 dark:bg-slate-800 border border-blue-400 rounded focus:outline-none"
                        />
                      ) : (
                        <span
                          onDoubleClick={() => {
                            setEditingId(task.id);
                            setEditText(task.text);
                          }}
                          className={`text-sm truncate select-none cursor-pointer flex-1 ${
                            task.completed ? 'line-through text-slate-400' : 'text-slate-800 dark:text-slate-200'
                          }`}
                          title="Double-click to edit"
                        >
                          {task.text}
                        </span>
                      )}
                    </div>

                    {/* Metadata & Controls */}
                    <div className="flex items-center gap-2">
                      <span className="text-[10px] px-2 py-0.5 rounded-full bg-slate-100 dark:bg-slate-800 text-slate-600 dark:text-slate-400 font-medium">
                        #{task.category}
                      </span>

                      {dueBadge && (
                        <span className={`text-[10px] px-2 py-0.5 rounded-md font-medium ${dueBadge.color}`}>
                          {dueBadge.label}
                        </span>
                      )}

                      <span className={`text-[10px] px-2 py-0.5 rounded-md border font-semibold ${priorityInfo.badge}`}>
                        {priorityInfo.label}
                      </span>

                      <button
                        onClick={() => dispatch({ type: 'DELETE_TASK', payload: { id: task.id } })}
                        className="p-1 text-slate-400 hover:text-rose-500 rounded opacity-0 group-hover:opacity-100 transition"
                        title="Delete Task"
                      >
                        ✕
                      </button>
                    </div>
                  </div>

                  {/* SUBTASKS & CHECKLIST PROGRESS */}
                  <div className="pl-7 pt-1">
                    {totalSubtasks > 0 && (
                      <div className="mb-2">
                        <div className="flex items-center justify-between text-[11px] text-slate-400 mb-1">
                          <span>Progress</span>
                          <span>{completedSubtasks}/{totalSubtasks}</span>
                        </div>
                        <div className="w-full bg-slate-100 dark:bg-slate-800 h-1 rounded-full overflow-hidden">
                          <div
                            className="bg-blue-600 h-full transition-all duration-300"
                            style={{ width: `${subtaskProgress}%` }}
                          />
                        </div>
                      </div>
                    )}

                    {/* Subtask Checkboxes */}
                    <div className="space-y-1">
                      {task.subtasks.map((st) => (
                        <label key={st.id} className="flex items-center gap-2 text-xs text-slate-600 dark:text-slate-400 cursor-pointer">
                          <input
                            type="checkbox"
                            checked={st.completed}
                            onChange={() => dispatch({ type: 'TOGGLE_SUBTASK', payload: { taskId: task.id, subtaskId: st.id } })}
                            className="w-3.5 h-3.5 rounded text-blue-600"
                          />
                          <span className={st.completed ? 'line-through text-slate-400' : ''}>{st.text}</span>
                        </label>
                      ))}
                    </div>

                    {/* Add Subtask Quick Input */}
                    <div className="mt-2 flex items-center gap-2">
                      <input
                        type="text"
                        placeholder="+ Add subtask (press Enter)"
                        value={newSubtaskInputs[task.id] || ''}
                        onChange={(e) => setNewSubtaskInputs({ ...newSubtaskInputs, [task.id]: e.target.value })}
                        onKeyDown={(e) => {
                          if (e.key === 'Enter' && (newSubtaskInputs[task.id] || '').trim()) {
                            e.preventDefault();
                            dispatch({
                              type: 'ADD_SUBTASK',
                              payload: { taskId: task.id, text: newSubtaskInputs[task.id].trim() },
                            });
                            setNewSubtaskInputs({ ...newSubtaskInputs, [task.id]: '' });
                          }
                        }}
                        className="text-xs bg-transparent text-slate-500 placeholder-slate-400 focus:outline-none border-b border-transparent focus:border-slate-300 dark:focus:border-slate-600 pb-0.5"
                      />
                    </div>
                  </div>
                </li>
              );
            })
          )}
        </ul>
      </div>

      {/* FLOATING UNDO TOAST (5s Auto-dismiss) */}
      {toast && (
        <div className="fixed bottom-6 right-6 bg-slate-900 text-white dark:bg-white dark:text-slate-950 px-4 py-3 rounded-xl shadow-2xl flex items-center gap-4 text-xs font-medium z-50 border border-slate-700 dark:border-slate-200">
          <span>{toast.message}</span>
          <button
            onClick={() => dispatch({ type: 'RESTORE_TASK' })}
            className="text-amber-400 dark:text-amber-600 font-bold hover:underline"
          >
            Undo
          </button>
          <button
            onClick={() => dispatch({ type: 'CLEAR_TOAST' })}
            className="text-slate-400 hover:text-white dark:hover:text-slate-950 ml-1"
          >
            ✕
          </button>
        </div>
      )}
    </div>
  );
}

```

---

### Key Architectural Highlights

1. **Deterministic State Transitions with `useReducer`:**

* Eliminates state race conditions across complex interactions (e.g., deleting a task triggers both array filtering and snapshot caching for the undo buffer in a single atomic dispatch).

1. **5-Second Undo Toast:**

* When an item is deleted, its previous position and data are preserved in `state.toast`. The user has 5 seconds to restore it exactly where it was before the timer cleans up.

1. **Subtasks & Progress Tracking:**

* Each task contains a `subtasks` array. Progress bars dynamically compute `(completedSubtasks / totalSubtasks) * 100` without triggering extra state updates.

1. **Accessible Keyboard Shortcuts:**

* Pressing `/` globally shifts focus directly to the search input, and `Escape` immediately drops focus or cancels an ongoing inline title edit.

1. **Portable Data Backup:**

* The Export button compiles state into a JSON Blob and initiates a dynamic browser download. The file input parses backup JSON arrays directly back into state.
