Here is a complete, self-contained **Task Tracker** component written in modern React (with hooks). It includes:

* Adding new tasks with basic validation.
* Toggling completion status.
* Deleting tasks.
* Filtering (`All`, `Active`, `Completed`).
* Persistent state using `localStorage`.
* Styled using clean, responsive Tailwind CSS (or inline styles/classes easily swappable).

---

### `TaskTracker.jsx`

```jsx
import React, { useState, useEffect } from 'react';

export default function TaskTracker() {
  // Initialize state from localStorage or fallback to default sample tasks
  const [tasks, setTasks] = useState(() => {
    const saved = localStorage.getItem('react_tasks');
    if (saved) {
      try {
        return JSON.parse(saved);
      } catch (e) {
        console.error('Failed to parse saved tasks', e);
      }
    }
    return [
      { id: 1, text: 'Set up Vite and Tailwind CSS', completed: true },
      { id: 2, text: 'Implement task filtering logic', completed: false },
      { id: 3, text: 'Connect state to localStorage', completed: false },
    ];
  });

  const [inputVal, setInputVal] = useState('');
  const [filter, setFilter] = useState('all'); // 'all' | 'active' | 'completed'

  // Persist tasks whenever they change
  useEffect(() => {
    localStorage.setItem('react_tasks', JSON.stringify(tasks));
  }, [tasks]);

  // Handler: Add a new task
  const handleAddTask = (e) => {
    e.preventDefault();
    const trimmed = inputVal.trim();
    if (!trimmed) return;

    const newTask = {
      id: crypto.randomUUID ? crypto.randomUUID() : Date.now(),
      text: trimmed,
      completed: false,
    };

    setTasks((prev) => [newTask, ...prev]);
    setInputVal('');
  };

  // Handler: Toggle complete status
  const handleToggleTask = (id) => {
    setTasks((prev) =>
      prev.map((task) =>
        task.id === id ? { ...task, completed: !task.completed } : task
      )
    );
  };

  // Handler: Delete task
  const handleDeleteTask = (id) => {
    setTasks((prev) => prev.filter((task) => task.id !== id));
  };

  // Filtered task computation
  const filteredTasks = tasks.filter((task) => {
    if (filter === 'active') return !task.completed;
    if (filter === 'completed') return task.completed;
    return true;
  });

  const activeCount = tasks.filter((t) => !t.completed).length;

  return (
    <div className="max-w-md mx-auto my-10 p-6 bg-white rounded-xl shadow-lg border border-slate-100 font-sans">
      <header className="mb-6">
        <h1 className="text-2xl font-bold text-slate-800">Task Tracker</h1>
        <p className="text-sm text-slate-500">
          {activeCount} {activeCount === 1 ? 'task' : 'tasks'} remaining
        </p>
      </header>

      {/* Input Form */}
      <form onSubmit={handleAddTask} className="flex gap-2 mb-6">
        <input
          type="text"
          placeholder="What needs to be done?"
          value={inputVal}
          onChange={(e) => setInputVal(e.target.value)}
          className="flex-1 px-4 py-2 text-sm border border-slate-200 rounded-lg focus:outline-none focus:ring-2 focus:ring-blue-500 transition"
        />
        <button
          type="submit"
          className="px-4 py-2 bg-blue-600 hover:bg-blue-700 text-white text-sm font-medium rounded-lg transition shadow-sm"
        >
          Add
        </button>
      </form>

      {/* Filter Tabs */}
      <div className="flex justify-center gap-2 mb-4 p-1 bg-slate-100 rounded-lg text-xs font-medium">
        {['all', 'active', 'completed'].map((type) => (
          <button
            key={type}
            onClick={() => setFilter(type)}
            className={`flex-1 py-1.5 capitalize rounded-md transition ${
              filter === type
                ? 'bg-white text-slate-800 shadow-sm font-semibold'
                : 'text-slate-500 hover:text-slate-700'
            }`}
          >
            {type}
          </button>
        ))}
      </div>

      {/* Task List */}
      <ul className="divide-y divide-slate-100">
        {filteredTasks.length === 0 ? (
          <li className="py-8 text-center text-sm text-slate-400">
            No {filter !== 'all' ? filter : ''} tasks found.
          </li>
        ) : (
          filteredTasks.map((task) => (
            <li
              key={task.id}
              className="group flex items-center justify-between py-3 gap-3 transition"
            >
              <label className="flex items-center gap-3 flex-1 cursor-pointer overflow-hidden">
                <input
                  type="checkbox"
                  checked={task.completed}
                  onChange={() => handleToggleTask(task.id)}
                  className="w-4 h-4 rounded text-blue-600 border-slate-300 focus:ring-blue-500 cursor-pointer"
                />
                <span
                  className={`text-sm truncate select-none ${
                    task.completed
                      ? 'line-through text-slate-400'
                      : 'text-slate-700'
                  }`}
                >
                  {task.text}
                </span>
              </label>

              <button
                onClick={() => handleDeleteTask(task.id)}
                title="Delete task"
                className="opacity-0 group-hover:opacity-100 p-1 text-slate-400 hover:text-red-500 rounded transition"
              >
                <svg
                  className="w-4 h-4"
                  fill="none"
                  stroke="currentColor"
                  viewBox="0 0 24 24"
                >
                  <path
                    strokeLinecap="round"
                    strokeLinejoin="round"
                    strokeWidth="2"
                    d="M19 7l-.867 12.142A2 2 0 0116.138 21H7.862a2 2 0 01-1.995-1.858L5 7m5 4v6m4-6v6m1-10V4a1 1 0 00-1-1h-4a1 1 0 00-1 1v3M4 7h16"
                  />
                </svg>
              </button>
            </li>
          ))
        )}
      </ul>
    </div>
  );
}

```

### Key React Concepts Used

1. **Lazy State Initialization (`useState(() => ...))`:** Prevents reading from `localStorage` on every render, executing the parsing only during initial mount.
2. **Derived State:** The filtered task list (`filteredTasks`) and remaining count (`activeCount`) are computed during render rather than kept in redundant state variables, preventing state desynchronization.
3. **Controlled Component:** The input box binds its value directly to `inputVal` and updates via `onChange`.
4. **Immutable Updates:** State setters update arrays using the spread operator (`[...prev]`) and array methods like `.map()` and `.filter()`.
