Here is the refactored, fully typed **TypeScript** implementation of the `FilterableProductTable` architecture.

It includes:

1. **Mock API integration with `useEffect`:** Simulates network latency, abort signals with `AbortController`, and handles loading/error states.
2. **Column sorting:** Clickable column headers (`Name`, `Price`, and `Stock`) with ascending/descending indicators.
3. **Strict TypeScript:** Strongly typed props, state interfaces, sort configurations, and product models.

---

### `FilterableProductTable.tsx`

```tsx
import React, { useState, useEffect, useMemo } from 'react';

// --- TYPE DEFINITIONS ---
export interface Product {
  id: string;
  category: string;
  price: number; // Stored as a raw numeric value for reliable sorting
  stocked: boolean;
  name: string;
}

export type SortField = 'name' | 'price' | 'stocked';
export type SortDirection = 'asc' | 'desc';

export interface SortConfig {
  field: SortField;
  direction: SortDirection;
}

interface ProductCategoryRowProps {
  category: string;
}

interface ProductRowProps {
  product: Product;
}

interface ProductTableProps {
  products: Product[];
  filterText: string;
  inStockOnly: string | boolean;
  sortConfig: SortConfig;
  onSortChange: (field: SortField) => void;
}

interface SearchBarProps {
  filterText: string;
  inStockOnly: boolean;
  onFilterTextChange: (text: string) => void;
  onInStockOnlyChange: (inStockOnly: boolean) => void;
}

// --- MOCK API SERVICE ---
const MOCK_DATA: Product[] = [
  { id: '1', category: 'Fruits', price: 1.0, stocked: true, name: 'Apple' },
  { id: '2', category: 'Fruits', price: 1.5, stocked: true, name: 'Dragonfruit' },
  { id: '3', category: 'Fruits', price: 2.0, stocked: false, name: 'Passionfruit' },
  { id: '4', category: 'Vegetables', price: 2.5, stocked: true, name: 'Spinach' },
  { id: '5', category: 'Vegetables', price: 4.0, stocked: false, name: 'Pumpkin' },
  { id: '6', category: 'Vegetables', price: 1.25, stocked: true, name: 'Peas' },
];

function fetchMockProducts(signal?: AbortSignal): Promise<Product[]> {
  return new Promise((resolve, reject) => {
    const timer = setTimeout(() => {
      if (signal?.aborted) {
        reject(new DOMException('Aborted', 'AbortError'));
      } else {
        resolve(MOCK_DATA);
      }
    }, 700);

    signal?.addEventListener('abort', () => {
      clearTimeout(timer);
      reject(new DOMException('Aborted', 'AbortError'));
    });
  });
}

// --- SUBCOMPONENTS ---

// 1. ProductCategoryRow
export const ProductCategoryRow: React.FC<ProductCategoryRowProps> = ({ category }) => {
  return (
    <tr>
      <th
        colSpan={3}
        className="pt-4 pb-1 text-left text-xs font-bold uppercase tracking-wider text-slate-500 border-b border-slate-200 bg-slate-50/50"
      >
        {category}
      </th>
    </tr>
  );
};

// 2. ProductRow
export const ProductRow: React.FC<ProductRowProps> = ({ product }) => {
  return (
    <tr className="hover:bg-slate-50/80 transition-colors">
      <td className="py-2.5 text-sm">
        <span className={product.stocked ? 'text-slate-800' : 'text-rose-500 font-medium'}>
          {product.name}
        </span>
      </td>
      <td className="py-2.5 text-sm text-slate-600 font-mono">
        ${product.price.toFixed(2)}
      </td>
      <td className="py-2.5 text-sm text-right">
        <span
          className={`inline-block px-2 py-0.5 text-xs rounded-full font-medium ${
            product.stocked
              ? 'bg-emerald-50 text-emerald-700 border border-emerald-200'
              : 'bg-rose-50 text-rose-700 border border-rose-200'
          }`}
        >
          {product.stocked ? 'In Stock' : 'Out of Stock'}
        </span>
      </td>
    </tr>
  );
};

// 3. ProductTable
export const ProductTable: React.FC<ProductTableProps> = ({
  products,
  filterText,
  inStockOnly,
  sortConfig,
  onSortChange,
}) => {
  // Filter and sort computation
  const filteredAndSortedProducts = useMemo(() => {
    return products
      .filter((product) => {
        const matchesQuery = product.name.toLowerCase().includes(filterText.toLowerCase());
        const matchesStock = inStockOnly ? product.stocked : true;
        return matchesQuery && matchesStock;
      })
      .slice()
      .sort((a, b) => {
        // Group by category first so category groupings remain intact
        if (a.category !== b.category) {
          return a.category.localeCompare(b.category);
        }

        const modifier = sortConfig.direction === 'asc' ? 1 : -1;
        if (sortConfig.field === 'name') {
          return a.name.localeCompare(b.name) * modifier;
        }
        if (sortConfig.field === 'price') {
          return (a.price - b.price) * modifier;
        }
        if (sortConfig.field === 'stocked') {
          return (Number(a.stocked) - Number(b.stocked)) * modifier;
        }
        return 0;
      });
  }, [products, filterText, inStockOnly, sortConfig]);

  const renderSortIndicator = (field: SortField) => {
    if (sortConfig.field !== field) {
      return <span className="text-slate-300 ml-1">↕</span>;
    }
    return <span className="text-blue-600 ml-1">{sortConfig.direction === 'asc' ? '↑' : '↓'}</span>;
  };

  // Build grouped rows
  const rows: React.ReactNode[] = [];
  let lastCategory: string | null = null;

  filteredAndSortedProducts.forEach((product) => {
    if (product.category !== lastCategory) {
      rows.push(
        <ProductCategoryRow
          category={product.category}
          key={`cat-${product.category}`}
        />
      );
      lastCategory = product.category;
    }
    rows.push(<ProductRow product={product} key={product.id} />);
  });

  return (
    <div className="overflow-x-auto">
      <table className="w-full text-left border-collapse">
        <thead>
          <tr className="border-b border-slate-200 select-none text-xs font-semibold text-slate-600 uppercase tracking-wider">
            <th
              onClick={() => onSortChange('name')}
              className="py-2.5 cursor-pointer hover:text-blue-600 transition"
            >
              Name {renderSortIndicator('name')}
            </th>
            <th
              onClick={() => onSortChange('price')}
              className="py-2.5 cursor-pointer hover:text-blue-600 transition"
            >
              Price {renderSortIndicator('price')}
            </th>
            <th
              onClick={() => onSortChange('stocked')}
              className="py-2.5 text-right cursor-pointer hover:text-blue-600 transition"
            >
              Stock {renderSortIndicator('stocked')}
            </th>
          </tr>
        </thead>
        <tbody className="divide-y divide-slate-100">
          {rows.length === 0 ? (
            <tr>
              <td colSpan={3} className="py-8 text-center text-sm text-slate-400">
                No matching products found.
              </td>
            </tr>
          ) : (
            rows
          )}
        </tbody>
      </table>
    </div>
  );
};

// 4. SearchBar
export const SearchBar: React.FC<SearchBarProps> = ({
  filterText,
  inStockOnly,
  onFilterTextChange,
  onInStockOnlyChange,
}) => {
  return (
    <form onSubmit={(e) => e.preventDefault()} className="mb-5 space-y-2.5">
      <div className="relative">
        <span className="absolute inset-y-0 left-3 flex items-center text-slate-400 text-xs">
          🔍
        </span>
        <input
          type="text"
          value={filterText}
          placeholder="Search products..."
          onChange={(e) => onFilterTextChange(e.target.value)}
          className="w-full pl-8 pr-3 py-2 text-sm border border-slate-200 rounded-lg focus:outline-none focus:ring-2 focus:ring-blue-500 bg-slate-50 transition"
        />
      </div>
      <label className="flex items-center gap-2 text-xs text-slate-600 cursor-pointer select-none">
        <input
          type="checkbox"
          checked={inStockOnly}
          onChange={(e) => onInStockOnlyChange(e.target.checked)}
          className="w-4 h-4 rounded text-blue-600 border-slate-300 focus:ring-blue-500 cursor-pointer"
        />
        Only show products in stock
      </label>
    </form>
  );
};

// 5. FilterableProductTable (Container)
export default function FilterableProductTable() {
  const [products, setProducts] = useState<Product[]>([]);
  const [isLoading, setIsLoading] = useState<boolean>(true);
  const [error, setError] = useState<string | null>(null);

  const [filterText, setFilterText] = useState<string>('');
  const [inStockOnly, setInStockOnly] = useState<boolean>(false);
  const [sortConfig, setSortConfig] = useState<SortConfig>({
    field: 'name',
    direction: 'asc',
  });

  useEffect(() => {
    const controller = new AbortController();
    setIsLoading(true);
    setError(null);

    fetchMockProducts(controller.signal)
      .then((data) => {
        setProducts(data);
        setIsLoading(false);
      })
      .catch((err: unknown) => {
        if (err instanceof DOMException && err.name === 'AbortError') {
          return; // Unmount cleanup
        }
        setError(err instanceof Error ? err.message : 'Failed to fetch products');
        setIsLoading(false);
      });

    return () => controller.abort();
  }, []);

  const handleSortChange = (field: SortField) => {
    setSortConfig((prev) => ({
      field,
      direction: prev.field === field && prev.direction === 'asc' ? 'desc' : 'asc',
    }));
  };

  return (
    <div className="max-w-md mx-auto my-10 p-6 bg-white rounded-xl shadow-lg border border-slate-100 font-sans">
      <header className="mb-5 pb-3 border-b border-slate-100 flex justify-between items-center">
        <h2 className="text-xl font-bold text-slate-800">Product Inventory</h2>
        {isLoading && (
          <span className="text-xs text-blue-600 animate-pulse font-medium">Loading...</span>
        )}
      </header>

      <SearchBar
        filterText={filterText}
        inStockOnly={inStockOnly}
        onFilterTextChange={setFilterText}
        onInStockOnlyChange={setInStockOnly}
      />

      {error ? (
        <div className="py-6 text-center text-sm text-rose-500 bg-rose-50 rounded-lg border border-rose-100">
          {error}
        </div>
      ) : isLoading ? (
        <div className="space-y-3 py-4">
          <div className="h-4 bg-slate-100 rounded w-1/3 animate-pulse" />
          <div className="h-6 bg-slate-100 rounded w-full animate-pulse" />
          <div className="h-6 bg-slate-100 rounded w-full animate-pulse" />
        </div>
      ) : (
        <ProductTable
          products={products}
          filterText={filterText}
          inStockOnly={inStockOnly}
          sortConfig={sortConfig}
          onSortChange={handleSortChange}
        />
      )}
    </div>
  );
}

```

### Highlights of Technical Improvements

* **Clean Sorting Within Categories:** The sort comparator explicitly compares `a.category.localeCompare(b.category)` first, ensuring rows are sorted inside their relevant category headers without splitting the sections across the table.
* **Safe Asynchronous Fetching:** Uses standard `AbortController` in the `useEffect` cleanup return function to cancel outgoing requests if the component unmounts before response completion.
* **Type-Safe Numerical Storage:** The mock model stores prices as `number` (e.g., `1.5` instead of `"$1.50"`), which allows mathematical comparisons (`a.price - b.price`) without string manipulation parsing. Formatting into `$1.50` occurs strictly at the presentation layer.
