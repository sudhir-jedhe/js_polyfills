### 1. Most Frequent Element / Character

* **Method 1: Hash Map / Frequency Object** ($O(N)$)

```javascript
function mostFrequent(arr) {
  const count = {};
  let maxItem = arr[0], maxCount = 0;
  for (const item of arr) {
    count[item] = (count[item] || 0) + 1;
    if (count[item] > maxCount) {
      maxCount = count[item];
      maxItem = item;
    }
  }
  return maxItem;
}

```

* **Method 2: `Array.prototype.reduce` with `Map**` ($O(N)$)

```javascript
const mostFrequentReduce = arr => 
  [...arr.reduce((map, el) => map.set(el, (map.get(el) || 0) + 1), new Map()).entries()]
    .reduce((a, b) => b[1] > a[1] ? b : a)[0];

```

### 2. First Non-Repeating Character (First Unique)

* **Method 1: Two-Pass Frequency Map** ($O(N)$)

```javascript
function firstNonRepeating(str) {
  const count = {};
  for (const char of str) count[char] = (count[char] || 0) + 1;
  for (const char of str) {
    if (count[char] === 1) return char;
  }
  return null;
}

```

* **Method 2: `indexOf` and `lastIndexOf**` ($O(N^2)$)

```javascript
const firstUnique = str => 
  [...str].find(c => str.indexOf(c) === str.lastIndexOf(c)) || null;
```

### 3. Find Duplicates in List

* **Method 1: `Set` Tracking** ($O(N)$)

```javascript
function findDuplicates(arr) {
  const seen = new Set();
  const duplicates = new Set();
  for (const item of arr) {
    if (seen.has(item)) duplicates.add(item);
    else seen.add(item);
  }
  return [...duplicates];
}

```

* **Method 2: `filter` and `indexOf**` ($O(N^2)$)

```javascript
const findDuplicatesFilter = arr => 
  [...new Set(arr.filter((item, index) => arr.indexOf(item) !== index))];

```

### 5. String Palindrome

* **Method 1: Two-Pointer Approach** ($O(N)$ time, $O(1)$ space)

```javascript
function isPalindrome(str) {
  const clean = str.toLowerCase().replace(/[\W_]/g, '');
  let left = 0, right = clean.length - 1;
  while (left < right) {
    if (clean[left] !== clean[right]) return false;
    left++;
    right--;
  }
  return true;
}

```

* **Method 2: Built-in Reverse Chain** ($O(N)$)

```javascript
const isPalindromeBuiltIn = str => {
  const clean = str.toLowerCase().replace(/[\W_]/g, '');
  return clean === clean.split('').reverse().join('');
};

```

### 7. Reverse Array

* **Method 1: In-Place Two Pointers** ($O(N)$ time, $O(1)$ space)

```javascript
function reverseArray(arr) {
  let left = 0, right = arr.length - 1;
  while (left < right) {
    [arr[left], arr[right]] = [arr[right], arr[left]];
    left++;
    right--;
  }
  return arr;
}

```

* **Method 2: Built-in In-place / Non-mutating**

```javascript
const reverseInPlace = arr => arr.reverse();
const reverseImmutable = arr => arr.toReversed(); // ES2023
```

### 8. Second Largest Number

* **Method 1: Single Pass Iteration** ($O(N)$ time, $O(1)$ space)

```javascript
function secondLargest(arr) {
  let first = -Infinity, second = -Infinity;
  for (const num of arr) {
    if (num > first) {
      second = first;
      first = num;
    } else if (num > second && num !== first) {
      second = num;
    }
  }
  return second === -Infinity ? null : second;
}

```

* **Method 2: Deduplication + Sorting** ($O(N \log N)$)

```javascript
const secondLargestSort = arr => {
  const unique = [...new Set(arr)].sort((a, b) => b - a);
  return unique.length >= 2 ? unique[1] : null;
};

```

### 11. Find Pair with Given Sum (Two Sum)

* **Method 1: Complement Hash Map** ($O(N)$ time, $O(N)$ space)

```javascript
function twoSum(arr, target) {
  const seen = new Map();
  for (let i = 0; i < arr.length; i++) {
    const complement = target - arr[i];
    if (seen.has(complement)) return [seen.get(complement), i];
    seen.set(arr[i], i);
  }
  return [];
}

```

* **Method 2: Two Pointers (For Sorted Arrays)** ($O(N)$ time, $O(1)$ space)

```javascript
function twoSumSorted(sortedArr, target) {
  let left = 0, right = sortedArr.length - 1;
  while (left < right) {
    const sum = sortedArr[left] + sortedArr[right];
    if (sum === target) return [sortedArr[left], sortedArr[right]];
    sum < target ? left++ : right--;
  }
  return [];
}

```

### 12. Intersection of 3 Arrays (Common Elements)

* **Method 1: `Set.prototype.has` Filter** ($O(N + M + K)$)

```javascript
function intersectionOf3(a1, a2, a3) {
  const set2 = new Set(a2);
  const set3 = new Set(a3);
  return [...new Set(a1.filter(item => set2.has(item) && set3.has(item)))];
}

```

* **Method 2: Three Pointers (For Sorted Arrays)** ($O(N + M + K)$ time, $O(1)$ space)

```javascript
function intersection3Sorted(a, b, c) {
  let i = 0, j = 0, k = 0;
  const res = [];
  while (i < a.length && j < b.length && k < c.length) {
    if (a[i] === b[j] && b[j] === c[k]) {
      res.push(a[i]);
      i++; j++; k++;
    } else if (a[i] < b[j]) i++;
    else if (b[j] < c[k]) j++;
    else k++;
  }
  return res;
}

```

### 13. Longest Common Prefix

* **Method 1: Horizontal Scanning** ($O(S)$)

```javascript
function longestCommonPrefix(strs) {
  if (!strs.length) return "";
  let prefix = strs[0];
  for (let i = 1; i < strs.length; i++) {
    while (!strs[i].startsWith(prefix)) {
      prefix = prefix.slice(0, -1);
      if (!prefix) return "";
    }
  }
  return prefix;
}

```

* **Method 2: Sorting Comparison (First vs. Last)** ($O(N \log N + M)$)

```javascript
function longestCommonPrefixSort(strs) {
  if (!strs.length) return "";
  strs.sort();
  const first = strs[0], last = strs[strs.length - 1];
  let i = 0;
  while (i < first.length && first[i] === last[i]) i++;
  return first.slice(0, i);
}

```

### 14. Find Missing Number / Character in Sequence

* **Method 1: Missing Number via Gauss Sum Formula** ($O(N)$ time, $O(1)$ space)

```javascript
function findMissingNumber(arr, n) {
  const expectedSum = (n * (n + 1)) / 2;
  const actualSum = arr.reduce((sum, el) => sum + el, 0);
  return expectedSum - actualSum;
}

```

* **Method 2: Missing Character via CharCode Offset** ($O(N)$)

```javascript
function findMissingChar(str) {
  for (let i = 0; i < str.length - 1; i++) {
    if (str.charCodeAt(i + 1) - str.charCodeAt(i) > 1) {
      return String.fromCharCode(str.charCodeAt(i) + 1);
    }
  }
  return null;
}

```

### 15. Rotate Array by `k` Positions

* **Method 1: In-Place Reverse Algorithm** ($O(N)$ time, $O(1)$ space)

```javascript
function rotateArray(arr, k) {
  k = k % arr.length;
  const reverse = (left, right) => {
    while (left < right) [arr[left++], arr[right--]] = [arr[right], arr[left]];
  };
  reverse(0, arr.length - 1);
  reverse(0, k - 1);
  reverse(k, arr.length - 1);
  return arr;
}

```

* **Method 2: `slice` and `concat**` ($O(N)$ time, $O(N)$ space)

```javascript
const rotateArraySlice = (arr, k) => {
  k = k % arr.length;
  return arr.slice(-k).concat(arr.slice(0, -k));
};

```

### 23. Check Prime Number

* **Method 1: Optimized Trial Division ($\sqrt{N}$)** ($O(\sqrt{N})$)

```javascript
function isPrime(num) {
  if (num <= 1) return false;
  if (num <= 3) return true;
  if (num % 2 === 0 || num % 3 === 0) return false;
  for (let i = 5; i * i <= num; i += 6) {
    if (num % i === 0 || num % (i + 2) === 0) return false;
  }
  return true;
}
```

### 24. Reverse a Number

```javascript
function reverseNumber(n) {
  let rev = 0, num = Math.abs(n);
  while (num > 0) {
    rev = rev * 10 + (num % 10);
    num = Math.floor(num / 10);
  }
  return n < 0 ? -rev : rev;
}

```

### 28. Sum of Digits

* **Method 1: Mathematical Modulo Loop** ($O(\log_{10} N)$ time, $O(1)$ space)

```javascript
function sumOfDigits(num) {
  let sum = 0, n = Math.abs(num);
  while (n > 0) {
    sum += n % 10;
    n = Math.floor(n / 10);
  }
  return sum;
}

```

* **Method 2: `reduce` over String Conversion**

```javascript

  Math.abs(num).toString().split('').reduce((sum, digit) => sum + Number(digit), 0);
```
