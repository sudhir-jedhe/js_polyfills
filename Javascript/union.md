### 1. `Array.prototype.with`

An ECMAScript 2023 feature that returns a **new array** with the element at the specified index replaced by a given value, without mutating the original array. Supports negative indexing from the end.

```typescript
// Polyfill / Functional equivalent
function arrayWith<T>(arr: readonly T[], index: number, value: T): T[] {
  const len = arr.length;
  // Handle negative indices: -1 points to the last element
  const actualIndex = index < 0 ? len + index : index;

  if (actualIndex < 0 || actualIndex >= len) {
    throw new RangeError(`Index ${index} is out of bounds for array of length ${len}`);
  }

  const copy = [...arr];
  copy[actualIndex] = value;
  return copy;
}

// Example:
const original = ["a", "b", "c", "d"];
const updated = arrayWith(original, 2, "Z"); // ["a", "b", "Z", "d"]
const fromEnd = arrayWith(original, -1, "X"); // ["a", "b", "c", "X"]
console.log(original); // ["a", "b", "c", "d"] (unmutated)

```

### 2. `String.prototype.repeat`

Constructs and returns a new string containing the specified number of copies of the original string concatenated together.

```typescript
// Polyfill / Functional equivalent
function stringRepeat(str: string, count: number): string {
  if (count < 0 || count === Infinity || Number.isNaN(count)) {
    throw new RangeError(`Invalid count value: ${count}`);
  }

  let n = Math.floor(count);
  if (n === 0 || str.length === 0) return "";

  // Efficient log(N) squaring algorithm
  let result = "";
  let base = str;

  while (n > 0) {
    if (n % 2 === 1) {
      result += base;
    }
    if (n > 1) {
      base += base;
    }
    n = Math.floor(n / 2);
  }

  return result;
}

// Example:
console.log(stringRepeat("abc", 3)); // "abcabcabc"
console.log(stringRepeat("*", 5));   // "*****"

```

### 3. `meanBy`

Computes the arithmetic mean (average) of the values generated by an iteratee/projection function over an array of items.

```typescript
export type Iteratee<T> = ((item: T) => number) | keyof T;

function meanBy<T>(array: readonly T[], iteratee: Iteratee<T>): number {
  if (array.length === 0) return NaN;

  const getValue = (item: T): number => {
    if (typeof iteratee === "function") {
      return iteratee(item);
    }
    return Number(item[iteratee]);
  };

  const sum = array.reduce((acc, curr) => acc + getValue(curr), 0);
  return sum / array.length;
}

// Example:
interface Student {
  name: string;
  score: number;
}

const students: Student[] = [
  { name: "Alice", score: 85 },
  { name: "Bob", score: 92 },
  { name: "Charlie", score: 78 },
];

console.log(meanBy(students, (s) => s.score)); // 85
console.log(meanBy(students, "score"));         // 85

```

### 4. `negate`

A higher-order function that takes a predicate function and returns a new predicate function that negates its boolean result.

```typescript
function negate<Args extends any[]>(
  predicate: (...args: Args) => boolean
): (...args: Args) => boolean {
  return (...args: Args) => !predicate(...args);
}

// Example:
const isEven = (n: number): boolean => n % 2 === 0;
const isOdd = negate(isEven);

const numbers = [1, 2, 3, 4, 5, 6];
console.log(numbers.filter(isOdd));  // [1, 3, 5]
console.log(numbers.filter(isEven)); // [2, 4, 6]

```

### 1. `union`

Combines multiple arrays and produces a flat array of unique values using default equality (`SameValueZero` / `Set`).

```javascript
function union(...arrays) {
  return Array.from(new Set(arrays.flat()));
}

// Example:
const arr1 = [1, 2, 3];
const arr2 = [2, 3, 4, 5];
console.log(union(arr1, arr2)); 
// Output: [1, 2, 3, 4, 5]

```

---

### 2. `unionWith`

Combines multiple arrays into one array of unique values, using a custom comparator function `(a, b) => boolean` to determine equality instead of default reference/value equality.

```javascript
function unionWith(comparator, ...arrays) {
  const combined = arrays.flat();
  return combined.reduce((acc, current) => {
    const exists = acc.some((item) => comparator(item, current));
    if (!exists) acc.push(current);
    return acc;
  }, []);
}

// Example:
const list1 = [{ x: 1, y: 2 }, { x: 2, y: 1 }];
const list2 = [{ x: 1, y: 2 }, { x: 3, y: 3 }];

const customComparator = (a, b) => a.x === b.x && a.y === b.y;

console.log(unionWith(customComparator, list1, list2));
// Output: [{ x: 1, y: 2 }, { x: 2, y: 1 }, { x: 3, y: 3 }]

```

### 3. `uniqueBy` / `uniqBy`

Deduplicates an array by extracting a derived key using an iteratee function (or property path) and ensuring that key is unique.

```javascript
function uniqueBy(array, iteratee) {
  const seen = new Set();
  const getKey = typeof iteratee === 'function' ? iteratee : (item) => item[iteratee];

  return array.filter((item) => {
    const key = getKey(item);
    if (seen.has(key)) {
      return false;
    }
    seen.add(key);
    return true;
  });
}

// Example (Using a property name):
const users = [
  { id: 1, name: 'Alice' },
  { id: 2, name: 'Bob' },
  { id: 1, name: 'Alice Duplicate' }
];

console.log(uniqueBy(users, 'id'));
// Output: [{ id: 1, name: 'Alice' }, { id: 2, name: 'Bob' }]

// Example (Using an iteratee function):
console.log(uniqueBy([2.1, 1.2, 2.3], Math.floor));
// Output: [2.1, 1.2]

```

### 4. `uniqueWith` / `uniqWith`

Deduplicates a single array using a custom comparator function `(a, b) => boolean`.

```javascript
function uniqueWith(array, comparator) {
  return array.reduce((acc, current) => {
    const exists = acc.some((item) => comparator(item, current));
    if (!exists) acc.push(current);
    return acc;
  }, []);
}

// Example:
const points = [
  { x: 1, y: 2 },
  { x: 2, y: 3 },
  { x: 1, y: 2 },
  { x: 4, y: 5 }
];

const isEqualPoint = (a, b) => a.x === b.x && a.y === b.y;

console.log(uniqueWith(points, isEqualPoint));
// Output: [{ x: 1, y: 2 }, { x: 2, y: 3 }, { x: 4, y: 5 }]

```

### Quick Comparison Summary

| Method           | Purpose                         | Mechanism                      | Time Complexity    |
| ---------------- | ------------------------------- | ------------------------------ | ------------------ |
| **`union`**      | Merge arrays, keep unique items | Value / Set equality           | $\mathcal{O}(N)$   |
| **`unionWith`**  | Merge arrays, keep unique items | Custom `comparator(a, b)`      | $\mathcal{O}(N^2)$ |
| **`uniqueBy`**   | Deduplicate single array        | Key extractor `iteratee(item)` | $\mathcal{O}(N)$   |
| **`uniqueWith`** | Deduplicate single array        | Custom `comparator(a, b)`      | $\mathcal{O}(N^2)$ |

Here are fully typed TypeScript implementations for **`intersection`**, **`difference`**, and **`groupBy`**, including support for multiple array inputs, custom extractors, and strict dictionary return types.

```typescript
type PropertyKey = string | number | symbol;

/**
 * 1. intersection
 * Returns an array of unique values that are included in all given arrays.
 */
export function intersection<T>(first: readonly T[], ...rest: readonly (readonly T[])[]): T[] {
  if (rest.length === 0) {
    return Array.from(new Set(first));
  }

  const restSets = rest.map((arr) => new Set(arr));
  const seen = new Set<T>();

  return first.filter((item) => {
    if (seen.has(item)) {
      return false;
    }
    const inAll = restSets.every((set) => set.has(item));
    if (inAll) {
      seen.add(item);
      return true;
    }
    return false;
  });
}

/**
 * 2. difference
 * Returns values from the first array that are not present in any subsequent arrays.
 */
export function difference<T>(first: readonly T[], ...rest: readonly (readonly unknown[])[]): T[] {
  const excludeSet = new Set(rest.flat());
  return first.filter((item) => !excludeSet.has(item));
}

/**
 * 3. groupBy
 * Groups elements of an array by keys returned by a key extractor function or property name.
 */
export function groupBy<T, K extends keyof T>(
  array: readonly T[],
  iteratee: K
): Record<T[K] & PropertyKey, T[]>;
export function groupBy<T, K extends PropertyKey>(
  array: readonly T[],
  iteratee: (item: T) => K
): Record<K, T[]>;
export function groupBy<T, K extends PropertyKey>(
  array: readonly T[],
  iteratee: keyof T | ((item: T) => K)
): Record<PropertyKey, T[]> {
  const getKey =
    typeof iteratee === "function"
      ? iteratee
      : (item: T) => item[iteratee] as unknown as K;

  return array.reduce<Record<PropertyKey, T[]>>((acc, item) => {
    const key = getKey(item);
    if (!acc[key]) {
      acc[key] = [];
    }
    acc[key].push(item);
    return acc;
  }, {});
}

```

Here are fully typed TypeScript implementations for **`intersectionWith`**, **`differenceWith`**, and **`differenceBy`**, supporting custom comparators, cross-type array comparisons, and flexible property/iteratee key extraction.

```typescript
type Comparator<T, U = T> = (a: T, b: U) => boolean;

/**
 * 1. intersectionWith
 * Returns unique elements from the first array that match at least one element
 * in each of the subsequent arrays according to a custom comparator predicate.
 */
export function intersectionWith<T, U = T>(
  comparator: Comparator<T, U>,
  first: readonly T[],
  ...rest: readonly (readonly U[])[]
): T[] {
  if (rest.length === 0) {
    return first.reduce<T[]>((acc, item) => {
      const exists = acc.some((seen) => comparator(seen, item as unknown as U));
      if (!exists) acc.push(item);
      return acc;
    }, []);
  }

  const result: T[] = [];

  for (const item of first) {
    // 1. Check if item has already been added to output (deduplication)
    const alreadyAdded = result.some((seen) =>
      comparator(seen, item as unknown as U)
    );
    if (alreadyAdded) continue;

    // 2. Check if item exists in every other array using the comparator
    const existsInAllRest = rest.every((arr) =>
      arr.some((other) => comparator(item, other))
    );

    if (existsInAllRest) {
      result.push(item);
    }
  }

  return result;
}

/**
 * 2. differenceWith
 * Returns elements from the first array that do NOT match any element
 * in any subsequent arrays according to a custom comparator predicate.
 */
export function differenceWith<T, U = T>(
  comparator: Comparator<T, U>,
  first: readonly T[],
  ...rest: readonly (readonly U[])[]
): T[] {
  const flattenedRest = rest.flat();

  return first.filter(
    (item) => !flattenedRest.some((other) => comparator(item, other))
  );
}

/**
 * 3. differenceBy
 * Returns elements from the first array whose derived keys (via property name or
 * iteratee function) are not present among the keys of subsequent arrays.
 */
export function differenceBy<T, K extends keyof T>(
  array: readonly T[],
  values: readonly (readonly T[])[],
  iteratee: K
): T[];
export function differenceBy<T, U, V>(
  array: readonly T[],
  values: readonly (readonly U[])[],
  iteratee: (item: T | U) => V
): T[];
export function differenceBy<T, U, V>(
  array: readonly T[],
  values: readonly (readonly U[])[],
  iteratee: keyof T | ((item: T | U) => V)
): T[] {
  const getKey =
    typeof iteratee === "function"
      ? iteratee
      : (item: T | U) => (item as T)[iteratee as keyof T];

  const excludedKeys = new Set(values.flat().map(getKey));

  return array.filter((item) => !excludedKeys.has(getKey(item)));
}

```

Here are fully typed TypeScript implementations for **`pickBy`**, **`omitBy`**, and **`mapKeys`**, supporting both type-guard predicates for strict narrowing and general boolean callbacks.

```typescript
type PropertyKey = string | number | symbol;

/**
 * 1. pickBy
 * Creates an object composed of the object properties that satisfy the predicate.
 * Supports custom type guards for type narrowing.
 */
export function pickBy<T extends object, S extends T[keyof T]>(
  object: T,
  predicate: (value: T[keyof T], key: keyof T) => value is S
): Partial<Record<keyof T, S>>;
export function pickBy<T extends object>(
  object: T,
  predicate: (value: T[keyof T], key: keyof T) => boolean
): Partial<T>;
export function pickBy<T extends object>(
  object: T,
  predicate: (value: T[keyof T], key: keyof T) => boolean
): Partial<T> {
  const result: Partial<T> = {};

  for (const key of Object.keys(object) as (keyof T)[]) {
    const value = object[key];
    if (predicate(value, key)) {
      result[key] = value;
    }
  }

  return result;
}

/**
 * 2. omitBy
 * Creates an object composed of the properties that do NOT satisfy the predicate.
 */
export function omitBy<T extends object>(
  object: T,
  predicate: (value: T[keyof T], key: keyof T) => boolean
): Partial<T> {
  const result: Partial<T> = {};

  for (const key of Object.keys(object) as (keyof T)[]) {
    const value = object[key];
    if (!predicate(value, key)) {
      result[key] = value;
    }
  }

  return result;
}

/**
 * 3. mapKeys
 * Creates an object with the same values and keys generated by running
 * each own enumerable string-keyed property through the iteratee.
 */
export function mapKeys<T extends object, K extends PropertyKey>(
  object: T,
  iteratee: (value: T[keyof T], key: keyof T) => K
): Record<K, T[keyof T]> {
  const result = {} as Record<K, T[keyof T]>;

  for (const key of Object.keys(object) as (keyof T)[]) {
    const value = object[key];
    const newKey = iteratee(value, key);
    result[newKey] = value;
  }

  return result;
}

```

Here are fully typed TypeScript implementations for **`mapValues`**, **`invert`**, and **`fromPairs`**, preserving mapped object keys and supporting strict tuple inference.

```typescript
type PropertyKey = string | number | symbol;

/**
 * 1. mapValues
 * Creates an object with the same keys as the source object and values
 * generated by running each own property through the iteratee.
 */
export function mapValues<T extends object, R>(
  object: T,
  iteratee: (value: T[keyof T], key: keyof T) => R
): { [K in keyof T]: R } {
  const result = {} as { [K in keyof T]: R };

  for (const key of Object.keys(object) as (keyof T)[]) {
    result[key] = iteratee(object[key], key);
  }

  return result;
}

/**
 * 2. invert
 * Inverts an object's keys and values.
 * If the values are string/number literals, it types the inverted keys strictly.
 */
export function invert<T extends Record<PropertyKey, PropertyKey>>(
  object: T
): { [K in keyof T as `${T[K]}`]: K } {
  const result = {} as { [K in keyof T as `${T[K]}`]: K };

  for (const key of Object.keys(object) as (keyof T)[]) {
    const value = object[key];
    (result as any)[String(value)] = key;
  }

  return result;
}

/**
 * 3. fromPairs
 * Converts key-value pairs (tuples) into an object.
 * Supports both strict tuple literal types and general array pairs.
 */
export function fromPairs<const T extends readonly (readonly [PropertyKey, unknown])[]>(
  pairs: T
): {
  [P in T[number] as P[0] extends PropertyKey ? P[0] : never]: P[1];
};
export function fromPairs<K extends PropertyKey, V>(
  pairs: readonly (readonly [K, V])[]
): Record<K, V>;
export function fromPairs(
  pairs: readonly (readonly [PropertyKey, unknown])[]
): Record<PropertyKey, unknown> {
  const result: Record<PropertyKey, unknown> = {};

  for (const [key, value] of pairs) {
    result[key] = value;
  }

  return result;
}
```

```typescript
/**
 * 1. cloneDeep
 * Recursively creates a deep copy of an object, array, Date, RegExp, Map, or Set.
 * Handles circular references safely using a WeakMap.
 */
export function cloneDeep<T>(value: T, hash = new WeakMap<object, any>()): T {
  if (value === null || typeof value !== "object") {
    return value;
  }

  if (value instanceof Date) {
    return new Date(value.getTime()) as unknown as T;
  }

  if (value instanceof RegExp) {
    return new RegExp(value.source, value.flags) as unknown as T;
  }

  if (hash.has(value)) {
    return hash.get(value);
  }

  if (value instanceof Map) {
    const result = new Map();
    hash.set(value, result);
    value.forEach((v, k) => result.set(cloneDeep(k, hash), cloneDeep(v, hash)));
    return result as unknown as T;
  }

  if (value instanceof Set) {
    const result = new Set();
    hash.set(value, result);
    value.forEach((v) => result.add(cloneDeep(v, hash)));
    return result as unknown as T;
  }

  if (Array.isArray(value)) {
    const result: any[] = [];
    hash.set(value, result);
    value.forEach((item, index) => {
      result[index] = cloneDeep(item, hash);
    });
    return result as unknown as T;
  }

  const result = Object.create(Object.getPrototypeOf(value));
  hash.set(value, result);
  for (const key of Reflect.ownKeys(value)) {
    const descriptor = Object.getOwnPropertyDescriptor(value, key);
    if (descriptor) {
      if ("value" in descriptor) {
        descriptor.value = cloneDeep((value as any)[key], hash);
      }
      Object.defineProperty(result, key, descriptor);
    }
  }

  return result as T;
}

/**
 * 2. merge
 * Recursively merges own and inherited enumerable string-keyed properties
 * of source objects into the destination object.
 */
export function merge<T extends object, U extends object>(
  target: T,
  source: U
): DeepMerge<T, U> {
  const result = cloneDeep(target) as any;

  for (const key of Object.keys(source)) {
    const sourceVal = (source as any)[key];
    const targetVal = result[key];

    if (
      sourceVal &&
      typeof sourceVal === "object" &&
      !Array.isArray(sourceVal) &&
      !(sourceVal instanceof Date) &&
      !(sourceVal instanceof RegExp)
    ) {
      if (targetVal && typeof targetVal === "object" && !Array.isArray(targetVal)) {
        result[key] = merge(targetVal, sourceVal);
      } else {
        result[key] = cloneDeep(sourceVal);
      }
    } else {
      result[key] = cloneDeep(sourceVal);
    }
  }

  return result;
}

 * 3. get
 * Gets the value at path of object. If the resolved value is undefined,
 * the defaultValue is returned in its place.
 */

  object: T,
  path: P
): PathValue<T, P>;

export function get<T extends object, D = undefined>(
  object: T,
  path: string,
  defaultValue?: D
): any {
  const keys = path.split(".");
  let current: any = object;

  for (const key of keys) {
    if (current == null) {
      return defaultValue;
    }
    current = current[key];
  }

  return current === undefined ? defaultValue : current;
}

/**
 * 4. set
 * Sets the value at path of object. If a portion of path doesn't exist, it's created.
 */
export function set<T extends object, P extends Path<T>>(
  object: T,
  path: P,
  value: PathValue<T, P>
): T;
export function set<T extends object>(
  object: T,
  path: string,
  value: any
): T;
export function set<T extends object>(
  object: T,
  path: string,
  value: any
): T {
  const keys = path.split(".");
  let current: any = object;

  for (let i = 0; i < keys.length - 1; i++) {
    const key = keys[i];
    if (!(key in current) || typeof current[key] !== "object" || current[key] === null) {
      // If the next key is numeric, create an array; otherwise an object
      current[key] = /^\d+$/.test(keys[i + 1]) ? [] : {};
    }
    current = current[key];
  }

  const finalKey = keys[keys.length - 1];
  current[finalKey] = value;
  return object;
}
```

```typescript
export interface DebouncedFunction<F extends (...args: any[]) => any> {
  (...args: Parameters<F>): void;
  cancel: () => void;
  flush: () => void;
}

export interface DebounceOptions {
  leading?: boolean;
  trailing?: boolean;
  maxWait?: number;
}

/**
 * 1. debounce
 * Delays invoking `fn` until after `wait` milliseconds have elapsed
 * since the last time the debounced function was invoked.
 */
export function debounce<F extends (...args: any[]) => any>(
  fn: F,
  wait: number,
  options: DebounceOptions = {}
): DebouncedFunction<F> {
  const { leading = false, trailing = true, maxWait } = options;

  let timerId: ReturnType<typeof setTimeout> | undefined;
  let maxTimerId: ReturnType<typeof setTimeout> | undefined;
  let lastArgs: Parameters<F> | undefined;
  let lastThis: any;
  let lastCallTime: number | undefined;
  let lastInvokeTime = 0;

  const invokeFunc = (time: number) => {
    const args = lastArgs;
    const thisArg = lastThis;

    lastArgs = undefined;
    lastThis = undefined;
    lastInvokeTime = time;

    if (args) {
      fn.apply(thisArg, args);
    }
  };

  const cancel = () => {
    if (timerId !== undefined) {
      clearTimeout(timerId);
      timerId = undefined;
    }
    if (maxTimerId !== undefined) {
      clearTimeout(maxTimerId);
      maxTimerId = undefined;
    }
    lastArgs = undefined;
    lastThis = undefined;
    lastCallTime = undefined;
    lastInvokeTime = 0;
  };

  const flush = () => {
    if (timerId !== undefined && lastArgs) {
      invokeFunc(Date.now());
      cancel();
    }
  };

  const debounced = function (this: any, ...args: Parameters<F>): void {
    const time = Date.now();
    const isInvoking = timerId === undefined;

    lastArgs = args;
    lastThis = this;
    lastCallTime = time;

    if (isInvoking) {
      if (leading) {
        invokeFunc(time);
      }
    }

    if (timerId !== undefined) {
      clearTimeout(timerId);
    }

    timerId = setTimeout(() => {
      if (trailing && lastArgs) {
        invokeFunc(Date.now());
      }
      timerId = undefined;
    }, wait);

    if (maxWait !== undefined && maxTimerId === undefined) {
      maxTimerId = setTimeout(() => {
        if (lastArgs) {
          invokeFunc(Date.now());
        }
        maxTimerId = undefined;
      }, maxWait);
    }
  };

  debounced.cancel = cancel;
  debounced.flush = flush;

  return debounced;
}

export interface ThrottledFunction<F extends (...args: any[]) => any> {
  (...args: Parameters<F>): void;
  cancel: () => void;
  flush: () => void;
}

export interface ThrottleOptions {
  leading?: boolean;
  trailing?: boolean;
}

/**
 * 2. throttle
 * Creates a throttled function that only invokes `fn` at most once
 * per every `wait` milliseconds.
 */
export function throttle<F extends (...args: any[]) => any>(
  fn: F,
  wait: number,
  options: ThrottleOptions = {}
): ThrottledFunction<F> {
  const { leading = true, trailing = true } = options;

  let timerId: ReturnType<typeof setTimeout> | undefined;
  let lastArgs: Parameters<F> | undefined;
  let lastThis: any;
  let lastCallTime = 0;

  const invokeFunc = (time: number) => {
    const args = lastArgs;
    const thisArg = lastThis;

    lastArgs = undefined;
    lastThis = undefined;
    lastCallTime = time;

    if (args) {
      fn.apply(thisArg, args);
    }
  };

  const cancel = () => {
    if (timerId !== undefined) {
      clearTimeout(timerId);
      timerId = undefined;
    }
    lastArgs = undefined;
    lastThis = undefined;
    lastCallTime = 0;
  };

  const flush = () => {
    if (timerId !== undefined && lastArgs) {
      invokeFunc(Date.now());
      cancel();
    }
  };

  const throttled = function (this: any, ...args: Parameters<F>): void {
    const now = Date.now();
    lastArgs = args;
    lastThis = this;

    if (!lastCallTime && !leading) {
      lastCallTime = now;
    }

    const remaining = wait - (now - lastCallTime);

    if (remaining <= 0 || remaining > wait) {
      if (timerId !== undefined) {
        clearTimeout(timerId);
        timerId = undefined;
      }
      invokeFunc(now);
    } else if (timerId === undefined && trailing) {
      timerId = setTimeout(() => {
        invokeFunc(Date.now());
        timerId = undefined;
      }, remaining);
    }
  };

  throttled.cancel = cancel;
  throttled.flush = flush;

  return throttled;
}

export interface MemoizedFunction<F extends (...args: any[]) => any> {
  (...args: Parameters<F>): ReturnType<F>;
  cache: Map<any, ReturnType<F>>;
  clear: () => void;
}

/**
 * 3. memoize
 * Creates a function that memoizes the result of `fn`.
 * Accepts an optional `resolver` to generate custom cache keys.
 */
export function memoize<F extends (...args: any[]) => any>(
  fn: F,
  resolver?: (...args: Parameters<F>) => any
): MemoizedFunction<F> {
  const cache = new Map<any, ReturnType<F>>();

  const memoized = function (this: any, ...args: Parameters<F>): ReturnType<F> {
    const key = resolver ? resolver(...args) : args[0];

    if (cache.has(key)) {
      return cache.get(key)!;
    }

    const result = fn.apply(this, args);
    cache.set(key, result);
    return result;
  };

  memoized.cache = cache;
  memoized.clear = () => cache.clear();

  return memoized;
}
```

```typescript
/**
 * 1. curry
 * Returns a curried equivalent of the provided function.
 * Supports passing arguments one by one or in partial batches.
 */
export function curry<T extends (...args: any[]) => any>(
  fn: T
): Curry<Parameters<T>, ReturnType<T>> {
  return function curried(this: any, ...args: any[]): any {
    if (args.length >= fn.length) {
      return fn.apply(this, args);
    }
    return function (this: any, ...nextArgs: any[]) {
      return curried.apply(this, [...args, ...nextArgs]);
    };
  } as Curry<Parameters<T>, ReturnType<T>>;
}

  /**
```
```typescript
   * Concurrency limit. Must be an integer >= 1.
   * @default Infinity
   */
  concurrency?: number;
  /**
   * When true, errors are collected and the entire run completes before rejecting
   * with an AggregateError.
   * @default false
   */

}

/**
 * 1. pMap
 * Maps iterable items concurrently over an async mapper function with a strict concurrency ceiling.
 */
export async function pMap<T, R>(
  iterable: Iterable<T> | readonly T[],
  mapper: (item: T, index: number) => Promise<R> | R,
  options: PMapOptions = {}
): Promise<R[]> {
  const { concurrency = Infinity, stopOnError = true } = options;

  if (concurrency < 1) {
    throw new TypeError("Concurrency must be at least 1");
  }

  const items = Array.from(iterable);
  const result: R[] = new Array(items.length);
  const errors: unknown[] = [];
  let currentIndex = 0;
  let isRejected = false;

  const next = async (): Promise<void> => {
    while (currentIndex < items.length) {
      if (isRejected && stopOnError) {
        return;
      }

      const index = currentIndex++;
      const item = items[index];

      try {
        result[index] = await mapper(item, index);
      } catch (err) {
        if (stopOnError) {
          isRejected = true;
          throw err;
        } else {
          errors.push(err);
        }
      }
    }
  };

  const poolSize = Math.min(concurrency, items.length);
  const workers: Promise<void>[] = [];

  for (let i = 0; i < poolSize; i++) {
    workers.push(next());
  }

  await Promise.all(workers);

  if (errors.length > 0) {
    throw new AggregateError(errors, "One or more promises failed in pMap");
  }

  return result;
}

export interface RetryOptions {
  /**
   * Maximum number of attempts (including initial call).
   * @default 3
   */

   * Base delay in milliseconds before the first retry.
   * @default 1000
   */
  delay?: number;

   * Exponential backoff factor.
   * @default 2
   */

   * Maximum delay in milliseconds between retries.
   * @default Infinity

   * Add randomized jitter (0 to 1 * delay) to prevent thundering herd.
   * @default true
   */
  jitter?: boolean;

   * Predicate callback to conditionally determine whether to retry.
   */

  /**
   * Lifecycle hook called on each failed attempt before waiting.
   */

/**
 * 2. retry
 * Wraps an async operation with automated exponential backoff, jitter, and retry guards.
 */
export async function retry<T>(
  fn: (attempt: number) => Promise<T> | T,
  options: RetryOptions = {}
): Promise<T> {
  const {
    retries = 3,
    delay = 1000,
    factor = 2,
    maxDelay = Infinity,
    jitter = true,
    shouldRetry = () => true,
    onFailedAttempt,
  } = options;

  let currentDelay = delay;

  for (let attempt = 1; attempt <= retries; attempt++) {
    try {
      return await fn(attempt);
    } catch (error) {
      if (attempt >= retries || !shouldRetry(error, attempt)) {
        throw error;
      }

      // Compute backoff interval with optional full jitter
      let waitTime = Math.min(currentDelay, maxDelay);
      if (jitter) {
        waitTime = Math.floor(Math.random() * waitTime);
      }

      if (onFailedAttempt) {
        onFailedAttempt(error, attempt, waitTime);
      }

      await new Promise((resolve) => setTimeout(resolve, waitTime));
      currentDelay = Math.min(currentDelay * factor, maxDelay);
    }
  }

  throw new Error("Retry loop exhausted without result");
}
```

```typescript
export type Releaser = () => void;

/**
 * 1. Semaphore
 * Controls access to a common pool of identical resources.
 */
export class Semaphore {
  private permits: number;
  private readonly queue: Array<(releaser: Releaser) => void> = [];

  constructor(permits: number) {
    if (permits < 1) {
      throw new RangeError("Semaphore initial permits must be at least 1");
    }
    this.permits = permits;
  }

  /**
   * Acquires a permit. Returns a release function to return the permit back to the pool.
   */
  public acquire(): Promise<Releaser> {
    if (this.permits > 0) {
      this.permits--;
      let released = false;
      const release: Releaser = () => {
        if (released) return;
        released = true;
        this.release();
      };
      return Promise.resolve(release);
    }

    return new Promise<Releaser>((resolve) => {
      this.queue.push((release) => resolve(release));
    });
  }

  /**
   * Releases a permit back to the pool or hands it off to the next waiting acquirer.
   */
  private release(): void {
    if (this.queue.length > 0) {
      const next = this.queue.shift()!;
      let released = false;
      const release: Releaser = () => {
        if (released) return;
        released = true;
        this.release();
      };
      next(release);
    } else {
      this.permits++;
    }
  }

  /**
   * Executes a callback within an acquired permit and automatically releases it upon completion.
   */
  public async runExclusive<T>(callback: () => Promise<T> | T): Promise<T> {
    const release = await this.acquire();
    try {
      return await callback();
    } finally {
      release();
    }
  }

  public get availablePermits(): number {
    return this.permits;
  }

  public get queueLength(): number {
    return this.queue.length;
  }
}

/**
 * 2. Mutex
 * Binary semaphore ensuring mutual exclusion (only one task runs at a time).
 */
export class Mutex {
  private readonly semaphore = new Semaphore(1);

  /**
   * Acquires the lock. Returns an unlock releaser function.
   */
  public acquire(): Promise<Releaser> {
    return this.semaphore.acquire();
  }

  /**
   * Executes an async task while holding the lock and automatically releases it.
   */
  public async runExclusive<T>(callback: () => Promise<T> | T): Promise<T> {
    return this.semaphore.runExclusive(callback);
  }

  /**
   * Returns true if the lock is currently held by an active task.
   */
  public isLocked(): boolean {
    return this.semaphore.availablePermits === 0;
  }
}

/**
 * 3. Barrier (Cyclic Barrier)
 * Enables a fixed number of concurrent tasks to wait for each other to reach a rendezvous point.
 */
export class Barrier {
  private readonly capacity: number;
  private count: number = 0;
  private waiters: Array<{ resolve: () => void; reject: (err: unknown) => void }> = [];
  private readonly onTrip?: () => Promise<void> | void;

  /**
   * @param capacity Number of parties that must call `wait()` to trip the barrier.
   * @param onTrip Optional action executed once each time all parties arrive, before releasing them.
   */
  constructor(capacity: number, onTrip?: () => Promise<void> | void) {
    if (capacity < 1) {
      throw new RangeError("Barrier capacity must be at least 1");
    }
    this.capacity = capacity;
    this.onTrip = onTrip;
  }

  /**
   * Waits until all parties have invoked `wait()` on this barrier.
   */
  public async wait(): Promise<void> {
    this.count++;

    if (this.count === this.capacity) {
      const currentWaiters = this.waiters;
      this.waiters = [];
      this.count = 0;

      try {
        if (this.onTrip) {
          await this.onTrip();
        }
        for (const waiter of currentWaiters) {
          waiter.resolve();
        }
      } catch (err) {
        for (const waiter of currentWaiters) {
          waiter.reject(err);
        }
        throw err;
      }
      return;
    }

    return new Promise<void>((resolve, reject) => {
      this.waiters.push({ resolve, reject });
    });
  }

  /**
   * Resets the barrier to its initial state, rejecting any currently waiting promises.
   */
  public reset(reason: string = "Barrier broken"): void {
    const currentWaiters = this.waiters;
    this.waiters = [];
    this.count = 0;
    for (const waiter of currentWaiters) {
      waiter.reject(new Error(reason));
    }
  }

  public get waitingCount(): number {
    return this.count;
  }
}
```

```typescript
   * @default 1
   */

interface QueueItem<T> {
  task: () => Promise<T> | T;
  priority: number;
  resolve: (value: T | PromiseLike<T>) => void;
  reject: (reason?: unknown) => void;
}

/**
 * 1. AsyncQueue
 * Priority-aware FIFO queue for scheduling and throttling concurrent async operations.
 */
export class AsyncQueue {
  private concurrency: number;
  private running = 0;
  private queue: QueueItem<any>[] = [];
  private paused = false;

  constructor(options: AsyncQueueOptions = {}) {
    const { concurrency = 1 } = options;
    if (concurrency < 1) {
      throw new RangeError("Concurrency must be at least 1");
    }
    this.concurrency = concurrency;
  }

  /**
   * Adds an asynchronous task to the queue.
   * Higher priority numbers execute before lower priority ones.
   */
  public add<T>(task: () => Promise<T> | T, priority: number = 0): Promise<T> {
    return new Promise<T>((resolve, reject) => {
      const item: QueueItem<T> = { task, priority, resolve, reject };

      // Insert sorted by priority descending
      const insertIndex = this.queue.findIndex((q) => q.priority < priority);
      if (insertIndex === -1) {
        this.queue.push(item);
      } else {
        this.queue.splice(insertIndex, 0, item);
      }

      this.processNext();
    });
  }

  private async processNext(): Promise<void> {
    if (this.paused || this.running >= this.concurrency || this.queue.length === 0) {
      return;
    }

    this.running++;
    const { task, resolve, reject } = this.queue.shift()!;

    try {
      const result = await task();
      resolve(result);
    } catch (err) {
      reject(err);
    } finally {
      this.running--;
      this.processNext();
    }
  }

  public pause(): void {
    this.paused = true;
  }

  public resume(): void {
    if (this.paused) {
      this.paused = false;
      for (let i = 0; i < this.concurrency; i++) {
        this.processNext();
      }
    }
  }

  public clear(): void {
    this.queue = [];
  }

  public get size(): number {
    return this.queue.length;
  }

  public get activeCount(): number {
    return this.running;
  }
}

/**
 * 2. EventBus
 * Strictly typed Publish-Subscribe event broker.
 */
export class EventBus<Events extends EventMap> {
  private listeners: {
    [K in keyof Events]?: Set<EventListener<Events[K]>>;
  } = {};

  /**
   * Subscribes a listener to an event. Returns an unsubscription callback.
   */
  public on<K extends keyof Events>(
    event: K,
    listener: EventListener<Events[K]>
  ): () => void {
    if (!this.listeners[event]) {
      this.listeners[event] = new Set();
    }
    this.listeners[event]!.add(listener);

    return () => this.off(event, listener);
  }

  /**
   * Subscribes a listener that automatically removes itself after its first invocation.
   */
  public once<K extends keyof Events>(
    event: K,
    listener: EventListener<Events[K]>
  ): () => void {
    const unsubscribe = this.on(event, (payload) => {
      unsubscribe();
      return listener(payload);
    });
    return unsubscribe;
  }

  /**
   * Removes a specific listener subscription.
   */
  public off<K extends keyof Events>(
    event: K,
    listener: EventListener<Events[K]>
  ): void {
    const handlers = this.listeners[event];
    if (handlers) {
      handlers.delete(listener);
      if (handlers.size === 0) {
        delete this.listeners[event];
      }
    }
  }

  /**
   * Synchronously invokes all listeners for an event.
   */
  public emit<K extends keyof Events>(event: K, payload: Events[K]): void {
    const handlers = this.listeners[event];
    if (handlers) {
      handlers.forEach((listener) => {
        try {
          listener(payload);
        } catch (err) {
          console.error(`Error in listener for event "${String(event)}":`, err);
        }
      });
    }
  }

  /**
   * Asynchronously invokes all listeners and waits for all to settle.
   */
  public async emitAsync<K extends keyof Events>(
    event: K,
    payload: Events[K]
  ): Promise<void> {
    const handlers = this.listeners[event];
    if (handlers) {
      const promises = Array.from(handlers).map((listener) => listener(payload));
      await Promise.all(promises);
    }
  }

  /**
   * Removes all listeners for a given event, or all events if omitted.
   */
  public removeAllListeners<K extends keyof Events>(event?: K): void {
    if (event) {
      delete this.listeners[event];
    } else {
      this.listeners = {};
    }
  }
}
```

```typescript
export class Observable<T> {
  constructor(
    private readonly subscribeFn: (observer: Observer<T>) => (() => void) | void
  ) {}

  /**
   * Subscribes an observer or next-callback to the Observable stream.
   */
  public subscribe(
    observerOrNext: Partial<Observer<T>> | ((val: T) => void)
  ): Subscription {
    let isClosed = false;

    const observer: Observer<T> =
      typeof observerOrNext === "function"
        ? { next: observerOrNext }
        : {
            next: (v) => observerOrNext.next?.(v),
            error: (e) => observerOrNext.error?.(e),
            complete: () => observerOrNext.complete?.(),
          };

    const safeObserver: Observer<T> = {
      next: (val) => {
        if (!isClosed) observer.next(val);
      },
      error: (err) => {
        if (!isClosed) {
          isClosed = true;
          observer.error?.(err);
        }
      },
      complete: () => {
        if (!isClosed) {
          isClosed = true;
          observer.complete?.();
        }
      },
    };

    const cleanup = this.subscribeFn(safeObserver);

    return {
      unsubscribe: () => {
        if (!isClosed) {
          isClosed = true;
          cleanup?.();
        }
      },
      get closed() {
        return isClosed;
      },
    };
  }

  /**
   * Chains reactive operators onto this Observable stream.
   */
  public pipe(): Observable<T>;
  public pipe<A>(op1: OperatorFunction<T, A>): Observable<A>;
  public pipe<A, B>(
    op1: OperatorFunction<T, A>,
    op2: OperatorFunction<A, B>
  ): Observable<B>;
  public pipe<A, B, C>(
    op1: OperatorFunction<T, A>,
    op2: OperatorFunction<A, B>,
    op3: OperatorFunction<B, C>
  ): Observable<C>;
  public pipe<A, B, C, D>(
    op1: OperatorFunction<T, A>,
    op2: OperatorFunction<A, B>,
    op3: OperatorFunction<B, C>,
    op4: OperatorFunction<C, D>
  ): Observable<D>;
  public pipe(...operations: OperatorFunction<any, any>[]): Observable<any> {
    return operations.reduce(
      (prev: Observable<any>, fn) => fn(prev),
```

```typescript
/**
 * Subject
 * Acts as both an Observable and an Observer. Multicasts values to multiple subscribers.
 */
export class Subject<T> extends Observable<T> implements Observer<T> {
  protected observers = new Set<Observer<T>>();
  protected isStopped = false;

  constructor() {
    super((observer) => {
      this.observers.add(observer);
      return () => {
        this.observers.delete(observer);
      };
    });
  }

  public next(value: T): void {
    if (this.isStopped) return;
    for (const observer of Array.from(this.observers)) {
      observer.next(value);
    }
  }

  public error(err: unknown): void {
    if (this.isStopped) return;
    this.isStopped = true;
    for (const observer of Array.from(this.observers)) {
      observer.error?.(err);
    }
    this.observers.clear();
  }

  public complete(): void {
    if (this.isStopped) return;
    this.isStopped = true;
    for (const observer of Array.from(this.observers)) {
      observer.complete?.();
    }
    this.observers.clear();
  }

  public get observed(): boolean {
    return this.observers.size > 0;
  }
}

/**
 * BehaviorSubject
 * Holds a current value and emits it immediately to new subscribers.
 */
export class BehaviorSubject<T> extends Subject<T> {
  constructor(private _value: T) {
    super();
  }

  public get value(): T {
    return this._value;
  }

  public override next(value: T): void {
    this._value = value;
    super.next(value);
  }

  public override subscribe(
    observerOrNext: Partial<Observer<T>> | ((val: T) => void)
  ): Subscription {
    const sub = super.subscribe(observerOrNext);
    if (!this.isStopped) {
      if (typeof observerOrNext === "function") {
        observerOrNext(this._value);
      } else {
        observerOrNext.next?.(this._value);
      }
    }
    return sub;
  }
}
```

```typescript
 * Emits only the first `count` values emitted by the source Observable.
 */
export function take<T>(count: number): OperatorFunction<T, T> {
  return (source: Observable<T>) =>
    new Observable<T>((observer) => {
      let seen = 0;
      if (count <= 0) {
        observer.complete?.();
        return;
      }
      const sub = source.subscribe({
        next: (val) => {
          seen++;
          if (seen <= count) {
            observer.next(val);
            if (seen === count) {
              observer.complete?.();
              sub.unsubscribe();
            }
          }
        },
        error: (err) => observer.error?.(err),
        complete: () => observer.complete?.(),
      });
      return sub.unsubscribe;
    });
```

```typescript
import { Observable, OperatorFunction, Subscription } from "./observable";

export type ObservableInput<T> = Observable<T> | Promise<T> | readonly T[];

/**
 * Helper to convert ObservableInput (Observable, Promise, or Array) into a standard Observable.
 */
export function from<T>(input: ObservableInput<T>): Observable<T> {
  if (input instanceof Observable) {
    return input;
  }

  // Handle Promises
  if (typeof (input as any)?.then === "function") {
    return new Observable<T>((observer) => {
      let isSubscribed = true;
      (input as Promise<T>)
        .then((val) => {
          if (isSubscribed) {
            observer.next(val);
            observer.complete?.();
          }
        })
        .catch((err) => {
          if (isSubscribed) {
            observer.error?.(err);
          }
        });
      return () => {
        isSubscribed = false;
      };
    });
  }

  // Handle Iterables / Arrays
  if (Array.isArray(input)) {
    return new Observable<T>((observer) => {
      for (const item of input) {
        observer.next(item);
      }
      observer.complete?.();
    });
  }

  throw new TypeError("Unsupported ObservableInput type");
}

/**
 * 1. mergeMap (flatMap)
 * Projects each source value to an Observable and merges the resulting Observables concurrently.
 */
export function mergeMap<T, R>(
  project: (value: T, index: number) => ObservableInput<R>,
  concurrent: number = Infinity
): OperatorFunction<T, R> {
  return (source: Observable<T>) =>
    new Observable<R>((observer) => {
      let sourceCompleted = false;
      let activeCount = 0;
      let index = 0;
      const buffer: { val: T; idx: number }[] = [];
      const innerSubs = new Set<Subscription>();

      const checkComplete = () => {
        if (sourceCompleted && activeCount === 0 && buffer.length === 0) {
          observer.complete?.();
        }
      };

      const subscribeInner = (val: T, idx: number) => {
        activeCount++;
        let innerObs: Observable<R>;

        try {
          innerObs = from(project(val, idx));
        } catch (err) {
          observer.error?.(err);
          return;
        }

        let sub: Subscription;
        sub = innerObs.subscribe({
          next: (innerVal) => observer.next(innerVal),
          error: (err) => observer.error?.(err),
          complete: () => {
            innerSubs.delete(sub);
            activeCount--;
            if (buffer.length > 0) {
              const nextItem = buffer.shift()!;
              subscribeInner(nextItem.val, nextItem.idx);
            } else {
              checkComplete();
            }
          },
        });

        innerSubs.add(sub);
      };

      const mainSub = source.subscribe({
        next: (val) => {
          const currentIdx = index++;
          if (activeCount < concurrent) {
            subscribeInner(val, currentIdx);
          } else {
            buffer.push({ val, idx: currentIdx });
          }
        },
        error: (err) => observer.error?.(err),
        complete: () => {
          sourceCompleted = true;
          checkComplete();
        },
      });

      return () => {
        mainSub.unsubscribe();
        innerSubs.forEach((sub) => sub.unsubscribe());
        innerSubs.clear();
      };
    });
}

/**
 * 2. switchMap
 * Projects each source value to an Observable, canceling the previous inner Observable subscription.
 */
export function switchMap<T, R>(
  project: (value: T, index: number) => ObservableInput<R>
): OperatorFunction<T, R> {
  return (source: Observable<T>) =>
    new Observable<R>((observer) => {
      let sourceCompleted = false;
      let index = 0;
      let innerSub: Subscription | undefined;
      let hasActiveInner = false;

      const checkComplete = () => {
        if (sourceCompleted && !hasActiveInner) {
          observer.complete?.();
        }
      };

      const mainSub = source.subscribe({
        next: (val) => {
          // Cancel previous ongoing inner subscription
          if (innerSub && !innerSub.closed) {
            innerSub.unsubscribe();
          }

          const currentIdx = index++;
          let innerObs: Observable<R>;

          try {
            innerObs = from(project(val, currentIdx));
          } catch (err) {
            observer.error?.(err);
            return;
          }

          hasActiveInner = true;
          innerSub = innerObs.subscribe({
            next: (innerVal) => observer.next(innerVal),
            error: (err) => observer.error?.(err),
            complete: () => {
              hasActiveInner = false;
              checkComplete();
            },
          });
        },
        error: (err) => observer.error?.(err),
        complete: () => {
          sourceCompleted = true;
          checkComplete();
        },
      });

      return () => {
        mainSub.unsubscribe();
        if (innerSub) {
          innerSub.unsubscribe();
        }
      };
    });
}

/**
 * 3. concatMap
 * Projects each source value to an Observable and serializes their execution in strict FIFO order.
 * Equivalent to mergeMap(project, 1).
 */
export function concatMap<T, R>(
  project: (value: T, index: number) => ObservableInput<R>
): OperatorFunction<T, R> {
  return mergeMap(project, 1);
}
```

```typescript
/**
 * 1. combineLatest
 * Combines multiple Observables to create an Observable whose values are calculated
 * from the latest values of each of its input Observables.
 * Emits once all sources have emitted at least once, then emits on every subsequent emission.
 */
export function combineLatest<T extends readonly ObservableInput<any>[]>(
  sources: [...T]
): Observable<ObservedValuesTuple<T>> {
  return new Observable<ObservedValuesTuple<T>>((observer) => {
    const len = sources.length;
    if (len === 0) {
      observer.next([] as unknown as ObservedValuesTuple<T>);
      observer.complete?.();
      return;
    }

    const values = new Array<any>(len);
    const hasEmitted = new Array<boolean>(len).fill(false);
    const completed = new Array<boolean>(len).fill(false);
    let emittedCount = 0;
    let completedCount = 0;

    const subscriptions: Subscription[] = sources.map((source, index) => {
      return from(source).subscribe({
        next: (val) => {
          values[index] = val;
          if (!hasEmitted[index]) {
            hasEmitted[index] = true;
            emittedCount++;
          }
          if (emittedCount === len) {
            observer.next([...values] as unknown as ObservedValuesTuple<T>);
          }
        },
        error: (err) => observer.error?.(err),
        complete: () => {
          completed[index] = true;
          completedCount++;
          if (completedCount === len) {
            observer.complete?.();
          }
        },
      });
    });

    return () => {
      subscriptions.forEach((sub) => sub.unsubscribe());
    };
  });
}

/**
 * 2. zip
 * Combines multiple Observables by merging their values in lockstep index order.
 * Emits an array of the nth values from each source once all sources have emitted an nth value.
 */
export function zip<T extends readonly ObservableInput<any>[]>(
  sources: [...T]
): Observable<ObservedValuesTuple<T>> {
  return new Observable<ObservedValuesTuple<T>>((observer) => {
    const len = sources.length;
    if (len === 0) {
      observer.complete?.();
      return;
    }

    const buffers: any[][] = Array.from({ length: len }, () => []);
    const completed = new Array<boolean>(len).fill(false);

    const subscriptions: Subscription[] = sources.map((source, index) => {
      return from(source).subscribe({
        next: (val) => {
          buffers[index].push(val);
          // Check if every buffer has at least 1 pending value
          if (buffers.every((buf) => buf.length > 0)) {
            const nextTuple = buffers.map((buf) => buf.shift()) as unknown as ObservedValuesTuple<T>;
            observer.next(nextTuple);
          }
        },
        error: (err) => observer.error?.(err),
        complete: () => {
          completed[index] = true;
          // If a stream completes and has empty queue, no further complete tuples can ever be formed
          if (buffers[index].length === 0) {
            observer.complete?.();
          }
        },
      });
    });

    return () => {
      subscriptions.forEach((sub) => sub.unsubscribe());
    };
  });
}

/**
 * 3. forkJoin
 * Accepts a tuple/array of Observables and waits for all of them to complete.
 * Once all sources complete, emits a tuple of the single LAST emitted value from each.
 */
export function forkJoin<T extends readonly ObservableInput<any>[]>(
  sources: [...T]
): Observable<ObservedValuesTuple<T>> {
  return new Observable<ObservedValuesTuple<T>>((observer) => {
    const len = sources.length;
    if (len === 0) {
      observer.complete?.();
      return;
    }

    const lastValues = new Array<any>(len);
    const hasEmitted = new Array<boolean>(len).fill(false);
    let completedCount = 0;

    const subscriptions: Subscription[] = sources.map((source, index) => {
      return from(source).subscribe({
        next: (val) => {
          lastValues[index] = val;
          hasEmitted[index] = true;
        },
        error: (err) => observer.error?.(err),
        complete: () => {
          if (!hasEmitted[index]) {
            // An inner source completed without emitting any value
            observer.complete?.();
            return;
          }
          completedCount++;
          if (completedCount === len) {
            observer.next([...lastValues] as unknown as ObservedValuesTuple<T>);
            observer.complete?.();
          }
        },
      });
    });

    return () => {
      subscriptions.forEach((sub) => sub.unsubscribe());
    };
  });
}
```

```typescript
import { Observable, OperatorFunction } from "./observable";

export interface ThrottleTimeConfig {
  /**
   * Whether to emit on the leading edge of the time window.
   * @default true
   */
  leading?: boolean;
  /**
   * Whether to emit the latest value on the trailing edge of the window.
   * @default false
   */
  trailing?: boolean;
}

/**
 * 1. debounceTime
 * Emits a notification from the source Observable only after a particular time span
 * has passed without another source emission.
 */
export function debounceTime<T>(duration: number): OperatorFunction<T, T> {
  return (source: Observable<T>) =>
    new Observable<T>((observer) => {
      let timerId: ReturnType<typeof setTimeout> | null = null;
      let lastValue: T | undefined;
      let hasValue = false;
      let isSourceComplete = false;

      const clearTimer = () => {
        if (timerId !== null) {
          clearTimeout(timerId);
          timerId = null;
        }
      };

      const emitValue = () => {
        if (hasValue) {
          observer.next(lastValue!);
          hasValue = false;
        }
        clearTimer();
        if (isSourceComplete) {
          observer.complete?.();
        }
      };

      const sub = source.subscribe({
        next: (val) => {
          clearTimer();
          lastValue = val;
          hasValue = true;
          timerId = setTimeout(emitValue, duration);
        },
        error: (err) => {
          clearTimer();
          observer.error?.(err);
        },
        complete: () => {
          isSourceComplete = true;
          if (!hasValue) {
            observer.complete?.();
          }
        },
      });

      return () => {
        clearTimer();
        sub.unsubscribe();
      };
    });
}

/**
 * 2. throttleTime
 * Emits a value from the source Observable, then ignores subsequent source
 * values for `duration` milliseconds.
 */
export function throttleTime<T>(
  duration: number,
  config: ThrottleTimeConfig = {}
): OperatorFunction<T, T> {
  const { leading = true, trailing = false } = config;

  return (source: Observable<T>) =>
    new Observable<T>((observer) => {
      let timerId: ReturnType<typeof setTimeout> | null = null;
      let trailingValue: T | undefined;
      let hasTrailingValue = false;

      const clearTimer = () => {
        if (timerId !== null) {
          clearTimeout(timerId);
          timerId = null;
        }
      };

      const startTimer = () => {
        timerId = setTimeout(() => {
          timerId = null;
          if (trailing && hasTrailingValue) {
            observer.next(trailingValue!);
            hasTrailingValue = false;
            startTimer();
          }
        }, duration);
      };

      const sub = source.subscribe({
        next: (val) => {
          if (timerId === null) {
            if (leading) {
              observer.next(val);
            }
            startTimer();
          } else if (trailing) {
            trailingValue = val;
            hasTrailingValue = true;
          }
        },
        error: (err) => {
          clearTimer();
          observer.error?.(err);
        },
        complete: () => {
          clearTimer();
          if (trailing && hasTrailingValue) {
            observer.next(trailingValue!);
          }
          observer.complete?.();
        },
      });

      return () => {
        clearTimer();
        sub.unsubscribe();
      };
    });
}

/**
 * 3. delay
 * Delays the emission of items from the source Observable by a given timeout.
 */
export function delay<T>(delayDuration: number): OperatorFunction<T, T> {
  return (source: Observable<T>) =>
    new Observable<T>((observer) => {
      const activeTimers = new Set<ReturnType<typeof setTimeout>>();
      let isSourceComplete = false;

      const checkComplete = () => {
        if (isSourceComplete && activeTimers.size === 0) {
          observer.complete?.();
        }
      };

      const sub = source.subscribe({
        next: (val) => {
          const timerId = setTimeout(() => {
            activeTimers.delete(timerId);
            observer.next(val);
            checkComplete();
          }, delayDuration);
          activeTimers.add(timerId);
        },
        error: (err) => {
          activeTimers.forEach(clearTimeout);
          activeTimers.clear();
          observer.error?.(err);
        },
        complete: () => {
          isSourceComplete = true;
          checkComplete();
        },
      });

      return () => {
        activeTimers.forEach(clearTimeout);
        activeTimers.clear();
        sub.unsubscribe();
      };
    });
}
```

```typescript
import { Observable, OperatorFunction, Subscription } from "./observable";

/**
 * 1. catchError
 * Catches errors on the source Observable by returning a new Observable or throwing an error.
 * Provides the caught error and the source Observable (for fallback or retry strategies).
 */
export function catchError<T, R>(
  selector: (err: unknown, caught: Observable<T>) => ObservableInput<R>
): OperatorFunction<T, T | R> {
  return (source: Observable<T>) =>
    new Observable<T | R>((observer) => {
      let sub: Subscription | undefined;
      let innerSub: Subscription | undefined;

      sub = source.subscribe({
        next: (val) => observer.next(val),
        error: (err) => {
          let fallbackObs: Observable<R>;
          try {
            fallbackObs = from(selector(err, source));
          } catch (selectorErr) {
            observer.error?.(selectorErr);
            return;
          }

          innerSub = fallbackObs.subscribe({
            next: (val) => observer.next(val),
            error: (fallbackErr) => observer.error?.(fallbackErr),
            complete: () => observer.complete?.(),
          });
        },
        complete: () => observer.complete?.(),
      });

      return () => {
        sub?.unsubscribe();
        innerSub?.unsubscribe();
      };
    });
}

/**
 * 2. retryWhen
 * Returns an Observable that mirrors the source, but if it calls error, it emits the error
 * to a notifier stream. If the notifier emits a value, the source resubscribes. If the notifier
 * errors or completes, this stream errors or completes.
 */
export function retryWhen<T>(
  notifier: (errors: Observable<unknown>) => Observable<unknown>
): OperatorFunction<T, T> {
  return (source: Observable<T>) =>
    new Observable<T>((observer) => {
      let sourceSub: Subscription | undefined;
      let notifierSub: Subscription | undefined;
      const errors$ = new Subject<unknown>();

      let isClosed = false;

      const subscribeToSource = () => {
        if (isClosed) return;

        sourceSub?.unsubscribe();
        sourceSub = source.subscribe({
          next: (val) => observer.next(val),
          error: (err) => {
            // Forward the error into the notifier pipeline
            errors$.next(err);
          },
          complete: () => {
            errors$.complete();
            observer.complete?.();
          },
        });
      };

      // Set up the notifier trigger stream
      const trigger$ = notifier(errors$);

      notifierSub = trigger$.subscribe({
        next: () => {
          // Whenever the notifier yields an item, retry by resubscribing
          subscribeToSource();
        },
```

```typescript
  accumulator: (acc: R | T, value: T, index: number) => R | T,
  seed?: R

    new Observable<R | T>((observer) => {

/**
 * 2. distinctUntilChanged
 * Emits all items emitted by the source Observable that are distinct by comparison
 * from the previous emitted item.
 */
export function distinctUntilChanged<T, K = T>(
  comparator?: (previous: K, current: K) => boolean,
  keySelector?: (value: T) => K
): OperatorFunction<T, T> {
  const compare = comparator ?? ((a: K, b: K) => a === b);
  const getKey = keySelector ?? ((v: T) => v as unknown as K);

  return (source: Observable<T>) =>
    new Observable<T>((observer) => {
      let previousKey: K;
      let hasPrevious = false;

      return source.subscribe({
        next: (value) => {
          let currentKey: K;
          try {
            currentKey = getKey(value);
          } catch (err) {
            observer.error?.(err);
            return;
          }

          let isDuplicate = false;
          if (hasPrevious) {
            try {
              isDuplicate = compare(previousKey, currentKey);
            } catch (err) {
              observer.error?.(err);
              return;
            }
          }

          if (!isDuplicate) {
            hasPrevious = true;
            previousKey = currentKey;
            observer.next(value);
          }
        },
        error: (err) => observer.error?.(err),
        complete: () => observer.complete?.(),
      }).unsubscribe;
    });
}
```

```typescript
import { Observable, OperatorFunction, Subscription, Subject } from "./observable";

export interface ShareReplayConfig {
  /**
   * Maximum number of items to retain in the replay buffer.
   * @default Infinity
   */
  bufferSize?: number;
  /**
   * Maximum time in milliseconds an item remains in the buffer.
   * @default Infinity
   */
  windowTime?: number;
  /**
   * If true, unsubscribing all observers resets the source connection and buffer.
   * @default false
   */
  refCount?: boolean;
}

interface ReplayEntry<T> {
  value: T;
  time: number;
}

/**
 * 1. shareReplay
 * Multicasts the source Observable with a replay buffer of specified size and window time.
 */
export function shareReplay<T>(
  bufferSize?: number
): OperatorFunction<T, T>;
export function shareReplay<T>(
  config: ShareReplayConfig
): OperatorFunction<T, T>;
export function shareReplay<T>(
  configOrBufferSize?: number | ShareReplayConfig
): OperatorFunction<T, T> {
  const config: ShareReplayConfig =
    typeof configOrBufferSize === "number"
      ? { bufferSize: configOrBufferSize, refCount: false }
      : configOrBufferSize ?? { bufferSize: Infinity, refCount: false };

  const { bufferSize = Infinity, windowTime = Infinity, refCount = false } = config;

  return (source: Observable<T>) => {
    const buffer: ReplayEntry<T>[] = [];
    const observers = new Set<{
      next: (val: T) => void;
      error?: (err: unknown) => void;
      complete?: () => void;
    }>();

    let sourceSub: Subscription | null = null;
    let isComplete = false;
    let hasError = false;
    let thrownError: unknown = null;

    const trimBuffer = (now: number) => {
      // Remove items that exceeded windowTime
      if (windowTime < Infinity) {
        while (buffer.length > 0 && now - buffer[0].time > windowTime) {
          buffer.shift();
        }
      }
      // Remove oldest items exceeding bufferSize
      while (buffer.length > bufferSize) {
        buffer.shift();
      }
    };

    const resetState = () => {
      sourceSub?.unsubscribe();
      sourceSub = null;
      buffer.length = 0;
      isComplete = false;
      hasError = false;
      thrownError = null;
    };

    return new Observable<T>((observer) => {
      const now = Date.now();
      trimBuffer(now);

      // 1. Replay buffered values to this new subscriber
      for (const entry of buffer) {
        observer.next(entry.value);
      }

      if (hasError) {
        observer.error?.(thrownError);
        return;
      }

      if (isComplete) {
        observer.complete?.();
        return;
      }

      observers.add(observer);

      // 2. Connect to the underlying source if not already connected
      if (!sourceSub) {
        sourceSub = source.subscribe({
          next: (val) => {
            const time = Date.now();
            buffer.push({ value: val, time });
            trimBuffer(time);

            for (const obs of Array.from(observers)) {
              obs.next(val);
            }
          },
          error: (err) => {
            hasError = true;
            thrownError = err;
            for (const obs of Array.from(observers)) {
              obs.error?.(err);
            }
            observers.clear();
          },
          complete: () => {
            isComplete = true;
            for (const obs of Array.from(observers)) {
              obs.complete?.();
            }
            observers.clear();
          },
        });
      }

      // 3. Cleanup on unsubscription
      return () => {
        observers.delete(observer);

        if (refCount && observers.size === 0) {
          resetState();
        }
      };
    });
  };
}

/**
 * 2. ConnectableObservable
 * An Observable that waits for its `connect()` method to be invoked before subscribing to source.
 */
export class ConnectableObservable<T> extends Observable<T> {
  private subject = new Subject<T>();
  private connection: Subscription | null = null;

  constructor(private readonly source: Observable<T>) {
    super((observer) => this.subject.subscribe(observer).unsubscribe);
  }

  /**
   * Connects the shared Subject to the underlying source Observable.
   */
  public connect(): Subscription {
    if (!this.connection || this.connection.closed) {
      this.connection = this.source.subscribe({
        next: (val) => this.subject.next(val),
        error: (err) => this.subject.error(err),
        complete: () => this.subject.complete(),
      });
    }
    return this.connection;
  }
}

/**
 * 3. publish
 * Converts an Observable into a ConnectableObservable using a shared Subject.
 */
export function publish<T>(): (source: Observable<T>) => ConnectableObservable<T> {
  return (source: Observable<T>) => new ConnectableObservable<T>(source);
}
```

```typescript
import { Observable, OperatorFunction } from "./observable";

/**
 * 1. bufferCount
 * Buffers the source Observable values until the size reaches `bufferSize`.
 * If `startBufferEvery` is provided, creates a new buffer every `startBufferEvery` emissions,
 * enabling sliding/overlapping or skipping buffers.
 */
export function bufferCount<T>(
  bufferSize: number,
  startBufferEvery: number | null = null
): OperatorFunction<T, T[]> {
  return (source: Observable<T>) =>
    new Observable<T[]>((observer) => {
      const buffers: T[][] = [];
      let count = 0;
      const step = startBufferEvery ?? bufferSize;

      return source.subscribe({
        next: (value) => {
          // Open a new buffer on interval
          if (count % step === 0) {
            buffers.push([]);
          }
          count++;

          // Push to all currently active open buffers
          for (let i = buffers.length - 1; i >= 0; i--) {
            const buf = buffers[i];
            buf.push(value);

            if (buf.length === bufferSize) {
              buffers.splice(i, 1);
              observer.next(buf);
            }
          }
        },
        error: (err) => observer.error?.(err),
        complete: () => {
          // Flush remaining open buffers
          while (buffers.length > 0) {
            const buf = buffers.shift()!;
            if (buf.length > 0) {
              observer.next(buf);
            }
          }
          observer.complete?.();
        },
      }).unsubscribe;
    });
}

/**
 * 2. bufferTime
 * Buffers the incoming values for a specified time span.
 * Supports periodic buffer opening interval (`bufferCreationInterval`) and maximum buffer capacity (`maxBufferSize`).
 */
export function bufferTime<T>(
  bufferTimeSpan: number,
  bufferCreationInterval: number | null = null,
  maxBufferSize: number = Infinity
): OperatorFunction<T, T[]> {
  return (source: Observable<T>) =>
    new Observable<T[]>((observer) => {
      let isClosed = false;

      // Case A: Non-overlapping buffers (interval is null or equals span)
      if (bufferCreationInterval == null || bufferCreationInterval === bufferTimeSpan) {
        let currentBuffer: T[] = [];
        let timerId: ReturnType<typeof setInterval> | null = null;

        const flush = () => {
          const buf = currentBuffer;
          currentBuffer = [];
          observer.next(buf);
        };

        timerId = setInterval(flush, bufferTimeSpan);

        const sub = source.subscribe({
          next: (value) => {
            currentBuffer.push(value);
            if (currentBuffer.length >= maxBufferSize) {
              flush();
            }
          },
          error: (err) => {
            if (timerId) clearInterval(timerId);
            observer.error?.(err);
          },
          complete: () => {
            if (timerId) clearInterval(timerId);
            observer.next(currentBuffer);
            observer.complete?.();
          },
        });

        return () => {
          if (timerId) clearInterval(timerId);
          sub.unsubscribe();
        };
      }

      // Case B: Overlapping or gap buffers with distinct creation intervals
      const buffers: { data: T[]; timer: ReturnType<typeof setTimeout> }[] = [];
      let creationTimer: ReturnType<typeof setInterval> | null = null;

      const startBuffer = () => {
        if (isClosed) return;
        const newBufferData: T[] = [];

        const closeTimer = setTimeout(() => {
          const idx = buffers.findIndex((b) => b.data === newBufferData);
          if (idx !== -1) {
            buffers.splice(idx, 1);
            observer.next(newBufferData);
          }
        }, bufferTimeSpan);

        buffers.push({ data: newBufferData, timer: closeTimer });
      };

      // Open initial buffer and setup periodic creator
      startBuffer();
      creationTimer = setInterval(startBuffer, bufferCreationInterval);

      const sub = source.subscribe({
        next: (value) => {
          for (let i = buffers.length - 1; i >= 0; i--) {
            const bufObj = buffers[i];
            bufObj.data.push(value);

            if (bufObj.data.length >= maxBufferSize) {
              clearTimeout(bufObj.timer);
              buffers.splice(i, 1);
              observer.next(bufObj.data);
            }
          }
        },
        error: (err) => {
          isClosed = true;
          if (creationTimer) clearInterval(creationTimer);
          buffers.forEach((b) => clearTimeout(b.timer));
          observer.error?.(err);
        },
        complete: () => {
          isClosed = true;
          if (creationTimer) clearInterval(creationTimer);
          buffers.forEach((b) => {
            clearTimeout(b.timer);
            observer.next(b.data);
          });
          buffers.length = 0;
          observer.complete?.();
        },
      });

      return () => {
        isClosed = true;
        if (creationTimer) clearInterval(creationTimer);
        buffers.forEach((b) => clearTimeout(b.timer));
        sub.unsubscribe();
      };
    });
}
```

```typescript
 * of size `windowSize`.

  startWindowEvery: number | null = null

            windows.push({ subject, count: 0 });

              win.subject.complete();
              windows.splice(i, 1);

          windows.length = 0;
          observer.complete?.();

        windows.forEach((w) => w.subject.complete());

      let creationTimer: ReturnType<typeof setInterval> | null = null;

          }
        }, windowTimeSpan);

        windows.push({ subject, size: 0, timer: closeTimer });
      };

      startWindow();
      creationTimer = setInterval(startWindow, windowCreationInterval);

      const sub = source.subscribe({
        next: (value) => {
          for (let i = windows.length - 1; i >= 0; i--) {
            const win = windows[i];
            win.subject.next(value);
            win.size++;

            if (win.size >= maxWindowSize) {
              clearTimeout(win.timer);
              win.subject.complete();
              windows.splice(i, 1);
            }
          }
        },
        error: (err) => {
          isClosed = true;
          if (creationTimer) clearInterval(creationTimer);
          windows.forEach((w) => {
            clearTimeout(w.timer);
            w.subject.error(err);
          });
          windows.length = 0;
          observer.error?.(err);
        },
        complete: () => {
          isClosed = true;
          if (creationTimer) clearInterval(creationTimer);
          windows.forEach((w) => {
            clearTimeout(w.timer);
            w.subject.complete();
          });
          windows.length = 0;
          observer.complete?.();
        },
      });

      return () => {
        isClosed = true;
        if (creationTimer) clearInterval(creationTimer);
        windows.forEach((w) => {
          clearTimeout(w.timer);
          w.subject.complete();
        });
        windows.length = 0;
        sub.unsubscribe();
      };
    });
}
```

Here are the fully typed TypeScript implementations for **`auditTime`** and **`sampleTime`**.

---

### Core Difference

* **`auditTime(duration)`**: Silence-window rate limiter. When a value arrives, a timer starts; when the timer expires, it emits the **most recent value** received during that window. If no values arrive, no timer runs.
* **`sampleTime(period)`**: Periodic sampler. A periodic clock runs continuously and emits the **latest value** received since the last sample tick. If no new value arrived during that period, nothing is emitted.

---

### Implementations

```typescript
import { Observable, OperatorFunction } from "./observable";

/**
 * 1. auditTime
 * Ignores source values for `duration` milliseconds, then emits the most recent
 * value from the source Observable.
 */
export function auditTime<T>(duration: number): OperatorFunction<T, T> {
  return (source: Observable<T>) =>
    new Observable<T>((observer) => {
      let timerId: ReturnType<typeof setTimeout> | null = null;
      let lastValue: T | undefined;
      let hasValue = false;

      const clearTimer = () => {
        if (timerId !== null) {
          clearTimeout(timerId);
          timerId = null;
        }
      };

      const sub = source.subscribe({
        next: (value) => {
          lastValue = value;
          hasValue = true;

          // Start timer only on the first arrival in a new window
          if (timerId === null) {
            timerId = setTimeout(() => {
              timerId = null;
              if (hasValue) {
                hasValue = false;
                observer.next(lastValue!);
              }
            }, duration);
          }
        },
        error: (err) => {
          clearTimer();
          observer.error?.(err);
        },
        complete: () => {
          clearTimer();
          observer.complete?.();
        },
      });

      return () => {
        clearTimer();
        sub.unsubscribe();
      };
    });
}

/**
 * 2. sampleTime
 * Periodically looks at the source Observable and emits whichever value it has
 * most recently emitted since the previous sampling, if any.
 */
export function sampleTime<T>(period: number): OperatorFunction<T, T> {
  return (source: Observable<T>) =>
    new Observable<T>((observer) => {
      let lastValue: T | undefined;
      let hasValue = false;
      let timerId: ReturnType<typeof setInterval> | null = null;

      timerId = setInterval(() => {
        if (hasValue) {
          hasValue = false;
          observer.next(lastValue!);
        }
      }, period);

      const sub = source.subscribe({
        next: (value) => {
          lastValue = value;
          hasValue = true;
        },
        error: (err) => {
          if (timerId !== null) clearInterval(timerId);
          observer.error?.(err);
        },
        complete: () => {
          if (timerId !== null) clearInterval(timerId);
          observer.complete?.();
        },
      });

      return () => {
        if (timerId !== null) clearInterval(timerId);
        sub.unsubscribe();
      };
    });
}
```

### Implementations

```typescript
import { Observable, Observer, OperatorFunction } from "./observable";

export type TapObserver<T> = Partial<Observer<T>> & {
  subscribe?: () => void;
  unsubscribe?: () => void;
  finalize?: () => void;
};

/**
 * 1. tap
 * Performs side-effects for notifications from the source Observable without
 * modifying the emitted items or timing.
 */
export function tap<T>(
  observerOrNext?: TapObserver<T> | ((value: T) => void),
  error?: (err: unknown) => void,
  complete?: () => void
): OperatorFunction<T, T> {
  const tapObserver: TapObserver<T> =
    typeof observerOrNext === "function"
      ? { next: observerOrNext, error, complete }
      : observerOrNext ?? {};

  return (source: Observable<T>) =>
    new Observable<T>((observer) => {
      tapObserver.subscribe?.();

      let isCleanedUp = false;
      const onFinalize = () => {
        if (!isCleanedUp) {
          isCleanedUp = true;
          tapObserver.finalize?.();
        }
      };

      const sub = source.subscribe({
        next: (value) => {
          try {
            tapObserver.next?.(value);
          } catch (err) {
            observer.error?.(err);
            return;
          }
          observer.next(value);
        },
        error: (err) => {
          try {
            tapObserver.error?.(err);
          } catch (tapErr) {
            observer.error?.(tapErr);
            onFinalize();
            return;
          }
          observer.error?.(err);
          onFinalize();
        },
        complete: () => {
          try {
            tapObserver.complete?.();
          } catch (tapErr) {
            observer.error?.(tapErr);
            onFinalize();
            return;
          }
          observer.complete?.();
          onFinalize();
        },
      });

      return () => {
        try {
          tapObserver.unsubscribe?.();
        } finally {
          onFinalize();
          sub.unsubscribe();
        }
      };
    });
}

/**
 * 2. finalize
 * Calls a specified callback function when the source terminates on complete,
 * on error, or when the consumer explicitly unsubscribes.
 */
export function finalize<T>(callback: () => void): OperatorFunction<T, T> {
  return (source: Observable<T>) =>
    new Observable<T>((observer) => {
      let isFinalized = false;

      const runCallback = () => {
        if (!isFinalized) {
          isFinalized = true;
          try {
            callback();
          } catch (err) {
            console.error("Error in finalize callback:", err);
          }
        }
      };

      const sub = source.subscribe({
        next: (val) => observer.next(val),
        error: (err) => {
          try {
            observer.error?.(err);
          } finally {
            runCallback();
          }
        },
        complete: () => {
          try {
            observer.complete?.();
          } finally {
            runCallback();
          }
        },
      });

      return () => {
        try {
          sub.unsubscribe();
        } finally {
          runCallback();
        }
      };
    });
}

/**
 * 3. defaultIfEmpty
 * Emits a specified default value if the source Observable completes without
 * emitting any next value.
 */
export function defaultIfEmpty<T, R = T>(
  defaultValue: R
): OperatorFunction<T, T | R> {
  return (source: Observable<T>) =>
    new Observable<T | R>((observer) => {
      let hasEmitted = false;

      return source.subscribe({
        next: (val) => {
          hasEmitted = true;
          observer.next(val);
        },
        error: (err) => observer.error?.(err),
        complete: () => {
          if (!hasEmitted) {
            observer.next(defaultValue);
          }
          observer.complete?.();
        },
      }).unsubscribe;
    });
}
```

```typescript
import { Observable, OperatorFunction } from "./observable";

export type Comparer<T> = (x: T, y: T) => number;

/**
 * 1. max
 * Emits the item with the maximum value from the source Observable once it completes.
 * Accepts an optional custom comparator `(x, y) => number`.
 */
export function max<T>(comparer?: Comparer<T>): OperatorFunction<T, T> {
  const compare: Comparer<T> =
    comparer ?? ((x: any, y: any) => (x > y ? 1 : x < y ? -1 : 0));

  return (source: Observable<T>) =>
    new Observable<T>((observer) => {
      let maxValue: T | undefined;
      let hasValue = false;

      return source.subscribe({
        next: (value) => {
          if (!hasValue) {
            maxValue = value;
            hasValue = true;
          } else {
            try {
              if (compare(value, maxValue!) > 0) {
                maxValue = value;
              }
            } catch (err) {
              observer.error?.(err);
            }
          }
        },
        error: (err) => observer.error?.(err),
        complete: () => {
          if (hasValue) {
            observer.next(maxValue!);
          }
          observer.complete?.();
        },
      }).unsubscribe;
    });
}

/**
 * 2. min
 * Emits the item with the minimum value from the source Observable once it completes.
 * Accepts an optional custom comparator `(x, y) => number`.
 */

  const compare: Comparer<T> =
    comparer ?? ((x: any, y: any) => (x > y ? 1 : x < y ? -1 : 0));

  return (source: Observable<T>) =>
    new Observable<T>((observer) => {
      let minValue: T | undefined;
      let hasValue = false;

      return source.subscribe({
        next: (value) => {
          if (!hasValue) {
            minValue = value;

                minValue = value;
              }
            } catch (err) {
              observer.error?.(err);
            }
          }
        },
        error: (err) => observer.error?.(err),
        complete: () => {
          if (hasValue) {
            observer.next(minValue!);
          }
          observer.complete?.();
        },
      }).unsubscribe;
    });
}

/**
 * 3. count
 * Counts the number of emissions that match the optional predicate function
 * and emits that count when the source completes.
 */
export function count<T>(
  predicate?: (value: T, index: number) => boolean
): OperatorFunction<T, number> {
  return (source: Observable<T>) =>
    new Observable<number>((observer) => {
      let total = 0;
      let index = 0;

      return source.subscribe({
        next: (value) => {
          const currentIndex = index++;
          if (!predicate) {
            total++;
          } else {
            try {
              if (predicate(value, currentIndex)) {
                total++;
              }
            } catch (err) {
              observer.error?.(err);
            }
          }
        },
        error: (err) => observer.error?.(err),
        complete: () => {
          observer.next(total);
          observer.complete?.();
        },
      }).unsubscribe;
    });
}
```

```typescript
/**
 * 1. every
 * Determines whether all items emitted by the source Observable satisfy the predicate.
 * Short-circuits and emits `false` immediately upon the first failure.
 */
export function every<T>(
  predicate: (value: T, index: number) => boolean
): OperatorFunction<T, boolean> {
  return (source: Observable<T>) =>
    new Observable<boolean>((observer) => {
      let index = 0;

      const sub = source.subscribe({
        next: (value) => {
          let matches = false;
          try {
            matches = predicate(value, index++);
          } catch (err) {
            observer.error?.(err);
            sub.unsubscribe();
            return;
          }

          if (!matches) {
            observer.next(false);
            observer.complete?.();
            sub.unsubscribe();
          }
        },
        error: (err) => observer.error?.(err),
        complete: () => {
          observer.next(true);
          observer.complete?.();
        },
      });

      return sub.unsubscribe;
    });
}
```

```typescript
import { Observable, OperatorFunction, Subscription } from "./observable";
import { from, ObservableInput } from "./flattening-operators";

/**
 * 1. takeUntil
 * Emits values from the source Observable until a `notifier` ObservableInput emits its first value.
 * Commonly used for stream lifecycle termination (e.g. component unmount).
 */
export function takeUntil<T>(
  notifier: ObservableInput<unknown>
): OperatorFunction<T, T> {
  return (source: Observable<T>) =>
    new Observable<T>((observer) => {
      let sourceSub: Subscription | undefined;
      let notifierSub: Subscription | undefined;

      // Subscribe to the notifier trigger stream
      notifierSub = from(notifier).subscribe({
        next: () => {
          observer.complete?.();
          cleanup();
        },
        error: (err) => {
          observer.error?.(err);
          cleanup();
        },
      });

      // Subscribe to the source stream
      sourceSub = source.subscribe({
        next: (val) => observer.next(val),
        error: (err) => {
          observer.error?.(err);
          cleanup();
        },
        complete: () => {
          observer.complete?.();
          cleanup();
        },
      });

      const cleanup = () => {
        sourceSub?.unsubscribe();
        notifierSub?.unsubscribe();
      };

      return cleanup;
    });
}

/**
 * 2. takeWhile
 * Emits values emitted by the source Observable so long as each value satisfies the predicate.
 * Completes as soon as the predicate returns false.
 * Supports inclusive emission of the failing value via `inclusive: true`.
 */
export function takeWhile<T, S extends T>(
  predicate: (value: T, index: number) => value is S,
  inclusive?: false
): OperatorFunction<T, S>;
export function takeWhile<T>(
  predicate: (value: T, index: number) => boolean,
  inclusive?: boolean
): OperatorFunction<T, T>;
export function takeWhile<T>(
  predicate: (value: T, index: number) => boolean,
  inclusive: boolean = false
): OperatorFunction<T, T> {
  return (source: Observable<T>) =>
    new Observable<T>((observer) => {
      let index = 0;

      const sub = source.subscribe({
        next: (value) => {
          let matched = false;
          try {
            matched = predicate(value, index++);
          } catch (err) {
            observer.error?.(err);
            sub.unsubscribe();
            return;
          }

          if (matched) {
            observer.next(value);
          } else {
            if (inclusive) {
              observer.next(value);
            }
            observer.complete?.();
            sub.unsubscribe();
          }
        },
        error: (err) => observer.error?.(err),
        complete: () => observer.complete?.(),
      });

      return sub.unsubscribe;
    });
}

/**
 * 3. skipUntil
 * Ignores all items emitted by the source Observable until a `notifier` ObservableInput
 * emits its first value.
 */
export function skipUntil<T>(
  notifier: ObservableInput<unknown>
): OperatorFunction<T, T> {
  return (source: Observable<T>) =>
    new Observable<T>((observer) => {
      let taking = false;
      let sourceSub: Subscription | undefined;
      let notifierSub: Subscription | undefined;

      notifierSub = from(notifier).subscribe({
        next: () => {
          taking = true;
          notifierSub?.unsubscribe();
        },
        error: (err) => {
          observer.error?.(err);
          cleanup();
        },
      });

      sourceSub = source.subscribe({
        next: (val) => {
          if (taking) {
            observer.next(val);
          }
        },
        error: (err) => {
          observer.error?.(err);
          cleanup();
        },
        complete: () => {
          observer.complete?.();
          cleanup();
        },
      });

      const cleanup = () => {
        sourceSub?.unsubscribe();
        notifierSub?.unsubscribe();
      };

      return cleanup;
    });
}
```

### Implementations

```typescript
import { Observable, OperatorFunction, Subscription } from "./observable";
import { from, ObservableInput } from "./flattening-operators";

export interface ThrottleConfig {
  /**
   * Whether to emit on the leading edge of the duration window.
   * @default true
   */
  leading?: boolean;
  /**
   * Whether to emit the latest value on the trailing edge of the duration window.
   * @default false
   */
  trailing?: boolean;
}

/**
 * 1. debounce
 * Emits a notification from the source Observable only after a duration determined
 * by another Observable (returned by `durationSelector`) has emitted a value.
 */
export function debounce<T>(
  durationSelector: (value: T) => ObservableInput<unknown>
): OperatorFunction<T, T> {
  return (source: Observable<T>) =>
    new Observable<T>((observer) => {
      let durationSub: Subscription | undefined;
      let lastValue: T | undefined;
      let hasValue = false;
      let isSourceComplete = false;

      const clearDuration = () => {
        durationSub?.unsubscribe();
        durationSub = undefined;
      };

      const emitValue = () => {
        if (hasValue) {
          observer.next(lastValue!);
          hasValue = false;
        }
        clearDuration();
        if (isSourceComplete) {
          observer.complete?.();
        }
      };

      const mainSub = source.subscribe({
        next: (val) => {
          clearDuration();
          lastValue = val;
          hasValue = true;

          let durationObs: Observable<unknown>;
          try {
            durationObs = from(durationSelector(val));
          } catch (err) {
            observer.error?.(err);
            return;
          }

          durationSub = durationObs.subscribe({
            next: emitValue,
            error: (err) => observer.error?.(err),
            complete: emitValue,
          });
        },
        error: (err) => {
          clearDuration();
          observer.error?.(err);
        },
        complete: () => {
          isSourceComplete = true;
          if (!hasValue) {
            observer.complete?.();
          }
        },
      });

      return () => {
        clearDuration();
        mainSub.unsubscribe();
      };
    });
}

/**
 * 2. throttle
 * Emits a value from the source Observable, then ignores subsequent source values
 * for a duration determined by an Observable returned by `durationSelector`.
 */
export function throttle<T>(
  durationSelector: (value: T) => ObservableInput<unknown>,
  config: ThrottleConfig = {}
): OperatorFunction<T, T> {
  const { leading = true, trailing = false } = config;

  return (source: Observable<T>) =>
    new Observable<T>((observer) => {
      let durationSub: Subscription | undefined;
      let trailingValue: T | undefined;
      let hasTrailingValue = false;
      let isSourceComplete = false;

      const startDuration = (value: T) => {
        let durationObs: Observable<unknown>;
        try {
          durationObs = from(durationSelector(value));
        } catch (err) {
          observer.error?.(err);
          return;
        }

        const onDurationEnd = () => {
          durationSub?.unsubscribe();
          durationSub = undefined;

          if (trailing && hasTrailingValue) {
            const nextVal = trailingValue!;
            hasTrailingValue = false;
            observer.next(nextVal);
            startDuration(nextVal);
          } else if (isSourceComplete) {
            observer.complete?.();
          }
        };

        durationSub = durationObs.subscribe({
          next: onDurationEnd,
          error: (err) => observer.error?.(err),
          complete: onDurationEnd,
        });
      };

      const mainSub = source.subscribe({
        next: (val) => {
          if (!durationSub) {
            if (leading) {
              observer.next(val);
            }
            startDuration(val);
          } else if (trailing) {
            trailingValue = val;
            hasTrailingValue = true;
          }
        },
        error: (err) => {
          durationSub?.unsubscribe();
          observer.error?.(err);
        },
        complete: () => {
          isSourceComplete = true;
          if (!durationSub) {
            observer.complete?.();
          }
        },
      });

      return () => {
        durationSub?.unsubscribe();
        mainSub.unsubscribe();
      };
    });
}

/**
 * 3. audit
 * Ignores source values for a duration determined by an Observable returned by `durationSelector`,
 * then emits the most recent value from the source Observable.
 */
export function audit<T>(
  durationSelector: (value: T) => ObservableInput<unknown>
): OperatorFunction<T, T> {
  return (source: Observable<T>) =>
    new Observable<T>((observer) => {
      let durationSub: Subscription | undefined;
      let lastValue: T | undefined;
      let hasValue = false;

      const clearDuration = () => {
        durationSub?.unsubscribe();
        durationSub = undefined;
      };

      const mainSub = source.subscribe({
        next: (val) => {
          lastValue = val;
          hasValue = true;

          if (!durationSub) {
            let durationObs: Observable<unknown>;
            try {
              durationObs = from(durationSelector(val));
            } catch (err) {
              observer.error?.(err);
              return;
            }

            const onDurationEnd = () => {
              clearDuration();
              if (hasValue) {
                hasValue = false;
                observer.next(lastValue!);
              }
            };

            durationSub = durationObs.subscribe({
              next: onDurationEnd,
              error: (err) => observer.error?.(err),
              complete: onDurationEnd,
            });
          }
        },
        error: (err) => {
          clearDuration();
          observer.error?.(err);
        },
        complete: () => {
          clearDuration();
          observer.complete?.();
        },
      });

      return () => {
        clearDuration();
        mainSub.unsubscribe();
      };
    });
}
```

### Marble Parser Implementation

The marble parser understands:

* `-`: 1 virtual time frame step (10 virtual units by default)
* `a-z`, `0-9`: `next` notification with the given value key
* `|`: Stream `complete` notification
* `#`: Stream `error` notification
* `()`: Grouped synchronous emissions occurring on the exact same frame
* `^`: Subscription start point (for hot observables)
* `!`: Unsubscription point (for subscription marble tracking)

```typescript
export class MarbleParser {
  public static readonly FRAME_TIME_FACTOR = 10;

  /**
   * Parses marble strings into an array of scheduled test messages.
   */
  public static parseMarbles<T = any>(
    marbles: string,
    values?: Record<string, T>,
    errorValue: unknown = "error"
  ): TestMessage<T>[] {
    const messages: TestMessage<T>[] = [];
    let frame = 0;
    let inGroup = false;

    for (let i = 0; i < marbles.length; i++) {
      const char = marbles[i];

      switch (char) {
        case " ":
          // Visual spacing, ignore
          break;
        case "-":
          frame += MarbleParser.FRAME_TIME_FACTOR;
          break;
        case "(":
          inGroup = true;
          break;
        case ")":
          inGroup = false;
          frame += MarbleParser.FRAME_TIME_FACTOR;
          break;
        case "|":
          messages.push({
            frame,
            notification: { kind: "C" },
          });
          if (!inGroup) frame += MarbleParser.FRAME_TIME_FACTOR;
          break;
        case "#":
          messages.push({
            frame,
            notification: { kind: "E", error: errorValue },
          });
          if (!inGroup) frame += MarbleParser.FRAME_TIME_FACTOR;
          break;
        case "^":
          // Hot stream subscription start marker (handled at scheduler level)
          break;
        default: {
          const val = values && char in values ? values[char] : (char as unknown as T);
          messages.push({
            frame,
            notification: { kind: "N", value: val },
          });
          if (!inGroup) frame += MarbleParser.FRAME_TIME_FACTOR;
          break;
        }
      }
    }

    return messages;
  }
}
```

```typescript
export class ColdObservable<T> extends Observable<T> {
  public messages: TestMessage<T>[];
  public subscriptions: SubscriptionLog[] = [];

  constructor(messages: TestMessage<T>[], private scheduler: TestScheduler) {
    super((observer: Observer<T>) => {
      const subscribedFrame = this.scheduler.frame;
      const index = this.subscriptions.push({
        subscribedFrame,
        unsubscribedFrame: Infinity,
      }) - 1;

      // Schedule all marble events relative to subscription time
      const scheduledActions: number[] = [];

      for (const msg of this.messages) {
        const actionTime = subscribedFrame + msg.frame;
        this.scheduler.schedule(() => {
          if (msg.notification.kind === "N") observer.next(msg.notification.value!);
          if (msg.notification.kind === "E") observer.error?.(msg.notification.error);
          if (msg.notification.kind === "C") observer.complete?.();
        }, actionTime);
      }

      return () => {
        this.subscriptions[index].unsubscribedFrame = this.scheduler.frame;
      };
    });

    this.messages = messages;
  }
}

export class HotObservable<T> extends Observable<T> {
  public messages: TestMessage<T>[];
  public subscriptions: SubscriptionLog[] = [];
  private observers: Observer<T>[] = [];

  constructor(messages: TestMessage<T>[], private scheduler: TestScheduler) {
    super((observer: Observer<T>) => {
      this.observers.push(observer);
      const index = this.subscriptions.push({
        subscribedFrame: this.scheduler.frame,
        unsubscribedFrame: Infinity,
      }) - 1;

      return () => {
        const obsIdx = this.observers.indexOf(observer);
        if (obsIdx !== -1) this.observers.splice(obsIdx, 1);
        this.subscriptions[index].unsubscribedFrame = this.scheduler.frame;
      };
    });

    this.messages = messages;

    // Schedule events in absolute virtual time
    for (const msg of messages) {
      this.scheduler.schedule(() => {
        for (const obs of Array.from(this.observers)) {
          if (msg.notification.kind === "N") obs.next(msg.notification.value!);
          if (msg.notification.kind === "E") obs.error?.(msg.notification.error);
          if (msg.notification.kind === "C") obs.complete?.();
        }
      }, msg.frame);
    }
  }
}
```

### `TestScheduler` Engine

```typescript
export class TestScheduler {
  public frame: number = 0;
  private actions: { work: () => void; frame: number }[] = [];

  constructor(
    private assertDeepEqual: (actual: any, expected: any) => void = (a, b) => {
      if (JSON.stringify(a) !== JSON.stringify(b)) {
        throw new Error(
          `Assertion Failure:\nExpected: ${JSON.stringify(b, null, 2)}\nActual:   ${JSON.stringify(a, null, 2)}`
        );
      }
    }
  ) {}

  public schedule(work: () => void, frame: number = this.frame): void {
    this.actions.push({ work, frame });
    this.actions.sort((a, b) => a.frame - b.frame);
  }

  public flush(): void {
    while (this.actions.length > 0) {
      const action = this.actions.shift()!;
      this.frame = action.frame;
      action.work();
    }
  }

  public createColdObservable<T = any>(
    marbles: string,
    values?: Record<string, T>,
    error?: unknown
  ): ColdObservable<T> {
    const messages = MarbleParser.parseMarbles<T>(marbles, values, error);
    return new ColdObservable(messages, this);
  }

  public createHotObservable<T = any>(
    marbles: string,
    values?: Record<string, T>,
    error?: unknown
  ): HotObservable<T> {
    const messages = MarbleParser.parseMarbles<T>(marbles, values, error);
    return new HotObservable(messages, this);
  }

  /**
   * Fluent expectation helper to assert output messages match expected marble diagrams.
   */
  public expectObservable<T>(
    observable: Observable<T>,
    unsubscriptionMarbles: string | null = null
  ): {
    toBe: (expectedMarbles: string, values?: Record<string, T>, error?: unknown) => void;
  } {
    const actualMessages: TestMessage<T>[] = [];
    let sub: Subscription;

    this.schedule(() => {
      sub = observable.subscribe({
        next: (value) => {
          actualMessages.push({
            frame: this.frame,
            notification: { kind: "N", value },
          });
        },
        error: (err) => {
          actualMessages.push({
            frame: this.frame,
            notification: { kind: "E", error: err },
          });
        },
        complete: () => {
          actualMessages.push({
            frame: this.frame,
            notification: { kind: "C" },
          });
        },
      });
    }, this.frame);

    if (unsubscriptionMarbles) {
      const unsubFrame = unsubscriptionMarbles.indexOf("!") * MarbleParser.FRAME_TIME_FACTOR;
      this.schedule(() => sub?.unsubscribe(), unsubFrame);
    }

    return {
      toBe: (expectedMarbles: string, values?: Record<string, T>, error?: unknown) => {
        const expectedMessages = MarbleParser.parseMarbles<T>(expectedMarbles, values, error);
        this.flush();
        this.assertDeepEqual(actualMessages, expectedMessages);
      },
    };
  }

  /**
   * Helper runner providing scoped helpers in a clean DSL.
   */
  public run<R>(
    callback: (helpers: {
      cold: typeof this.createColdObservable;
      hot: typeof this.createHotObservable;
      expectObservable: typeof this.expectObservable;
      flush: () => void;
    }) => R
  ): R {
    this.frame = 0;
    this.actions = [];
    const result = callback({
      cold: this.createColdObservable.bind(this),
      hot: this.createHotObservable.bind(this),
      expectObservable: this.expectObservable.bind(this),
      flush: this.flush.bind(this),
    });
    this.flush();
    return result;
  }
}

```
