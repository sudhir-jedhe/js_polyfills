Group an Array of Objects by Multiple Keys (TypeScript)

const employees = [
  { id: 1, department: "Engineering", role: "Frontend", name: "Alice" },
  { id: 2, department: "Engineering", role: "Backend", name: "Bob" },
  { id: 3, department: "Engineering", role: "Frontend", name: "Charlie" },
  { id: 4, department: "HR", role: "Recruiter", name: "David" },
  { id: 5, department: "HR", role: "Recruiter", name: "Emma" },
];

Convert this into:
{
  Engineering: {
    Frontend: [...],
    Backend: [...]
  },
  HR: {
    Recruiter: [...]
  }
}
Using strict TypeScript generics with full type safety.

Here is the fully type-safe, generic solution using recursive conditional types to group by an arbitrary sequence of keys.

---

### Implementation & Recursive Type Definition

```typescript
type PropertyKey = string | number | symbol;

/**
 * Recursive type to compute nested record structure based on the provided keys tuple.
 * e.g., GroupByResult<T, ['department', 'role']> => Record<T['department'], Record<T['role'], T[]>>
 */
export type GroupByResult<T, Keys extends readonly (keyof T)[]> = Keys extends [
  infer First extends keyof T,
  ...infer Rest extends readonly (keyof T)[]
]
  ? Record<
      T[First] extends PropertyKey ? T[First] : string,
      GroupByResult<T, Rest>
    >
  : T[];

/**
 * Groups an array of objects by an arbitrary sequence of keys.
 */
export function groupByMultipleKeys<
  T extends Record<PropertyKey, any>,
  const Keys extends readonly (keyof T)[]
>(array: readonly T[], keys: Keys): GroupByResult<T, Keys> {
  if (keys.length === 0) {
    return array as unknown as GroupByResult<T, Keys>;
  }

  const [currentKey, ...remainingKeys] = keys;
  const result: Record<PropertyKey, any> = {};

  for (const item of array) {
    const groupVal = item[currentKey];
    if (!result[groupVal]) {
      result[groupVal] = [];
    }
    result[groupVal].push(item);
  }

  // Recursively group deeper levels if more keys remain
  if (remainingKeys.length > 0) {
    for (const key of Object.keys(result)) {
      result[key] = groupByMultipleKeys(result[key], remainingKeys as any);
    }
  }

  return result as GroupByResult<T, Keys>;
}

```

Flatten Any Nested Object
Convert
{
  user: {
    profile: {
      name: "John"
    }
  }
}
into
{
  "user.profile.name": "John"
}
with proper TypeScript types.

Here is the fully type-safe implementation using TypeScript recursive template literal types and intersection mapping.

---

### Type Definition & Implementation

```typescript
// Helper to identify plain objects (and exclude primitives, functions, arrays, null, Date, etc.)
type IsPlainObject<T> = T extends object
  ? T extends Function | any[] | Date | RegExp | Set<any> | Map<any, any>
    ? false
    : true
  : false;

// Flattens nested keys into dot-separated paths mapped to their leaf values
type FlattenUnion<T, KeyPrefix extends string = ''> = {
  [K in keyof T & string]: IsPlainObject<T[K]> extends true
    ? FlattenUnion<T[K], KeyPrefix extends '' ? K : `${KeyPrefix}.${K}`>
    : { [P in KeyPrefix extends '' ? K : `${KeyPrefix}.${K}`]: T[K] };
}[keyof T & string];

// Converts union of single-property objects { 'a': 1 } | { 'b': 2 } into an intersection / single object { 'a': 1, 'b': 2 }
type UnionToIntersection<U> = (U extends any ? (k: U) => void : never) extends (
  k: infer I
) => void
  ? I
  : never;

// Prettify helper to collapse intersection types into a clean readable object type
type Id<T> = { [K in keyof T]: T[K] } & {};

export type FlattenObject<T> = Id<UnionToIntersection<FlattenUnion<T>>>;

/**
 * Recursively flattens a nested object into dot-notation paths.
 */
export function flattenObject<T extends Record<string, any>>(
  obj: T,
  prefix = ''
): FlattenObject<T> {
  const result: Record<string, any> = {};

  for (const [key, value] of Object.entries(obj)) {
    const newKey = prefix ? `${prefix}.${key}` : key;

    if (
      value !== null &&
      typeof value === 'object' &&
      !Array.isArray(value) &&
      !(value instanceof Date) &&
      !(value instanceof RegExp)
    ) {
      Object.assign(result, flattenObject(value, newKey));
    } else {
      result[newKey] = value;
    }
  }

  return result as FlattenObject<T>;
}

```

### Type Definition & Implementation

```typescript
type PropertyKey = string | number | symbol;

/**
 * Computes a nested Record type: Record<T[K1], Record<T[K2], T[]>>
 */
export type GroupByTwoKeys<
  T,
  K1 extends keyof T,
  K2 extends keyof T
> = Record<
  T[K1] extends PropertyKey ? T[K1] : string,
  Record<
    T[K2] extends PropertyKey ? T[K2] : string,
    T[]
  >
>;

/**
 * Groups an array of objects by two specified keys into a two-level nested object.
 */
export function groupByTwoKeys<
  T extends Record<PropertyKey, any>,
  K1 extends keyof T,
  K2 extends keyof T
>(
  array: readonly T[],
  primaryKey: K1,
  secondaryKey: K2
): GroupByTwoKeys<T, K1, K2> {
  const result: Record<PropertyKey, Record<PropertyKey, T[]>> = {};

  for (const item of array) {
    const primaryVal = item[primaryKey];
    const secondaryVal = item[secondaryKey];

    if (!result[primaryVal]) {
      result[primaryVal] = {};
    }

    if (!result[primaryVal][secondaryVal]) {
      result[primaryVal][secondaryVal] = [];
    }

    result[primaryVal][secondaryVal].push(item);
  }

  return result as GroupByTwoKeys<T, K1, K2>;
}

```

**6) Creating a `readonly` Property Inside an Interface**

Use the `readonly` modifier prefix before the property name. Once assigned during object initialization, the property cannot be reassigned:

```typescript
interface User {
  readonly id: string;
  name: string;
}

const user: User = { id: "usr_101", name: "Alice" };
user.name = "Bob";     // Allowed
user.id = "usr_102";   // ❌ Cannot assign to 'id' because it is a read-only property.

```

**8) Key Differences: `interface` vs. `type**`

| Feature                 | `interface`                                                              | `type`                                                   |
| ----------------------- | ------------------------------------------------------------------------ | -------------------------------------------------------- |
| **Declaration Merging** | **Supported** (Multiple declarations with the same name merge into one). | **Not supported** (Duplicate identifier error).          |
| **Primitives & Unions** | Cannot name primitive types or union types directly.                     | Can define unions (`type ID = string                     | number`), tuples, and primitives. |
| **Extension Syntax**    | Uses `extends` keyword (`interface B extends A`).                        | Uses intersection operator `&` (`type B = A & { ... }`). |
| **Computed Properties** | Limited in older versions, standard in modern TS.                        | Fully supported via mapped types (`[K in Keys]: ...`).   |
| **Primary Use Case**    | Object shapes, OOP contracts, public library APIs.                       | Unions, complex transforms, mapped types, utility types. |

**9) Creating a Type-Safe Reusable Utility Function Using Generics**

Use generic type parameters with constraints (`extends`) to infer and preserve exact types:

```typescript
/**
 * Plucks a specific property from every object in an array type-safely.
 */
function pluck<T, K extends keyof T>(array: readonly T[], key: K): T[K][] {
  return array.map((item) => item[key]);
}

const users = [
  { id: 1, name: "Alice", age: 30 },
  { id: 2, name: "Bob", age: 25 },
];

const names = pluck(users, "name"); // Type inferred as string[]
const ages = pluck(users, "age");   // Type inferred as number[]

// @ts-expect-error
pluck(users, "invalidKey");        // ❌ Error: Argument of type '"invalidKey"' is not assignable to parameter of type '"id" | "name" | "age"'.

```

**10) Call Stack vs. Web APIs vs. Callback Queue vs. Event Loop**

```
┌─────────────────────────────────────────────────────────────┐
│ 1. CALL STACK (Single-threaded execution)                   │
│    Executes synchronous code frames (LIFO)                  │
└──────────────────────────────┬──────────────────────────────┘
                               │ (Calls async API like setTimeout/fetch)
                               ▼
┌─────────────────────────────────────────────────────────────┐
│ 2. WEB APIs (Browser / C++ threads)                         │
│    Handles timers, network I/O, DOM events concurrently     │
└──────────────────────────────┬──────────────────────────────┘
                               │ (Timer expires / request completes)
                               ▼
┌─────────────────────────────────────────────────────────────┐
│ 3. TASK QUEUES (FIFO)                                       │
│    • Microtask Queue: Promises (.then), MutationObserver    │
│    • Macrotask / Callback Queue: setTimeout, setInterval, UI│
└──────────────────────────────┬──────────────────────────────┘
                               │
                               ▼
┌─────────────────────────────────────────────────────────────┐
│ 4. EVENT LOOP                                               │
│    Continuously checks if the Call Stack is empty.          │
│    Flushes ALL Microtasks -> Flushes ONE Macrotask -> Paint │
└─────────────────────────────────────────────────────────────┘

```

* **Call Stack:** The LIFO (Last In, First Out) stack provided by the JS engine (like V8) where current execution contexts run synchronously.
* **Web APIs:** Browser-provided background threads (DOM, `fetch`, `setTimeout`, indexedDB) that execute tasks outside the single JS thread.
* **Callback / Task Queue:** The FIFO (First In, First Out) holding areas where completed Web API callbacks wait to run.
* **Microtask Queue:** Highest priority (Promises, `queueMicrotask`). Emptied completely after every stack execution.
* **Macrotask Queue:** Lower priority (`setTimeout`, `setInterval`, I/O).

* **Event Loop:** The coordination mechanism that continuously checks if the Call Stack is empty. Once empty, it drains all microtasks, executes the next available macrotask, and allows the browser to paint.
