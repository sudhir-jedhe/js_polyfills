Here are two standard ways to implement the validation, depending on whether you want to exclude `NaN`:

### Solution 1: `typeof` check (Standard)

```javascript
function add(n1, n2) {
  if (typeof n1 !== "number" || typeof n2 !== "number") {
    throw new TypeError("Both arguments must be numbers");
  }
  return n1 + n2;
}

console.log(add(23, 23));   // 46
console.log(add("a", 23));  // Uncaught TypeError: Both arguments must be numbers

```

---

### Solution 2: `typeof` + `NaN` check (Safer)

In JavaScript, `typeof NaN === "number"`, so checking `Number.isNaN()` ensures invalid calculations don't slip through:

```javascript
function add(n1, n2) {
  if (
    typeof n1 !== "number" || 
    typeof n2 !== "number" || 
    Number.isNaN(n1) || 
    Number.isNaN(n2)
  ) {
    throw new TypeError("Both arguments must be valid numbers");
  }
  return n1 + n2;
}

```

Here are the two ways to implement this using variable arguments:

### 1. Modern Rest Parameters (`...args`)

```javascript
function add(...args) {
  if (args.length === 0 || args.some((arg) => typeof arg !== "number" || Number.isNaN(arg))) {
    throw new TypeError("All arguments must be valid numbers");
  }

  return args.reduce((sum, current) => sum + current, 0);
}

console.log(add(23, 23));       // 46
console.log(add(10, 20, 30));   // 60
console.log(add("a", 23));      // Uncaught TypeError: All arguments must be valid numbers

```

---

### 2. Traditional `arguments` Object

```javascript
function add() {
  const args = Array.from(arguments);

  if (args.length === 0 || args.some((arg) => typeof arg !== "number" || Number.isNaN(arg))) {
    throw new TypeError("All arguments must be valid numbers");
  }

  return args.reduce((sum, current) => sum + current, 0);
}

console.log(add(23, 23));       // 46
console.log(add("a", 23));      // Uncaught TypeError: All arguments must be valid numbers

```
